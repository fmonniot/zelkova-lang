1//! The canonical representation of a zelkova programs is a translation of a local
2//! source into the broader world.
3//!
4//! This phase is where we integrate the local parsed module into the rest of the
5//! program. We do the following steps:
6//!
7//! - Resolve all imports
8//! - Qualify all `Name` (eg. a local value `test` in a `Mod.A` module will be renamed `Mod.A.test`)
9//! - Checks that exported names are actually present in the module
10//! - Checks there is none cyclic dependency between this module and others (Might be done earlier, let's see)
11//!
12//! Note that we use `HashMap`'s a lot in this module's structures. This is because later phases will
13//! want to have cheap access to the different components of a `Module`.
14//!
15//! TODO Rename this to core ? I feel it's going to be te main internal representation of the language.

39/// A resolved module
40#[derive(Debug)]
41pub struct Module {
42    pub name: ModuleName,
43    pub exports: Exports,
44    /// Operator name to infix details
45    pub infixes: HashMap<Name, Infix>,
46    pub types: HashMap<Name, UnionType>,
47    pub values: HashMap<Name, Value>,
48    /// True when this module is a JavaScript binding module.
49    /// Such modules have synthetic placeholder bodies and must not be type-checked.
50    pub binding_javascript: bool,
51}
52
53impl Module {
54    /// Build the trimmed-down view of this module other modules import against.
55    ///
56    /// `file` is the [`SourceFileId`] this module was read from, which is what lets
57    /// a later module's diagnostic point back into *this* module's source — see
58    /// [`Interface::file`](super::Interface::file). It is a parameter rather than
59    /// something this method could work out because a `canonical::Module` does not
60    /// know it: only driver code does, and the sole caller
61    /// (`dependencies::ModuleWalker::check_in_order`) is driver code that has the
62    /// map from module name to file in hand. A caller with nothing to give — every
63    /// test that hand-builds an interface — passes `None`, and
64    /// [`Interface::source_span`](super::Interface::source_span) then declines to
65    /// build a cross-module label rather than building one at the wrong place.
66    pub fn to_interface(&self, file: Option<SourceFileId>) -> super::Interface {
67        let values = self
68            .values
69            .iter()
70            .filter_map(|(name, value)| match value {
71                Value::Value { .. } => None,
72                Value::TypedValue { tpe, span, .. } => Some((name.clone(), (*span, tpe.clone()))),
73            })
74            .collect();
75
76        super::Interface {
77            module_name: self.name.clone(),
78            values,
79            unions: self.types.clone(),
80            infixes: self.infixes.clone(),
81            file,
82        }
83    }
84}

134/// A canonical type.
135///
136/// # Why there is no span on this type, or on [`TypeConstructor`]
137///
138/// Every other canonical node this module builds carries a [`NodeSpan`] taken from
139/// the parser node it came from. A `Type` deliberately does not.
140///
141/// The reason `ERR-3` recorded is that a `Type` does not always come from the module
142/// being canonicalized: `Type::from_parser_type` resolves a name through the
143/// [`Environment`], which clones types straight out of the [`Interface`]s of the
144/// modules this one imports, so the `Type` handed back may well have been *written
145/// in a different file* — and a bare span on it would be read as a position in the
146/// importing module's source. `ERR-5` settled that half: a [`SpanLabel`] can now
147/// carry a file of its own, so "written in another file" is no longer the obstacle.
148///
149/// The other half is unfinished work, not an impossibility. `parser::Type` carries a
150/// [`NodeSpan`] per node like every other parser node, and `Type::from_parser_type`
151/// is the recursive walk over it — `tpe.span` is in hand at every level, and each
152/// branch of an `Arrow` arrives as its own spanned `parser::Type`. Those spans are
153/// discarded here by choice. Keeping them means a span field on every canonical
154/// `Type` node, plus a decision about what a `Type` cloned out of an `Environment`
155/// should then report — where it was written, or where it was used. Nobody has
156/// written that.
157///
158/// What exists instead is coarser and enough for the diagnostics that exist: the
159/// *declaration* a `Type` is the type of — [`Value`]'s own `span`, [`UnionType`]'s,
160/// [`Infix`]'s — is what a diagnostic wants to underline ("defined here"), not a
161/// position inside the type expression itself. [`Interface::values`] pairs each
162/// value's `Type` with that declaration's [`NodeSpan`], and
163/// [`Interface::source_span`] pairs it with the file, once [`Interface::file`] is
164/// set. A type error still points at the whole declaration that failed rather than
165/// the sub-expression that disagrees, which is `ERR-4`, a separate limit in the
166/// typer's own `Term`/`Constraint` translation.
167#[derive(Debug, Clone, PartialEq)]
168pub enum Type {
169    Variable(Name),
170    Type(Name, Vec<Type>),
171    // Record
172    // Unit
173    Arrow(Box<Type>, Box<Type>),
174    /// A tuple type. Zelkova keeps Elm's restriction of two or three elements,
175    /// which [`Tuple`] carries in its shape.
176    Tuple(Tuple<Type>),
177    // Alias
178}
179
180impl Type {
181    /// The one conversion in this module that reads a span and produces none.
182    ///
183    /// `parser::Type` carries a [`NodeSpan`] like every other parser node, and it is
184    /// dropped here on purpose — see this type's documentation for why a canonical
185    /// `Type` holds no span at all, and what `ERR-5` built instead.
186    fn from_parser_type(env: &dyn Environment, tpe: &parser::Type) -> Result<Type, Error> {
187        match &tpe.kind {
188            parser::TypeKind::Unqualified(name, vars) => match env.find_type(name) {
189                Some(t) => Ok(t.clone()),
190                None => {
191                    let types = vars
192                        .iter()
193                        .map(|t| Type::from_parser_type(env, t))
194                        .collect::<Result<Vec<_>, Error>>()?;
195
196                    // TODO Insert back into Environment ?
197                    Ok(Type::Type(name.clone(), types))
198                }
199            },
200            parser::TypeKind::Arrow(t1, t2) => Ok(Type::Arrow(
201                Box::new(Type::from_parser_type(env, t1)?),
202                Box::new(Type::from_parser_type(env, t2)?),
203            )),
204            parser::TypeKind::Variable(n) => Ok(Type::Variable(n.clone())),
205            parser::TypeKind::Tuple(tuple) => Ok(Type::Tuple(
206                tuple.try_map(|t| Type::from_parser_type(env, t))?,
207            )),
208        }
209    }
210
211    // TODO Write some tests
212    fn to_linear_types(tpe: &Type) -> Vec<Type> {
213        match tpe {
214            Type::Arrow(a, b) => {
215                let mut next = Type::to_linear_types(b);
216
217                next.insert(0, *a.clone());
218
219                next
220            }
221            _ => vec![tpe.clone()],
222        }
223    }
224}

264/// A canonical pattern, and where it was written.
265///
266/// Same shape as [`parser::Pattern`] — a [`NodeSpan`] beside a kind — and for the
267/// same reason: the children stay plain `Pattern`s, so a reader matches `&p.kind`.
268#[derive(Debug, PartialEq)]
269pub struct Pattern {
270    pub span: NodeSpan,
271    pub kind: PatternKind,

274#[derive(Debug, PartialEq)]
275pub enum PatternKind {
276    Anything,
277    Variable(Name), // TODO Name or QualName ?
278    Int(i64),
279    Float(f64),
280    Char(char),
281    Bool(bool),
282    /// A tuple pattern. Zelkova keeps Elm's restriction of two or three
283    /// elements, which [`Tuple`] carries in its shape.
284    Tuple(Tuple<Pattern>),
285
286    Constructor {
287        ctor: TypeConstructor,
288        args: Vec<Pattern>,
289    },
290}

292impl Pattern {
293    /// A pattern canonicalized from source, keeping the parser node's position.
294    pub fn new(span: NodeSpan, kind: PatternKind) -> Pattern {
295        Pattern { span, kind }
296    }
297
298    /// A pattern with no position — hand-built by a test. See [`NodeSpan`].
299    pub fn bare(kind: PatternKind) -> Pattern {
300        Pattern {
301            span: NodeSpan::none(),
302            kind,
303        }
304    }
305
306    fn from_parser(p: &parser::Pattern, env: &dyn Environment) -> Result<Pattern, Error> {
307        let kind = match &p.kind {
308            parser::PatternKind::Anything => PatternKind::Anything,
309            parser::PatternKind::Variable(name) => PatternKind::Variable(name.clone()),
310            parser::PatternKind::Literal(parser::Literal::Int(i)) => PatternKind::Int(*i),
311            parser::PatternKind::Literal(parser::Literal::Float(f)) => PatternKind::Float(*f),
312            parser::PatternKind::Literal(parser::Literal::Char(c)) => PatternKind::Char(*c),
313            parser::PatternKind::Literal(parser::Literal::Bool(b)) => PatternKind::Bool(*b),
314            parser::PatternKind::Tuple(tuple) => {
315                PatternKind::Tuple(tuple.try_map(|p| Pattern::from_parser(p, env))?)
316            }
317            parser::PatternKind::Constructor(name, args) => {
318                // `p.span` covers the constructor and its arguments, which is the
319                // text a "no such constructor" caret should sit under.
320                let ctor = env
321                    .find_type_constructor(name)
322                    .ok_or_else(|| {
323                        let suggestion =
324                            suggest_name(name, env.type_constructor_names().into_iter());
325                        Error::VariantNotFound(
326                            env.module_name().qualify_name(name),
327                            p.span,
328                            suggestion,
329                        )
330                    })?
331                    .clone();
332
333                let args = args
334                    .iter()
335                    .map(|p| Pattern::from_parser(p, env))
336                    .collect::<Result<Vec<_>, Error>>()?;
337
338                PatternKind::Constructor { ctor, args }
339            }
340        };
341
342        Ok(Pattern::new(p.span, kind))
343    }
344}

346/// A canonical expression, and where it was written.
347///
348/// Same shape as [`parser::Expression`] — a [`NodeSpan`] beside a kind — and for the
349/// same reason: the children stay `Box<Expression>`, so a reader matches `&e.kind`.
350#[derive(Debug, PartialEq)]
351pub struct Expression {
352    pub span: NodeSpan,
353    pub kind: ExpressionKind,
354}
355
356// TODO Find a way to detect recursive functions (even indirect recursivity,
357// eg. `a` calls `b` calls `a`)
358/// Expression is an optimized version for checks and caches.
359///
360/// Elm declare those expressions:
361/// ```haskell
362/// data Expr_
363///   = VarLocal Name
364///   | VarTopLevel ModuleName.Canonical Name
365///   | VarKernel Name Name
366///   | VarForeign ModuleName.Canonical Name Annotation
367///   | VarCtor CtorOpts ModuleName.Canonical Name Index.ZeroBased Annotation
368///   | VarDebug ModuleName.Canonical Name Annotation
369///   | VarOperator Name ModuleName.Canonical Name Annotation -- CACHE real name for optimization
370///   | Chr ES.String
371///   | Str ES.String
372///   | Int Int
373///   | Float EF.Float
374///   | List [Expr]
375///   | Negate Expr
376///   | Binop Name ModuleName.Canonical Name Annotation Expr Expr -- CACHE real name for optimization
377///   | Lambda [Pattern] Expr
378///   | Call Expr [Expr]
379///   | If [(Expr, Expr)] Expr
380///   | Let Def Expr
381///   | LetRec [Def] Expr
382///   | LetDestruct Pattern Expr Expr
383///   | Case Expr [CaseBranch]
384///   | Accessor Name
385///   | Access Expr (A.Located Name)
386///   | Update Name Expr (Map.Map Name FieldUpdate)
387///   | Record (Map.Map Name Expr)
388///   | Unit
389///   | Tuple Expr Expr (Maybe Expr)
390/// ```
391#[derive(Debug, PartialEq)]
392pub enum ExpressionKind {
393    VarLocal(Name),
394    VarTopLevel(QualName),
395    VarKernel(QualName),
396    VarForeign(QualName, Type),
397    VarConstructor(QualName, Type),
398    Char(char),
399    Int(i64),
400    Float(f64),
401    Bool(bool),
402    // List
403    // Lambda
404    Apply(Box<Expression>, Box<Expression>),
405    If(Box<Expression>, Box<Expression>, Box<Expression>),
406    // Let
407    // LetRec
408    // LetDestruct (eg. `(a,b) = someTuple`)
409    Case(Box<Expression>, Vec<CaseBranch>),
410    // Accessor
411    // Access
412    // Update (record)
413    // Unit
414    /// A tuple expression. Zelkova keeps Elm's restriction of two or three
415    /// elements, which [`Tuple`] carries in its shape.
416    Tuple(Tuple<Expression>),
417}
418
419impl Expression {
420    /// An expression canonicalized from source, keeping the parser node's position.
421    pub fn new(span: NodeSpan, kind: ExpressionKind) -> Expression {
422        Expression { span, kind }
423    }
424
425    /// An expression with no position: hand-built by a test, or synthesised by the
426    /// compiler with nothing in the user's text behind it. See [`NodeSpan`].
427    pub fn bare(kind: ExpressionKind) -> Expression {
428        Expression {
429            span: NodeSpan::none(),
430            kind,
431        }
432    }
433
434    fn from_parser(e: &parser::Expression, env: &dyn Environment) -> Result<Expression, Error> {
435        // Every arm builds a kind and every kind gets `e.span`, so a name the
436        // environment cannot resolve is underlined where it was written rather than
437        // somewhere up the tree.
438        let kind = match &e.kind {
439            parser::ExpressionKind::Lit(parser::Literal::Int(i)) => ExpressionKind::Int(*i),
440            parser::ExpressionKind::Lit(parser::Literal::Float(f)) => ExpressionKind::Float(*f),
441            parser::ExpressionKind::Lit(parser::Literal::Char(c)) => ExpressionKind::Char(*c),
442            parser::ExpressionKind::Lit(parser::Literal::Bool(b)) => ExpressionKind::Bool(*b),
443            parser::ExpressionKind::Variable(name) => {
444                match env.find_value(name).ok_or_else(|| {
445                    let suggestion = suggest_name(name, env.value_names().into_iter());
446                    Error::VariableNotFound(
447                        env.module_name().qualify_name(name),
448                        e.span,
449                        suggestion,
450                    )
451                })? {
452                    ValueType::Local => ExpressionKind::VarLocal(name.clone()),
453                    ValueType::TopLevel => {
454                        ExpressionKind::VarTopLevel(env.module_name().qualify_name(name))
455                    }
456                    ValueType::Foreign(m, _source, tpe) => {
457                        ExpressionKind::VarForeign(m.qualify_name(name), tpe.clone())
458                    }
459                    ValueType::Foreigns(candidates) => {
460                        return Err(Error::AmbiguousVariables(
461                            name.clone(),
462                            candidates.clone(),
463                            e.span,
464                        ))
465                    }
466                }
467            }
468            parser::ExpressionKind::TypeConstructor(name) => {
469                let ctor = env.find_type_constructor(name).ok_or_else(|| {
470                    let suggestion = suggest_name(name, env.type_constructor_names().into_iter());
471                    Error::VariantNotFound(env.module_name().qualify_name(name), e.span, suggestion)
472                })?;
473
474                let tpe = if ctor.type_parameters.is_empty() {
475                    Type::Type(ctor.tpe.clone(), vec![])
476                } else {
477                    // TODO Rework that part. ctor.types is only for the type parameters of the constructor, not for the overall type.
478                    let mut iter = ctor.type_parameters.iter().rev();
479                    let first = iter.next().unwrap().clone();
480
481                    // TODO tests this, 99.999% I'm wrong about it (like everytime I try to implement arrows, plus
482                    // the foldr in this particular case)
483                    let tpe = iter.fold(first, |acc, t| {
484                        Type::Arrow(Box::new(t.clone()), Box::new(acc))
485                    });
486
487                    Type::Arrow(
488                        Box::new(Type::Type(ctor.tpe.clone(), vec![])),
489                        Box::new(tpe),
490                    )
491                };
492
493                let name = name
494                    .to_qual()
495                    .unwrap_or_else(|| env.module_name().qualify_name(name));
496
497                ExpressionKind::VarConstructor(name, tpe)
498            }
499            parser::ExpressionKind::Application(a, b) => {
500                let a = Expression::from_parser(a, env)?;
501                let b = Expression::from_parser(b, env)?;
502
503                ExpressionKind::Apply(Box::new(a), Box::new(b))
504            }
505            parser::ExpressionKind::Tuple(tuple) => {
506                ExpressionKind::Tuple(tuple.try_map(|e| Expression::from_parser(e, env))?)
507            }
508            parser::ExpressionKind::Case(expr, branches) => {
509                let expr = Expression::from_parser(expr, env)?;
510
511                let b = branches.iter().map::<Result<CaseBranch, Error>, _>(|cb| {
512                    let pattern = Pattern::from_parser(&cb.pattern, env)?;
513                    let mut scoped = env.new_scope();
514
515                    scoped.expose_pattern(&pattern);
516
517                    let expression = Expression::from_parser(&cb.expression, &scoped)?;
518
519                    Ok(CaseBranch {
520                        pattern,
521                        expression,
522                        span: cb.span,
523                    })
524                });
525
526                let branches = collect_accumulate(b)?;
527
528                ExpressionKind::Case(Box::new(expr), branches)
529            }
530            parser::ExpressionKind::If(cond, then, els) => {
531                let cond = Expression::from_parser(cond, env)?;
532                let then = Expression::from_parser(then, env)?;
533                let els = Expression::from_parser(els, env)?;
534
535                ExpressionKind::If(Box::new(cond), Box::new(then), Box::new(els))
536            }
537        };
538
539        Ok(Expression::new(e.span, kind))
540    }
541}

553/// Everything canonicalization can reject.
554///
555/// # Why only some variants carry a span
556///
557/// A variant carries a [`NodeSpan`] when its construction site has one in hand — it
558/// is looking at a `parser::Import`, `parser::Infix`, `parser::UnionType`,
559/// `parser::Function` or `parser::Exposed`, all of which the grammar gives a span.
560/// Those are the variants a diagnostic can put a caret under, and today that is every
561/// variant below except three: the two group variants, `EnvironmentErrors` and
562/// `Many`, have no position of their own and flatten their members' labels instead;
563/// and `InvalidTupleSize` carries none for an unrelated reason — see its own doc
564/// comment. Writing `NodeSpan::none()` into a variant that lacks a real position
565/// would say "this error has a position we happen not to know", which is a lie;
566/// leaving the field off says "this error has nowhere to point", which is true, and
567/// the reporter renders it as message-plus-notes with no caret.
568#[derive(Debug)]
569pub enum Error {
570    /// A name in the `module … exposing (…)` header that nothing in the module
571    /// declares, and where that name was written (`ERR-9`).
572    ExportNotFound(Name, ExportType, NodeSpan),
573    EnvironmentErrors(Vec<EnvError>),
574    /// (infix, function), and where the `infix` declaration was written
575    InfixReferenceInvalidValue(Name, Name, NodeSpan),
576    BindingPatternsInvalidLen(NodeSpan),
577    /// A declaration with a type annotation and no body, and where the annotation
578    /// was written — which is the only part of it there is to point at.
579    NoBindings(NodeSpan),
580    /// A name used as a value that nothing in scope declares, where it was
581    /// written — the identifier alone, not the declaration around it — and,
582    /// when one name in scope is a close enough typo-distance match, a
583    /// suggestion for what was meant (`ERR-7`).
584    VariableNotFound(QualName, NodeSpan, Option<Name>),
585    /// A name exposed unqualified by more than one imported module, and where it
586    /// was used. Each candidate module is paired with where the name is declared
587    /// there — `Some` when that module's `Interface` knows both its file and the
588    /// declaration's span, `None` for a hand-built interface (a test) or one built
589    /// before its module's file was known (`ERR-5`).
590    AmbiguousVariables(Name, Vec<(ModuleName, Option<super::SourceSpan>)>, NodeSpan),
591    /// A constructor used in an expression or a pattern that nothing in scope
592    /// declares, where it was written, and an optional "did you mean …?"
593    /// suggestion (`ERR-7`).
594    VariantNotFound(QualName, NodeSpan, Option<Name>),
595    /// Nothing constructs this today — `Environment::find_type_constructor` returns
596    /// at most one constructor per name, so it has no way to report an ambiguity.
597    /// It is the designated rejection path once it can, and carries the span the
598    /// construction site would have, alongside each candidate's declaration
599    /// location — see [`Error::AmbiguousVariables`].
600    AmbiguousVariants(Name, Vec<(ModuleName, Option<super::SourceSpan>)>, NodeSpan),
601    /// A tuple type, pattern or expression had a size other than 2 or 3 (the
602    /// only sizes the language supports).
603    ///
604    /// Nothing constructs this today, and it is kept deliberately. Both ASTs
605    /// now hold their tuples in [`Tuple`], which cannot represent another
606    /// arity, and the grammar has one production per arity — so a bad tuple is
607    /// a parse error and never reaches canonicalization. This variant is the
608    /// designated rejection path should a future source of tuples (a REPL, a
609    /// desugaring pass) build one from a list.
610    InvalidTupleSize(usize),
611    /// A function was declared with multiple bindings (multi-clause definitions),
612    /// which the compiler does not support yet.
613    MultipleBindingsUnsupported(Name, NodeSpan),
614
615    // Binding module
616    InfixDeclared(Name, NodeSpan),
617    TypeDeclared(Name, NodeSpan),
618    NoTypeInBinding(Name, NodeSpan),
619
620    // Utility error
621    Many(Vec<Error>),
622}

624/// Canonicalization errors name source constructs — a value, a type, an operator —
625/// so their messages can be written in the same words the user wrote.
626///
627/// The variants whose construction site had a declaration in hand also point at it;
628/// see the enum's own documentation for why the others do not.
629impl PhaseError for Error {
630    fn message(&self) -> String {
631        match self {
632            Error::ExportNotFound(name, tpe, _) => format!(
633                "`{}` is exposed by this module but no {} of that name is declared in it",
634                name,
635                export_type_noun(tpe)
636            ),
637            Error::EnvironmentErrors(errors) => match errors.as_slice() {
638                [only] => only.message(),
639                many => format!("{} of this module's imports could not be resolved", many.len()),
640            },
641            Error::InfixReferenceInvalidValue(infix, function, _) => format!(
642                "the infix operator `{}` is declared as `{}`, which is not a value declared in this module",
643                infix, function
644            ),
645            Error::BindingPatternsInvalidLen(_) => {
646                "the arguments of this declaration do not line up with its type annotation"
647                    .to_owned()
648            }
649            Error::NoBindings(_) => {
650                "this declaration has a type annotation but no body".to_owned()
651            }
652            Error::VariableNotFound(name, _, _) => {
653                format!("cannot find a value named `{}`", name.to_name())
654            }
655            Error::AmbiguousVariables(name, _, _) => {
656                format!("`{}` is exposed by several imported modules", name)
657            }
658            Error::VariantNotFound(name, _, _) => {
659                format!("cannot find a type constructor named `{}`", name.to_name())
660            }
661            Error::AmbiguousVariants(name, _, _) => format!(
662                "the type constructor `{}` is exposed by several imported modules",
663                name
664            ),
665            Error::InvalidTupleSize(size) => format!(
666                "a tuple has two or three elements, this one has {}",
667                size
668            ),
669            Error::MultipleBindingsUnsupported(name, _) => format!(
670                "`{}` is declared over several bindings, which is not supported yet",
671                name
672            ),
673            Error::InfixDeclared(name, _) => format!(
674                "a `module javascript` facade cannot declare an infix operator, but declares `{}`",
675                name
676            ),
677            Error::TypeDeclared(name, _) => format!(
678                "a `module javascript` facade cannot declare a type, but declares `{}`",
679                name
680            ),
681            Error::NoTypeInBinding(name, _) => format!(
682                "`{}` has no type annotation, and a `module javascript` facade is annotations only",
683                name
684            ),
685            Error::Many(errors) => match errors.as_slice() {
686                [only] => only.message(),
687                many => format!("{} errors while canonicalizing this module", many.len()),
688            },
689        }
690    }
691
692    fn labels(&self) -> Vec<SpanLabel> {
693        // One helper per shape: a variant either has a span and one thing to say
694        // about it, or it delegates to the errors it wraps.
695        let primary = |span: &NodeSpan, message: &str| match span.span() {
696            Some(span) => vec![SpanLabel {
697                span,
698                message: message.to_owned(),
699                primary: true,
700                file: None,
701            }],
702            None => Vec::new(),
703        };
704
705        // A secondary label pointing at where one ambiguous candidate is declared,
706        // in *its own* module's file rather than the one being checked — the
707        // mechanism `ERR-5` adds. `None` when that candidate's `Interface` cannot
708        // say (a hand-built interface, or one built before its file was known)
709        // yields no label for that candidate rather than one at the wrong place.
710        let ambiguous_candidates = |candidates: &[(ModuleName, Option<super::SourceSpan>)]| {
711            candidates
712                .iter()
713                .filter_map(|(module, source)| {
714                    source.map(|s| SpanLabel {
715                        span: s.span,
716                        message: format!("also exposed here, by `{}`", module.name()),
717                        primary: false,
718                        file: Some(s.file),
719                    })
720                })
721                .collect::<Vec<_>>()
722        };
723
724        match self {
725            Error::ExportNotFound(name, _, span) => primary(
726                span,
727                &format!("`{}` is not declared anywhere in this module", name),
728            ),
729            Error::InfixReferenceInvalidValue(_, _, span) => primary(span, "declared here"),
730            // The four that name an identifier: the caret sits under the name the
731            // user wrote, which is the whole point of spanning expressions and
732            // patterns rather than only declarations. `VariableNotFound` and
733            // `VariantNotFound` append their suggestion, when they have one, to
734            // this same label rather than a free-floating note, so the caret and
735            // the suggestion agree about which name is meant (`ERR-7`).
736            Error::VariableNotFound(_, span, suggestion) => primary(
737                span,
738                &format!(
739                    "no value of this name is in scope{}",
740                    suggestion_suffix(suggestion)
741                ),
742            ),
743            Error::AmbiguousVariables(_, candidates, span) => {
744                let mut labels = primary(span, "this name is ambiguous");
745                labels.extend(ambiguous_candidates(candidates));
746                labels
747            }
748            Error::VariantNotFound(_, span, suggestion) => primary(
749                span,
750                &format!(
751                    "no type constructor of this name is in scope{}",
752                    suggestion_suffix(suggestion)
753                ),
754            ),
755            Error::AmbiguousVariants(_, candidates, span) => {
756                let mut labels = primary(span, "this type constructor is ambiguous");
757                labels.extend(ambiguous_candidates(candidates));
758                labels
759            }
760            Error::BindingPatternsInvalidLen(span) => primary(span, "declared here"),
761            Error::NoBindings(span) => primary(span, "this annotation has no body"),
762            Error::MultipleBindingsUnsupported(_, span) => primary(span, "declared here"),
763            Error::InfixDeclared(_, span) => primary(span, "declared here"),
764            Error::TypeDeclared(_, span) => primary(span, "declared here"),
765            Error::NoTypeInBinding(_, span) => primary(span, "declared here"),
766            // A group has no position of its own; the errors it swallowed do.
767            Error::EnvironmentErrors(errors) => errors.iter().flat_map(|e| e.labels()).collect(),
768            Error::Many(errors) => errors.iter().flat_map(|e| e.labels()).collect(),
769            _ => Vec::new(),
770        }
771    }
772
773    fn notes(&self) -> Vec<String> {
774        match self {
775            Error::AmbiguousVariables(_, candidates, _)
776            | Error::AmbiguousVariants(_, candidates, _) => {
777                vec![format!(
778                    "it is exposed by: {}",
779                    candidates
780                        .iter()
781                        .map(|(m, _)| m.name().to_string())
782                        .collect::<Vec<_>>()
783                        .join(", ")
784                )]
785            }
786            // A group renders as a summary, so every message it swallowed becomes a
787            // note. A group of one is rendered by its own message and adds nothing.
788            Error::EnvironmentErrors(errors) => match errors.as_slice() {
789                [only] => only.notes(),
790                many => many.iter().flat_map(|e| e.message_and_notes()).collect(),
791            },
792            Error::Many(errors) => match errors.as_slice() {
793                [only] => only.notes(),
794                many => many.iter().flat_map(|e| e.message_and_notes()).collect(),
795            },
796            _ => Vec::new(),
797        }
798    }
799}

801/// A "did you mean `X`?" suffix for a label message, when a suggestion was
802/// found — empty otherwise, so callers can always append the result without
803/// checking `is_some()` first (`ERR-7`). Mirrors the identically-named helper
804/// in `environment.rs`; kept separate rather than shared because the two
805/// modules' `EnvError`/`Error` types are unrelated and neither should reach
806/// into the other for a two-line formatter.
807fn suggestion_suffix(suggestion: &Option<Name>) -> String {
808    match suggestion {
809        Some(name) => format!(" — did you mean `{}`?", name),
810        None => String::new(),
811    }

835/// Transform a given `parser::Module` into a `canonical::Module`
836pub fn canonicalize(
837    package: &PackageName,
838    interfaces: &HashMap<Name, Interface>,
839    source: &parser::Module,
840) -> Result<Module, Vec<Error>> {
841    let name = ModuleName {
842        package: package.clone(),
843        name: source.name.clone(),
844    };
845
846    let mut errors: Vec<Error> = vec![];
847    let mut env =
848        new_environment(&name, interfaces, &source.imports).map_err(|e| vec![e.into()])?;
849
850    let (infixes, types, values) = if source.binding_javascript {
851        // Javascript modules run a parallel canonicalization process as the constraints are a bit different:
852        // - Only functions without bindings are authorized.
853        // - Infixes and types are forbidden.
854        // The idea being to have the js module be a facade for the actual Javascript module.
855        // Assuming Json types are part of the prelude, this should goes well with the restriction on what types are available for bindings.
856
857        // Verify no infix present
858        if !source.infixes.is_empty() {
859            let e = source
860                .infixes
861                .iter()
862                .map(|i| Error::InfixDeclared(i.operator.clone(), i.span));
863            errors.extend(e);
864        }
865        // Verify no types present
866        if !source.types.is_empty() {
867            let e = source
868                .types
869                .iter()
870                .map(|t| Error::TypeDeclared(t.name.clone(), t.span));
871            errors.extend(e);
872        }
873
874        // Iterate on values
875        let iter = source.functions.iter().map(|function| {
876            // Make sure there is no binding
877            if !function.bindings.is_empty() {
878                //println!("bindings = {:?} (js module)", function.bindings);
879                Err(Error::BindingPatternsInvalidLen(function.span))? // TODO More specific error
880            }
881
882            // Make sure there is a type
883            let tpe = function
884                .tpe
885                .as_ref()
886                .ok_or_else(|| Error::NoTypeInBinding(function.name.clone(), function.span))?;
887            let tpe = Type::from_parser_type(&env, tpe)?;
888
889            let name = function.name.clone();
890            // TODO Think how it's going to be represented. Currently canonical values assume an expression is present
891            //      I'd like to not introduce a trait or new struct for binding. Should we fake an expression or create
892            //      a new type of value ? New type of value will be annoying for regular modules as they aren't present
893            //      there. Fake expression might be ok as MVP. We have to make sure that binding module are removed from
894            //      some phase of the compilation pipeline.
895            let value = Value::TypedValue {
896                name: name.clone(),
897                patterns: vec![],
898                // A `module javascript` facade has no body in the source, so this
899                // stand-in has nothing to point at (see the TODO above).
900                body: Expression::bare(ExpressionKind::Bool(true)),
901                tpe,
902                span: function.span,
903                annotation_span: function.annotation_span,
904            };
905
906            Ok((name, value))
907        });
908        let values = crate::utils::collect_accumulate(iter).unwrap_or_else(|err| {
909            errors.extend(err);
910            HashMap::new()
911        });
912
913        (HashMap::new(), HashMap::new(), values)
914    } else {
915        // Because we are rewriting infixes in this phase, we must do this check before
916        // resolving values.
917        let infixes =
918            do_infixes(&source.infixes, &mut env, &source.functions).unwrap_or_else(|err| {
919                errors.extend(err);
920                HashMap::new()
921            });
922
923        let types = do_types(&env, &source.types).unwrap_or_else(|err| {
924            errors.extend(err);
925            HashMap::new()
926        });
927
928        for (n, t) in types.iter() {
929            env.insert_union_type(n.clone(), t.clone());
930        }
931
932        trace!("Environment after do_types: {:#?}", env);
933
934        // TODO Should I manage infixes rewrite here too ?
935        // Yes I should do it here
936        let values = do_values(&mut env, &source.functions).unwrap_or_else(|err| {
937            errors.extend(err);
938            HashMap::new()
939        });
940
941        (infixes, types, values)
942    };
943
944    // We do exports at the end, and verify that all exported value do
945    // have a reference within the current module
946    let exports = do_exports(&source.exposing, &env).unwrap_or_else(|err| {
947        errors.extend(err);
948        Exports::Everything // Never exposed, as we will return the errors instead
949    });
950
951    if errors.is_empty() {
952        Ok(Module {
953            name,
954            exports,
955            infixes,
956            types,
957            values,
958            binding_javascript: source.binding_javascript,
959        })
960    } else {
961        Err(errors)
962    }
963}

965fn do_values(
966    env: &mut RootEnvironment,
967    functions: &[parser::Function],
968) -> Result<HashMap<Name, Value>, Vec<Error>> {
969    // Before resolving expressions, we store the top-level values in the environment.
970    // We do so first because their expression below could refer to them.
971    for f in functions.iter() {
972        env.insert_top_level_value(f.name.clone());
973    }
974
975    let iter = functions.iter().map(|function| {
976        // Bindings to expression
977
978        // TODO Better error message with position of mismatch
979        // TODO Error when bindings is empty
980        let bindings_size = function
981            .bindings
982            .iter()
983            .all(|v| v.patterns.len() == function.bindings[0].patterns.len());
984
985        if !bindings_size {
986            //println!("bindings = {:?} (bindings_size)", function.bindings);
987            Err(Error::BindingPatternsInvalidLen(function.span))?
988        }
989
990        let (patterns, body): (Vec<Pattern>, Expression) = match function.bindings.len() {
991            0 => Err(Error::NoBindings(function.span)),
992            1 => {
993                // if one binding, we can convert directly to canonical format
994                let binding = &function.bindings[0];
995
996                let mut scoped = env.new_scope();
997
998                let patterns: Vec<Pattern> = binding
999                    .patterns
1000                    .iter()
1001                    .map(|p| Pattern::from_parser(p, env))
1002                    .collect::<Result<Vec<_>, Error>>()?;
1003
1004                for p in &patterns {
1005                    scoped.expose_pattern(p);
1006                }
1007
1008                // Maybe create a case_branch function and make it common with Expression::Case ?
1009                // Or maybe not at the case_branch level, as here we can have multiple patterns
1010                // whereas cases cannot.
1011                // eg. a: Int -> Int -> Int  ==>  a b c = b + c
1012                //println!("Env before transforming expression: {:?}", scoped);
1013                let body = Expression::from_parser(&binding.body, &scoped)?;
1014
1015                Ok((patterns, body))
1016            }
1017            _ => {
1018                // if multiple bindings, we need to create synthetics variables and put all bindings into a case expression
1019                Err(Error::MultipleBindingsUnsupported(
1020                    function.name.clone(),
1021                    function.span,
1022                ))
1023            }
1024        }?;
1025
1026        let name = function.name.clone();
1027
1028        match &function.tpe {
1029            Some(t) => {
1030                let tpe = Type::from_parser_type(env, t)?;
1031                let linear = Type::to_linear_types(&tpe);
1032
1033                // Linear is a list of types making the function. Because it includes the return type,
1034                // it will always be bigger than the number of patterns by one.
1035                if !patterns.is_empty() && (linear.len() - 1 != patterns.len()) {
1036                    // TODO Better error message
1037                    debug!(
1038                        "linear = {:#?}\nbindings = {:#?} (linear.len ({}) != patterns.len ({}))",
1039                        linear,
1040                        function.bindings,
1041                        linear.len(),
1042                        patterns.len()
1043                    );
1044                    Err(Error::BindingPatternsInvalidLen(function.span))?
1045                }
1046
1047                let patterns = patterns.into_iter().zip(linear).collect();
1048
1049                Ok((
1050                    function.name.clone(),
1051                    Value::TypedValue {
1052                        name,
1053                        patterns,
1054                        body,
1055                        tpe,
1056                        span: function.span,
1057                        annotation_span: function.annotation_span,
1058                    },
1059                ))
1060            }
1061            None => Ok((
1062                function.name.clone(),
1063                Value::Value {
1064                    name,
1065                    patterns,
1066                    body,
1067                    span: function.span,
1068                },
1069            )),
1070        }
1071    });
1072
1073    collect_accumulate(iter)
1074}

1076fn do_types(
1077    env: &dyn Environment,
1078    types: &[parser::UnionType],
1079) -> Result<HashMap<Name, UnionType>, Vec<Error>> {
1080    let iter = types.iter().map(|tpe| {
1081        let tpe_name = tpe.name.clone();
1082        let variables = tpe.type_arguments.clone();
1083
1084        trace!("do_types(in:{:?})", tpe);
1085
1086        // variants are represented as parser::Type::Unqualified. Other types
1087        // can be safely ignored in this context.
1088
1089        let variants = tpe
1090            .variants
1091            .iter()
1092            .filter_map(|t| match &t.kind {
1093                // TODO It might actually make more sense to put Type::from_parser_type
1094                // on `Environment`.
1095                parser::TypeKind::Unqualified(name, vars) => Some((name, vars)),
1096                _ => None,
1097            })
1098            .map(|(name, vars)| {
1099                let type_parameters = vars
1100                    .iter()
1101                    .map(|t| Type::from_parser_type(env, t))
1102                    .collect::<Result<Vec<_>, Error>>()?;
1103
1104                Ok(TypeConstructor {
1105                    name: name.clone(),
1106                    type_parameters,
1107                    tpe: tpe_name.clone(),
1108                })
1109            })
1110            .collect::<Result<Vec<_>, Error>>()?;
1111
1112        Ok((
1113            tpe_name,
1114            UnionType {
1115                variables,
1116                variants,
1117                span: tpe.span,
1118            },
1119        ))
1120    });
1121
1122    collect_accumulate(iter)
1123}

1125fn do_infixes(
1126    infixes: &[parser::Infix],
1127    env: &mut RootEnvironment,
1128    functions: &[parser::Function],
1129) -> Result<HashMap<Name, Infix>, Vec<Error>> {
1130    let iter = infixes.iter().map(|infix| {
1131        let op_name = infix.operator.clone();
1132        let function_name = infix.function_name.clone();
1133
1134        let function_exist = functions
1135            .iter()
1136            .find(|f| f.name == infix.function_name)
1137            .is_some();
1138
1139        if function_exist {
1140            let infix = Infix {
1141                associativity: infix.associativity,
1142                precedence: infix.precedence,
1143                function_name,
1144                span: infix.span,
1145            };
1146
1147            env.insert_local_infix(op_name.clone(), infix.clone());
1148
1149            Ok((op_name, infix))
1150        } else {
1151            Err(Error::InfixReferenceInvalidValue(
1152                op_name,
1153                function_name,
1154                infix.span,
1155            ))
1156        }
1157    });
1158
1159    collect_accumulate(iter)
1160}

1162// `BUG-8`: only the `Operator` arm below checks that the exposed name actually
1163// exists — a `Lower` or `Upper` name in a module's own `exposing (...)` header is
1164// accepted unconditionally even when nothing by that name is declared.
1165fn do_exports(
1166    source_exposing: &parser::Exposing,
1167    env: &dyn Environment,
1168) -> Result<Exports, Vec<Error>> {
1169    match source_exposing {
1170        parser::Exposing::Open => Ok(Exports::Everything),
1171        parser::Exposing::Explicit(exposed) => {
1172            let specifics = exposed.iter().map(|exposed| match &exposed.kind {
1173                parser::ExposedKind::Lower(name) => Ok((name.clone(), ExportType::Value)),
1174                parser::ExposedKind::Upper(name, parser::Privacy::Public) => {
1175                    Ok((name.clone(), ExportType::UnionPublic))
1176                }
1177                parser::ExposedKind::Upper(name, parser::Privacy::Private) => {
1178                    Ok((name.clone(), ExportType::UnionPrivate))
1179                }
1180                parser::ExposedKind::Operator(name) => {
1181                    if env.local_infix_exists(name) {
1182                        Ok((name.clone(), ExportType::Infix))
1183                    } else {
1184                        Err(Error::ExportNotFound(
1185                            name.clone(),
1186                            ExportType::Infix,
1187                            exposed.span,
1188                        ))
1189                    }
1190                }
1191            });
1192
1193            let specifics = collect_accumulate(specifics)?;
1194
1195            Ok(Exports::Specifics(specifics))
1196        }
1197    }

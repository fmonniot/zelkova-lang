1//! Collecting a package's tests.
2//!
3//! **A test is a value a module under `tests/` exposes whose type is `zelkova-test`'s
4//! `Test`** — [*What a test is*](../../docs/spec/packages.md#what-a-test-is). This
5//! module is the pass that finds them: given the [`Interface`]s of the modules of a
6//! package's `tests/` root that checked, [`collect`] returns, per module, the sorted names
7//! of the values that have that type.
8//!
9//! It does not run anything, and does not decide how a runner reports what it finds —
10//! that is [`zelkova_test_runner`](crate)'s, the only caller this one is written for.
11//!
12//! # Identified by type, never by spelling
13//!
14//! `Test` is found by its full [`QualName`] — package, module and name — the same way
15//! a [`Scalar`](zelkova_compiler::scalars::Scalar) is
16//! ([`DEC-15`](../../docs/decisions/dec-15.md) decision 1 is that precedent). A
17//! package that declares its own `Test` type does not get its values run:
18//! `MyPackage.Test` and `zelkova-test:Test.Test` are two different declarations that
19//! happen to share a spelling, and only a whole-`QualName` comparison tells them apart.
20//! A package that does not depend on `zelkova-test` at all simply has nothing that can
21//! carry the type, so [`collect`] finds nothing in it and raises no error over that.
22
23use zelkova_compiler::canonical::Type;
24use zelkova_compiler::name::{Name, QualName};
25use zelkova_compiler::{Interface, ModuleName};
26
27/// The package that declares `Test`. Nothing outside the test modules names it: the
28/// checking modules do not know `zelkova-test` exists.
29pub const TEST_PACKAGE: &str = "zelkova-test";
30
31/// The module of [`TEST_PACKAGE`] that declares `Test`, and the name of the type it
32/// declares: both are spelled `Test`.
33const TEST: &str = "Test";
34
35/// Whether `tpe` is exactly `Test`, applied to no arguments.
36///
37/// A type variable, an arrow, a tuple, `Test` applied to an argument, and a type of
38/// any of those shapes declared under a different name all fail this — only
39/// [`Type::Type`] naming the exact declaration [`is_test_type`] accepts, with an empty
40/// argument list, is a test. `Test` has none of its own since it is not generic, so a
41/// value could otherwise only reach this by naming a different, unrelated type that
42/// happens to unify with none of the above; canonicalization already rules that out
43/// for anything that isn't `Type::Type`.
44fn is_test(tpe: &Type) -> bool {
45    matches!(tpe, Type::Type(name, args) if is_test_type(name) && args.is_empty())
46}
47
48/// Whether `name` is the declaration of `Test` in `zelkova-test`'s `Test` module, compared
49/// field by field: the package, the module and the name, all three.
50fn is_test_type(name: &QualName) -> bool {
51    name.package().as_str() == TEST_PACKAGE
52        && name.module_name().as_str() == TEST
53        && name.unqualified_name().as_str() == TEST
54}
55
56/// One test module's collected tests: its own name, and the sorted names of the
57/// values it exposes whose type is `Test`.
58#[derive(Debug, Clone, PartialEq, Eq)]
59pub struct ModuleTests {
60    pub module: ModuleName,
61    /// Sorted by name, so a runner built on this reports tests in one order run to
62    /// run rather than whatever order a `HashMap` happened to iterate in.
63    pub tests: Vec<Name>,
64}
65
66/// The tests each of `modules` exposes.
67///
68/// `modules` is one checked [`Interface`] per module of a package's `tests/` root, which
69/// is what a build that compiled the tests hands back. One
70/// [`ModuleTests`] comes back per module, in the same order, even when its `tests` is
71/// empty — a module that exposes no `Test` is still a module the build held, and
72/// dropping it here would leave a caller unable to tell "no tests" from "not
73/// compiled".
74///
75/// Every `tests` is empty, for every module, when `zelkova-test` is not among the
76/// packages this build reached: nothing can have declared a value of a type nothing
77/// in the build can name, so nothing is found. That is not this pass's error to
78/// raise — a package is free to hold no tests at all
79/// ([*What a test is*](../../docs/spec/packages.md#what-a-test-is)).
80pub fn collect(modules: &[Interface]) -> Vec<ModuleTests> {
81    modules
82        .iter()
83        .map(|interface| {
84            let mut tests: Vec<Name> = interface
85                .values
86                .iter()
87                .filter(|(_, (_, tpe))| is_test(tpe))
88                .map(|(name, _)| name.clone())
89                .collect();
90            tests.sort_by(|a, b| a.as_str().cmp(b.as_str()));
91
92            ModuleTests {
93                module: interface.module_name.clone(),
94                tests,
95            }
96        })
97        .collect()
98}

100#[cfg(test)]
101mod tests {
102    use super::*;
103    use std::collections::HashMap;
104    use zelkova_compiler::PackageName;
105    use zelkova_syntax::position::NodeSpan;

107    fn interface(package: &str, module: &str, values: Vec<(&str, Type)>) -> Interface {
108        Interface {
109            module_name: ModuleName::new(PackageName::new(package).unwrap(), Name::new(module)),
110            values: values
111                .into_iter()
112                .map(|(name, tpe)| (Name::new(name), (NodeSpan::none(), tpe)))
113                .collect(),
114            unions: HashMap::new(),
115            opaque_unions: Default::default(),
116            infixes: HashMap::new(),
117            infix_functions: HashMap::new(),
118            arities: HashMap::new(),
119            file: None,
120            incomplete: false,
121        }
122    }
123
124    fn real_test() -> Type {
125        Type::Type(
126            QualName::in_module(PackageName::new(TEST_PACKAGE).unwrap(), "Test", "Test"),
127            vec![],
128        )
129    }
130
131    fn int() -> Type {
132        Type::Type(
133            QualName::in_module(PackageName::core(), "Basics", "Int"),
134            vec![],
135        )
136    }

138    /// The shape the ticket's acceptance test pins end to end
139    /// (`crates/zelkova/tests/pipeline.rs::a_test_is_found_by_its_qualname_not_its_spelling`), at the
140    /// unit level: a real `Test`, an unrelated `Int`, and a value of some other
141    /// package's own type spelled `Test` too — only the first is collected, and the
142    /// unrelated one never enters the sort.
143    ///
144    /// Each comparison of `is_test_type` has a decoy that differs from the real type in
145    /// that field alone: `acme:Test.Test` (package), `zelkova-test:Other.Test` (module) and
146    /// `zelkova-test:Test.Other` (name); `acme:AppTest.Test` differs in two.
147    ///
148    /// Mutation-checked one comparison at a time, replacing each with `true`: every one
149    /// lets its own decoy in and goes red.
150    #[test]
151    fn only_the_real_test_type_is_collected() {
152        let decoy_type = Type::Type(
153            QualName::in_module(PackageName::new("acme").unwrap(), "AppTest", "Test"),
154            vec![],
155        );
156        let wrong_package = Type::Type(
157            QualName::in_module(PackageName::new("acme").unwrap(), "Test", "Test"),
158            vec![],
159        );
160        let wrong_module = Type::Type(
161            QualName::in_module(PackageName::new(TEST_PACKAGE).unwrap(), "Other", "Test"),
162            vec![],
163        );
164        let wrong_name = Type::Type(
165            QualName::in_module(PackageName::new(TEST_PACKAGE).unwrap(), "Test", "Other"),
166            vec![],
167        );
168        let modules = vec![interface(
169            "acme",
170            "AppTest",
171            vec![
172                ("addsUp", real_test()),
173                ("helper", int()),
174                ("decoyTest", decoy_type),
175                ("wrongPackage", wrong_package),
176                ("wrongModule", wrong_module),
177                ("wrongName", wrong_name),
178            ],
179        )];
180
181        let collected = collect(&modules);
182
183        assert_eq!(collected.len(), 1);
184        assert_eq!(
185            collected[0].tests,
186            vec![Name::new("addsUp")],
187            "only the exposed value of the real `Test` type must be collected"
188        );
189    }

191    /// Four tests of one module come back sorted, not in whatever order the
192    /// `HashMap` they were read from happened to iterate.
193    ///
194    /// Mutation-checked by dropping the `sort_by` call: with only two entries this was
195    /// flaky rather than reliably red, since a two-entry `HashMap` may already iterate
196    /// in order by chance — dropping the call went green in 4 of 10 runs. Four entries,
197    /// inserted out of order, went red in 10 of 10 runs of the same check.
198    #[test]
199    fn tests_of_one_module_are_sorted_by_name() {
200        let modules = vec![interface(
201            "acme",
202            "AppTest",
203            vec![
204                ("zLast", real_test()),
205                ("mSecond", real_test()),
206                ("bThird", real_test()),
207                ("aFirst", real_test()),
208            ],
209        )];
210
211        let collected = collect(&modules);
212
213        assert_eq!(
214            collected[0].tests,
215            vec![
216                Name::new("aFirst"),
217                Name::new("bThird"),
218                Name::new("mSecond"),
219                Name::new("zLast"),
220            ]
221        );
222    }

224    /// A module with no `Test`-typed value still gets an entry, with an empty list —
225    /// dropping it would leave a caller unable to tell "no tests" from "not
226    /// compiled".
227    ///
228    /// Mutation-checked by turning `map` into `filter_map` returning `None` for an
229    /// empty `tests`: the module then disappears from `collected` and the length
230    /// assertion goes red.
231    #[test]
232    fn a_module_with_no_tests_still_gets_an_entry() {
233        let modules = vec![interface("acme", "AppTest", vec![("helper", int())])];
234
235        let collected = collect(&modules);
236
237        assert_eq!(collected.len(), 1);
238        assert!(collected[0].tests.is_empty());
239    }

241    /// [`TEST_PACKAGE`] is compared against a package name's spelling rather than built
242    /// into a `PackageName`, so nothing on the way checks it; this does. A constant that
243    /// broke the package-name rule would name a package no manifest can declare, and no
244    /// value would ever be a test.
245    ///
246    /// Mutation-checked by spelling the constant `zelkova_test`: `new` rejects the
247    /// underscore and this goes red.
248    #[test]
249    fn the_test_package_name_is_a_legal_one() {
250        assert!(PackageName::new(TEST_PACKAGE).is_ok());
251    }

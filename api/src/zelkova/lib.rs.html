1//! Building a package: the half of `zelkova compile`, `zelkova test` and `zelkova run`
2//! that comes after the check.
3//!
4//! [`zelkova_compiler::check_package`] and [`zelkova_compiler::check_package_with_tests`] check a package
5//! and every package it depends on, and print, write and render nothing. This module
6//! takes what they hand back and does the rest: it prints each status line to stderr,
7//! emits a build that checked and writes it, and renders every error to stderr.
8//!
9//! # Emitting and writing
10//!
11//! Once every module of every package has checked, each one is emitted as JavaScript
12//! (`zelkova_js::emit`) and, if that failed nowhere either, the build is written to
13//! `build/out/js/` (`output::write`). A build with any error writes nothing. A build that
14//! also compiled the tests writes a second, complete tree at `build/test/js/` — the
15//! runtime, then one directory per package, holding every package of the build (a
16//! test-only one included) and the root's `tests/` modules beside its `src/` ones — so
17//! that tree alone is what a test runner ever reads from.
18//!
19//! Which file is a facade's companion is this module's to find, not the check's: a
20//! checked module comes back as a [`CheckedSource`] naming the source root it was read
21//! under, and [`zelkova_js::module_file`] names the companion below it ([*A facade names
22//! a boundary, not a
23//! backend*](../docs/spec/interop.md#a-facade-names-a-boundary-not-a-backend)).
24//!
25//! Every error is rendered through `as_diagnostic`, and the status lines are printed
26//! before any of them.

50/// Which of the root package's source roots a build compiles: [`zelkova_compiler::check_package`]'s
51/// `src/` alone, or [`zelkova_compiler::check_package_with_tests`]'s `src/` and `tests/`.
52#[derive(Debug, Clone, Copy, PartialEq, Eq)]
53enum TestRoot {
54    /// `src/` alone.
55    Skipped,
56    /// Both roots.
57    Compiled,
58}

60/// Every way building a package can fail: its check, or emitting, writing or running what
61/// checked.
62///
63/// The check's own errors are [`CompilationError`]s, each wrapped in
64/// [`Check`](BuildError::Check); the other variants are what only a build can raise.
65#[derive(Debug)]
66pub enum BuildError {
67    /// An error of the check. On its own — returned rather than accumulated — it is one
68    /// raised before there was a build to check, the root's manifest or the resolution of
69    /// its dependencies, and is unrendered.
70    Check(CompilationError),
71    /// The named module checked and could not be emitted as JavaScript.
72    Emit(Vec<zelkova_js::Error>, Name),
73    /// A file of the build's output could not be written. Raised only once every module
74    /// of the build has checked and emitted, since nothing is written before that.
75    Output(output::Error),
76    /// A package's tests were compiled and could not be run: `zelkova test` could not
77    /// write its entry point or could not run `node`. A test that ran and did not pass is
78    /// not this; it is the exit code of the run.
79    TestRun(zelkova_test_runner::Error),
80    /// A package's program could not be run: `zelkova run` was pointed at a package with no
81    /// `main`, or could not write its entry point or could not run `node`. A program that
82    /// ran and aborted is not this; it is the exit code of the run.
83    ProgramRun(program_runner::Error),
84    /// An [`Emit`](BuildError::Emit) together with the file its module was read from, so
85    /// that its labels have a file to point into. It wraps nothing else: an error of the
86    /// check carries its own file inside [`Check`](BuildError::Check), as a
87    /// [`CompilationError::InFile`].
88    InFile(Box<BuildError>, SourceFileId),
89    /// Every error a build accumulated, each already rendered to stderr.
90    ///
91    /// Flat: each error of the check is one [`Check`](BuildError::Check) member, and a
92    /// [`CompilationError::Many`] is never wrapped — its members are. A build does not
93    /// stop on the first failure, so that one broken module cannot hide the diagnostics
94    /// of the others; this is how that accumulation becomes a failure again at its end,
95    /// with the typed errors still intact for the caller to inspect.
96    Many(Vec<BuildError>),
97}
98
99impl BuildError {
100    /// Turn this error into the diagnostic the user reads.
101    ///
102    /// A [`Check`](BuildError::Check) renders through [`CompilationError::as_diagnostic`];
103    /// every other variant through [`phase_diagnostic`] or [`plain_diagnostic`], the two
104    /// functions [`CompilationError::as_diagnostic`] shares with this one.
105    pub fn as_diagnostic(&self) -> Diagnostic<SourceFileId> {
106        self.as_diagnostic_in(None)
107    }
108
109    /// The name of the module this error belongs to, when it has one.
110    pub fn module(&self) -> Option<&Name> {
111        match self {
112            BuildError::Emit(_, module) => Some(module),
113            BuildError::Check(error) => error.module(),
114            BuildError::InFile(inner, _) => inner.module(),
115            _ => None,
116        }
117    }
118
119    /// `as_diagnostic`, carrying the file the error's module was read from, which only an
120    /// [`InFile`](BuildError::InFile) wrapper supplies.
121    fn as_diagnostic_in(&self, file: Option<SourceFileId>) -> Diagnostic<SourceFileId> {
122        match self {
123            BuildError::Check(error) => error.as_diagnostic(),
124            BuildError::InFile(inner, id) => inner.as_diagnostic_in(Some(*id)),
125            BuildError::Emit(errors, module) => {
126                phase_diagnostic(module, "code generation", errors, file)
127            }
128            // A path on disk, not a place in any source, so there is nothing to label.
129            BuildError::Output(error) => plain_diagnostic(error),
130            // Like `Output`, about the machine and not about any source.
131            BuildError::TestRun(error) => plain_diagnostic(error),
132            BuildError::ProgramRun(error) => plain_diagnostic(error),
133            // Each member was rendered with its own carets before this was built, so this
134            // summarises rather than flattening their labels: drawing every caret of the
135            // build a second time under one headline is what flattening would do.
136            BuildError::Many(errors) => plain_diagnostic(&Summary { errors, file }),
137        }
138    }
139}

141/// What a [`BuildError::Many`] renders as: one headline counting its members, and each
142/// member's own headline as a note.
143struct Summary<'a> {
144    errors: &'a [BuildError],
145    file: Option<SourceFileId>,
146}
147
148impl PhaseError for Summary<'_> {
149    fn message(&self) -> String {
150        format!(
151            "compilation failed with {} error{}",
152            self.errors.len(),
153            if self.errors.len() == 1 { "" } else { "s" }
154        )
155    }
156
157    fn notes(&self) -> Vec<String> {
158        self.errors
159            .iter()
160            .map(|e| e.as_diagnostic_in(self.file).message)
161            .collect()
162    }
163}

165/// What `?` goes through for an error raised before a build's accumulator exists — the
166/// root's manifest, the resolution — which comes back bare and unrendered.
167impl From<CompilationError> for BuildError {
168    fn from(error: CompilationError) -> Self {
169        BuildError::Check(error)
170    }

173/// Compile the package rooted at `package_dir` — a directory holding a `zelkova.toml`
174/// manifest beside a `src/` source root, per
175/// [`docs/spec/packages.md`](../docs/spec/packages.md#what-a-package-is) — and every
176/// package it depends on.
177///
178/// Its `tests/` root is not compiled; [`compile_package_with_tests`] is that build. A
179/// package is compiled from source the same way whether it is the one asked for or a
180/// dependency of it, and the whole build shares one file database and one error
181/// accumulator: an error in any package of it makes this return `Err`.
182///
183/// A build that checks writes its JavaScript to `build/out/js/` beside `package_dir`'s
184/// manifest — see [`compile_package_into`] for what that tree holds. One that emitted
185/// any error writes nothing.
186///
187/// The check itself is [`zelkova_compiler::check_package`]'s. This prints its status lines to stderr, emits
188/// and writes the build, and renders every error to stderr before returning it inside
189/// [`BuildError::Many`]. An error raised before any source is read — the manifest,
190/// the resolution — is returned bare and unrendered.
191pub fn compile_package(package_dir: &Path) -> Result<(), BuildError> {
192    compile(
193        package_dir,
194        TestRoot::Skipped,
195        &package_dir.join(BUILD_DIRECTORY),
196        &Overlay::new(),
197    )
198    .map(|_| ())
199}

201/// The directory a build's output goes to, beside the root package's manifest and never
202/// beside a source it read ([*The compiler's
203/// interface*](../docs/spec/toolchain.md#the-compilers-interface)).
204pub const BUILD_DIRECTORY: &str = "build";

206/// The tree a build that compiled the tests writes below `build_dir`: `build_dir/test/js/`,
207/// laid out like `build_dir/out/js/`. Both the write of that tree and [`test()`], which hands
208/// it to [`zelkova_test_runner::run`], name it through here.
209pub fn test_tree(build_dir: &Path) -> PathBuf {
210    build_dir.join("test").join("js")
211}

213/// [`compile_package`], writing its output below `build_dir` rather than below the
214/// package's own `build/`.
215///
216/// The output is one tree, `<build_dir>/out/js/`: the runtime at its root, then one
217/// directory per package of the build holding one `.mjs` file per module of that
218/// package, named after the module within its own package, and each facade's companion
219/// beside the facade ([`zelkova_js`]'s *Paths* section has the names, [`DEC-18` decision
220/// 5](../docs/decisions/dec-18.md#5--output-is-written-per-package-beside-the-root-manifest)
221/// the reasons). Nothing is written until every module of every package has checked and
222/// emitted.
223pub fn compile_package_into(package_dir: &Path, build_dir: &Path) -> Result<(), BuildError> {
224    compile(package_dir, TestRoot::Skipped, build_dir, &Overlay::new()).map(|_| ())
225}

227/// [`compile_package`], compiling the package's `tests/` root as well as its `src/`.
228///
229/// This is what running a package's own tests needs, and the only thing that ever reads
230/// a `tests/` root: the tests of the build's other packages are not compiled, because
231/// nothing outside a package reads its tests
232/// ([*Running a package's tests*](../docs/spec/toolchain.md#running-a-packages-tests)).
233/// A package holding no `tests/` at all compiles exactly as it does through
234/// [`compile_package`].
235///
236/// It compiles the tests and does not run them: what makes a declaration a test is
237/// [its type](../docs/spec/packages.md#what-a-test-is), and running one is
238/// [`test()`]'s. What it hands back on success is the `Interface` of each of
239/// the root's own `tests/` modules that checked — never a test-only package's, and never
240/// `src/`'s — so a caller can find which of their exposed values are tests without a phase
241/// dropping the checked modules once they are emitted. `test_collection::collect` is that
242/// pass, and `zelkova_test_runner::run` is its caller. Empty when the package holds no `tests/` at
243/// all.
244///
245/// A test module and every `test-dependency`'s modules are checked and, unlike a plain
246/// build, written — to a tree of their own, `build/test/js/`, laid out exactly like
247/// `build/out/js/` and holding every package of the build (a test-only one included) plus the
248/// root's `tests/` modules beside its `src/` ones. `build/out/js/` itself is left exactly as
249/// [`compile_package`] would have written it: a test module never turns up there, so a
250/// plain build run afterwards never finds one left behind by a run that also compiled the
251/// tests ([`GEN-18`](../docs/tickets/README.md)).
252pub fn compile_package_with_tests(package_dir: &Path) -> Result<Vec<Interface>, BuildError> {
253    compile_package_with_tests_into(package_dir, &package_dir.join(BUILD_DIRECTORY))
254}
255
256/// [`compile_package_with_tests`], writing below `build_dir` rather than below the
257/// package's own `build/` — the same relationship [`compile_package_into`] has to
258/// [`compile_package`].
259pub fn compile_package_with_tests_into(
260    package_dir: &Path,
261    build_dir: &Path,
262) -> Result<Vec<Interface>, BuildError> {
263    compile(package_dir, TestRoot::Compiled, build_dir, &Overlay::new())
264}

266/// Compile the package rooted at `package_dir` with its tests
267/// ([`compile_package_with_tests`]), run every test it holds under `node`
268/// ([`zelkova_test_runner::run`], from the tree [`test_tree`] names), and answer the exit code the
269/// process should end with.
270///
271/// The answer is `Ok(0)` when every test passed and when the package holds none. It is
272/// `node`'s own exit code otherwise, which is non-zero exactly when a test did not pass or
273/// `node` itself failed. Anything that stops the tests being run — a build that did not
274/// compile, `node` missing — is an `Err`, and a build that did not compile never reaches
275/// `node`.
276pub fn test(package_dir: &Path) -> Result<i32, BuildError> {
277    let interfaces = compile_package_with_tests(package_dir)?;
278    zelkova_test_runner::run(&test_tree(&package_dir.join(BUILD_DIRECTORY)), &interfaces)
279        .map_err(BuildError::TestRun)
280}

282/// Print one [`Status`] line to `writer`. Failing to write a status line is not itself a
283/// compilation failure, so the write results are deliberately discarded rather than
284/// unwrapped.
285fn print_status(writer: &mut StandardStream, status: &Status) {
286    let (color, label) = if status.success {
287        (Color::Green, "success")
288    } else {
289        (Color::Red, "failure")
290    };
291    let _ = writer.set_color(ColorSpec::new().set_bold(true).set_fg(Some(color)));
292    let _ = write!(writer, "{}", label);
293    let _ = writer.reset();
294    let _ = writeln!(writer, " {}", status.text);
295}

297/// Each error of the check as one [`BuildError::Check`], with a
298/// [`CompilationError::Many`] replaced by its members, at any depth. This is what keeps
299/// [`BuildError::Many`] flat by construction rather than by `check` never building a `Many`.
300fn check_errors(errors: Vec<CompilationError>) -> Vec<BuildError> {
301    errors
302        .into_iter()
303        .flat_map(|error| match error {
304            CompilationError::Many(members) => check_errors(members),
305            other => vec![BuildError::Check(other)],
306        })
307        .collect()
308}

310/// The CLI half of every `compile_package` variant: check the package
311/// ([`zelkova_compiler::check_package`] or [`zelkova_compiler::check_package_with_tests`], as `tests`
312/// says), print its status lines, emit and write a build that checked, and render every error to stderr.
313///
314/// The status lines are printed once the check has finished rather than as each phase
315/// ends. For a check that returns, the bytes `compile_package` itself writes to stderr are
316/// the same as before, in the same order: every status line, then every diagnostic. Log
317/// output (`RUST_LOG`) and a panic's message no longer interleave with them, and a panic
318/// during checking loses the status lines already recorded.
319fn compile(
320    package_dir: &Path,
321    tests: TestRoot,
322    build_dir: &Path,
323    overlay: &Overlay,
324) -> Result<Vec<Interface>, BuildError> {
325    let PackageCheck {
326        package: root_package,
327        errors,
328        sources,
329        modules,
330        test_dependency_modules,
331        test_modules,
332        // A build emits only what checked; these are for a reader of a module's tree.
333        failing: _,
334        status,
335    } = match tests {
336        TestRoot::Skipped => zelkova_compiler::check_package(package_dir, overlay)?,
337        TestRoot::Compiled => zelkova_compiler::check_package_with_tests(package_dir, overlay)?,
338    };
339
340    // The build's one accumulator: each error of the check on its own, then whatever
341    // emitting and writing the build adds. An empty one is what makes this return `Ok`.
342    let mut errors: Vec<BuildError> = check_errors(errors);
343
344    // Error reporter
345    let mut writer = StandardStream::stderr(ColorChoice::Auto);
346    let config = codespan_reporting::term::Config {
347        tab_width: 2,
348        ..codespan_reporting::term::Config::default()
349    };
350
351    for line in &status {
352        print_status(&mut writer, line);
353    }
354
355    // What `compile_package_with_tests` hands back: the `Interface` of each of the
356    // root's own `tests/` modules that checked — never a test-only package's, and
357    // never `src/`'s. This is what lets a caller find a package's tests
358    // (`test_collection::collect` is that pass) without a phase dropping the checked
359    // modules on the floor once they have been emitted. Stays empty for a build that
360    // did not ask for the tests, or whose root `tests/` did not check.
361    let mut root_test_interfaces: Vec<Interface> = Vec::new();
362
363    // Generate code, only for a build in which nothing failed. Every module of
364    // both trees is emitted before anything is written, so a module that cannot be
365    // emitted — in `src/` or in `tests/` — also leaves the build with no output at all.
366    if errors.is_empty() {
367        let checked: Vec<ModuleToEmit> = modules.into_iter().map(to_module_to_emit).collect();
368        let mut root_tests_checked: Vec<ModuleToEmit> =
369            test_modules.into_iter().map(to_module_to_emit).collect();
370
371        // A test companion imports the companion it checks by its path in the source
372        // tree, across the two roots, which reaches nothing in the build
373        // ([*Testing a companion*](../docs/spec/interop.md#testing-a-companion)). Each
374        // companion of the root's `src/` is one it may check, and `checked` holds every
375        // one of them.
376        let targets: Vec<&Name> = checked
377            .iter()
378            .filter(|to_emit| {
379                to_emit.companion.is_some()
380                    && to_emit.module.canonical.name.package() == &root_package
381            })
382            .map(|to_emit| to_emit.module.canonical.name.name())
383            .collect();
384        for to_emit in root_tests_checked
385            .iter_mut()
386            .filter(|to_emit| to_emit.companion.is_some())
387        {
388            let checks = to_emit.module.canonical.name.name();
389            to_emit.companion_imports = targets
390                .iter()
391                .map(|target| zelkova_js::test_companion_import(checks, target))
392                .collect();
393        }
394        // Built from the same `ModuleToEmit`s `test_tree_modules` is about to take,
395        // before that move: an `Interface` is cheap to clone off a `CheckedModule`
396        // that is otherwise about to be consumed by emission.
397        root_test_interfaces = root_tests_checked
398            .iter()
399            .map(|to_emit| to_emit.module.to_interface(to_emit.file))
400            .collect();
401
402        // Every module the *test* tree (`build/test/js/`) needs beyond what `checked`
403        // already holds: each test-only package's modules, and the root's `tests/` modules.
404        // Empty for a build that did not ask for the tests.
405        let mut test_tree_modules: Vec<ModuleToEmit> = test_dependency_modules
406            .into_iter()
407            .map(to_module_to_emit)
408            .collect();
409        test_tree_modules.extend(root_tests_checked);
410
411        debug!("phase: codegen");
412        // Every union of the build, read by a facade's boundary checks: a facade may name
413        // a union any module of the build declares, and a test-only package's or the
414        // root's `tests/` modules are part of the build their facades see.
415        let unions = zelkova_js::Unions::of(
416            checked
417                .iter()
418                .chain(test_tree_modules.iter())
419                .map(|to_emit| &to_emit.module),
420        );
421        let files = emit_build(checked, &unions, &mut errors);
422
423        // A build that also compiled the tests writes a second, complete tree at
424        // `<build_dir>/test/js/`, laid out exactly like `<build_dir>/out/js/` — the runtime,
425        // then one directory per package — but holding every package of the build (a
426        // test-only one included) and the root's `tests/` modules beside its `src/` ones
427        // (`docs/decisions/dec-18.md#5--output-is-written-per-package-beside-the-root-manifest`).
428        // `files` already holds everything a plain build would have written — the
429        // runtime, the root's `src/` and every plain dependency's modules — so the test
430        // tree reuses it rather than emitting those modules a second time, and only
431        // `test_tree_modules` (the test-only packages' and the root's `tests/`) is new
432        // work. That work happens here, before `out/js/` is written, so a module of the test
433        // tree that cannot be emitted blocks both writes.
434        let test_files = (tests == TestRoot::Compiled).then(|| {
435            debug!("phase: codegen (tests)");
436            let mut test_files = files.clone();
437            test_files.extend(emit_modules(test_tree_modules, &unions, &mut errors));
438            test_files
439        });
440
441        if errors.is_empty() {
442            debug!("phase: write the build");
443            errors.extend(
444                output::write(&build_dir.join("out").join("js"), &files)
445                    .into_iter()
446                    .map(BuildError::Output),
447            );
448        }
449
450        // Gated on `errors.is_empty()` a second time so a plain build's own failure to
451        // write `out/js/` — an unlikely I/O error, not a checking one — does not also
452        // attempt the test tree.
453        if let Some(test_files) = test_files.filter(|_| errors.is_empty()) {
454            debug!("phase: write the test build");
455            errors.extend(
456                output::write(&test_tree(build_dir), &test_files)
457                    .into_iter()
458                    .map(BuildError::Output),
459            );
460        }
461    }
462
463    // Report everything we accumulated, then let that accumulation decide the
464    // return value. Rendering the errors and returning `Ok` regardless was `BUG-1`.
465    for error in &errors {
466        // A rendering failure must not mask the compilation failure we are about to
467        // return, and there is nowhere left to report it to, so it is dropped.
468        let _ = term::emit_to_write_style(
469            &mut writer.lock(),
470            &config,
471            &sources,
472            &error.as_diagnostic(),
473        );
474    }
475
476    if errors.is_empty() {
477        Ok(root_test_interfaces)
478    } else {
479        Err(BuildError::Many(errors))
480    }
481}

483/// A module that checked, with what emitting and writing it needs beyond the module
484/// itself.
485struct ModuleToEmit {
486    module: CheckedModule,
487    /// The file it was read from, which an emission error's labels point into.
488    file: Option<SourceFileId>,
489    /// For a facade, its JavaScript companion when one sits beside its `.zel` source —
490    /// a file of the same base name in the same directory
491    /// ([*A facade names a boundary, not a
492    /// backend*](../docs/spec/interop.md#a-facade-names-a-boundary-not-a-backend)).
493    /// `None` for every other module, and for a facade with no companion, which
494    /// [`zelkova_js::emit`] refuses.
495    companion: Option<std::path::PathBuf>,
496    /// The imports its companion spells for the source tree, each beside the specifier
497    /// that replaces it in the build ([`zelkova_js::test_companion_import`]). Empty for
498    /// every module but a facade of the root package's `tests/` with a companion, which
499    /// [`compile`] fills in once it knows the companions of `src/` that one may import.
500    companion_imports: Vec<(String, String)>,
501}
502
503/// What emitting and writing one checked module needs: its companion when it is a facade
504/// with one sitting beside its source under its `root_dir`, so a facade under `tests/`
505/// finds its companion there rather than under `src/`
506/// ([*Testing a companion*](../docs/spec/interop.md#testing-a-companion)).
507fn to_module_to_emit(checked: CheckedSource) -> ModuleToEmit {
508    let CheckedSource {
509        module,
510        file,
511        root_dir,
512    } = checked;
513    let companion = Some(root_dir.join(zelkova_js::module_file(module.canonical.name.name())))
514        .filter(|path| module.ir.foreign && path.is_file());
515    ModuleToEmit {
516        file,
517        companion,
518        companion_imports: Vec::new(),
519        module,
520    }
521}

523/// Every file a build writes: the runtime, and each module of `checked` as the text
524/// [`zelkova_js::emit`] gives it, with its facade's companion beside it.
525///
526/// A module that cannot be emitted pushes its errors onto `errors`, tagged with the
527/// file it came from, and every other module is still emitted so that one refusal
528/// cannot hide the next. The caller writes the files only when `errors` stays empty.
529fn emit_build(
530    checked: Vec<ModuleToEmit>,
531    unions: &zelkova_js::Unions,
532    errors: &mut Vec<BuildError>,
533) -> Vec<output::File> {
534    let mut files = vec![output::File {
535        path: zelkova_js::RUNTIME_FILE.into(),
536        contents: output::Contents::Text(zelkova_js::RUNTIME.to_string()),
537    }];
538    files.extend(emit_modules(checked, unions, errors));
539    files
540}
541
542/// [`emit_build`], without the runtime file at the front.
543///
544/// [`emit_build`] calls it for the plain build's own modules. A test build calls it a
545/// second time for the modules a plain build never emits — each test-only package's,
546/// and the root's `tests/` — which is why it is factored out: the test tree already has
547/// the runtime, since it starts from a clone of the plain build's own files, and those
548/// extra modules must not emit a second runtime file to sit unused beside the first.
549fn emit_modules(
550    checked: Vec<ModuleToEmit>,
551    unions: &zelkova_js::Unions,
552    errors: &mut Vec<BuildError>,
553) -> Vec<output::File> {
554    let mut files = Vec::new();
555
556    for ModuleToEmit {
557        module,
558        file,
559        companion,
560        companion_imports,
561    } in checked
562    {
563        let name = module.canonical.name.clone();
564        let package_dir = std::path::PathBuf::from(name.package().as_str());
565
566        match zelkova_js::emit(&module, companion.is_some(), unions) {
567            Ok(text) => {
568                files.push(output::File {
569                    path: package_dir.join(zelkova_js::module_file(name.name())),
570                    contents: output::Contents::Text(text),
571                });
572
573                if let Some(companion) = companion {
574                    let contents = if companion_imports.is_empty() {
575                        output::Contents::Copy(companion)
576                    } else {
577                        output::Contents::Rewritten {
578                            from: companion,
579                            imports: companion_imports,
580                        }
581                    };
582                    files.push(output::File {
583                        path: package_dir.join(zelkova_js::companion_file(name.name())),
584                        contents,
585                    });
586                }
587            }
588            Err(emit_errors) => {
589                let error = BuildError::Emit(emit_errors, name.name().clone());
590                errors.push(match file {
591                    Some(id) => BuildError::InFile(Box::new(error), id),
592                    None => error,
593                });
594            }
595        }
596    }
597
598    files
599}

605    /// A `CompilationError::Many` in the check's errors becomes one `Check` per member,
606    /// however deeply it was nested, and a `Many` is never itself wrapped. A plain error
607    /// beside it stays as it was, in order.
608    ///
609    /// Mutation-checked by turning `check_errors` back into a plain
610    /// `.map(BuildError::Check)`: the first assertion sees one `Check` where three are
611    /// expected and this goes red.
612    #[test]
613    fn a_many_of_the_check_is_flattened_into_its_members() {
614        let leaf = || CompilationError::Manifest(vec![]);
615        let nested = CompilationError::Many(vec![
616            leaf(),
617            CompilationError::Many(vec![CompilationError::Resolution(vec![])]),
618        ]);
619
620        let flat = check_errors(vec![nested, leaf()]);
621
622        assert_eq!(flat.len(), 3);
623        assert!(matches!(
624            flat[0],
625            BuildError::Check(CompilationError::Manifest(_))
626        ));
627        assert!(matches!(
628            flat[1],
629            BuildError::Check(CompilationError::Resolution(_))
630        ));
631        assert!(matches!(
632            flat[2],
633            BuildError::Check(CompilationError::Manifest(_))
634        ));
635        assert!(!flat
636            .iter()
637            .any(|e| matches!(e, BuildError::Check(CompilationError::Many(_)))));
638    }
639
640    /// `module` is `Some` for an `Emit` and for a `Check` of one module, `None` for an
641    /// error that is about no one module.
642    ///
643    /// Mutation-checked by making the `Emit` arm return `None`: the first assertion
644    /// goes red.
645    #[test]
646    fn module_names_the_module_of_an_emit_error() {
647        let emit = BuildError::Emit(vec![], Name::new("Main"));
648        assert_eq!(emit.module(), Some(&Name::new("Main")));
649
650        let check = BuildError::Check(CompilationError::Canonical(vec![], Name::new("Other")));
651        assert_eq!(check.module(), Some(&Name::new("Other")));
652
653        assert_eq!(BuildError::Many(vec![]).module(), None);

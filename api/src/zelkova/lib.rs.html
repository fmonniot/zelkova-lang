1//! Building a package: the half of `zelkova compile`, `zelkova test` and `zelkova run`
2//! that comes after the check.
3//!
4//! [`zelkova_compiler::check_package`] and [`zelkova_compiler::check_package_with_tests`] check a package
5//! and every package it depends on, and print, write and render nothing. This module
6//! takes what they hand back and does the rest: it prints each status line to stderr,
7//! emits a build that checked and writes it, and renders every error to stderr.
8//!
9//! # Emitting and writing
10//!
11//! Once every module of every package has checked, the specialisations of the whole build
12//! are found (`zelkova_compiler::ir::specialise`, over the modules of both trees at once), and
13//! each module is emitted as JavaScript
14//! (`zelkova_js::emit`) and, if that failed nowhere either, the build is written to
15//! `build/out/js/` (`output::write`). A build with any error writes nothing. A build that
16//! also compiled the tests writes a second, complete tree at `build/test/js/` — the
17//! runtime, then one directory per package, holding every package of the build (a
18//! test-only one included) and the root's `tests/` modules beside its `src/` ones — so
19//! that tree alone is what a test runner ever reads from.
20//!
21//! Which file is a facade's companion is this module's to find, not the check's: a
22//! checked module comes back as a [`CheckedSource`] naming the source root it was read
23//! under, and [`zelkova_js::module_file`] names the companion below it ([*A facade names
24//! a boundary, not a
25//! backend*](../docs/spec/interop.md#a-facade-names-a-boundary-not-a-backend)).
26//!
27//! Every error is rendered through `as_diagnostic`, and the status lines are printed
28//! before any of them.

54/// Which of the root package's source roots a build compiles: [`zelkova_compiler::check_package`]'s
55/// `src/` alone, or [`zelkova_compiler::check_package_with_tests`]'s `src/` and `tests/`.
56#[derive(Debug, Clone, Copy, PartialEq, Eq)]
57enum TestRoot {
58    /// `src/` alone.
59    Skipped,
60    /// Both roots.
61    Compiled,
62}

64/// Every way building a package can fail: its check, or emitting, writing or running what
65/// checked.
66///
67/// The check's own errors are [`CompilationError`]s, each wrapped in
68/// [`Check`](BuildError::Check); the other variants are what only a build can raise.
69#[derive(Debug)]
70pub enum BuildError {
71    /// An error of the check. On its own — returned rather than accumulated — it is one
72    /// raised before there was a build to check, the root's manifest or the resolution of
73    /// its dependencies, and is unrendered.
74    Check(CompilationError),
75    /// The named module checked, and the specialisations the build needs could not be found
76    /// for it: a constrained function whose specialisations never end
77    /// ([`SpecialiseError::Unbounded`]), or one of the errors that can only mean the type
78    /// checker accepted what it should not have. Raised once every module of the build has
79    /// checked, and the build writes nothing.
80    Specialise(Vec<SpecialiseError>, Name),
81    /// The named module checked and could not be emitted as JavaScript.
82    Emit(Vec<zelkova_js::Error>, Name),
83    /// A file of the build's output could not be written. Raised only once every module
84    /// of the build has checked and emitted, since nothing is written before that.
85    Output(output::Error),
86    /// A package's tests were compiled and could not be run: `zelkova test` could not
87    /// write its entry point or could not run `node`. A test that ran and did not pass is
88    /// not this; it is the exit code of the run.
89    TestRun(zelkova_test_runner::Error),
90    /// A package's program could not be run: `zelkova run` was pointed at a package with no
91    /// `main`, or could not write its entry point or could not run `node`. A program that
92    /// ran and aborted is not this; it is the exit code of the run.
93    ProgramRun(program_runner::Error),
94    /// An [`Emit`](BuildError::Emit) or a [`Specialise`](BuildError::Specialise) together
95    /// with the file its module was read from, so that its labels have a file to point into.
96    /// It wraps nothing else: an error of the check carries its own file inside
97    /// [`Check`](BuildError::Check), as a [`CompilationError::InFile`].
98    InFile(Box<BuildError>, SourceFileId),
99    /// Every error a build accumulated, each already rendered to stderr.
100    ///
101    /// Flat: each error of the check is one [`Check`](BuildError::Check) member, and a
102    /// [`CompilationError::Many`] is never wrapped — its members are. A build does not
103    /// stop on the first failure, so that one broken module cannot hide the diagnostics
104    /// of the others; this is how that accumulation becomes a failure again at its end,
105    /// with the typed errors still intact for the caller to inspect.
106    Many(Vec<BuildError>),
107}
108
109impl BuildError {
110    /// Turn this error into the diagnostic the user reads.
111    ///
112    /// A [`Check`](BuildError::Check) renders through [`CompilationError::as_diagnostic`];
113    /// every other variant through [`phase_diagnostic`] or [`plain_diagnostic`], the two
114    /// functions [`CompilationError::as_diagnostic`] shares with this one.
115    pub fn as_diagnostic(&self) -> Diagnostic<SourceFileId> {
116        self.as_diagnostic_in(None)
117    }
118
119    /// The name of the module this error belongs to, when it has one.
120    pub fn module(&self) -> Option<&Name> {
121        match self {
122            BuildError::Emit(_, module) | BuildError::Specialise(_, module) => Some(module),
123            BuildError::Check(error) => error.module(),
124            BuildError::InFile(inner, _) => inner.module(),
125            _ => None,
126        }
127    }
128
129    /// `as_diagnostic`, carrying the file the error's module was read from, which only an
130    /// [`InFile`](BuildError::InFile) wrapper supplies.
131    fn as_diagnostic_in(&self, file: Option<SourceFileId>) -> Diagnostic<SourceFileId> {
132        match self {
133            BuildError::Check(error) => error.as_diagnostic(),
134            BuildError::InFile(inner, id) => inner.as_diagnostic_in(Some(*id)),
135            BuildError::Specialise(errors, module) => {
136                phase_diagnostic(module, "specialisation", errors, file)
137            }
138            BuildError::Emit(errors, module) => {
139                phase_diagnostic(module, "code generation", errors, file)
140            }
141            // A path on disk, not a place in any source, so there is nothing to label.
142            BuildError::Output(error) => plain_diagnostic(error),
143            // Like `Output`, about the machine and not about any source.
144            BuildError::TestRun(error) => plain_diagnostic(error),
145            BuildError::ProgramRun(error) => plain_diagnostic(error),
146            // Each member was rendered with its own carets before this was built, so this
147            // summarises rather than flattening their labels: drawing every caret of the
148            // build a second time under one headline is what flattening would do.
149            BuildError::Many(errors) => plain_diagnostic(&Summary { errors, file }),
150        }
151    }
152}
153
154/// What a [`BuildError::Many`] renders as: one headline counting its members, and each
155/// member's own headline as a note.
156struct Summary<'a> {
157    errors: &'a [BuildError],
158    file: Option<SourceFileId>,
159}
160
161impl PhaseError for Summary<'_> {
162    fn message(&self) -> String {
163        format!(
164            "compilation failed with {} error{}",
165            self.errors.len(),
166            if self.errors.len() == 1 { "" } else { "s" }
167        )
168    }
169
170    fn notes(&self) -> Vec<String> {
171        self.errors
172            .iter()
173            .map(|e| e.as_diagnostic_in(self.file).message)
174            .collect()
175    }
176}

178/// What `?` goes through for an error raised before a build's accumulator exists — the
179/// root's manifest, the resolution — which comes back bare and unrendered.
180impl From<CompilationError> for BuildError {
181    fn from(error: CompilationError) -> Self {
182        BuildError::Check(error)
183    }

186/// Compile the package rooted at `package_dir` — a directory holding a `zelkova.toml`
187/// manifest beside a `src/` source root, per
188/// [`docs/spec/packages.md`](../docs/spec/packages.md#what-a-package-is) — and every
189/// package it depends on.
190///
191/// Its `tests/` root is not compiled; [`compile_package_with_tests`] is that build. A
192/// package is compiled from source the same way whether it is the one asked for or a
193/// dependency of it, and the whole build shares one file database and one error
194/// accumulator: an error in any package of it makes this return `Err`.
195///
196/// A build that checks writes its JavaScript to `build/out/js/` beside `package_dir`'s
197/// manifest — see [`compile_package_into`] for what that tree holds. One that emitted
198/// any error writes nothing.
199///
200/// The check itself is [`zelkova_compiler::check_package`]'s. This prints its status lines to stderr, emits
201/// and writes the build, and renders every error to stderr before returning it inside
202/// [`BuildError::Many`]. An error raised before any source is read — the manifest,
203/// the resolution — is returned bare and unrendered.
204pub fn compile_package(package_dir: &Path) -> Result<(), BuildError> {
205    compile(
206        package_dir,
207        TestRoot::Skipped,
208        &package_dir.join(BUILD_DIRECTORY),
209        &Overlay::new(),
210    )
211    .map(|_| ())
212}

214/// The directory a build's output goes to, beside the root package's manifest and never
215/// beside a source it read ([*The compiler's
216/// interface*](../docs/spec/toolchain.md#the-compilers-interface)).
217pub const BUILD_DIRECTORY: &str = "build";

219/// The tree a build that compiled the tests writes below `build_dir`: `build_dir/test/js/`,
220/// laid out like `build_dir/out/js/`. Both the write of that tree and [`test()`], which hands
221/// it to [`zelkova_test_runner::run`], name it through here.
222pub fn test_tree(build_dir: &Path) -> PathBuf {
223    build_dir.join("test").join("js")
224}

226/// [`compile_package`], writing its output below `build_dir` rather than below the
227/// package's own `build/`.
228///
229/// The output is one tree, `<build_dir>/out/js/`: the runtime at its root, then one
230/// directory per package of the build holding one `.mjs` file per module of that
231/// package, named after the module within its own package, and each facade's companion
232/// beside the facade ([`zelkova_js`]'s *Paths* section has the names, [`DEC-18` decision
233/// 5](../docs/decisions/dec-18.md#5--output-is-written-per-package-beside-the-root-manifest)
234/// the reasons). Nothing is written until every module of every package has checked and
235/// emitted.
236pub fn compile_package_into(package_dir: &Path, build_dir: &Path) -> Result<(), BuildError> {
237    compile(package_dir, TestRoot::Skipped, build_dir, &Overlay::new()).map(|_| ())
238}

240/// [`compile_package`], compiling the package's `tests/` root as well as its `src/`.
241///
242/// This is what running a package's own tests needs, and the only thing that ever reads
243/// a `tests/` root: the tests of the build's other packages are not compiled, because
244/// nothing outside a package reads its tests
245/// ([*Running a package's tests*](../docs/spec/toolchain.md#running-a-packages-tests)).
246/// A package holding no `tests/` at all compiles exactly as it does through
247/// [`compile_package`].
248///
249/// It compiles the tests and does not run them: what makes a declaration a test is
250/// [its type](../docs/spec/packages.md#what-a-test-is), and running one is
251/// [`test()`]'s. What it hands back on success is the `Interface` of each of
252/// the root's own `tests/` modules that checked — never a test-only package's, and never
253/// `src/`'s — so a caller can find which of their exposed values are tests without a phase
254/// dropping the checked modules once they are emitted. `test_collection::collect` is that
255/// pass, and `zelkova_test_runner::run` is its caller. Empty when the package holds no `tests/` at
256/// all.
257///
258/// A test module and every `test-dependency`'s modules are checked and, unlike a plain
259/// build, written — to a tree of their own, `build/test/js/`, laid out exactly like
260/// `build/out/js/` and holding every package of the build (a test-only one included) plus the
261/// root's `tests/` modules beside its `src/` ones. `build/out/js/` itself is left exactly as
262/// [`compile_package`] would have written it: a test module never turns up there, so a
263/// plain build run afterwards never finds one left behind by a run that also compiled the
264/// tests ([`GEN-18`](../docs/tickets/README.md)).
265pub fn compile_package_with_tests(package_dir: &Path) -> Result<Vec<Interface>, BuildError> {
266    compile_package_with_tests_into(package_dir, &package_dir.join(BUILD_DIRECTORY))
267}
268
269/// [`compile_package_with_tests`], writing below `build_dir` rather than below the
270/// package's own `build/` — the same relationship [`compile_package_into`] has to
271/// [`compile_package`].
272pub fn compile_package_with_tests_into(
273    package_dir: &Path,
274    build_dir: &Path,
275) -> Result<Vec<Interface>, BuildError> {
276    compile(package_dir, TestRoot::Compiled, build_dir, &Overlay::new())
277}

279/// Compile the package rooted at `package_dir` with its tests
280/// ([`compile_package_with_tests`]), run every test it holds under `node`
281/// ([`zelkova_test_runner::run`], from the tree [`test_tree`] names), and answer the exit code the
282/// process should end with.
283///
284/// The answer is `Ok(0)` when every test passed and when the package holds none. It is
285/// `node`'s own exit code otherwise, which is non-zero exactly when a test did not pass or
286/// `node` itself failed. Anything that stops the tests being run — a build that did not
287/// compile, `node` missing — is an `Err`, and a build that did not compile never reaches
288/// `node`.
289pub fn test(package_dir: &Path) -> Result<i32, BuildError> {
290    let interfaces = compile_package_with_tests(package_dir)?;
291    zelkova_test_runner::run(&test_tree(&package_dir.join(BUILD_DIRECTORY)), &interfaces)
292        .map_err(BuildError::TestRun)
293}

295/// Print one [`Status`] line to `writer`. Failing to write a status line is not itself a
296/// compilation failure, so the write results are deliberately discarded rather than
297/// unwrapped.
298fn print_status(writer: &mut StandardStream, status: &Status) {
299    let (color, label) = if status.success {
300        (Color::Green, "success")
301    } else {
302        (Color::Red, "failure")
303    };
304    let _ = writer.set_color(ColorSpec::new().set_bold(true).set_fg(Some(color)));
305    let _ = write!(writer, "{}", label);
306    let _ = writer.reset();
307    let _ = writeln!(writer, " {}", status.text);
308}

310/// Each error of the check as one [`BuildError::Check`], with a
311/// [`CompilationError::Many`] replaced by its members, at any depth. This is what keeps
312/// [`BuildError::Many`] flat by construction rather than by `check` never building a `Many`.
313fn check_errors(errors: Vec<CompilationError>) -> Vec<BuildError> {
314    errors
315        .into_iter()
316        .flat_map(|error| match error {
317            CompilationError::Many(members) => check_errors(members),
318            other => vec![BuildError::Check(other)],
319        })
320        .collect()
321}

323/// The CLI half of every `compile_package` variant: check the package
324/// ([`zelkova_compiler::check_package`] or [`zelkova_compiler::check_package_with_tests`], as `tests`
325/// says), print its status lines, emit and write a build that checked, and render every error to stderr.
326///
327/// The status lines are printed once the check has finished rather than as each phase
328/// ends. For a check that returns, the bytes `compile_package` itself writes to stderr are
329/// the same as before, in the same order: every status line, then every diagnostic. Log
330/// output (`RUST_LOG`) and a panic's message no longer interleave with them, and a panic
331/// during checking loses the status lines already recorded.
332fn compile(
333    package_dir: &Path,
334    tests: TestRoot,
335    build_dir: &Path,
336    overlay: &Overlay,
337) -> Result<Vec<Interface>, BuildError> {
338    let PackageCheck {
339        package: root_package,
340        errors,
341        sources,
342        modules,
343        test_dependency_modules,
344        test_modules,
345        // A build emits only what checked; these are for a reader of a module's tree.
346        failing: _,
347        status,
348    } = match tests {
349        TestRoot::Skipped => zelkova_compiler::check_package(package_dir, overlay)?,
350        TestRoot::Compiled => zelkova_compiler::check_package_with_tests(package_dir, overlay)?,
351    };
352
353    // The build's one accumulator: each error of the check on its own, then whatever
354    // emitting and writing the build adds. An empty one is what makes this return `Ok`.
355    let mut errors: Vec<BuildError> = check_errors(errors);
356
357    // Error reporter
358    let mut writer = StandardStream::stderr(ColorChoice::Auto);
359    let config = codespan_reporting::term::Config {
360        tab_width: 2,
361        ..codespan_reporting::term::Config::default()
362    };
363
364    for line in &status {
365        print_status(&mut writer, line);
366    }
367
368    // What `compile_package_with_tests` hands back: the `Interface` of each of the
369    // root's own `tests/` modules that checked — never a test-only package's, and
370    // never `src/`'s. This is what lets a caller find a package's tests
371    // (`test_collection::collect` is that pass) without a phase dropping the checked
372    // modules on the floor once they have been emitted. Stays empty for a build that
373    // did not ask for the tests, or whose root `tests/` did not check.
374    let mut root_test_interfaces: Vec<Interface> = Vec::new();
375
376    // Generate code, only for a build in which nothing failed. Every module of
377    // both trees is emitted before anything is written, so a module that cannot be
378    // emitted — in `src/` or in `tests/` — also leaves the build with no output at all.
379    if errors.is_empty() {
380        let mut checked: Vec<ModuleToEmit> = modules.into_iter().map(to_module_to_emit).collect();
381        let mut root_tests_checked: Vec<ModuleToEmit> =
382            test_modules.into_iter().map(to_module_to_emit).collect();
383
384        // A test companion imports the companion it checks by its path in the source
385        // tree, across the two roots, which reaches nothing in the build
386        // ([*Testing a companion*](../docs/spec/interop.md#testing-a-companion)). Each
387        // companion of the root's `src/` is one it may check, and `checked` holds every
388        // one of them.
389        let targets: Vec<&Name> = checked
390            .iter()
391            .filter(|to_emit| {
392                to_emit.companion.is_some()
393                    && to_emit.module.canonical.name.package() == &root_package
394            })
395            .map(|to_emit| to_emit.module.canonical.name.name())
396            .collect();
397        for to_emit in root_tests_checked
398            .iter_mut()
399            .filter(|to_emit| to_emit.companion.is_some())
400        {
401            let checks = to_emit.module.canonical.name.name();
402            to_emit.companion_imports = targets
403                .iter()
404                .map(|target| zelkova_js::test_companion_import(checks, target))
405                .collect();
406        }
407        // Built from the same `ModuleToEmit`s `test_tree_modules` is about to take,
408        // before that move: an `Interface` is cheap to clone off a `CheckedModule`
409        // that is otherwise about to be consumed by emission.
410        root_test_interfaces = root_tests_checked
411            .iter()
412            .map(|to_emit| to_emit.module.to_interface(to_emit.file))
413            .collect();
414
415        // Every module the *test* tree (`build/test/js/`) needs beyond what `checked`
416        // already holds: each test-only package's modules, and the root's `tests/` modules.
417        // Empty for a build that did not ask for the tests.
418        let mut test_tree_modules: Vec<ModuleToEmit> = test_dependency_modules
419            .into_iter()
420            .map(to_module_to_emit)
421            .collect();
422        test_tree_modules.extend(root_tests_checked);
423
424        debug!("phase: specialise");
425        let specialised = specialise_build(&mut checked, &mut test_tree_modules, &mut errors);
426
427        debug!("phase: codegen");
428        // Every union of the build, read by a facade's boundary checks: a facade may name
429        // a union any module of the build declares, and a test-only package's or the
430        // root's `tests/` modules are part of the build their facades see.
431        let unions = zelkova_js::Unions::of(
432            checked
433                .iter()
434                .chain(test_tree_modules.iter())
435                .map(|to_emit| &to_emit.module),
436        );
437        // A build whose specialisations could not be found has references that still ask
438        // for an instance, and emitting it would only add an error for each.
439        let files = if specialised {
440            emit_build(checked, &unions, &mut errors)
441        } else {
442            Vec::new()
443        };
444
445        // A build that also compiled the tests writes a second, complete tree at
446        // `<build_dir>/test/js/`, laid out exactly like `<build_dir>/out/js/` — the runtime,
447        // then one directory per package — but holding every package of the build (a
448        // test-only one included) and the root's `tests/` modules beside its `src/` ones
449        // (`docs/decisions/dec-18.md#5--output-is-written-per-package-beside-the-root-manifest`).
450        // `files` already holds everything a plain build would have written — the
451        // runtime, the root's `src/` and every plain dependency's modules — so the test
452        // tree reuses it rather than emitting those modules a second time, and only
453        // `test_tree_modules` (the test-only packages' and the root's `tests/`) is new
454        // work. That work happens here, before `out/js/` is written, so a module of the test
455        // tree that cannot be emitted blocks both writes.
456        let test_files = (tests == TestRoot::Compiled).then(|| {
457            debug!("phase: codegen (tests)");
458            let mut test_files = files.clone();
459            if specialised {
460                test_files.extend(emit_modules(test_tree_modules, &unions, &mut errors));
461            }
462            test_files
463        });
464
465        if errors.is_empty() {
466            debug!("phase: write the build");
467            errors.extend(
468                output::write(&build_dir.join("out").join("js"), &files)
469                    .into_iter()
470                    .map(BuildError::Output),
471            );
472        }
473
474        // Gated on `errors.is_empty()` a second time so a plain build's own failure to
475        // write `out/js/` — an unlikely I/O error, not a checking one — does not also
476        // attempt the test tree.
477        if let Some(test_files) = test_files.filter(|_| errors.is_empty()) {
478            debug!("phase: write the test build");
479            errors.extend(
480                output::write(&test_tree(build_dir), &test_files)
481                    .into_iter()
482                    .map(BuildError::Output),
483            );
484        }
485    }
486
487    // Report everything we accumulated, then let that accumulation decide the
488    // return value. Rendering the errors and returning `Ok` regardless was `BUG-1`.
489    for error in &errors {
490        // A rendering failure must not mask the compilation failure we are about to
491        // return, and there is nowhere left to report it to, so it is dropped.
492        let _ = term::emit_to_write_style(
493            &mut writer.lock(),
494            &config,
495            &sources,
496            &error.as_diagnostic(),
497        );
498    }
499
500    if errors.is_empty() {
501        Ok(root_test_interfaces)
502    } else {
503        Err(BuildError::Many(errors))
504    }
505}

507/// Find the specialisations of the whole build ([`zelkova_compiler::ir::specialise`]) and
508/// resolve every reference that asks for an instance, in the modules of both trees at once:
509/// `checked`, which the plain tree is, and `test_tree_modules`, which the test tree holds as
510/// well. A module's specialisations are made for it alone, so what the plain tree's modules
511/// hold does not depend on the test tree having been read with them; the pass is run once
512/// because both trees are emitted from these modules.
513///
514/// An error is pushed onto `errors`, tagged with the file the module it is written in was read
515/// from, and `false` answers that the modules are not specialised.
516fn specialise_build(
517    checked: &mut [ModuleToEmit],
518    test_tree_modules: &mut [ModuleToEmit],
519    errors: &mut Vec<BuildError>,
520) -> bool {
521    let files: HashMap<ModuleName, Option<SourceFileId>> = checked
522        .iter()
523        .chain(test_tree_modules.iter())
524        .map(|to_emit| (to_emit.module.canonical.name.clone(), to_emit.file))
525        .collect();
526
527    let mut modules: Vec<&mut CheckedModule> = checked
528        .iter_mut()
529        .chain(test_tree_modules.iter_mut())
530        .map(|to_emit| &mut to_emit.module)
531        .collect();
532
533    match zelkova_compiler::ir::specialise(&mut modules) {
534        Ok(()) => true,
535        Err(failures) => {
536            for failure in failures {
537                let file = files.get(&failure.module).copied().flatten();
538                let error = BuildError::Specialise(failure.errors, failure.module.name().clone());
539                errors.push(match file {
540                    Some(id) => BuildError::InFile(Box::new(error), id),
541                    None => error,
542                });
543            }
544            false
545        }
546    }
547}

549/// A module that checked, with what emitting and writing it needs beyond the module
550/// itself.
551struct ModuleToEmit {
552    module: CheckedModule,
553    /// The file it was read from, which an emission error's labels point into.
554    file: Option<SourceFileId>,
555    /// For a facade, its JavaScript companion when one sits beside its `.zel` source —
556    /// a file of the same base name in the same directory
557    /// ([*A facade names a boundary, not a
558    /// backend*](../docs/spec/interop.md#a-facade-names-a-boundary-not-a-backend)).
559    /// `None` for every other module, and for a facade with no companion, which
560    /// [`zelkova_js::emit`] refuses.
561    companion: Option<std::path::PathBuf>,
562    /// The imports its companion spells for the source tree, each beside the specifier
563    /// that replaces it in the build ([`zelkova_js::test_companion_import`]). Empty for
564    /// every module but a facade of the root package's `tests/` with a companion, which
565    /// [`compile`] fills in once it knows the companions of `src/` that one may import.
566    companion_imports: Vec<(String, String)>,
567}
568
569/// What emitting and writing one checked module needs: its companion when it is a facade
570/// with one sitting beside its source under its `root_dir`, so a facade under `tests/`
571/// finds its companion there rather than under `src/`
572/// ([*Testing a companion*](../docs/spec/interop.md#testing-a-companion)).
573fn to_module_to_emit(checked: CheckedSource) -> ModuleToEmit {
574    let CheckedSource {
575        module,
576        file,
577        root_dir,
578    } = checked;
579    let companion = Some(root_dir.join(zelkova_js::module_file(module.canonical.name.name())))
580        .filter(|path| module.ir.foreign && path.is_file());
581    ModuleToEmit {
582        file,
583        companion,
584        companion_imports: Vec::new(),
585        module,
586    }
587}
588
589/// Every file a build writes: the runtime, and each module of `checked` as the text
590/// [`zelkova_js::emit`] gives it, with its facade's companion beside it.
591///
592/// A module that cannot be emitted pushes its errors onto `errors`, tagged with the
593/// file it came from, and every other module is still emitted so that one refusal
594/// cannot hide the next. The caller writes the files only when `errors` stays empty.
595fn emit_build(
596    checked: Vec<ModuleToEmit>,
597    unions: &zelkova_js::Unions,
598    errors: &mut Vec<BuildError>,
599) -> Vec<output::File> {
600    let mut files = vec![output::File {
601        path: zelkova_js::RUNTIME_FILE.into(),
602        contents: output::Contents::Text(zelkova_js::RUNTIME.to_string()),
603    }];
604    files.extend(emit_modules(checked, unions, errors));
605    files
606}
607
608/// [`emit_build`], without the runtime file at the front.
609///
610/// [`emit_build`] calls it for the plain build's own modules. A test build calls it a
611/// second time for the modules a plain build never emits — each test-only package's,
612/// and the root's `tests/` — which is why it is factored out: the test tree already has
613/// the runtime, since it starts from a clone of the plain build's own files, and those
614/// extra modules must not emit a second runtime file to sit unused beside the first.
615fn emit_modules(
616    checked: Vec<ModuleToEmit>,
617    unions: &zelkova_js::Unions,
618    errors: &mut Vec<BuildError>,
619) -> Vec<output::File> {
620    let mut files = Vec::new();
621
622    for ModuleToEmit {
623        module,
624        file,
625        companion,
626        companion_imports,
627    } in checked
628    {
629        let name = module.canonical.name.clone();
630        let package_dir = std::path::PathBuf::from(name.package().as_str());
631
632        match zelkova_js::emit(&module, companion.is_some(), unions) {
633            Ok(text) => {
634                files.push(output::File {
635                    path: package_dir.join(zelkova_js::module_file(name.name())),
636                    contents: output::Contents::Text(text),
637                });
638
639                if let Some(companion) = companion {
640                    let contents = if companion_imports.is_empty() {
641                        output::Contents::Copy(companion)
642                    } else {
643                        output::Contents::Rewritten {
644                            from: companion,
645                            imports: companion_imports,
646                        }
647                    };
648                    files.push(output::File {
649                        path: package_dir.join(zelkova_js::companion_file(name.name())),
650                        contents,
651                    });
652                }
653            }
654            Err(emit_errors) => {
655                let error = BuildError::Emit(emit_errors, name.name().clone());
656                errors.push(match file {
657                    Some(id) => BuildError::InFile(Box::new(error), id),
658                    None => error,
659                });
660            }
661        }
662    }
663
664    files
665}

667#[cfg(test)]
668mod tests {
669    use super::*;
670
671    /// A `CompilationError::Many` in the check's errors becomes one `Check` per member,
672    /// however deeply it was nested, and a `Many` is never itself wrapped. A plain error
673    /// beside it stays as it was, in order.
674    ///
675    /// Mutation-checked by turning `check_errors` back into a plain
676    /// `.map(BuildError::Check)`: the first assertion sees one `Check` where three are
677    /// expected and this goes red.
678    #[test]
679    fn a_many_of_the_check_is_flattened_into_its_members() {
680        let leaf = || CompilationError::Manifest(vec![]);
681        let nested = CompilationError::Many(vec![
682            leaf(),
683            CompilationError::Many(vec![CompilationError::Resolution(vec![])]),
684        ]);
685
686        let flat = check_errors(vec![nested, leaf()]);
687
688        assert_eq!(flat.len(), 3);
689        assert!(matches!(
690            flat[0],
691            BuildError::Check(CompilationError::Manifest(_))
692        ));
693        assert!(matches!(
694            flat[1],
695            BuildError::Check(CompilationError::Resolution(_))
696        ));
697        assert!(matches!(
698            flat[2],
699            BuildError::Check(CompilationError::Manifest(_))
700        ));
701        assert!(!flat
702            .iter()
703            .any(|e| matches!(e, BuildError::Check(CompilationError::Many(_)))));
704    }
705
706    /// `module` is `Some` for an `Emit` and for a `Check` of one module, `None` for an
707    /// error that is about no one module.
708    ///
709    /// Mutation-checked by making the `Emit` arm return `None`: the first assertion
710    /// goes red.
711    #[test]
712    fn module_names_the_module_of_an_emit_error() {
713        let emit = BuildError::Emit(vec![], Name::new("Main"));
714        assert_eq!(emit.module(), Some(&Name::new("Main")));
715
716        let check = BuildError::Check(CompilationError::Canonical(vec![], Name::new("Other")));
717        assert_eq!(check.module(), Some(&Name::new("Other")));
718
719        assert_eq!(BuildError::Many(vec![]).module(), None);
720    }
721}

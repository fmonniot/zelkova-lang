1//! Walk package's module dependencies
2//!
3//! Within a package, each `Module` will depends on other `Module`s. The aim
4//! of the dependencies module is to provide a way to sort the modules in a dependent
5//! way, allowing parallel action on which module. We will also find out existing
6//! cyclic dependencies and return all of them as error.
7//!
8//! We are currently using the excellent `petgraph` crate for sorting our dependencies
9//! and discovering cycle. We currently use the results of the cycle discovery to drive
10//! how we execute the processing. If we want to introduce parallel processing of the
11//! different module, we will have to be a bit more thorough because we currently don't
12//! have any information _at processing time_ to know when to block the check of some
13//! modules. We don't know how we will do most of that processing, so it's a bit
14//! premature to actually implement multi threading here.
15//! Let's see how it goes first on a single thread and we can revisit this decision
16//! later on.
17//!

47/// One `import` that forms an edge of a dependency cycle.
48///
49/// `from` imports `to` — the module the note lists right after it, wrapping
50/// around to the cycle's start for the last edge — and `span`/`file` are where
51/// that specific `import` line was written, when known.
52///
53/// `file` comes from the `module_files` map [`ModuleWalker::new`] is handed by
54/// `compile_package`, the only place that knows which file a parsed module came
55/// from — the same reasoning `ERR-5` used for [`crate::compiler::Interface::file`].
56/// It is `None` for every edge built in this module's own tests, which hand-build
57/// modules with nothing on disk behind them.
58#[derive(Debug, PartialEq, Clone)]
59pub struct CycleEdge {
60    pub from: Name,
61    pub to: Name,
62    pub span: NodeSpan,
63    pub file: Option<SourceFileId>,
64}

66/// A dependency cycle: one loop through a strongly-connected component of the
67/// import graph, alongside the `import` that created each edge between consecutive
68/// modules (wrapping around to the first), and any component member that loop does
69/// not pass through.
70#[derive(Debug, PartialEq, Clone)]
71pub struct Cycle {
72    /// *One* cycle within the strongly-connected component, in cycle order —
73    /// what the summary note is built from. Not necessarily every module in the
74    /// component: a component with several overlapping loops has more than one
75    /// cycle in it, and picking the shortest one through a single start node (see
76    /// [`cycle_walk`]) can leave others out. Finding a cycle that covers every
77    /// member is the Hamiltonian cycle problem, and no such cycle need exist, so
78    /// this does not try; the members left out are in [`Cycle::others`] instead,
79    /// and the note names them so a user does not break one loop and immediately
80    /// hit the next one hiding in the same component.
81    pub path: Vec<Name>,
82    /// The other modules in the same component: genuinely part of the circular
83    /// dependency, but not on the particular cycle `path` names. Empty whenever
84    /// `path` already covers the component, which every fixture and every real
85    /// import graph seen so far is.
86    pub others: Vec<Name>,
87    /// `edges[i]` is the `import` written in `path[i]` that names `path[i + 1]`
88    /// (or `path[0]`, for the last edge). Same length as `path`, and every entry
89    /// is a real `import`, because [`cycle_walk`] only ever returns a path whose
90    /// consecutive pairs are edges of the graph.
91    pub edges: Vec<CycleEdge>,
92}
93
94#[derive(Debug, PartialEq)]
95pub enum Error {
96    CycleDetected(Vec<Cycle>),
97}

99/// A dependency cycle belongs to the package, not to any one module, so its message
100/// names the modules involved rather than being prefixed with one of them, and its
101/// labels — one per `import` that forms an edge — each carry their own file rather
102/// than relying on a fallback the way a single-module phase error can
103/// (`CompilationError::DependenciesError` renders this without one, see there).
104impl crate::compiler::PhaseError for Error {
105    fn message(&self) -> String {
106        match self {
107            Error::CycleDetected(cycles) => format!(
108                "{} circular dependenc{} between modules",
109                cycles.len(),
110                if cycles.len() == 1 { "y" } else { "ies" }
111            ),
112        }
113    }
114
115    fn notes(&self) -> Vec<String> {
116        match self {
117            // Each cycle is written back to its start, so it reads as the loop it is:
118            // `A -> B -> A`. A component that holds more than the one loop gets a
119            // second note naming the members that loop leaves out, so breaking it
120            // does not just uncover the next cycle in the same component
121            // (`Cycle::others`).
122            Error::CycleDetected(cycles) => cycles
123                .iter()
124                .flat_map(|cycle| {
125                    let mut path: Vec<String> =
126                        cycle.path.iter().map(|name| name.to_string()).collect();
127                    if let Some(first) = path.first().cloned() {
128                        path.push(first);
129                    }
130                    let mut notes = vec![format!("cycle: {}", path.join(" -> "))];
131
132                    if !cycle.others.is_empty() {
133                        let others: Vec<String> =
134                            cycle.others.iter().map(|name| name.to_string()).collect();
135                        notes.push(format!(
136                            "also part of this circular dependency, on another loop through the same modules: {}",
137                            others.join(", ")
138                        ));
139                    }
140
141                    notes
142                })
143                .collect(),
144        }
145    }
146
147    fn labels(&self) -> Vec<SpanLabel> {
148        match self {
149            Error::CycleDetected(cycles) => cycles
150                .iter()
151                .flat_map(|cycle| {
152                    cycle.edges.iter().filter_map(|edge| {
153                        // An edge with no span (a hand-built test module) or no known
154                        // file (nothing in the real pipeline, since `module_files`
155                        // covers every parsed module) has nowhere honest to point;
156                        // it is dropped rather than guessed at, the same rule
157                        // `Interface::source_span` follows.
158                        let span = edge.span.span()?;
159                        let file = edge.file?;
160                        Some(SpanLabel {
161                            span,
162                            // Every edge is equally the cause — breaking any one
163                            // of them breaks the loop — so a label states only its
164                            // own edge and leaves the loop to the note. Saying
165                            // "closing the cycle" here would put that claim on all
166                            // N labels when only the last edge closes it.
167                            message: format!("`{}` imports `{}` here", edge.from, edge.to),
168                            primary: true,
169                            file: Some(file),
170                        })
171                    })
172                })
173                .collect(),
174        }
175    }
176}

178/// Walk `members` — one strongly-connected component `tarjan_scc` found, so every
179/// node in it can reach every other — into an actual directed cycle through
180/// `start`: a sequence of modules where each one really does import the next, and
181/// the last one imports the first.
182///
183/// `tarjan_scc`'s own ordering of a component does not promise this: it is
184/// membership, not a walk, so consecutive entries are not necessarily connected by
185/// an edge, and the note used to print a "cycle" the user could not follow. That is
186/// what `ERR-6` set out to fix, and `build_cycle` needs it to be true a second time
187/// over, because it looks up the `import` behind each consecutive pair and an edge
188/// that does not exist has no `import` to point a label at.
189///
190/// So the guarantee has to hold by construction, not by luck. This is a breadth-first
191/// search from `start` over in-component edges only, stopping at the first node found
192/// to import `start` back; the path is then reconstructed backwards along the BFS
193/// tree. Because BFS pops nodes in non-decreasing distance order, that is the
194/// *shortest* cycle through `start`, and because a BFS tree path never repeats a
195/// node, it is a simple one. A greedy walk that just takes the first unvisited
196/// neighbour cannot promise this — in a component with two loops sharing a node
197/// (`B` imports `C` and `E`; `C` imports `D`; `D` imports `B`; `E` imports `B`) it
198/// wanders into `E`, dead-ends, and hands back `[D, B, E]` even though `E` does not
199/// import `D`.
200///
201/// The cycle it finds may be a strict subset of `members` — see [`Cycle::path`] for
202/// why that is not fixable in general and what is reported instead.
203///
204/// `start` must be one of `members`. Callers get that from the `partition` in
205/// [`ModuleWalker::new`], which is also where the component is known to be
206/// non-empty; taking it as a parameter is what keeps this function off an unchecked
207/// `members[0]`.
208///
209/// The returned path is empty only if `start` lies on no cycle at all, which a
210/// component `tarjan_scc` reported with more than one member cannot produce.
211fn cycle_walk(
212    graph: &DiGraph<&Module, ()>,
213    members: &[NodeIndex],
214    start: NodeIndex,
215) -> Vec<NodeIndex> {
216    let member_set: HashSet<NodeIndex> = members.iter().copied().collect();
217
218    // How each node was first reached. `start` is deliberately absent: it is the
219    // target, so reaching it is what ends the search rather than something to
220    // record a predecessor for.
221    let mut predecessor: HashMap<NodeIndex, NodeIndex> = HashMap::new();
222    let mut queue: VecDeque<NodeIndex> = VecDeque::from([start]);
223    // The node that closes the loop: the first one popped that imports `start`.
224    let mut closing: Option<NodeIndex> = None;
225
226    'search: while let Some(current) = queue.pop_front() {
227        for next in graph.neighbors(current).filter(|n| member_set.contains(n)) {
228            if next == start {
229                closing = Some(current);
230                break 'search;
231            }
232
233            // First time we reach `next` is along a shortest path to it, so a
234            // later, longer way in is ignored.
235            if predecessor.contains_key(&next) {
236                continue;
237            }
238
239            predecessor.insert(next, current);
240            queue.push_back(next);
241        }
242    }
243
244    let mut path = Vec::new();
245
246    if let Some(closing) = closing {
247        let mut node = closing;
248        path.push(node);
249
250        // Back up the BFS tree to `start`. `predecessor` was built by this search,
251        // so every node on that chain has one until `start` is reached; the `None`
252        // arm cannot fire, and stopping is the right thing if it somehow did.
253        while node != start {
254            match predecessor.get(&node) {
255                Some(&previous) => {
256                    node = previous;
257                    path.push(node);
258                }
259                None => break,
260            }
261        }
262
263        path.reverse();
264    }
265
266    path
267}

269/// Turn one strongly-connected component into the [`Cycle`] a diagnostic renders:
270/// the module path `cycle_walk` finds, the `import` behind each edge of it, and the
271/// component members that path leaves out.
272///
273/// `start` is the node the cycle is found through; [`ModuleWalker::new`] takes it
274/// from the component itself, which is how this avoids indexing one.
275fn build_cycle(
276    graph: &DiGraph<&Module, ()>,
277    members: &[NodeIndex],
278    start: NodeIndex,
279    module_files: &HashMap<Name, SourceFileId>,
280) -> Cycle {
281    let node_path = cycle_walk(graph, members, start);
282    let on_path: HashSet<NodeIndex> = node_path.iter().copied().collect();
283
284    let path: Vec<Name> = node_path
285        .iter()
286        .map(|&idx| graph[idx].name.clone())
287        .collect();
288
289    let mut others: Vec<Name> = members
290        .iter()
291        .filter(|idx| !on_path.contains(idx))
292        .map(|&idx| graph[idx].name.clone())
293        .collect();
294    // `tarjan_scc`'s ordering within a component is an implementation detail; a
295    // note the user reads should not vary with it. Sorted on the identifier text
296    // rather than through an `Ord` on `Name`, which `name.rs` deliberately does not
297    // derive.
298    others.sort_by(|l, r| l.as_str().cmp(r.as_str()));
299
300    let edges = node_path
301        .iter()
302        .enumerate()
303        .map(|(i, &idx)| {
304            let from_module = graph[idx];
305            let to_idx = node_path[(i + 1) % node_path.len()];
306            let to_name = graph[to_idx].name.clone();
307
308            // The specific `import …` line in `from_module` that names `to_name` —
309            // there may be several imports in `from_module`, but at most one of
310            // them names this particular neighbour. `cycle_walk` guarantees this is
311            // a real edge, so the only reason to find nothing is a hand-built
312            // module in this file's own tests, which carry no source text.
313            let import = from_module.imports.iter().find(|imp| imp.name == to_name);
314
315            CycleEdge {
316                from: from_module.name.clone(),
317                to: to_name,
318                span: import.map(|imp| imp.span).unwrap_or_else(NodeSpan::none),
319                file: import.and_then(|_| module_files.get(&from_module.name).copied()),
320            }
321        })
322        .collect();
323
324    Cycle {
325        path,
326        others,
327        edges,
328    }
329}

331/// Put the [default imports](crate::compiler::default_imports) into the import
332/// graph, so a module that never wrote `import Basics` is still checked after it.
333///
334/// Canonicalization resolves an implicit import against an `Interface` that already
335/// exists, and nothing but this graph decides which module is checked first — so
336/// without these edges whether `1 + 2` resolved would come down to the order the
337/// source files happened to load in.
338///
339/// Two entries are skipped, and both are the reason this is not simply "an edge per
340/// module per default":
341///
342/// - a module **named** on the list gets none, because `Basics` cannot import
343///   itself and `Maybe`/`Result` would import each other
344///   ([`default_imports::is_default`]);
345/// - an edge whose target already depends on the importer is dropped, because
346///   adding it would close a loop. `Basics` imports the `Js.Basics` facade, so
347///   `Js.Basics` does not implicitly import `Basics` back.
348///
349/// The second test is made against the graph *as built so far* rather than against
350/// the written imports alone, so no addition here can introduce a cycle that was
351/// not already written in the source. `tarjan_scc` below still reports one that
352/// was.
353///
354/// Both skips agree with [`default_imports::implicit_imports`], which drops the
355/// same import for its own reason — a dropped edge means the target is checked
356/// *later*, so its interface is not available when the importer is canonicalized.
357///
358/// # Why the loops are nested target-first
359///
360/// Refusing an edge against the graph *so far* means the edges already added are
361/// part of what the next test sees, so this pass can talk itself out of an edge it
362/// would otherwise have allowed. The nesting decides which edge wins that race, and
363/// only one nesting makes the winner a property of the package rather than of the
364/// alphabet.
365///
366/// Iterating importer-first did not. A package where `Basics` imports `Aux` and
367/// `Maybe` imports `Zed` gave `Aux` its `Maybe` — nothing needs it, but nothing
368/// forbade it either — and that edge made `Basics` reach `Zed` through
369/// `Aux -> Maybe -> Zed`, so `Zed -> Basics` was refused and `Zed` could no longer
370/// spell `+`. Renaming `Aux` to sort *after* `Zed` compiled the same package.
371/// `tests/fixtures/package_default_import_priority/` is that package.
372///
373/// Iterating target-first, in [`default_imports::DEFAULT_IMPORTS`] order, gives the
374/// race a fixed outcome twice over:
375///
376/// - **Across targets**, every `Basics` edge is settled before any `Maybe` edge
377///   exists to constrain the graph, so a conflict is resolved in favour of whichever
378///   entry the list writes first. That is the list's own stated priority, and it is
379///   the same list on every package.
380/// - **Within one target**, the importers cannot interfere with each other at all.
381///   Adding `m -> d` creates new paths only from nodes that reach `m` to nodes `d`
382///   reaches, so a *new* path `d ⇝ m'` would have to run `d ⇝ m -> d ⇝ m'` and
383///   therefore require `d ⇝ m` to have held already — in which case `m -> d` was
384///   refused rather than added. So every importer of `d` is tested against the graph
385///   as it stood when `d`'s pass began, and the set of edges added for `d` does not
386///   depend on the order they were tested in.
387///
388/// The same argument run with `m' = m` is why no edge added here can close a loop.
389///
390/// Importers are still visited in name order, but now only so that the edges land
391/// in the graph in a fixed sequence: `names` is a `HashMap`, and `tarjan_scc` breaks
392/// ties between unordered modules by insertion order, so without the sort the
393/// *reported* check order would drift between runs even though the edge set did not.
394fn add_default_import_edges(graph: &mut DiGraph<&Module, ()>, names: &HashMap<&Name, NodeIndex>) {
395    let mut importers: Vec<(&Name, NodeIndex)> =
396        names.iter().map(|(&name, &idx)| (name, idx)).collect();
397    importers.sort_by(|left, right| left.0.as_str().cmp(right.0.as_str()));
398
399    for default in crate::compiler::default_imports::DEFAULT_IMPORTS {
400        let target = default.name();
401        let Some(&target_idx) = names.get(&target) else {
402            continue;
403        };
404
405        for &(module_name, module_idx) in &importers {
406            if crate::compiler::default_imports::is_default(module_name) {
407                continue;
408            }
409
410            // A written import already carries the edge, and replaces the implicit
411            // import anyway.
412            if graph[module_idx].imports.iter().any(|i| i.name == target) {
413                continue;
414            }
415
416            if petgraph::algo::has_path_connecting(&*graph, target_idx, module_idx, None) {
417                continue;
418            }
419
420            graph.add_edge(module_idx, target_idx, ());
421        }
422    }
423}

425impl<'a> ModuleWalker<'a> {
426    /// `module_files` is how a cycle's edges learn which file each `import` was
427    /// written in (`CycleEdge::file`) — this is driver code, the same as
428    /// `check_in_order` below, so it is handed the map `compile_package` already
429    /// built while parsing rather than trying to discover it itself. Every test in
430    /// this module passes an empty map: their modules are hand-built, with no file
431    /// on disk to name, and `CycleEdge::file` degrades to `None` for them exactly
432    /// as documented there.
433    pub fn new(
434        modules: &'a [Module],
435        module_files: &HashMap<Name, SourceFileId>,
436    ) -> Result<ModuleWalker<'a>, Error> {
437        let mut graph = DiGraph::new();
438
439        let mut names = HashMap::new();
440
441        // First populate the graph with all nodes, and keep a reference from Path to the node index
442        for module in modules.iter() {
443            let idx = graph.add_node(module);
444            names.insert(&module.name, idx);
445        }
446
447        // Now that all modules have been inserted, we can create edges
448        for module in modules.iter() {
449            let m_idx = names.get(&module.name).unwrap();
450
451            // When building the dependency graph, we exclude all imports for
452            // modules not in this package. The reason is simple: a dependent
453            // package cannot depends on the current package, so no cycle to detect
454            for dep in module
455                .imports
456                .iter()
457                .filter_map(|import| names.get(&import.name))
458            {
459                graph.add_edge(*m_idx, *dep, ());
460            }
461        }
462
463        add_default_import_edges(&mut graph, &names);
464
465        // Find the strongly connected graphs (scc), if there are more than one node per scc
466        // it means there is a circular dependency.
467        let scc = petgraph::algo::tarjan_scc(&graph);
468
469        let (cycles, deps): (Vec<_>, Vec<_>) = scc.iter().partition(|&v| v.len() > 1);
470
471        if cycles.is_empty() {
472            let modules = deps.into_iter().flatten().map(|&idx| graph[idx]).collect();
473
474            Ok(ModuleWalker { modules })
475        } else {
476            // `partition` above already established every component here has more
477            // than one member, so `first()` is always `Some`; taking the start node
478            // from it here, rather than indexing inside `cycle_walk`, is what keeps
479            // that guarantee next to the check that establishes it.
480            let c = cycles
481                .into_iter()
482                .filter_map(|members| {
483                    members
484                        .first()
485                        .map(|&start| build_cycle(&graph, members, start, module_files))
486                })
487                .collect();
488
489            Err(Error::CycleDetected(c))
490        }
491    }
492
493    /// Given a package name, a set of existing interfaces and a checker function,
494    /// check each module in its dependencies order.
495    ///
496    /// Every module is checked regardless of whether an earlier one failed: the
497    /// successes are returned alongside the errors rather than being discarded as
498    /// soon as one module fails (`BUG-2`). A successfully checked module still has
499    /// its `Interface` inserted into `interfaces` as it goes, so later modules keep
500    /// resolving against earlier ones even when some sibling failed.
501    ///
502    /// This is the narrower half of a wider idea, still open: making partial
503    /// progress *within* one failing module (accumulating more errors from it
504    /// rather than stopping at its first) instead of only across modules as this
505    /// does. That would need `check` itself to return something like
506    /// `Result<(Module, Errors), Errors>`, and scoped-fail semantics through
507    /// canonicalization to produce it.
508    ///
509    /// `module_files` is how a checked module's `Interface` learns the file it was
510    /// read from (`Interface::file`, `ERR-5`) — `check` itself is a phase-orchestrating
511    /// function and, like every phase, never knows it. This method is driver code
512    /// rather than a phase, the same as `compile_package`, so it is the one that can
513    /// look the id up and stamp it on before the interface goes into the shared map.
514    /// A module missing from `module_files` — there is none in the real pipeline,
515    /// since only a module that already parsed reaches here — simply leaves that
516    /// interface's `file` as `None`, same as a hand-built one.
517    #[allow(clippy::type_complexity)]
518    pub fn check_in_order<E>(
519        &self,
520        package: &crate::compiler::PackageName,
521        interfaces: &mut HashMap<Name, crate::compiler::Interface>,
522        module_files: &HashMap<Name, crate::compiler::source::files::SourceFileId>,
523        check: fn(
524            package: &crate::compiler::PackageName,
525            interfaces: &HashMap<Name, crate::compiler::Interface>,
526            source: &crate::compiler::parser::Module,
527        ) -> Result<super::canonical::Module, E>,
528    ) -> (Vec<crate::compiler::canonical::Module>, Vec<E>) {
529        let mut modules = Vec::new();
530        let mut errors = Vec::new();
531
532        for module in self.modules.iter() {
533            match check(package, interfaces, module) {
534                Ok(m) => {
535                    // Once we have successfuly checked a module, we can add it to the available interfaces
536                    // for the following modules.
537                    let iface_name = m.name.name().clone();
538                    // Driver code, so this is where the module's file is known: the
539                    // interface carries it so a *later* module's diagnostic can point
540                    // back into this one's source (`ERR-5`).
541                    let iface = m.to_interface(module_files.get(&module.name).copied());
542                    debug!("Inserting {} with value {:?}", iface_name, iface);
543                    interfaces.insert(iface_name, iface);
544
545                    modules.push(m);
546                }
547                Err(err) => errors.push(err),
548            }
549        }
550
551        (modules, errors)
552    }
553}

677    /// The property `cycle_walk` is supposed to guarantee, asserted directly against
678    /// the source modules rather than against `path` restated as `edges`: every
679    /// consecutive pair in `path`, wrapping around from the last back to the first,
680    /// is an `import` actually written in the earlier module.
681    ///
682    /// Checking `Cycle::edges` instead would pin nothing — `build_cycle` derives
683    /// `from`/`to` from consecutive `path` entries unconditionally, and the two
684    /// fields that could tell a real edge from a fabricated one are both dead in
685    /// these fixtures (`file` is always `None` with an empty `module_files`, and
686    /// `NodeSpan`'s `PartialEq` always returns `true`, per CLAUDE.md).

688        assert!(
689            cycle.path.len() > 1,

692        );
693
694        for (i, from) in cycle.path.iter().enumerate() {
695            let to = &cycle.path[(i + 1) % cycle.path.len()];
696            let from_module = modules
697                .iter()
698                .find(|m| &m.name == from)
699                .unwrap_or_else(|| panic!("`{}` is one of the fixture modules", from));
700
701            assert!(
702                from_module.imports.iter().any(|imp| &imp.name == to),
703                "`{}` does not import `{}`, so the reported cycle {:?} is not a walk \
704                 this graph's edges can produce",
705                from,
706                to,
707                cycle.path
708            );
709        }

773    /// Every ordering of `specs` — a list of `(module name, its imports)` — handed to
774    /// `ModuleWalker::new` in turn, with `assert` run on each resulting `Cycle`.
775    ///
776    /// The declaration order is what decides `tarjan_scc`'s ordering of a component,
777    /// and therefore which member `build_cycle` starts its search from. A fixture
778    /// written in one fixed order only ever exercises one start node, and a walk that
779    /// is wrong from *some* start nodes will pass it by luck — which is exactly how
780    /// the dead-end below went unnoticed. Running every permutation removes the luck.
781    fn for_every_declaration_order(
782        specs: &[(&str, Vec<&str>)],
783        assert: impl Fn(&Cycle, &[Module], &[&str]),
784    ) {
785        // Lexicographic permutations of the indices into `specs`, so the whole set is
786        // covered without pulling in a crate for it.
787        let mut order: Vec<usize> = (0..specs.len()).collect();
788
789        loop {
790            let modules: Vec<Module> = order
791                .iter()
792                .map(|&i| module(specs[i].0, specs[i].1.clone()))
793                .collect();
794            let names: Vec<&str> = order.iter().map(|&i| specs[i].0).collect();
795            let module_files = HashMap::new();
796
797            let res = ModuleWalker::new(&modules, &module_files)
798                .expect_err("every fixture here has a cycle in it");
799
800            let Error::CycleDetected(cycles) = &res;
801            assert_eq!(
802                cycles.len(),
803                1,
804                "one component, so one cycle, in declaration order {:?}: {:?}",
805                names,
806                cycles
807            );
808
809            assert(&cycles[0], &modules, &names);
810
811            // Next permutation.
812            let Some(pivot) = (0..order.len() - 1)
813                .rev()
814                .find(|&i| order[i] < order[i + 1])
815            else {
816                break;
817            };
818            let successor = (pivot + 1..order.len())
819                .rev()
820                .find(|&i| order[i] > order[pivot])
821                .expect("`pivot` was chosen because `pivot + 1` qualifies");
822            order.swap(pivot, successor);
823            order[pivot + 1..].reverse();
824        }
825    }

827    /// The counterexample that sank the original greedy walk: a component holding
828    /// two loops that share a node.
829    ///
830    /// ```text
831    /// b imports c, e      b -> c -> d -> b
832    /// c imports d         b -> e -> b
833    /// d imports b
834    /// e imports b
835    /// ```
836    ///
837    /// `{b, c, d, e}` is one component. The greedy walk started at whichever member
838    /// `tarjan_scc` listed first, took the first *unvisited* in-component neighbour
839    /// at each step, and could walk itself into a corner: starting from `d` it went
840    /// to `b`, from `b` to `e`, and `e`'s only neighbour `b` was already visited — so
841    /// it stopped and returned `[d, b, e]`, whose wraparound `e -> d` is not an edge
842    /// of this graph at all. The note then claimed `cycle: d -> b -> e -> d`, stating
843    /// something false about the user's source, and `build_cycle` found no `import`
844    /// behind that step so its label was silently dropped.
845    ///
846    /// Whether the start node is `d` depends on the declaration order, hence
847    /// `for_every_declaration_order`.
848    ///
849    /// Mutation-checked against exactly that: restoring the greedy walk turns this
850    /// test red on `e` not importing `d`.

853        let specs = [
854            ("b", vec!["c", "e"]),
855            ("c", vec!["d"]),
856            ("d", vec!["b"]),
857            ("e", vec!["b"]),
858        ];
859
860        for_every_declaration_order(&specs, |cycle, modules, order| {
861            assert_path_is_a_real_cycle(cycle, modules);
862            // Guard against the property being satisfied vacuously by a walk that
863            // stopped one step in.

1068    /// The implicit edges `add_default_import_edges` adds, as slot pairs.
1069    ///
1070    /// Slots rather than names, so that two namings of one package shape produce
1071    /// directly comparable answers.
1072    fn implicit_edges(names: &[&str], written: &[Vec<usize>]) -> Vec<(usize, usize)> {
1073        let modules: Vec<Module> = names
1074            .iter()
1075            .enumerate()
1076            .map(|(i, n)| {
1077                let deps: Vec<&str> = written[i].iter().map(|&j| names[j]).collect();
1078                module(*n, deps)
1079            })
1080            .collect();
1081
1082        let mut graph = DiGraph::new();
1083        let mut idx_of = HashMap::new();
1084        let mut slot_of = HashMap::new();
1085        for (slot, m) in modules.iter().enumerate() {
1086            let idx = graph.add_node(m);
1087            idx_of.insert(&m.name, idx);
1088            slot_of.insert(idx, slot);
1089        }
1090        for m in modules.iter() {
1091            let from = idx_of[&m.name];
1092            for &to in m.imports.iter().filter_map(|i| idx_of.get(&i.name)) {
1093                graph.add_edge(from, to, ());
1094            }
1095        }
1096
1097        let before = graph.edge_count();
1098        add_default_import_edges(&mut graph, &idx_of);
1099
1100        let mut added: Vec<(usize, usize)> = graph
1101            .edge_indices()
1102            .skip(before)
1103            .filter_map(|e| graph.edge_endpoints(e))
1104            .map(|(from, to)| (slot_of[&from], slot_of[&to]))
1105            .collect();
1106        added.sort_unstable();
1107        added
1108    }

1122    /// `DEFAULT_IMPORTS`, the rest not — and then renames only the *non*-default

1124    /// module would be a different package, so those names stay put.

1131        // ones on every run, or a failure here could not be reproduced.

1133        let mut rnd = move |n: usize| {

1135                .wrapping_mul(6364136223846793005)
1136                .wrapping_add(1442695040888963407);

1141        let pool = [
1142            "Aux", "Bee", "Cee", "Dee", "Eee", "Mmm", "Nnn", "Yak", "Zed",
1143        ];
1144
1145        for trial in 0..2000 {
1146            let default_count = rnd(4);
1147            let other_count = 2 + rnd(5);
1148            let total = default_count + other_count;
1149
1150            let mut default_names: Vec<&str> = Vec::new();
1151            while default_names.len() < default_count {
1152                let candidate = defaults[rnd(defaults.len())];
1153                if !default_names.contains(&candidate) {
1154                    default_names.push(candidate);
1155                }
1156            }
1157            let mut other_names: Vec<&str> = Vec::new();
1158            while other_names.len() < other_count {
1159                let candidate = pool[rnd(pool.len())];
1160                if !other_names.contains(&candidate) {
1161                    other_names.push(candidate);
1162                }
1163            }
1164
1165            // Which slots hold a module named on the list.
1166            let mut is_default = vec![false; total];
1167            let mut remaining = default_count;
1168            for (slot, flag) in is_default.iter_mut().enumerate() {
1169                if remaining > 0 && (total - slot == remaining || rnd(2) == 0) {
1170                    *flag = true;
1171                    remaining -= 1;
1172                }
1173            }
1174
1175            // Written imports, acyclic by construction: a slot only imports lower ones.
1176            let written: Vec<Vec<usize>> = (0..total)
1177                .map(|slot| (0..slot).filter(|_| rnd(3) == 0).collect())
1178                .collect();
1179
1180            let naming = |others: &[&'static str]| -> Vec<&'static str> {
1181                let mut defaults = default_names.iter();
1182                let mut others = others.iter();
1183                (0..total)
1184                    .map(|slot| {
1185                        if is_default[slot] {
1186                            *defaults.next().unwrap()
1187                        } else {
1188                            *others.next().unwrap()
1189                        }
1190                    })
1191                    .collect()
1192            };
1193
1194            let expected = implicit_edges(&naming(&other_names), &written);
1195
1196            for shift in 1..other_names.len() {
1197                let mut renamed = other_names.clone();
1198                renamed.rotate_left(shift);
1199
1200                assert_eq!(
1201                    implicit_edges(&naming(&renamed), &written),
1202                    expected,
1203                    "trial {} renamed {:?} to {:?}, written imports {:?}",
1204                    trial,
1205                    other_names,
1206                    renamed,
1207                    written
1208                );
1209            }
1210        }
1211    }

1//! Walk package's module dependencies
2//!
3//! Within a package, each `Module` will depends on other `Module`s. The aim
4//! of the dependencies module is to provide a way to sort the modules in a dependent
5//! way, allowing parallel action on which module. We will also find out existing
6//! cyclic dependencies and return all of them as error.
7//!
8//! We are currently using the excellent `petgraph` crate for sorting our dependencies
9//! and discovering cycle. We currently use the results of the cycle discovery to drive
10//! how we execute the processing. If we want to introduce parallel processing of the
11//! different module, we will have to be a bit more thorough because we currently don't
12//! have any information _at processing time_ to know when to block the check of some
13//! modules. We don't know how we will do most of that processing, so it's a bit
14//! premature to actually implement multi threading here.
15//! Let's see how it goes first on a single thread and we can revisit this decision
16//! later on.
17//!

29// TODO Improve this with barrier on each module, to be able to parallelize
30// the processing down thel line.
31pub struct ModuleWalker<'a> {
32    /// The list of modules to process. We loose the barrier orders
33    /// so we can't really process them in parallel at the moment.
34    modules: Vec<&'a Module>,
35    /// Whether the package these modules belong to declares one of the eight
36    /// default imports ([`default_imports::declares_a_default`](crate::compiler::default_imports::declares_a_default)),
37    /// asked once of the package and used for both the import graph
38    /// (`add_default_import_edges`) and every `check` call `check_in_order` makes, so
39    /// `implicit_imports` is asked the same question rather than each module
40    /// recomputing it from a module list it never sees.
41    declares_a_default: bool,
42}

54/// One `import` that forms an edge of a dependency cycle.
55///
56/// `from` imports `to` — the module the note lists right after it, wrapping
57/// around to the cycle's start for the last edge — and `span`/`file` are where
58/// that specific `import` line was written, when known.
59///
60/// `file` comes from the `module_files` map [`ModuleWalker::new`] is handed by
61/// `compile_package`, the only place that knows which file a parsed module came
62/// from — the same reasoning `ERR-5` used for [`crate::compiler::Interface::file`].
63/// It is `None` for every edge built in this module's own tests, which hand-build
64/// modules with nothing on disk behind them.
65#[derive(Debug, PartialEq, Clone)]
66pub struct CycleEdge {
67    pub from: Name,
68    pub to: Name,
69    pub span: NodeSpan,
70    pub file: Option<SourceFileId>,
71}

73/// A dependency cycle: one loop through a strongly-connected component of the
74/// import graph, alongside the `import` that created each edge between consecutive
75/// modules (wrapping around to the first), and any component member that loop does
76/// not pass through.
77#[derive(Debug, PartialEq, Clone)]
78pub struct Cycle {
79    /// *One* cycle within the strongly-connected component, in cycle order —
80    /// what the summary note is built from. Not necessarily every module in the
81    /// component: a component with several overlapping loops has more than one
82    /// cycle in it, and picking the shortest one through a single start node (see
83    /// [`cycle_walk`]) can leave others out. Finding a cycle that covers every
84    /// member is the Hamiltonian cycle problem, and no such cycle need exist, so
85    /// this does not try; the members left out are in [`Cycle::others`] instead,
86    /// and the note names them so a user does not break one loop and immediately
87    /// hit the next one hiding in the same component.
88    pub path: Vec<Name>,
89    /// The other modules in the same component: genuinely part of the circular
90    /// dependency, but not on the particular cycle `path` names. Empty whenever
91    /// `path` already covers the component, which every fixture and every real
92    /// import graph seen so far is.
93    pub others: Vec<Name>,
94    /// `edges[i]` is the `import` written in `path[i]` that names `path[i + 1]`
95    /// (or `path[0]`, for the last edge). Same length as `path`, and every entry
96    /// is a real `import`, because [`cycle_walk`] only ever returns a path whose
97    /// consecutive pairs are edges of the graph.
98    pub edges: Vec<CycleEdge>,
99}
100
101#[derive(Debug, PartialEq)]
102pub enum Error {
103    CycleDetected(Vec<Cycle>),
104}

106/// A dependency cycle belongs to the package, not to any one module, so its message
107/// names the modules involved rather than being prefixed with one of them, and its
108/// labels — one per `import` that forms an edge — each carry their own file rather
109/// than relying on a fallback the way a single-module phase error can
110/// (`CompilationError::DependenciesError` renders this without one, see there).
111impl crate::compiler::PhaseError for Error {
112    fn message(&self) -> String {
113        match self {
114            Error::CycleDetected(cycles) => format!(
115                "{} circular dependenc{} between modules",
116                cycles.len(),
117                if cycles.len() == 1 { "y" } else { "ies" }
118            ),
119        }
120    }
121
122    fn notes(&self) -> Vec<String> {
123        match self {
124            // Each cycle is written back to its start, so it reads as the loop it is:
125            // `A -> B -> A`. A component that holds more than the one loop gets a
126            // second note naming the members that loop leaves out, so breaking it
127            // does not just uncover the next cycle in the same component
128            // (`Cycle::others`).
129            Error::CycleDetected(cycles) => cycles
130                .iter()
131                .flat_map(|cycle| {
132                    let mut path: Vec<String> =
133                        cycle.path.iter().map(|name| name.to_string()).collect();
134                    if let Some(first) = path.first().cloned() {
135                        path.push(first);
136                    }
137                    let mut notes = vec![format!("cycle: {}", path.join(" -> "))];
138
139                    if !cycle.others.is_empty() {
140                        let others: Vec<String> =
141                            cycle.others.iter().map(|name| name.to_string()).collect();
142                        notes.push(format!(
143                            "also part of this circular dependency, on another loop through the same modules: {}",
144                            others.join(", ")
145                        ));
146                    }
147
148                    notes
149                })
150                .collect(),
151        }
152    }
153
154    fn labels(&self) -> Vec<SpanLabel> {
155        match self {
156            Error::CycleDetected(cycles) => cycles
157                .iter()
158                .flat_map(|cycle| {
159                    cycle.edges.iter().filter_map(|edge| {
160                        // An edge with no span (a hand-built test module) or no known
161                        // file (nothing in the real pipeline, since `module_files`
162                        // covers every parsed module) has nowhere honest to point;
163                        // it is dropped rather than guessed at, the same rule
164                        // `Interface::source_span` follows.
165                        let span = edge.span.span()?;
166                        let file = edge.file?;
167                        Some(SpanLabel {
168                            span,
169                            // Every edge is equally the cause — breaking any one
170                            // of them breaks the loop — so a label states only its
171                            // own edge and leaves the loop to the note. Saying
172                            // "closing the cycle" here would put that claim on all
173                            // N labels when only the last edge closes it.
174                            message: format!("`{}` imports `{}` here", edge.from, edge.to),
175                            primary: true,
176                            file: Some(file),
177                        })
178                    })
179                })
180                .collect(),
181        }
182    }
183}

185/// Walk `members` — one strongly-connected component `tarjan_scc` found, so every
186/// node in it can reach every other — into an actual directed cycle through
187/// `start`: a sequence of modules where each one really does import the next, and
188/// the last one imports the first.
189///
190/// `tarjan_scc`'s own ordering of a component does not promise this: it is
191/// membership, not a walk, so consecutive entries are not necessarily connected by
192/// an edge, and the note used to print a "cycle" the user could not follow. That is
193/// what `ERR-6` set out to fix, and `build_cycle` needs it to be true a second time
194/// over, because it looks up the `import` behind each consecutive pair and an edge
195/// that does not exist has no `import` to point a label at.
196///
197/// So the guarantee has to hold by construction, not by luck. This is a breadth-first
198/// search from `start` over in-component edges only, stopping at the first node found
199/// to import `start` back; the path is then reconstructed backwards along the BFS
200/// tree. Because BFS pops nodes in non-decreasing distance order, that is the
201/// *shortest* cycle through `start`, and because a BFS tree path never repeats a
202/// node, it is a simple one. A greedy walk that just takes the first unvisited
203/// neighbour cannot promise this — in a component with two loops sharing a node
204/// (`B` imports `C` and `E`; `C` imports `D`; `D` imports `B`; `E` imports `B`) it
205/// wanders into `E`, dead-ends, and hands back `[D, B, E]` even though `E` does not
206/// import `D`.
207///
208/// The cycle it finds may be a strict subset of `members` — see [`Cycle::path`] for
209/// why that is not fixable in general and what is reported instead.
210///
211/// `start` must be one of `members`. Callers get that from the `partition` in
212/// [`ModuleWalker::new`], which is also where the component is known to be
213/// non-empty; taking it as a parameter is what keeps this function off an unchecked
214/// `members[0]`.
215///
216/// The returned path is empty only if `start` lies on no cycle at all, which a
217/// component `tarjan_scc` reported with more than one member cannot produce.
218fn cycle_walk(
219    graph: &DiGraph<&Module, ()>,
220    members: &[NodeIndex],
221    start: NodeIndex,
222) -> Vec<NodeIndex> {
223    let member_set: HashSet<NodeIndex> = members.iter().copied().collect();
224
225    // How each node was first reached. `start` is deliberately absent: it is the
226    // target, so reaching it is what ends the search rather than something to
227    // record a predecessor for.
228    let mut predecessor: HashMap<NodeIndex, NodeIndex> = HashMap::new();
229    let mut queue: VecDeque<NodeIndex> = VecDeque::from([start]);
230    // The node that closes the loop: the first one popped that imports `start`.
231    let mut closing: Option<NodeIndex> = None;
232
233    'search: while let Some(current) = queue.pop_front() {
234        for next in graph.neighbors(current).filter(|n| member_set.contains(n)) {
235            if next == start {
236                closing = Some(current);
237                break 'search;
238            }
239
240            // First time we reach `next` is along a shortest path to it, so a
241            // later, longer way in is ignored.
242            if predecessor.contains_key(&next) {
243                continue;
244            }
245
246            predecessor.insert(next, current);
247            queue.push_back(next);
248        }
249    }
250
251    let mut path = Vec::new();
252
253    if let Some(closing) = closing {
254        let mut node = closing;
255        path.push(node);
256
257        // Back up the BFS tree to `start`. `predecessor` was built by this search,
258        // so every node on that chain has one until `start` is reached; the `None`
259        // arm cannot fire, and stopping is the right thing if it somehow did.
260        while node != start {
261            match predecessor.get(&node) {
262                Some(&previous) => {
263                    node = previous;
264                    path.push(node);
265                }
266                None => break,
267            }
268        }
269
270        path.reverse();
271    }
272
273    path
274}

276/// Turn one strongly-connected component into the [`Cycle`] a diagnostic renders:
277/// the module path `cycle_walk` finds, the `import` behind each edge of it, and the
278/// component members that path leaves out.
279///
280/// `start` is the node the cycle is found through; [`ModuleWalker::new`] takes it
281/// from the component itself, which is how this avoids indexing one.
282fn build_cycle(
283    graph: &DiGraph<&Module, ()>,
284    members: &[NodeIndex],
285    start: NodeIndex,
286    module_files: &HashMap<Name, SourceFileId>,
287) -> Cycle {
288    let node_path = cycle_walk(graph, members, start);
289    let on_path: HashSet<NodeIndex> = node_path.iter().copied().collect();
290
291    let path: Vec<Name> = node_path
292        .iter()
293        .map(|&idx| graph[idx].name.clone())
294        .collect();
295
296    let mut others: Vec<Name> = members
297        .iter()
298        .filter(|idx| !on_path.contains(idx))
299        .map(|&idx| graph[idx].name.clone())
300        .collect();
301    // `tarjan_scc`'s ordering within a component is an implementation detail; a
302    // note the user reads should not vary with it. Sorted on the identifier text
303    // rather than through an `Ord` on `Name`, which `name.rs` deliberately does not
304    // derive.
305    others.sort_by(|l, r| l.as_str().cmp(r.as_str()));
306
307    let edges = node_path
308        .iter()
309        .enumerate()
310        .map(|(i, &idx)| {
311            let from_module = graph[idx];
312            let to_idx = node_path[(i + 1) % node_path.len()];
313            let to_name = graph[to_idx].name.clone();
314
315            // The specific `import …` line in `from_module` that names `to_name` —
316            // there may be several imports in `from_module`, but at most one of
317            // them names this particular neighbour. `cycle_walk` guarantees this is
318            // a real edge, so the only reason to find nothing is a hand-built
319            // module in this file's own tests, which carry no source text.
320            let import = from_module.imports.iter().find(|imp| imp.name == to_name);
321
322            CycleEdge {
323                from: from_module.name.clone(),
324                to: to_name,
325                span: import.map(|imp| imp.span).unwrap_or_else(NodeSpan::none),
326                file: import.and_then(|_| module_files.get(&from_module.name).copied()),
327            }
328        })
329        .collect();
330
331    Cycle {
332        path,
333        others,
334        edges,
335    }
336}

338/// Put the [default imports](crate::compiler::default_imports) into the import
339/// graph, so a module that never wrote `import Basics` is still checked after it.
340///
341/// Canonicalization resolves an implicit import against an `Interface` that already
342/// exists, and nothing but this graph decides which module is checked first — so
343/// without these edges whether `1 + 2` resolved would come down to the order the
344/// source files happened to load in.
345///
346/// `package_declares_a_default` is the package's own answer, taken from every module of
347/// both its source roots: a package that declares one of the eight receives none of
348/// them, and `names` here is one root's worth of modules rather than the package's.
349fn add_default_import_edges(
350    graph: &mut DiGraph<&Module, ()>,
351    names: &HashMap<&Name, NodeIndex>,
352    package_declares_a_default: bool,
353) {
354    if package_declares_a_default {
355        return;
356    }
357
358    let mut importers: Vec<(&Name, NodeIndex)> =
359        names.iter().map(|(&name, &idx)| (name, idx)).collect();
360    importers.sort_by(|left, right| left.0.as_str().cmp(right.0.as_str()));
361
362    for default in crate::compiler::default_imports::DEFAULT_IMPORTS {
363        let target = default.name();
364        let Some(&target_idx) = names.get(&target) else {
365            continue;
366        };
367
368        for &(module_name, module_idx) in &importers {
369            if crate::compiler::default_imports::is_default(module_name) {
370                continue;
371            }
372
373            // A written import already carries the edge, and replaces the implicit
374            // import anyway.
375            if graph[module_idx].imports.iter().any(|i| i.name == target) {
376                continue;
377            }
378
379            if petgraph::algo::has_path_connecting(&*graph, target_idx, module_idx, None) {
380                continue;
381            }
382
383            graph.add_edge(module_idx, target_idx, ());
384        }
385    }
386}

388impl<'a> ModuleWalker<'a> {
389    /// `module_files` is how a cycle's edges learn which file each `import` was
390    /// written in (`CycleEdge::file`) — this is driver code, the same as
391    /// `check_in_order` below, so it is handed the map `compile_package` already
392    /// built while parsing rather than trying to discover it itself. Every test in
393    /// this module passes an empty map: their modules are hand-built, with no file
394    /// on disk to name, and `CycleEdge::file` degrades to `None` for them exactly
395    /// as documented there.
396    pub fn new(
397        modules: &'a [Module],
398        module_files: &HashMap<Name, SourceFileId>,
399    ) -> Result<ModuleWalker<'a>, Error> {
400        // Whether the package declares a default is a question about the package, and
401        // this is the answer for a caller whose `modules` are the whole of one.
402        let declares_a_default =
403            crate::compiler::default_imports::declares_a_default(modules.iter().map(|m| &m.name));
404
405        ModuleWalker::new_for_root(modules, module_files, declares_a_default)
406    }
407
408    /// A walker over the modules of *one* of a package's two source roots.
409    ///
410    /// A package has a `src/` and a `tests/`, and each is walked and checked on its own,
411    /// so the modules handed here are half a package and cannot answer a question about
412    /// the whole of one. `package_declares_a_default` is that question — whether this
413    /// package declares one of the eight [default
414    /// imports](crate::compiler::default_imports) and so receives none of them — asked
415    /// once by [`compile_package`](crate::compiler::compile_package) over both roots,
416    /// which is the only place that has seen both.
417    ///
418    /// The import graph is built from `modules` alone either way. An import naming a
419    /// module of the other root resolves through an
420    /// [`Interface`](crate::compiler::Interface) or not at all, the same as an import
421    /// naming another package's module.
422    pub fn new_for_root(
423        modules: &'a [Module],
424        module_files: &HashMap<Name, SourceFileId>,
425        package_declares_a_default: bool,
426    ) -> Result<ModuleWalker<'a>, Error> {
427        let mut graph = DiGraph::new();
428
429        let mut names = HashMap::new();
430
431        // First populate the graph with all nodes, and keep a reference from Path to the node index
432        for module in modules.iter() {
433            let idx = graph.add_node(module);
434            names.insert(&module.name, idx);
435        }
436
437        // Now that all modules have been inserted, we can create edges
438        for module in modules.iter() {
439            let m_idx = names.get(&module.name).unwrap();
440
441            // When building the dependency graph, we exclude all imports for
442            // modules not in this package. The reason is simple: a dependent
443            // package cannot depends on the current package, so no cycle to detect
444            for dep in module
445                .imports
446                .iter()
447                .filter_map(|import| names.get(&import.name))
448            {
449                graph.add_edge(*m_idx, *dep, ());
450            }
451        }
452
453        add_default_import_edges(&mut graph, &names, package_declares_a_default);
454
455        // Find the strongly connected graphs (scc), if there are more than one node per scc
456        // it means there is a circular dependency.
457        let scc = petgraph::algo::tarjan_scc(&graph);
458
459        let (cycles, deps): (Vec<_>, Vec<_>) = scc.iter().partition(|&v| v.len() > 1);
460
461        if cycles.is_empty() {
462            let modules = deps.into_iter().flatten().map(|&idx| graph[idx]).collect();
463
464            Ok(ModuleWalker {
465                modules,
466                declares_a_default: package_declares_a_default,
467            })
468        } else {
469            // `partition` above already established every component here has more
470            // than one member, so `first()` is always `Some`; taking the start node
471            // from it here, rather than indexing inside `cycle_walk`, is what keeps
472            // that guarantee next to the check that establishes it.
473            let c = cycles
474                .into_iter()
475                .filter_map(|members| {
476                    members
477                        .first()
478                        .map(|&start| build_cycle(&graph, members, start, module_files))
479                })
480                .collect();
481
482            Err(Error::CycleDetected(c))
483        }
484    }

486    /// Given a package name, a set of existing interfaces and a checker function,
487    /// check each module in its dependencies order.
488    ///
489    /// Every module is checked regardless of whether an earlier one failed: the
490    /// successes are returned alongside the errors rather than being discarded as
491    /// soon as one module fails (`BUG-2`). A successfully checked module still has
492    /// its `Interface` inserted into `interfaces` as it goes, so later modules keep
493    /// resolving against earlier ones even when some sibling failed.
494    ///
495    /// This is the narrower half of a wider idea, still open: making partial
496    /// progress *within* one failing module (accumulating more errors from it
497    /// rather than stopping at its first) instead of only across modules as this
498    /// does. That would need `check` itself to return something like
499    /// `Result<(Module, Errors), Errors>`, and scoped-fail semantics through
500    /// canonicalization to produce it.
501    ///
502    /// `module_files` is how a checked module's `Interface` learns the file it was
503    /// read from (`Interface::file`, `ERR-5`) — `check` itself is a phase-orchestrating
504    /// function and, like every phase, never knows it. This method is driver code
505    /// rather than a phase, the same as `compile_package`, so it is the one that can
506    /// look the id up and stamp it on before the interface goes into the shared map.
507    /// A module missing from `module_files` — there is none in the real pipeline,
508    /// since only a module that already parsed reaches here — simply leaves that
509    /// interface's `file` as `None`, same as a hand-built one.
510    ///
511    /// `check` takes `self.declares_a_default` as its last argument — the same
512    /// package-level answer `add_default_import_edges` used to build the import
513    /// graph this walker holds — so every module reaches `implicit_imports` with
514    /// the answer for its own package rather than recomputing it from a module
515    /// list a phase never sees.
516    #[allow(clippy::type_complexity)]
517    pub fn check_in_order<M: crate::compiler::Checked, E>(
518        &self,
519        package: &crate::compiler::PackageName,
520        interfaces: &mut HashMap<Name, crate::compiler::Interface>,
521        module_files: &HashMap<Name, crate::compiler::source::files::SourceFileId>,
522        check: fn(
523            package: &crate::compiler::PackageName,
524            interfaces: &HashMap<Name, crate::compiler::Interface>,
525            source: &crate::compiler::parser::Module,
526            declares_a_default: bool,
527        ) -> Result<M, E>,
528    ) -> (Vec<M>, Vec<E>) {
529        let mut modules = Vec::new();
530        let mut errors = Vec::new();
531
532        for module in self.modules.iter() {
533            match check(package, interfaces, module, self.declares_a_default) {
534                Ok(m) => {
535                    // Once we have successfuly checked a module, we can add it to the available interfaces
536                    // for the following modules.
537                    let iface_name = m.name().name().clone();
538                    // Driver code, so this is where the module's file is known: the
539                    // interface carries it so a *later* module's diagnostic can point
540                    // back into this one's source (`ERR-5`).
541                    let iface = m.to_interface(module_files.get(&module.name).copied());
542                    debug!("Inserting {} with value {:?}", iface_name, iface);
543                    interfaces.insert(iface_name, iface);
544
545                    modules.push(m);
546                }
547                Err(err) => errors.push(err),
548            }
549        }
550
551        (modules, errors)
552    }
553}

591    /// A `CycleEdge` matching what `module()` above produces: no span, since
592    /// there is no source text behind a hand-built import, and no file, since
593    /// these tests pass an empty `module_files` map to `ModuleWalker::new`.
594    fn edge<S: Into<String>>(from: S, to: S) -> CycleEdge {
595        CycleEdge {
596            from: Name::new(from),
597            to: Name::new(to),

684    /// The property `cycle_walk` is supposed to guarantee, asserted directly against
685    /// the source modules rather than against `path` restated as `edges`: every
686    /// consecutive pair in `path`, wrapping around from the last back to the first,
687    /// is an `import` actually written in the earlier module.
688    ///
689    /// Checking `Cycle::edges` instead would pin nothing — `build_cycle` derives
690    /// `from`/`to` from consecutive `path` entries unconditionally, and the two
691    /// fields that could tell a real edge from a fabricated one are both dead in
692    /// these fixtures (`file` is always `None` with an empty `module_files`, and
693    /// `NodeSpan`'s `PartialEq` always returns `true`, per CLAUDE.md).
694    fn assert_path_is_a_real_cycle(cycle: &Cycle, modules: &[Module]) {
695        assert!(
696            cycle.path.len() > 1,
697            "a cycle between modules needs at least two of them, got {:?}",
698            cycle.path
699        );
700
701        for (i, from) in cycle.path.iter().enumerate() {
702            let to = &cycle.path[(i + 1) % cycle.path.len()];
703            let from_module = modules
704                .iter()
705                .find(|m| &m.name == from)
706                .unwrap_or_else(|| panic!("`{}` is one of the fixture modules", from));
707
708            assert!(
709                from_module.imports.iter().any(|imp| &imp.name == to),
710                "`{}` does not import `{}`, so the reported cycle {:?} is not a walk \
711                 this graph's edges can produce",
712                from,
713                to,
714                cycle.path
715            );
716        }
717    }

780    /// Every ordering of `specs` — a list of `(module name, its imports)` — handed to
781    /// `ModuleWalker::new` in turn, with `assert` run on each resulting `Cycle`.
782    ///
783    /// The declaration order is what decides `tarjan_scc`'s ordering of a component,
784    /// and therefore which member `build_cycle` starts its search from. A fixture
785    /// written in one fixed order only ever exercises one start node, and a walk that
786    /// is wrong from *some* start nodes will pass it by luck — which is exactly how
787    /// the dead-end below went unnoticed. Running every permutation removes the luck.
788    fn for_every_declaration_order(
789        specs: &[(&str, Vec<&str>)],
790        assert: impl Fn(&Cycle, &[Module], &[&str]),
791    ) {
792        // Lexicographic permutations of the indices into `specs`, so the whole set is
793        // covered without pulling in a crate for it.
794        let mut order: Vec<usize> = (0..specs.len()).collect();
795
796        loop {
797            let modules: Vec<Module> = order
798                .iter()
799                .map(|&i| module(specs[i].0, specs[i].1.clone()))
800                .collect();
801            let names: Vec<&str> = order.iter().map(|&i| specs[i].0).collect();
802            let module_files = HashMap::new();
803
804            let res = ModuleWalker::new(&modules, &module_files)
805                .expect_err("every fixture here has a cycle in it");
806
807            let Error::CycleDetected(cycles) = &res;
808            assert_eq!(
809                cycles.len(),
810                1,
811                "one component, so one cycle, in declaration order {:?}: {:?}",
812                names,
813                cycles
814            );
815
816            assert(&cycles[0], &modules, &names);
817
818            // Next permutation.
819            let Some(pivot) = (0..order.len() - 1)
820                .rev()
821                .find(|&i| order[i] < order[i + 1])
822            else {
823                break;
824            };
825            let successor = (pivot + 1..order.len())
826                .rev()
827                .find(|&i| order[i] > order[pivot])
828                .expect("`pivot` was chosen because `pivot + 1` qualifies");
829            order.swap(pivot, successor);
830            order[pivot + 1..].reverse();
831        }
832    }

834    /// The counterexample that sank the original greedy walk: a component holding
835    /// two loops that share a node.
836    ///
837    /// ```text
838    /// b imports c, e      b -> c -> d -> b
839    /// c imports d         b -> e -> b
840    /// d imports b
841    /// e imports b
842    /// ```
843    ///
844    /// `{b, c, d, e}` is one component. The greedy walk started at whichever member
845    /// `tarjan_scc` listed first, took the first *unvisited* in-component neighbour
846    /// at each step, and could walk itself into a corner: starting from `d` it went
847    /// to `b`, from `b` to `e`, and `e`'s only neighbour `b` was already visited — so
848    /// it stopped and returned `[d, b, e]`, whose wraparound `e -> d` is not an edge
849    /// of this graph at all. The note then claimed `cycle: d -> b -> e -> d`, stating
850    /// something false about the user's source, and `build_cycle` found no `import`
851    /// behind that step so its label was silently dropped.
852    ///
853    /// Whether the start node is `d` depends on the declaration order, hence
854    /// `for_every_declaration_order`.
855    ///
856    /// Mutation-checked against exactly that: restoring the greedy walk turns this
857    /// test red on `e` not importing `d`.
858    #[test]
859    fn cycle_closes_on_a_real_import_when_two_loops_share_a_module() {
860        let specs = [
861            ("b", vec!["c", "e"]),
862            ("c", vec!["d"]),
863            ("d", vec!["b"]),
864            ("e", vec!["b"]),
865        ];
866
867        for_every_declaration_order(&specs, |cycle, modules, order| {
868            assert_path_is_a_real_cycle(cycle, modules);
869            // Guard against the property being satisfied vacuously by a walk that
870            // stopped one step in.
871            assert!(
872                cycle.path.len() >= 2,
873                "declaration order {:?} produced {:?}",
874                order,
875                cycle.path
876            );
877        });
878    }

885    /// b imports d, c      b -> c -> b
886    /// c imports b
887    /// d imports a
888    /// ```
889    ///
890    /// `{a, b, c, d}` is one component holding two loops that overlap only at `b`.
891    /// No cycle covers all four, so whichever one `cycle_walk` reports leaves
892    /// members out — and a user who breaks only the loop they were shown walks
893    /// straight into the other. `Cycle::others` is how the ones left out stay
894    /// visible, and `notes()` renders them.
895    ///
896    /// Mutation-checked by hardcoding `others: vec![]` in `build_cycle`: the union
897    /// assertion below goes red.

901

903            ("a", vec!["b"]),
904            ("b", vec!["d", "c"]),
905            ("c", vec!["b"]),
906            ("d", vec!["a"]),
907        ];
908
909        for_every_declaration_order(&specs, |cycle, modules, order| {

932            // the note, not just the struct, has to say what.
933            assert!(

937                order,
938                cycle.path
939            );

941            let notes = Error::CycleDetected(vec![cycle.clone()]).notes();

943                notes.iter().any(|n| {
944                    n.starts_with("also part of this circular dependency")
945                        && cycle.others.iter().all(|o| n.contains(o.as_str()))

1076    /// The implicit edges `add_default_import_edges` adds, as slot pairs.
1077    ///
1078    /// Slots rather than names, so that two namings of one package shape produce
1079    /// directly comparable answers.
1080    fn implicit_edges(names: &[&str], written: &[Vec<usize>]) -> Vec<(usize, usize)> {
1081        let modules: Vec<Module> = names
1082            .iter()
1083            .enumerate()
1084            .map(|(i, n)| {
1085                let deps: Vec<&str> = written[i].iter().map(|&j| names[j]).collect();
1086                module(*n, deps)
1087            })
1088            .collect();
1089
1090        let mut graph = DiGraph::new();
1091        let mut idx_of = HashMap::new();
1092        let mut slot_of = HashMap::new();
1093        for (slot, m) in modules.iter().enumerate() {
1094            let idx = graph.add_node(m);
1095            idx_of.insert(&m.name, idx);
1096            slot_of.insert(idx, slot);
1097        }
1098        for m in modules.iter() {
1099            let from = idx_of[&m.name];
1100            for &to in m.imports.iter().filter_map(|i| idx_of.get(&i.name)) {
1101                graph.add_edge(from, to, ());
1102            }
1103        }
1104
1105        let before = graph.edge_count();
1106        // The same question `ModuleWalker::new` asks of a whole package: these names
1107        // are the package here, so they are what it is asked of.
1108        let declares_a_default =
1109            crate::compiler::default_imports::declares_a_default(idx_of.keys().copied());
1110        add_default_import_edges(&mut graph, &idx_of, declares_a_default);
1111
1112        let mut added: Vec<(usize, usize)> = graph
1113            .edge_indices()
1114            .skip(before)
1115            .filter_map(|e| graph.edge_endpoints(e))
1116            .map(|(from, to)| (slot_of[&from], slot_of[&to]))
1117            .collect();
1118        added.sort_unstable();
1119        added
1120    }

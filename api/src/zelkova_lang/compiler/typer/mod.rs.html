1//! This module contains the type checker pass of the language
2//!
3//! It works with the source AST and will perform two jobs:
4//! - type checks the different declarations and expression
5//! - infer the types when not declared in the source
6//!
7//! I have no idea how that works; so bear with me while I explore
8//! the space, make mistake and (hopefully) learn something :)
9//!
10//! Some papers on type inference:
11//! - http://steshaw.org/hm/hindley-milner.pdf
12//! - https://pdfs.semanticscholar.org/8983/233b3dff2c5b94efb31235f62bddc22dc899.pdf
13//! - http://gallium.inria.fr/~fpottier/publis/fpottier-elaboration.pdf
14//! - http://gallium.inria.fr/~fpottier/publis/emlti-final.pdf
15//!
16//! A type inference problem consists of a type environment Γ , an expression t, and a type T of kind ?
17//!
18//! Constraint generation rules:
19//!
20//! - Equation 1: ⟦x : T⟧ = x ≼ T
21//!   "x has type T if and only if T is an instance of the type scheme associated with x"
22//!   Important part: There is no relation to the typing environment Γ, instead x appears free (and will be bound to Γ later)
23//!
24//! - Equation 2: ⟦λz.t : T⟧ = ∃X1X2.(let z : X1 in ⟦t : X2⟧ ∧ X1 → X2 ≤ T)
25//!   "λz.t has type T if and only if, for some X1 and X2,
26//!   (i) under the assumption that z has type X1, t has type X2, and
27//!   (ii) T is a supertype of X1 → X2."
28//!   z and t types must be fresh (can't generally guess them). They are _existentially_ bound because we are going to
29//!   solve their values. Note that z is _not_ fresh in the condition (i).
30//!
31//! - Equation 3: ⟦t1 t2 : T⟧ = ∃X2.(⟦t1 : X2 → T⟧ ∧ ⟦t2 : X2⟧)
32//!   "t1 t2 has type T if and only if, for some X2, t1 has type X2 → T and t2 has type X2"
33//!
34//! - Equation 4: ⟦let z = t1 in t2 : T⟧ = let z : ∀X[⟦t1 : X⟧].X in ⟦t2 : T⟧
35//!   "let z = t1 in t2 has type T if and only if, under the assumption that z has every type X such that ⟦t1 : X⟧ holds, t2 has type T"
36//!
37//!

55/// Why two types were required to match.
56///
57/// A constraint on its own is a pair of types, which is everything inference needs
58/// and nothing a reader can act on. The reason is what turns "cannot match `Bool`
59/// with `Int`" into something located: *this branch* has type `Bool`, and `Int` is
60/// what was expected *because of this annotation*. Elm calls this a `Reason`, rustc
61/// an `ObligationCause`.
62#[derive(Debug, Clone, Copy, PartialEq, Eq)]
63pub enum Reason {
64    /// A declaration's type annotation, which its body has to satisfy.
65    Annotation,
66    /// A literal's own type: `'a'` is a `Char`.
67    Literal,
68    /// A function's type is its parameter's type arrow its body's type.
69    FunctionShape,
70    /// What is applied has to be a function from the argument's type.
71    Application,
72    /// The condition of an `if` has to be a `Bool`.
73    IfCondition,
74    /// Every branch of an `if` has the type of the whole `if`.
75    IfBranch,
76    /// Every branch of a `case` has the type of the whole `case`.
77    CaseBranch,
78    /// A pattern has to match the type of the expression being matched on.
79    CasePattern,
80    /// A parameter written as a pattern has to match the type of the argument it takes.
81    ParameterPattern,
82    /// The body of a declaration that wrote a parameter as a pattern has the type the
83    /// declaration returns. It is the one branch of the match that parameter became,
84    /// and is reported as what the source wrote: a body, not a `case` branch.
85    DeclarationBody,
86    /// A `let` binding has the type of the value bound to it.
87    LetBinding,
88    /// A `let` has the type of its body.
89    LetBody,
90    /// A tuple's type is the tuple of its elements' types.
91    TupleElements,
92}
93
94impl Reason {
95    /// What goes under the caret when the constraint carrying this reason is the one
96    /// that failed: what the underlined text *is*, in the vocabulary of the source.
97    ///
98    /// It names no type, and that is not terseness. By the time a constraint fails,
99    /// `unify` has substituted solutions into both of its sides and may have
100    /// decomposed it into a component of the types the source actually mentions — so
101    /// neither side is reliably "the type of the text under this caret" any more.
102    /// The headline already prints both types; a label that named the wrong one
103    /// would be worse than a label that names none. The example that forced this:
104    /// `answer : Int` with body `true` fails on the literal's own constraint *after*
105    /// `Int` was substituted into it, and reading the type off that side produced
106    /// "this literal has type `Int`".
107    fn describes(&self) -> &'static str {
108        match self {
109            Reason::Annotation => "this type annotation",
110            Reason::Literal => "this literal",
111            Reason::FunctionShape => "this function",
112            Reason::Application => "the expression being applied",
113            Reason::IfCondition => "this condition",
114            Reason::IfBranch => "this branch of the `if`",
115            Reason::CaseBranch => "this branch of the `case`",
116            Reason::CasePattern => "this pattern",
117            Reason::ParameterPattern => "this pattern",
118            Reason::DeclarationBody => "the body of this declaration",
119            Reason::LetBinding => "the value bound here",
120            Reason::LetBody => "the body of this `let`",
121            Reason::TupleElements => "this tuple",
122        }
123    }
124
125    /// What goes under the caret when this reason is not the failure itself but the
126    /// explanation for one side of it — see [`Origin::explanation`].
127    ///
128    /// No type is interpolated here, deliberately. Provenance records which
129    /// constraint brought a type into another one; it does not prove that the type
130    /// printed in the failing constraint is still literally the one this constraint
131    /// carried, and a label is not the place to guess.
132    fn explains(&self) -> &'static str {
133        match self {
134            Reason::Annotation => "expected because of this type annotation",
135            Reason::Literal => "expected because of this literal",
136            Reason::FunctionShape => "expected because of this function",
137            Reason::Application => "expected because of this application",
138            Reason::IfCondition => "expected because this is an `if` condition",
139            Reason::IfBranch => "expected because of this branch",
140            Reason::CaseBranch => "expected because of this branch",
141            Reason::CasePattern => "expected because of this pattern",
142            Reason::ParameterPattern => "expected because of this pattern",
143            Reason::DeclarationBody => "expected because of this declaration's body",
144            Reason::LetBinding => "expected because of this value",
145            Reason::LetBody => "expected because of this `let` body",
146            Reason::TupleElements => "expected because of this tuple",
147        }
148    }
149
150    /// The rule that was broken, when naming it says something the labels do not.
151    fn note(&self) -> Option<&'static str> {
152        match self {
153            Reason::Annotation => {
154                Some("a declaration's body must have the type its annotation declares")
155            }
156            Reason::IfCondition => Some("the condition of an `if` must be a `Bool`"),
157            Reason::IfBranch => Some("every branch of an `if` must have the same type"),
158            Reason::CaseBranch => Some("every branch of a `case` must have the same type"),
159            Reason::CasePattern => Some(
160                "every pattern of a `case` must match the type of the expression it matches on",
161            ),
162            Reason::ParameterPattern => {
163                Some("a parameter's pattern must match the type of the argument it takes")
164            }
165            _ => None,
166        }
167    }
168}

170/// One piece of source text, and why it required a type: the answer to "where did
171/// this type come from".
172///
173/// Deliberately flat — no chain. A cause names the constraint that *introduced* a
174/// type, never one that relayed it, and that is arranged when the cause is built
175/// rather than by walking a chain afterwards. See `Origin::cause_of`.
176// No `Eq`: `NodeSpan`'s `PartialEq` is deliberately blind (see its documentation), so
177// equality here is a claim about the reason and not about the position.
178#[derive(Debug, Clone, Copy, PartialEq)]
179pub struct Cause {
180    /// Why that constraint required the two types to match.
181    pub reason: Reason,
182    /// The source text it is about.
183    pub span: NodeSpan,
184}
185
186/// Which side of a constraint a type sits on.
187///
188/// Unification is symmetric and does not care; provenance does. Whether a solved
189/// type was read off the left or the right of the constraint that solved it is what
190/// decides whether that constraint is where the type came from, or merely where a
191/// substitution put it — see [`Origin::cause_of`].
192#[derive(Debug, Clone, Copy, PartialEq, Eq)]
193enum Side {
194    Left,
195    Right,
196}
197
198/// Where a constraint came from, and — once inference has moved types around — where
199/// each of its two types came from.
200#[derive(Debug, Clone, PartialEq)]
201pub struct Origin {
202    /// Why the two types were required to match.
203    pub reason: Reason,
204    /// The source text this constraint is about. A failure here draws its caret at
205    /// this span.
206    pub span: NodeSpan,
207    /// The constraint whose solution first rewrote the left type, if any.
208    ///
209    /// `unify` solves a constraint by substituting a type for a variable and then
210    /// applying that substitution to every constraint left. Once that has happened,
211    /// this side holds a type this constraint never mentioned: it came from wherever
212    /// the substitution did. That is the fact a reader needs — "`Int`, because of the
213    /// annotation two lines up" — and it is what the secondary label of a type error
214    /// renders.
215    ///
216    /// The *first* rewrite is kept and later ones are dropped: the first substitution
217    /// to reach a side is the one that brought a foreign type into it, and later ones
218    /// only rewrite what is already there.
219    left_from: Option<Cause>,
220    /// The same, for the right type.
221    right_from: Option<Cause>,
222}
223
224impl Origin {
225    fn new(reason: Reason, span: NodeSpan) -> Origin {
226        Origin {
227            reason,
228            span,
229            left_from: None,
230            right_from: None,
231        }
232    }
233
234    /// This constraint, as the explanation for a type it introduced itself.
235    fn own_cause(&self) -> Cause {
236        Cause {
237            reason: self.reason,
238            span: self.span,
239        }
240    }
241
242    /// Where the type on `side` came from: whatever rewrote that side, or — when
243    /// nothing did — this constraint itself.
244    ///
245    /// This is the whole of the provenance rule, and it is a rule about *sides*.
246    /// When `unify` solves `t := T` from a constraint, `T` was read off one side of
247    /// it. If that side is a type the constraint was written with, the constraint is
248    /// the answer. If a previous solution had rewritten that side, the constraint is
249    /// only relaying a type, and the answer is whatever rewrote it — which is already
250    /// flat, so the credit passes straight through and no chain is ever built.
251    ///
252    /// Crediting a constraint that merely relayed a type is how `result : Bool` /
253    /// `result = not 42` came to blame the annotation: the `Bool` that `42` fails
254    /// against is `not`'s parameter type, which the application constraint carries on
255    /// its *left*, while the annotation had only rewritten its right.
256    fn cause_of(&self, side: Side) -> Cause {
257        let rewritten = match side {
258            Side::Left => self.left_from,
259            Side::Right => self.right_from,
260        };
261
262        rewritten.unwrap_or_else(|| self.own_cause())
263    }
264
265    /// Record that a solution rewrote one side of this constraint, keeping whatever
266    /// was already recorded for that side. See [`Origin::left_from`].
267    fn rewritten(&mut self, side: Side, cause: Cause) {
268        let slot = match side {
269            Side::Left => &mut self.left_from,
270            Side::Right => &mut self.right_from,
271        };
272
273        if slot.is_none() {
274            *slot = Some(cause);
275        }
276    }
277
278    /// Where the type this constraint clashed on came from, when the constraint is
279    /// not itself where it came from.
280    ///
281    /// This is what a diagnostic's secondary label and its rule note are both read
282    /// off. The left side is preferred because that is the declared or expected side
283    /// by convention (see `Constraint`), so "expected because of …" reads about the
284    /// right one; the right side answers when only it was rewritten.
285    pub fn explanation(&self) -> Option<Cause> {
286        self.left_from.or(self.right_from)
287    }
288}

290/// What went wrong, and — for everything unification can raise — where.
291///
292/// Each variant carries the [`Origin`] of the constraint that failed, so a type
293/// error can put its caret under the sub-expression that disagrees instead of across
294/// the declaration containing it. The declaration is still named by [`Error`], one
295/// level up, which is the frame that knows it.
296#[derive(Debug)]
297pub enum ErrorKind {
298    /// Two types were required to match and do not.
299    ///
300    /// The two are written in the order the message reads them out — declared side
301    /// first where the source had one. Neither is reliably "the type of the text the
302    /// origin points at": by the time a constraint fails, unification has substituted
303    /// into both sides and may have decomposed the pair the source actually wrote.
304    // The origins and the types are boxed because an `ErrorKind` is the `Err` half of
305    // a `Result` threaded through the whole of `unify`, and every *successful* return
306    // pays for the size of the largest variant. `Origin` carries a `Cause` per side,
307    // and a `Type` carries a whole `QualName` for every union it names.
308    UnificationFailed {
309        left: Box<Type>,
310        right: Box<Type>,
311        origin: Box<Origin>,
312    },
313    /// A type variable would have to occur inside its own solution.
314    CircularType {
315        /// The type the variable would have had to contain itself in. Boxed for the
316        /// reason above.
317        tpe: Box<Type>,
318        origin: Box<Origin>,
319    },
320    /// A name the typer's environment does not know. `type_check` turns this into a
321    /// [`Solved::UnboundName`] rather than an [`Error`]; that variant says why.
322    UnboundVariable {
323        name: String,
324        /// Where the name was written.
325        span: NodeSpan,
326    },
327}
328
329impl ErrorKind {
330    /// The provenance of the constraint that failed, when the failure came from
331    /// unification at all.
332    fn origin(&self) -> Option<&Origin> {
333        match self {
334            ErrorKind::UnificationFailed { origin, .. }
335            | ErrorKind::CircularType { origin, .. } => Some(origin.as_ref()),
336            ErrorKind::UnboundVariable { .. } => None,
337        }
338    }
339
340    fn message(&self) -> String {
341        match self {
342            ErrorKind::UnificationFailed { left, right, .. } => {
343                // Two modules may each declare a union of the same name, and the
344                // source spells both of them bare. Written that way the sentence
345                // uses one word for two declarations and says nothing; the module
346                // that declared each is what tells them apart, so both sides take
347                // it — both, because a sentence that qualifies one side and not the
348                // other reads as if only one of them had a module.
349                if AdtNames::for_all([left.as_ref(), right.as_ref()]) == AdtNames::Qualified {
350                    format!(
351                        "cannot match `{}` with `{}`",
352                        Qualified(left),
353                        Qualified(right)
354                    )
355                } else {
356                    format!("cannot match `{}` with `{}`", left, right)
357                }
358            }
359            // One type, but it can hold the collision on its own: the variable's
360            // solution is built out of whatever it was unified against, which may
361            // be two same-named unions from two modules.
362            ErrorKind::CircularType { tpe, .. } => {
363                let tpe: &dyn std::fmt::Display = match AdtNames::for_all([tpe.as_ref()]) {
364                    AdtNames::Qualified => &Qualified(tpe),
365                    AdtNames::Unqualified => &**tpe,
366                };
367
368                format!(
369                    "circular type: a type variable would have to contain itself in `{}`",
370                    tpe
371                )
372            }
373            ErrorKind::UnboundVariable { name, .. } => {
374                format!("cannot find a value named `{}`", name)
375            }
376        }
377    }
378}

380/// A type error: what went wrong, where, and in which declaration.
381///
382/// # Where the positions come from
383///
384/// The typer does not check the canonical AST directly — it translates it into its
385/// own [`Term`] language — but that translation now carries the canonical node's
386/// [`NodeSpan`] along, and each constraint keeps the span of the term that produced
387/// it. So a unification failure knows the sub-expression it is about, and
388/// [`labels`](PhaseError::labels) draws the caret there.
389///
390/// The declaration's own span is still kept as `span`, for the one case that has no
391/// finer answer: an error whose origin came from a hand-built term, or from a
392/// canonical node the parser never spanned. Then the label degrades to the whole
393/// declaration rather than disappearing.
394#[derive(Debug)]
395pub struct Error {
396    pub kind: ErrorKind,
397    /// Where the declaration the error was found in was written. Used only as the
398    /// fallback described above.
399    pub span: NodeSpan,
400    /// That declaration's name, which is what the note names: in a module of a
401    /// hundred declarations, a caret is not much use without it when the diagnostic
402    /// is rendered without a file (see `compile_package`).
403    pub declaration: Name,
404}
405
406/// Type errors are about types, and [`Type`]'s `Display` writes them the way the
407/// source does — so the message can name both sides of a mismatch instead of
408/// dumping the typer's internal representation.
409impl PhaseError for Error {
410    fn message(&self) -> String {
411        self.kind.message()
412    }
413
414    fn notes(&self) -> Vec<String> {
415        let mut notes = vec![format!("in the declaration of `{}`", self.declaration)];
416
417        // The rule that was broken, taken from the failing constraint and, if that
418        // one has nothing to add, from what explains it. `if`'s "every branch must
419        // have the same type" is on the branch constraint; "a body must have the
420        // type its annotation declares" is on the annotation, which is normally the
421        // explanation rather than the failure.
422        if let Some(origin) = self.kind.origin() {
423            let rule = origin
424                .reason
425                .note()
426                .or_else(|| origin.explanation().and_then(|c| c.reason.note()));
427
428            if let Some(rule) = rule {
429                notes.push(rule.to_owned());
430            }
431        }
432
433        notes
434    }
435
436    /// A caret under the text that disagrees, and — when inference can say where the
437    /// type it disagrees with came from — a second, secondary one under that.
438    ///
439    /// Falls back to underlining the whole declaration when the failing constraint
440    /// has no position, which is what a term built by hand, or one translated from a
441    /// canonical node with no span, produces.
442    fn labels(&self) -> Vec<SpanLabel> {
443        let mut labels = Vec::new();
444
445        match &self.kind {
446            ErrorKind::UnboundVariable { span, .. } => {
447                if let Some(span) = span.span() {
448                    labels.push(SpanLabel {
449                        span,
450                        message: "not found in this scope".to_owned(),
451                        primary: true,
452                        file: None,
453                    });
454                }
455            }
456            kind => {
457                if let Some(origin) = kind.origin() {
458                    if let Some(span) = origin.span.span() {
459                        labels.push(SpanLabel {
460                            span,
461                            message: origin.reason.describes().to_owned(),
462                            primary: true,
463                            file: None,
464                        });
465                    }
466
467                    // Only worth drawing once the primary one exists: on its own it
468                    // would be a caret under the annotation with nothing to contrast
469                    // it against. And not worth drawing at all when it lands on the
470                    // same text — a literal that is its own explanation renders as two
471                    // carets under one word saying the same thing twice.
472                    if let (Some(primary), Some(because)) = (labels.first(), origin.explanation()) {
473                        match because.span.span() {
474                            Some(span) if span != primary.span => labels.push(SpanLabel {
475                                span,
476                                message: because.reason.explains().to_owned(),
477                                primary: false,
478                                file: None,
479                            }),
480                            _ => (),
481                        }
482                    }
483                }
484            }
485        }
486
487        if labels.is_empty() {
488            if let Some(span) = self.span.span() {
489                labels.push(SpanLabel {
490                    span,
491                    message: format!("in `{}`", self.declaration),
492                    primary: true,
493                    file: None,
494                });
495            }
496        }
497
498        labels
499    }
500}

502/// Type check one canonical module and hand back what it solved, reporting every value
503/// whose inference produced a reportable error rather than stopping at the first — the
504/// shape `compile_package` is built to accumulate.
505///
506/// # What comes back
507///
508/// One [`Solved`] per declaration, keyed the way `module.values` is. A declaration the
509/// typer could not type is present and says so; none is ever merely absent, because
510/// absent is indistinguishable from checked-and-fine to whatever reads this next
511/// ([`DEC-18` decision 1](../../../docs/decisions/dec-18.md)). The map is only returned
512/// at all when no declaration failed: a module with type errors answers with them, so
513/// there is no half-checked module to interpret.
514///
515/// The term inside [`Solved::Typed`] is the one `annotate` built, with `unify`'s final
516/// substitution applied to every node rather than to the declaration's own type alone.
517/// That interior is the point — a backend needs the type of each sub-expression, not
518/// just of the declaration containing it.
519///
520/// # Where an [`Error`] is built
521///
522/// Here, and only here: this is the last frame that holds the `canonical::Value`, so
523/// it is the only one that knows the declaration's name and its span. What went
524/// *wrong* and *where inside the declaration* comes up from inference on the
525/// [`ErrorKind`]; see [`Error`].
526///
527/// # What a declaration is checked against
528///
529/// `interfaces` is the map `module` was canonicalized against. Every value and union
530/// it exposes is put in the typer's environment beside this module's own, so a
531/// declaration that forwards an imported value, builds an imported constructor or
532/// matches on one is checked like any other. Each use of one of those names gets a
533/// fresh instance of its declared type, so one declaration can use `Just` or
534/// `Maybe.withDefault` at two types — see [`Types`].
535pub fn type_check(
536    module: &Module,
537    interfaces: &HashMap<Name, Interface>,
538) -> Result<HashMap<Name, Solved>, Vec<Error>> {
539    // A `module foreign` facade uses synthetic placeholder bodies, so there is nothing
540    // to infer — but every declaration still has to be accounted for, so each is
541    // returned saying why it has no term.
542    if module.binding_foreign {
543        return Ok(module
544            .values
545            .keys()
546            .map(|name| (name.clone(), Solved::NoBody))
547            .collect());
548    }
549
550    // Start at a high offset to avoid collisions with the counter inside
551    // Types::new() (which starts at 10) used during inference.
552    let mut counter = 10_000u32;
553
554    // First pass: build global env from every declared type in reach — the values
555    // each imported interface exposes, then this module's own annotated values.
556    let mut global: HashMap<String, Type> = HashMap::new();
557
558    // An imported value is keyed the way a `VarForeign` reference spells it: its name
559    // qualified by the module that declared it. That is the only key, so it cannot
560    // collide with a local name or with a same-named value of another module. An
561    // operator's backing function the header did not expose by name is in
562    // `infix_functions` rather than `values`, and an operator resolves to a
563    // `VarForeign` naming it all the same.
564    for interface in interfaces.values() {
565        for (name, (_, tpe)) in interface.values.iter().chain(&interface.infix_functions) {
566            let mut var_map = HashMap::new();
567            if let Some(typer_tpe) = canonical_type_to_typer_type(tpe, &mut var_map, &mut counter) {
568                let qname = interface.module_name.qualify_name(name).to_name();
569                global.insert(qname.as_str().to_string(), typer_tpe);
570            }
571        }
572    }
573
574    for (name, value) in &module.values {
575        if let canonical::Value::TypedValue { tpe, .. } = value {
576            let mut var_map = HashMap::new();
577            if let Some(typer_tpe) = canonical_type_to_typer_type(tpe, &mut var_map, &mut counter) {
578                // Add both qualified (e.g. "Test.not") and unqualified (e.g. "not") names
579                let qname = module
580                    .name
581                    .qualify_name(name)
582                    .to_name()
583                    .as_str()
584                    .to_string();
585                global.insert(qname, typer_tpe.clone());
586                global.insert(name.as_str().to_string(), typer_tpe);
587            }
588        }
589    }
590
591    // What the canonical AST is read against: every union in reach, their
592    // constructors and the arity of each of this module's declarations. See
593    // [`Translation`].
594    let translation = Translation::of(module, interfaces);
595
596    // Second pass: add constructor types to global from every union in reach, this
597    // module's own and every imported interface's alike.
598    for (type_name, union_type) in &translation.unions {
599        // Fresh type vars for each ADT type parameter (e.g. "a" in Maybe a)
600        let mut adt_var_map: HashMap<String, TypeVariable> = HashMap::new();
601        for tv_name in &union_type.variables {
602            counter += 1;
603            adt_var_map.insert(tv_name.as_str().to_string(), TypeVariable { id: counter });
604        }
605
606        // Build result type: Adt(type_name, [TypeVar for each param])
607        let result_args: Vec<Type> = union_type
608            .variables
609            .iter()
610            .map(|v| Type::Variable(adt_var_map[v.as_str()].clone()))
611            .collect();
612        let result_type = Type::Adt(type_name.clone(), result_args);
613
614        for ctor in &union_type.variants {
615            let ctor_type = if ctor.type_parameters.is_empty() {
616                result_type.clone()
617            } else {
618                let mut translate_var_map = adt_var_map.clone();
619                let params: Vec<Type> = ctor
620                    .type_parameters
621                    .iter()
622                    .filter_map(|t| {
623                        canonical_type_to_typer_type(t, &mut translate_var_map, &mut counter)
624                    })
625                    .collect();
626                if params.len() != ctor.type_parameters.len() {
627                    continue; // untranslatable param type, skip this constructor
628                }
629                // Build Fun type: p1 -> p2 -> ... -> result_type
630                params
631                    .into_iter()
632                    .rev()
633                    .fold(result_type.clone(), |acc, p| Type::Fun {
634                        param_tpe: Box::new(p),
635                        return_tpe: Box::new(acc),
636                    })
637            };
638
639            // Registered under the qualified name a `VarConstructor` spells —
640            // "Maybe.Just", named by the module that declared the union — and, for
641            // this module's own unions only, under the bare name too.
642            let Some(qname) = ctor.name.qualify_with_name(&type_name.module_name()) else {
643                continue;
644            };
645            if module.name.qualify_name(&type_name.unqualified_name()) == *type_name {
646                global.insert(ctor.name.as_str().to_string(), ctor_type.clone());
647            }
648            global.insert(qname.to_name().as_str().to_string(), ctor_type);
649        }
650    }
651
652    // Third pass: check each value. A value that fails is recorded and the pass
653    // moves on, so one broken declaration cannot hide the others.
654    let mut errors: Vec<Error> = vec![];
655    let mut solved: HashMap<Name, Solved> = HashMap::new();
656    for (name, value) in &module.values {
657        let Some((term, annotation)) =
658            value_to_term_and_annotation(value, &translation, &mut counter)
659        else {
660            // Unsupported construct: nothing was checked, and the entry says so
661            // rather than the declaration going missing — see [`Solved`].
662            solved.insert(name.clone(), Solved::Untranslatable { span: value.span() });
663            continue;
664        };
665
666        // The annotation is handed to inference rather than checked against its
667        // result afterwards, which is what lets a mismatch *inside* the body know
668        // that the type it failed against came from the annotation. Checking the two
669        // whole types at the end could only ever say "this declaration is `Int` and
670        // its body is something else", with the caret across the lot.
671        match infer_annotated(term, global.clone(), annotation) {
672            // An unbound variable here is a hole in the typer's environment, not a
673            // mistake in the source — see [`Solved::UnboundName`].
674            Err(ErrorKind::UnboundVariable {
675                name: unbound,
676                span,
677            }) => {
678                solved.insert(
679                    name.clone(),
680                    Solved::UnboundName {
681                        name: unbound,
682                        span,
683                    },
684                );
685            }
686            Err(kind) => errors.push(Error {
687                kind,
688                span: value.span(),
689                declaration: name.clone(),
690            }),
691            Ok(term) => {
692                solved.insert(name.clone(), Solved::Typed(term));
693            }
694        }
695    }
696
697    if errors.is_empty() {
698        Ok(solved)
699    } else {
700        Err(errors)
701    }
702}

706/// Union declarations, keyed by the qualified name of each declaration.
707///
708/// Keyed that way rather than by the spelling, so that a lookup answers "is this the
709/// declaration that module named?" and not "is something spelled like that in reach?".
710/// The two differ for every imported constructor whose type shares a name with a local
711/// one, which is what `BUG-35` closed.
712type Unions<'a> = HashMap<QualName, &'a canonical::UnionType>;
713
714/// Everything the translation from the canonical AST reads besides the expression in
715/// front of it.
716///
717/// Each of the three answers a question the canonical node cannot: which union a
718/// constructor belongs to and where in it, and how many arguments a call has to supply
719/// before it is a direct call. All three are facts of the module and of what it
720/// imports, which is why they are gathered once here — the declaration being translated
721/// is the only thing that changes between calls.
722struct Translation<'a> {
723    /// Every union in reach — this module's own, and every one an imported
724    /// [`Interface`] exposes — keyed by the qualified name that identifies each
725    /// declaration rather than by the spelling the `type` line wrote. A constructor
726    /// carries the qualified name of the type it builds, so this key is what finds the
727    /// declaration a constructor in a pattern belongs to — see [`translate_pattern`].
728    ///
729    /// An interface's union is only as complete as its module exposed it: an opaque
730    /// one crosses with no constructors, which is also why no canonical constructor
731    /// can name one.
732    unions: Unions<'a>,
733    /// The constructors of every union in [`unions`](Self::unions), keyed the way a
734    /// `VarConstructor` spells one, each with the place in its declaration both
735    /// backends need.
736    constructors: HashMap<QualName, Constructor>,
737    /// How many parameters each of this module's declarations was written with.
738    ///
739    /// This is the callee's arity at a call site naming one of them, which is what
740    /// decides an application's [`Saturation`]. The rule itself — parameter count, not
741    /// arrow count — is [`canonical::Value::arity`], which `ir::build` reads too.
742    arities: HashMap<Name, usize>,
743}
744
745impl<'a> Translation<'a> {
746    /// The translation for `module`, checked against `interfaces`.
747    ///
748    /// The interfaces' unions go in first and this module's own after them, so that if
749    /// the map somehow held an interface of the module under check, the declarations
750    /// in front of the typer are the ones that win.
751    fn of(module: &'a Module, interfaces: &'a HashMap<Name, Interface>) -> Translation<'a> {
752        let mut unions: Unions<'a> = HashMap::new();
753
754        for interface in interfaces.values() {
755            for (name, union_type) in &interface.unions {
756                unions.insert(interface.module_name.qualify_name(name), union_type);
757            }
758        }
759
760        for (name, union_type) in &module.types {
761            unions.insert(module.name.qualify_name(name), union_type);
762        }
763
764        let constructors = constructors_of(&unions);
765
766        let arities = module
767            .values
768            .iter()
769            .map(|(name, value)| (name.clone(), value.arity()))
770            .collect();
771
772        Translation {
773            unions,
774            constructors,
775            arities,
776        }
777    }
778
779    /// A translation that knows these unions and nothing else: no arity is known, so
780    /// every application it produces is [`Saturation::Partial`].
781    #[cfg(test)]
782    fn of_types(unions: Unions<'a>) -> Translation<'a> {
783        let constructors = constructors_of(&unions);
784
785        Translation {
786            unions,
787            constructors,
788            arities: HashMap::new(),
789        }
790    }
791
792    /// How many arguments the callee of an application spine takes, when this module
793    /// knows.
794    ///
795    /// `None` is what makes an application [`Saturation::Partial`], and it is the honest
796    /// answer three times over: a local is a value rather than a declaration and has no
797    /// arity at all; an imported value's arity belongs to the module that declared it
798    /// and is not in the canonical AST here; and a callee that is itself an expression —
799    /// the result of a `case`, say — is a value too. A backend that cannot prove a call
800    /// saturated goes through `$curry`, which is correct for every one of them.
801    fn callee_arity(&self, callee: &canonical::Expression) -> Option<usize> {
802        match &callee.kind {
803            canonical::ExpressionKind::VarTopLevel(qname) => {
804                self.arities.get(&qname.unqualified_name()).copied()
805            }
806            canonical::ExpressionKind::VarConstructor(qname, _) => {
807                self.constructors.get(qname).map(|ctor| ctor.arity)
808            }
809            _ => None,
810        }
811    }
812}
813
814/// Every constructor of `unions`, keyed the way a `VarConstructor` spells one: the
815/// constructor's name qualified by the module that declared the union it builds.
816fn constructors_of(unions: &Unions) -> HashMap<QualName, Constructor> {
817    let mut constructors = HashMap::new();
818
819    for (union, union_type) in unions {
820        for variant in crate::compiler::ir::variants_of(union_type) {
821            // A union is named by its declaring module, so that module is where its
822            // constructors are named from too. `qualify_with_name` only declines an
823            // empty name, which a parsed declaration never has.
824            let Some(name) = variant.name.qualify_with_name(&union.module_name()) else {
825                continue;
826            };
827
828            constructors.insert(
829                name,
830                Constructor {
831                    union: union.clone(),
832                    name: variant.name,
833                    index: variant.index,
834                    arity: variant.arity,
835                },
836            );
837        }
838    }
839
840    constructors
841}

843/// Convert a canonical type to the typer's simplified Type representation.
844/// The match covers all four `canonical::Type` variants — `Variable`, `Arrow`, `Tuple`
845/// (either arity), and `Type` including named types with parameters — and every arm's
846/// own base case returns `Some`; a `None` only ever arises by propagating up from a
847/// nested recursive call. As of today no `canonical::Type` shape actually reaches such
848/// a case, so the function always returns `Some`. The `Option` return stays in place for
849/// when a genuinely unrepresentable variant (e.g. records, aliases) is added.
850///
851/// `var_map` maps named type variables (e.g. "a") to consistent TypeVariable
852/// ids, so that `a -> a` produces the same variable on both sides.
853///
854/// # How a name decides which type it becomes
855///
856/// A [`canonical::Type::Type`] names its declaration in full — `Widget.Size`, not
857/// `Size` (`AST-4`) — and the whole of that name is what picks the arm. A nullary
858/// type whose qualified name is [a scalar the compiler
859/// knows](super::scalars) becomes the matching [`Type::Literal`]; everything else
860/// becomes a [`Type::Adt`], carrying that name whole.
861///
862/// For a name that resolved, that is the declaring module's. For one that resolved to
863/// nothing it is the module under check, applied to the whole *written* spelling, dots
864/// and all, because `Type::from_parser_type` hands the undivided name to
865/// `ModuleName::qualify_name` rather than splitting a module half off it — so an
866/// unresolved `Missing.Thing` written in `Test` arrives here as `Test.Missing.Thing`
867/// and does not collapse onto the local `Test.Thing`.
868pub(crate) fn canonical_type_to_typer_type(
869    tpe: &canonical::Type,
870    var_map: &mut HashMap<String, TypeVariable>,
871    counter: &mut u32,
872) -> Option<Type> {
873    match tpe {
874        canonical::Type::Variable(name) => {
875            let tv = var_map.entry(name.as_str().to_string()).or_insert_with(|| {
876                *counter += 1;
877                TypeVariable { id: *counter }
878            });
879            Some(Type::Variable(tv.clone()))
880        }
881        canonical::Type::Arrow(a, b) => {
882            let a = canonical_type_to_typer_type(a, var_map, counter)?;
883            let b = canonical_type_to_typer_type(b, var_map, counter)?;
884            Some(Type::Fun {
885                param_tpe: Box::new(a),
886                return_tpe: Box::new(b),
887            })
888        }
889        // `Tuple::try_map` keeps the arity attached to the value instead of
890        // re-deriving it here, the same way the parser → canonical conversions
891        // in `canonical/mod.rs` do.
892        canonical::Type::Tuple(tuple) => {
893            let elements = tuple
894                .try_map(|elem| canonical_type_to_typer_type(elem, var_map, counter).ok_or(()))
895                .ok()?;
896            Some(Type::Tuple(elements))
897        }
898        canonical::Type::Type(name, args) => {
899            if args.is_empty() {
900                if let Some(literal) = scalar_literal(name) {
901                    return Some(Type::Literal(literal));
902                }
903            }
904
905            let converted: Option<Vec<Type>> = args
906                .iter()
907                .map(|a| canonical_type_to_typer_type(a, var_map, counter))
908                .collect();
909            Some(Type::Adt(name.clone(), converted?))
910        }
911    }
912}

914/// The literal type the typer gives a [scalar](super::scalars), if `name` is the
915/// qualified name of one it has a literal type for.
916///
917/// Three of the five appear here. The other two do not:
918///
919/// - [`scalars::BOOL`] is a scalar *and* an ordinary union, so `Bool` in an annotation
920///   takes the [`Type::Adt`] path every other declaration takes and meets `True` and
921///   `False` there. [`bool_type`] is the same type, built for the three `Bool`s no
922///   source spells.
923/// - [`scalars::STRING`] would gain a [`TypeLiteral`] variant the day there is a string
924///   literal to give a type to; a variant nothing constructs would be a type the
925///   unifier could name in an error and no source could produce.
926fn scalar_literal(name: &QualName) -> Option<TypeLiteral> {
927    const LITERALS: &[(scalars::Scalar, TypeLiteral)] = &[
928        (scalars::INT, TypeLiteral::Int),
929        (scalars::FLOAT, TypeLiteral::Float),
930        (scalars::CHAR, TypeLiteral::Char),
931    ];
932
933    LITERALS
934        .iter()
935        .find(|(scalar, _)| scalar.declares(name))
936        .map(|(_, literal)| literal.clone())
937}

939/// The type of a `Bool`: the union [`scalars::BOOL`] names, with no arguments.
940///
941/// `Bool` is [a scalar and an ordinary union at
942/// once](../../../docs/spec/types.md#scalar-types) — the compiler knows its
943/// representation and nothing about its structure — so this is the very type
944/// `canonical_type_to_typer_type` produces for an annotation naming `Basics.Bool`, and
945/// the type `Basics` registers `True` and `False` at.
946///
947/// Three constructs need a `Bool` the source did not spell: an [`if`
948/// condition](../../../docs/spec/expressions.md#if--then--else), the `true`/`false`
949/// keywords, and a `true`/`false` pattern. They name `Basics.Bool` and nothing else, so
950/// a module declaring its own `type Bool` does not satisfy them
951/// ([`DEC-15`](../../../docs/decisions/dec-15.md) decisions 1 and 5).
952pub(super) fn bool_type() -> Type {
953    Type::Adt(scalars::BOOL.qual_name(), vec![])
954}

956/// Convert a canonical expression to a Term, keeping the position it was written at.
957///
958/// Returns None for constructs the inference engine doesn't yet handle (a `VarKernel`
959/// reference, complex patterns inside a `Case`), and for a constructor of a union
960/// neither this module nor an interface in [`Translation`] declares.
961///
962/// Every arm attaches `expr.span` to the term it builds. That is the whole of what
963/// `ERR-4` needed from this function: a constraint can only point at a
964/// sub-expression if the term that produced it remembers where it came from.
965///
966/// # What is recorded here and nowhere else
967///
968/// This is the only moment at which a reference's *kind* is known. A local, a
969/// top-level, an imported value and a constructor are four different things to emit,
970/// and the spelling each leaves behind is bare for one of them and qualified for the
971/// rest — so a backend handed only the string could not tell them apart
972/// ([`DEC-18` decision
973/// 1](../../../docs/decisions/dec-18.md#1--the-backend-reads-a-typed-ir-and-the-typer-is-what-produces-it)).
974/// Each becomes a [`Reference`] carrying both the lookup key inference uses and what
975/// the name is. An application spine records the same way whether it is
976/// [saturated](Saturation), which is a question about the spine and not about any one
977/// `Apply` node.
978fn canonical_expr_to_term(
979    expr: &canonical::Expression,
980    translation: &Translation,
981    counter: &mut u32,
982) -> Option<Term> {
983    let kind = match &expr.kind {
984        // Carried at the canonical AST's own width: [`Int` is 64
985        // bits](../../../docs/spec/evaluation-semantics.md#numbers), and a term is what
986        // code is generated from, so narrowing here would emit a different number than
987        // the one that was written.
988        canonical::ExpressionKind::Int(i) => TermKind::Int(*i),
989        canonical::ExpressionKind::Bool(b) => TermKind::Bool(*b),
990        canonical::ExpressionKind::Char(c) => TermKind::Char(*c),
991        canonical::ExpressionKind::Float(f) => TermKind::Float(*f),
992        canonical::ExpressionKind::VarLocal(name) => {
993            TermKind::Identifier(Reference::local(name.as_str()))
994        }
995        canonical::ExpressionKind::VarTopLevel(qname) => TermKind::Identifier(Reference {
996            name: qname.to_name().as_str().to_string(),
997            kind: ReferenceKind::TopLevel(qname.clone()),
998        }),
999        // A value another module declares. Its type is the one its module's interface
1000        // declared, which `type_check` registers under this same qualified name.
1001        canonical::ExpressionKind::VarForeign(qname, _) => TermKind::Identifier(Reference {
1002            name: qname.to_name().as_str().to_string(),
1003            kind: ReferenceKind::Foreign(qname.clone()),
1004        }),
1005        // A constructor builds a tagged value rather than reading a binding, so it
1006        // carries its place in its declaration. That place comes from the union,
1007        // which is in `translation.constructors` whether this module declared it or
1008        // an imported interface did. The type the canonical node carries is not read:
1009        // the one `type_check` registers is built from the union's own variables.
1010        canonical::ExpressionKind::VarConstructor(qname, _) => {
1011            let ctor = translation.constructors.get(qname)?;
1012
1013            TermKind::Identifier(Reference {
1014                name: qname.to_name().as_str().to_string(),
1015                kind: ReferenceKind::Constructor(ctor.clone()),
1016            })
1017        }
1018        // The spine is walked as a whole rather than one node at a time, because
1019        // whether a call is saturated is a fact about the callee and the number of
1020        // arguments reaching it. Each node keeps its own canonical span, so a type
1021        // error still points where it did.
1022        canonical::ExpressionKind::Apply(_, _) => {
1023            let (callee, applications) = spine(expr);
1024            let arity = translation.callee_arity(callee);
1025
1026            let mut term = canonical_expr_to_term(callee, translation, counter)?;
1027
1028            for (supplied, application) in applications.iter().enumerate() {
1029                // `spine` collects `Apply` nodes and nothing else.
1030                let canonical::ExpressionKind::Apply(_, argument) = &application.kind else {
1031                    return None;
1032                };
1033
1034                let argument = canonical_expr_to_term(argument, translation, counter)?;
1035                // The node that consumes the callee's last argument is the direct
1036                // call; everything before it is a partial application and everything
1037                // after it applies whatever that call returned.
1038                let saturation = match arity {
1039                    Some(arity) if arity == supplied + 1 => Saturation::Saturated,
1040                    _ => Saturation::Partial,
1041                };
1042
1043                term = Term {
1044                    span: application.span,
1045                    kind: TermKind::Apply {
1046                        fun: Box::new(term),
1047                        arg: Box::new(argument),
1048                        saturation,
1049                    },
1050                };
1051            }
1052
1053            return Some(term);
1054        }
1055        canonical::ExpressionKind::If(cond, t, f) => {
1056            let cond = canonical_expr_to_term(cond, translation, counter)?;
1057            let t = canonical_expr_to_term(t, translation, counter)?;
1058            let f = canonical_expr_to_term(f, translation, counter)?;
1059            TermKind::If {
1060                cond: Box::new(cond),
1061                true_branch: Box::new(t),
1062                false_branch: Box::new(f),
1063            }
1064        }
1065        canonical::ExpressionKind::Tuple(tuple) => {
1066            let elements = tuple
1067                .try_map(|elem| canonical_expr_to_term(elem, translation, counter).ok_or(()))
1068                .ok()?;
1069            TermKind::Tuple(elements)
1070        }
1071        canonical::ExpressionKind::Case(scrutinee_expr, branches) => {
1072            let scrutinee = canonical_expr_to_term(scrutinee_expr, translation, counter)?;
1073            let term_branches: Vec<(TermPattern, Box<Term>)> = branches
1074                .iter()
1075                .map(|cb| {
1076                    let pattern = translate_pattern(&cb.pattern, translation, counter)?;
1077                    let body = canonical_expr_to_term(&cb.expression, translation, counter)?;
1078                    Some((pattern, Box::new(body)))
1079                })
1080                .collect::<Option<Vec<_>>>()?;
1081            TermKind::Case {
1082                scrutinee: Box::new(scrutinee),
1083                branches: term_branches,
1084                form: CaseForm::Expression,
1085            }
1086        }
1087        // Not yet supported: VarKernel
1088        _ => return None,
1089    };
1090
1091    Some(Term {
1092        span: expr.span,
1093        kind,
1094    })
1095}

1097/// An application spine: what is being applied, and the `Apply` nodes that apply it,
1098/// innermost first.
1099///
1100/// `f a b` is `Apply(Apply(f, a), b)` in the canonical AST, and one `Apply` node on its
1101/// own cannot say whether the call it is part of supplies everything `f` takes. This is
1102/// what turns the nesting back into a callee and a count. `applications[i]` supplies the
1103/// `i + 1`th argument, and the last of them is `expr` itself.
1104fn spine(expr: &canonical::Expression) -> (&canonical::Expression, Vec<&canonical::Expression>) {
1105    let mut applications = Vec::new();
1106    let mut callee = expr;
1107
1108    while let canonical::ExpressionKind::Apply(fun, _) = &callee.kind {
1109        applications.push(callee);
1110        callee = fun;
1111    }
1112
1113    applications.reverse();
1114
1115    (callee, applications)
1116}

1118/// Translate a canonical pattern into a `TermPattern`. Returns `None` for unsupported
1119/// pattern shapes.
1120///
1121/// The pattern keeps its own span, separate from the branch body's: a `case` branch
1122/// whose pattern does not match what is being matched on is about the pattern, and
1123/// the caret belongs there rather than under the expression in the `case … of` line.
1124fn translate_pattern(
1125    pattern: &canonical::Pattern,
1126    translation: &Translation,
1127    counter: &mut u32,
1128) -> Option<TermPattern> {
1129    let kind = match &pattern.kind {
1130        canonical::PatternKind::Anything => TermPatternKind::Anything,
1131        canonical::PatternKind::Variable(name) => {
1132            // The binding's actual type will be unified with the scrutinee type in annotate.
1133            TermPatternKind::Bind(name.as_str().to_string())
1134        }
1135        canonical::PatternKind::Bool(value) => TermPatternKind::Literal {
1136            tpe: bool_type(),
1137            value: LiteralValue::Bool(*value),
1138        },
1139        canonical::PatternKind::Int(value) => TermPatternKind::Literal {
1140            tpe: Type::Literal(TypeLiteral::Int),
1141            value: LiteralValue::Int(*value),
1142        },
1143        canonical::PatternKind::Char(value) => TermPatternKind::Literal {
1144            tpe: Type::Literal(TypeLiteral::Char),
1145            value: LiteralValue::Char(*value),
1146        },
1147        // `Basics`' own `True` and `False` are tested by value, exactly as `true` and
1148        // `false` are, so a backend meets one vocabulary for a `Bool` whichever
1149        // spelling the source used — a `case` mixing them included. The type is the
1150        // same one the constructor would have constrained the scrutinee to. A module's
1151        // own `type Bool = True | False` is not `Basics.Bool` and is left alone.
1152        canonical::PatternKind::Constructor { ctor, args }
1153            if args.is_empty() && ctor.tpe == scalars::BOOL.qual_name() =>
1154        {
1155            let value = match ctor.name.as_str() {
1156                "True" => true,
1157                "False" => false,
1158                _ => return None,
1159            };
1160            TermPatternKind::Literal {
1161                tpe: bool_type(),
1162                value: LiteralValue::Bool(value),
1163            }
1164        }
1165        canonical::PatternKind::Constructor { ctor, args } => {
1166            // Look up the parent union to get its type variables. `ctor.tpe` names
1167            // the declaration the constructor builds, module included, and `unions`
1168            // is keyed the same way — so this finds the one declaration the
1169            // constructor belongs to, this module's or an imported one.
1170            let union_type = translation.unions.get(&ctor.tpe)?;
1171
1172            // Which case of that union this pattern matches. Unification needs only
1173            // the union; a decision tree and a WIT `variant` both need the case, and
1174            // the declaration is the only thing that can say where it sits.
1175            let index = union_type
1176                .variants
1177                .iter()
1178                .position(|variant| variant.name == ctor.name)?;
1179
1180            // Create fresh type vars for each ADT type parameter.
1181            let mut adt_var_map: HashMap<String, TypeVariable> = HashMap::new();
1182            for tv_name in &union_type.variables {
1183                *counter += 1;
1184                adt_var_map.insert(tv_name.as_str().to_string(), TypeVariable { id: *counter });
1185            }
1186
1187            // Build the ADT result type args from the fresh vars.
1188            let adt_args: Vec<Type> = union_type
1189                .variables
1190                .iter()
1191                .map(|v| Type::Variable(adt_var_map[v.as_str()].clone()))
1192                .collect();
1193
1194            // Translate each constructor type parameter (reuses the same fresh vars).
1195            let param_types: Vec<Type> = ctor
1196                .type_parameters
1197                .iter()
1198                .filter_map(|t| canonical_type_to_typer_type(t, &mut adt_var_map, counter))
1199                .collect();
1200            if param_types.len() != ctor.type_parameters.len() {
1201                return None;
1202            }
1203
1204            // One sub-pattern per argument, so an argument's position is its place in
1205            // the list.
1206            let args = args
1207                .iter()
1208                .zip(param_types)
1209                .map(|(arg_pattern, param_type)| {
1210                    translate_sub_pattern(arg_pattern, param_type, translation, counter)
1211                })
1212                .collect::<Option<Vec<_>>>()?;
1213
1214            TermPatternKind::Constructor {
1215                ctor: Constructor {
1216                    union: ctor.tpe.clone(),
1217                    name: ctor.name.clone(),
1218                    index,
1219                    arity: ctor.type_parameters.len(),
1220                },
1221                adt_args,
1222                args,
1223            }
1224        }
1225        // Each element gets a fresh type, and the matched value has to be the tuple of
1226        // them. An element is held to the same limit a constructor's argument is (see
1227        // `translate_sub_pattern`). A tuple nested in a tuple is the stand-in for an
1228        // untranslatable declaration in `tests/typer.rs`, `tests/ir.rs` and
1229        // `tests/javascript.rs` — the only nested shape that parses today — so lifting
1230        // this limit turns all three red, and each needs a new stand-in.
1231        canonical::PatternKind::Tuple(elements) => {
1232            let elements = elements
1233                .try_map(|element| {
1234                    *counter += 1;
1235                    let tpe = Type::Variable(TypeVariable { id: *counter });
1236                    translate_sub_pattern(element, tpe, translation, counter).ok_or(())
1237                })
1238                .ok()?;
1239
1240            TermPatternKind::Tuple { elements }
1241        }
1242        _ => return None, // Float patterns — not yet supported
1243    };
1244
1245    Some(TermPattern {
1246        span: pattern.span,
1247        kind,
1248    })
1249}

1322/// Wrap a body Term in nested Fun nodes for each parameter, outermost first.
1323///
1324/// A `Fun` binds one name, so a parameter written as a variable or `_` is bound as it
1325/// is. A parameter written as any other pattern is bound under the name
1326/// [`pattern_parameter`] gives its position, and matched by a single-branch
1327/// [`CaseForm::Parameter`] `Case` on that name: `first (x, _) = x` is translated as
1328/// `first $0 = case $0 of (x, _) -> x` would be. The pattern goes through
1329/// [`translate_pattern`], the same translation a `case` branch's gets, so a parameter
1330/// admits exactly the patterns a branch does. Returns `None` when it does not.
1331///
1332/// Every `Fun` is outside every `Case`. That keeps the first `arity` nodes of the term
1333/// the declaration's parameters and nothing else, which is what `ir::build` reads them
1334/// off by. The `Case`s nest in parameter order, the first outermost, so a name two
1335/// patterned parameters both bind is the later one's in the body — which is how
1336/// canonicalization resolved it. A pattern's names also shadow a same-named parameter
1337/// written as a variable, whichever comes first — and when the pattern comes first,
1338/// that is the opposite of canonicalization, which lets the later parameter win:
1339/// `f (x, _) x = x` reads the pattern's `x` here and the plain parameter's there. Both
1340/// are a name bound twice in one clause, which the language rejects and the compiler
1341/// does not yet (`LANG-18`).
1342///
1343/// Each `Fun` spans its parameter through the body it wraps, so a function whose
1344/// declared shape does not match its definition is underlined from the parameter
1345/// that starts it rather than across the annotation as well. A `Case` built here spans
1346/// its pattern alone: it is what the source wrote in place of a plain parameter.
1347fn wrap_with_patterns<'a>(
1348    patterns: impl Iterator<Item = &'a canonical::Pattern>,
1349    body: Term,
1350    translation: &Translation,
1351    counter: &mut u32,
1352) -> Option<Term> {
1353    let body_span = body.span;
1354    let mut params: Vec<(String, NodeSpan)> = vec![];
1355    let mut matched: Vec<(String, TermPattern)> = vec![];
1356
1357    for (position, pattern) in patterns.enumerate() {
1358        match &pattern.kind {
1359            canonical::PatternKind::Variable(name) => {
1360                params.push((name.as_str().to_string(), pattern.span))
1361            }
1362            canonical::PatternKind::Anything => params.push(("_".to_string(), pattern.span)),
1363            _ => {
1364                let name = pattern_parameter(position);
1365                let translated = translate_pattern(pattern, translation, counter)?;
1366                params.push((name.clone(), pattern.span));
1367                matched.push((name, translated));
1368            }
1369        }
1370    }
1371
1372    let body = matched
1373        .into_iter()
1374        .rev()
1375        .fold(body, |acc, (name, pattern)| {
1376            let span = pattern.span;
1377            Term {
1378                span,
1379                kind: TermKind::Case {
1380                    scrutinee: Box::new(Term {
1381                        span,
1382                        kind: TermKind::Identifier(Reference::local(name)),
1383                    }),
1384                    branches: vec![(pattern, Box::new(acc))],
1385                    form: CaseForm::Parameter,
1386                },
1387            }
1388        });
1389
1390    let term = params
1391        .into_iter()
1392        .rev()
1393        .fold(body, |acc, (param, span)| Term {
1394            span: span.merge(body_span),
1395            kind: TermKind::Fun {
1396                param,
1397                body: Box::new(acc),
1398            },
1399        });
1400
1401    Some(term)
1402}

1466/// How the name of a [`Type::Adt`] is written out.
1467///
1468/// A type is normally quoted the way the source spells it, which for a union is its
1469/// bare name. That is ambiguous exactly when one message names two declarations that
1470/// share a spelling, and [`ErrorKind::message`] switches to the qualified form there.
1471#[derive(Debug, Clone, Copy, PartialEq, Eq)]
1472enum AdtNames {
1473    /// `Size`.
1474    Unqualified,
1475    /// `Widget.Size`.
1476    Qualified,
1477}
1478
1479impl AdtNames {
1480    /// How the types a single message is about have to be written for that message
1481    /// to distinguish the declarations it names.
1482    ///
1483    /// The question is not whether the *types* are spelled alike — `A.Size` against
1484    /// `Lib.Size A.Size` renders as `Size` against `Size Size`, which differs as
1485    /// text while still using one word for three declarations. It is whether any two
1486    /// of the unions named anywhere in those types are different declarations with
1487    /// the same bare name; if so every union in the sentence is qualified, since
1488    /// qualifying only the colliding pair would read as if the rest had no module.
1489    fn for_all<'a>(types: impl IntoIterator<Item = &'a Type>) -> AdtNames {
1490        let mut names: Vec<&QualName> = Vec::new();
1491
1492        for tpe in types {
1493            tpe.collect_adt_names(&mut names);
1494        }
1495
1496        let collides = names.iter().enumerate().any(|(i, name)| {
1497            names[i + 1..]
1498                .iter()
1499                .any(|other| *other != *name && other.unqualified_name() == name.unqualified_name())
1500        });
1501
1502        if collides {
1503            AdtNames::Qualified
1504        } else {
1505            AdtNames::Unqualified
1506        }
1507    }
1508}

1552impl Type {
1553    /// Every union named anywhere in this type, outermost first, appended to `out`.
1554    ///
1555    /// A union's arguments are types in their own right and may name unions of their
1556    /// own, so this recurses rather than reading the head alone. [`AdtNames::for_all`]
1557    /// is what it exists for: deciding how to write a type means looking at every
1558    /// name the rendering will contain, not only the one at the top.
1559    fn collect_adt_names<'a>(&'a self, out: &mut Vec<&'a QualName>) {
1560        match self {
1561            Type::Literal(_) | Type::Number | Type::Variable(_) => {}
1562            Type::Fun {
1563                param_tpe,
1564                return_tpe,
1565            } => {
1566                param_tpe.collect_adt_names(out);
1567                return_tpe.collect_adt_names(out);
1568            }
1569            Type::Tuple(Tuple::Two(a, b)) => {
1570                a.collect_adt_names(out);
1571                b.collect_adt_names(out);
1572            }
1573            Type::Tuple(Tuple::Three(a, b, c)) => {
1574                a.collect_adt_names(out);
1575                b.collect_adt_names(out);
1576                c.collect_adt_names(out);
1577            }
1578            Type::Adt(name, args) => {
1579                out.push(name);
1580                for arg in args {
1581                    arg.collect_adt_names(out);
1582                }
1583            }
1584        }
1585    }
1586
1587    /// The body of both renderings: the same text either way, except for how a union
1588    /// is named. See [`AdtNames`], and [`Display`](std::fmt::Display) for why unions
1589    /// are normally written bare.
1590    fn write(&self, f: &mut std::fmt::Formatter<'_>, names: AdtNames) -> std::fmt::Result {
1591        /// The same, wrapped in parentheses.
1592        fn parenthesised(
1593            tpe: &Type,
1594            f: &mut std::fmt::Formatter<'_>,
1595            names: AdtNames,
1596        ) -> std::fmt::Result {
1597            write!(f, "(")?;
1598            tpe.write(f, names)?;
1599            write!(f, ")")
1600        }
1601
1602        match self {
1603            Type::Literal(TypeLiteral::Int) => write!(f, "Int"),
1604            Type::Literal(TypeLiteral::Char) => write!(f, "Char"),
1605            Type::Literal(TypeLiteral::Float) => write!(f, "Float"),
1606            Type::Number => write!(f, "number"),
1607            Type::Variable(TypeVariable { id }) => write!(f, "t{}", id),
1608            // The parameter of a function type is parenthesised when it is itself a
1609            // function, because `->` is right-associative: `(a -> b) -> c` and
1610            // `a -> b -> c` are different types.
1611            Type::Fun {
1612                param_tpe,
1613                return_tpe,
1614            } => {
1615                match **param_tpe {
1616                    Type::Fun { .. } => parenthesised(param_tpe, f, names)?,
1617                    _ => param_tpe.write(f, names)?,
1618                }
1619                write!(f, " -> ")?;
1620                return_tpe.write(f, names)
1621            }
1622            Type::Tuple(Tuple::Two(a, b)) => {
1623                write!(f, "( ")?;
1624                a.write(f, names)?;
1625                write!(f, ", ")?;
1626                b.write(f, names)?;
1627                write!(f, " )")
1628            }
1629            Type::Tuple(Tuple::Three(a, b, c)) => {
1630                write!(f, "( ")?;
1631                a.write(f, names)?;
1632                write!(f, ", ")?;
1633                b.write(f, names)?;
1634                write!(f, ", ")?;
1635                c.write(f, names)?;
1636                write!(f, " )")
1637            }
1638            Type::Adt(name, args) => {
1639                match names {
1640                    AdtNames::Unqualified => write!(f, "{}", name.unqualified_name())?,
1641                    AdtNames::Qualified => write!(f, "{}", name.to_name())?,
1642                }
1643                for arg in args {
1644                    // Same reason as above: an argument that is itself applied or a
1645                    // function needs parentheses to stay the same type when re-read.
1646                    write!(f, " ")?;
1647                    match arg {
1648                        Type::Adt(_, inner) if !inner.is_empty() => parenthesised(arg, f, names)?,
1649                        Type::Fun { .. } => parenthesised(arg, f, names)?,
1650                        _ => arg.write(f, names)?,
1651                    }
1652                }
1653                Ok(())
1654            }
1655        }
1656    }
1657}

1659/// Two types that have to match, and why.
1660///
1661/// # What the two sides mean
1662///
1663/// Unification treats them symmetrically; the order is a rendering convention, and
1664/// the only thing that reads it is the headline — *cannot match `left` with `right`*.
1665/// `collect` therefore writes the declared or expected side first where the source
1666/// has one (an annotation's type, a pattern's type) and the inferred side second, so
1667/// the sentence comes out in the order a reader expects.
1668///
1669/// What the sides are emphatically *not* is a way to tell which type belongs to the
1670/// text at `origin.span`. By the time a constraint fails, `unify` has substituted
1671/// other constraints' solutions into both sides, and may have decomposed it into a
1672/// component of the types the source mentioned. That is why the labels name the
1673/// [`Reason`] and leave the types to the headline — see [`Reason::describes`].
1674///
1675/// # Why these are held in a `Vec` and not a `HashSet`
1676///
1677/// They used to be a `HashSet<Constraint>`, back when a constraint was a bare pair of
1678/// types. An origin makes that collection wrong twice over. Deduplication now
1679/// discards *provenance*: two constraints with equal types but different origins are
1680/// one entry, and which origin survives is whichever was inserted first. And the
1681/// order a `HashSet` yields is unspecified, so which constraint `unify` reaches first
1682/// — and therefore which one is reported when several are unsatisfiable — would vary
1683/// between runs of the same compiler on the same file.
1684///
1685/// A `Vec` fixes both, and buys a third thing: source order. The annotation is
1686/// pushed first, so its type is substituted into the body's constraints before they
1687/// are solved, which is what lets a mismatch deep in the body say that `Int` came
1688/// from the annotation. The cost is that duplicate constraints are no longer
1689/// collapsed, which is a few more `unify` steps on terms that repeat a type.
1690#[derive(Debug, Clone, PartialEq)]
1691struct Constraint {
1692    left: Type,
1693    right: Type,
1694    origin: Origin,
1695}
1696
1697impl Constraint {
1698    fn new(left: Type, right: Type, reason: Reason, span: NodeSpan) -> Constraint {
1699        Constraint {
1700            left,
1701            right,
1702            origin: Origin::new(reason, span),
1703        }
1704    }
1705
1706    /// A constraint between two components of this one — the parameters of two
1707    /// function types being matched, say — which is about the same source text and
1708    /// was required for the same reason.
1709    ///
1710    /// The origin is inherited whole, side provenance included: if a substitution
1711    /// rewrote the left type, it rewrote whatever the left type decomposes into. That
1712    /// is an over-approximation — a substitution that reached only the return half of
1713    /// an arrow is credited with the parameter half too — but it errs towards naming
1714    /// a constraint that did carry a type in, which is the direction that keeps a
1715    /// caret on the source rather than on the compiler's working.
1716    fn component(&self, left: Type, right: Type) -> Constraint {
1717        Constraint {
1718            left,
1719            right,
1720            origin: self.origin.clone(),
1721        }
1722    }
1723}

1725/// What one type variable was solved to, and where that type came from.
1726///
1727/// The cause is not used by inference at all. It is carried so that when this
1728/// solution is substituted into another constraint and *that* constraint then fails,
1729/// the failure can say where the type it failed against came from — see
1730/// [`Origin::left_from`].
1731///
1732/// It is a [`Cause`] and not an `Origin` because the question it answers is already
1733/// settled: [`Origin::cause_of`] resolved, at the moment the variable was solved,
1734/// whether the solving constraint introduced this type or was handed it. Nothing
1735/// downstream has to walk anything.
1736// No `Eq`: a `Cause` holds a `NodeSpan`, whose `PartialEq` is deliberately blind
1737// (see its documentation), so equality here is a claim about the types and the
1738// reasons, not about the positions.
1739#[derive(Debug, PartialEq, Clone)]
1740struct Solution {
1741    tpe: Type,
1742    cause: Cause,
1743}
1744
1745#[derive(Debug, PartialEq)]
1746struct Substitution {
1747    solutions: HashMap<TypeVariable, Solution>,
1748}
1749
1750impl Substitution {
1751    // constructors
1752
1753    fn empty() -> Substitution {
1754        Substitution {
1755            solutions: HashMap::new(),
1756        }
1757    }
1758
1759    fn one(tvar: TypeVariable, tpe: Type, cause: Cause) -> Substitution {
1760        let mut sub = Substitution::empty();
1761
1762        sub.solutions.insert(tvar, Solution { tpe, cause });
1763
1764        sub
1765    }

1767    // methods

1769    /// Rewrite a constraint with everything solved so far, recording for each side
1770    /// which solution first reached it.
1771    ///
1772    /// The two sides are tracked apart, because that is what tells a relayed type
1773    /// from an introduced one later on — see [`Origin::cause_of`]. A solution that
1774    /// rewrites only the right side has said nothing about the left, and treating it
1775    /// as though it had is what made a type error blame an annotation that was not
1776    /// load-bearing.
1777    ///
1778    /// The solutions are visited in type-variable order rather than in `HashMap`
1779    /// order: "first" has to mean the same thing on every run, or the secondary label
1780    /// of a diagnostic would move between compilations of an unchanged file. Ids are
1781    /// handed out as inference walks the term, so that order is roughly the order the
1782    /// user wrote things in.
1783    fn apply(&self, c: &Constraint) -> Constraint {
1784        let mut origin = c.origin.clone();
1785
1786        let mut solutions: Vec<_> = self.solutions.iter().collect();
1787        solutions.sort_by_key(|(tvar, _)| tvar.id);
1788
1789        for (tvar, solution) in solutions {
1790            // A solution that does not mention a variable a side uses rewrites
1791            // nothing there, and explains nothing about it either.
1792            if occurs(tvar, &c.left) {
1793                origin.rewritten(Side::Left, solution.cause);
1794            }
1795            if occurs(tvar, &c.right) {
1796                origin.rewritten(Side::Right, solution.cause);
1797            }
1798        }
1799
1800        Constraint {
1801            left: self.apply_type(&c.left),
1802            right: self.apply_type(&c.right),
1803            origin,
1804        }
1805    }
1806
1807    fn apply_type(&self, tpe: &Type) -> Type {
1808        self.solutions
1809            .iter()
1810            .fold(tpe.clone(), |tpe, (tvar, solution)| {
1811                Substitution::substitute(tpe, tvar, &solution.tpe)
1812            })
1813    }
1814
1815    /// Rewrite **every** type in a typed term with everything solved — the *zonk*.
1816    ///
1817    /// `annotate` gives each node a fresh inference variable and unification solves
1818    /// those variables somewhere else entirely, so until this runs a node's `tpe` is a
1819    /// `t17` that means nothing on its own. Applying the substitution to the root's
1820    /// type alone is enough to answer "what type is this declaration", which is all
1821    /// inference ever needed; it leaves every node below still holding a variable,
1822    /// which is not enough to generate code from
1823    /// ([`DEC-18` decision 1](../../../docs/decisions/dec-18.md)).
1824    ///
1825    /// The types a pattern carries are rewritten too: a constructor pattern's
1826    /// arguments and the bindings it introduces are the types the branch body's
1827    /// variables were bound at.
1828    fn apply_term(&self, term: TypedTerm) -> TypedTerm {
1829        let kind = match term.kind {
1830            kind @ (TypedTermKind::Int(_)
1831            | TypedTermKind::Bool(_)
1832            | TypedTermKind::Char(_)
1833            | TypedTermKind::Float(_)
1834            | TypedTermKind::Identifier(_)) => kind,
1835            TypedTermKind::Fun { param, body } => TypedTermKind::Fun {
1836                param: self.apply_binder(param),
1837                body: Box::new(self.apply_term(*body)),
1838            },
1839            TypedTermKind::Apply {
1840                fun,
1841                arg,
1842                saturation,
1843            } => TypedTermKind::Apply {
1844                fun: Box::new(self.apply_term(*fun)),
1845                arg: Box::new(self.apply_term(*arg)),
1846                saturation,
1847            },
1848            TypedTermKind::If {
1849                cond,
1850                true_branch,
1851                false_branch,
1852            } => TypedTermKind::If {
1853                cond: Box::new(self.apply_term(*cond)),
1854                true_branch: Box::new(self.apply_term(*true_branch)),
1855                false_branch: Box::new(self.apply_term(*false_branch)),
1856            },
1857            TypedTermKind::Let {
1858                binding,
1859                value,
1860                body,
1861            } => TypedTermKind::Let {
1862                binding: self.apply_binder(binding),
1863                value: Box::new(self.apply_term(*value)),
1864                body: Box::new(self.apply_term(*body)),
1865            },
1866            TypedTermKind::Tuple(Tuple::Two(a, b)) => {
1867                TypedTermKind::Tuple(Tuple::two(self.apply_term(*a), self.apply_term(*b)))
1868            }
1869            TypedTermKind::Tuple(Tuple::Three(a, b, c)) => TypedTermKind::Tuple(Tuple::three(
1870                self.apply_term(*a),
1871                self.apply_term(*b),
1872                self.apply_term(*c),
1873            )),
1874            TypedTermKind::Case {
1875                scrutinee,
1876                branches,
1877                form,
1878            } => TypedTermKind::Case {
1879                scrutinee: Box::new(self.apply_term(*scrutinee)),
1880                branches: branches
1881                    .into_iter()
1882                    .map(|(pattern, body)| {
1883                        (
1884                            self.apply_pattern(pattern),
1885                            Box::new(self.apply_term(*body)),
1886                        )
1887                    })
1888                    .collect(),
1889                form,
1890            },
1891        };
1892
1893        TypedTerm {
1894            span: term.span,
1895            tpe: self.apply_type(&term.tpe),
1896            kind,
1897        }
1898    }
1899
1900    fn apply_binder(&self, binder: TypeBinder) -> TypeBinder {
1901        TypeBinder {
1902            tpe: self.apply_type(&binder.tpe),
1903            name: binder.name,
1904        }
1905    }
1906
1907    fn apply_pattern(&self, pattern: TermPattern) -> TermPattern {
1908        let kind = match pattern.kind {
1909            kind @ (TermPatternKind::Anything | TermPatternKind::Bind(_)) => kind,
1910            TermPatternKind::Literal { tpe, value } => TermPatternKind::Literal {
1911                tpe: self.apply_type(&tpe),
1912                value,
1913            },
1914            TermPatternKind::Constructor {
1915                ctor,
1916                adt_args,
1917                args,
1918            } => TermPatternKind::Constructor {
1919                ctor,
1920                adt_args: adt_args.iter().map(|a| self.apply_type(a)).collect(),
1921                args: args
1922                    .into_iter()
1923                    .map(|arg| self.apply_sub_pattern(arg))
1924                    .collect(),
1925            },
1926            TermPatternKind::Tuple { elements } => TermPatternKind::Tuple {
1927                elements: elements.map(|element| self.apply_sub_pattern(element.clone())),
1928            },
1929        };
1930
1931        TermPattern {
1932            span: pattern.span,
1933            kind,
1934        }
1935    }
1936
1937    fn apply_sub_pattern(&self, sub: SubPattern) -> SubPattern {
1938        SubPattern {
1939            tpe: self.apply_type(&sub.tpe),
1940            pattern: self.apply_pattern(sub.pattern),
1941        }
1942    }
1943
1944    fn substitute(tpe: Type, tvar: &TypeVariable, replacement: &Type) -> Type {
1945        match tpe {
1946            Type::Literal(_) | Type::Number => tpe,
1947            Type::Fun {
1948                param_tpe,
1949                return_tpe,
1950            } => Type::Fun {
1951                param_tpe: Box::new(Substitution::substitute(*param_tpe, tvar, replacement)),
1952                return_tpe: Box::new(Substitution::substitute(*return_tpe, tvar, replacement)),
1953            },
1954            Type::Tuple(Tuple::Two(a, b)) => Type::Tuple(Tuple::two(
1955                Substitution::substitute(*a, tvar, replacement),
1956                Substitution::substitute(*b, tvar, replacement),
1957            )),
1958            Type::Tuple(Tuple::Three(a, b, c)) => Type::Tuple(Tuple::three(
1959                Substitution::substitute(*a, tvar, replacement),
1960                Substitution::substitute(*b, tvar, replacement),
1961                Substitution::substitute(*c, tvar, replacement),
1962            )),
1963            Type::Adt(name, args) => Type::Adt(
1964                name,
1965                args.into_iter()
1966                    .map(|a| Substitution::substitute(a, tvar, replacement))
1967                    .collect(),
1968            ),
1969            Type::Variable(tvar2) if tvar == &tvar2 => replacement.clone(),
1970            tpe @ Type::Variable(_) => tpe,
1971        }
1972    }
1973
1974    fn merge(&self, other: Substitution) -> Substitution {
1975        // This merge means we should try sub_tail first, and then sub_head
1976        // Merging other in self means we apply `other` substitution to `self` solutions
1977        // When merging, we want `other` solutions to take precedences over `self` solutions
1978
1979        let self_solutions = self.solutions.iter().map(|(k, v)| {
1980            (
1981                k.clone(),
1982                Solution {
1983                    tpe: other.apply_type(&v.tpe),
1984                    // The cause says where this variable's type came from, which
1985                    // rewriting the type does not change.
1986                    cause: v.cause,
1987                },
1988            )
1989        });
1990
1991        let mut sub = Substitution::empty();
1992
1993        sub.solutions.extend(self_solutions);
1994        sub.solutions.extend(other.solutions);
1995
1996        sub
1997    }
1998}

2000/// Does `tvar` appear anywhere inside `tpe`?
2001///
2002/// Two callers, for two different reasons: `unify_variable` uses it as the occurs
2003/// check that keeps it from building an infinite type, and `Substitution::apply` uses
2004/// it to tell whether a solution actually rewrites a given constraint — which is what
2005/// decides whether that solution explains one of the constraint's types.
2006fn occurs(tvar: &TypeVariable, tpe: &Type) -> bool {
2007    match tpe {
2008        Type::Fun {
2009            param_tpe,
2010            return_tpe,
2011        } => occurs(tvar, param_tpe) || occurs(tvar, return_tpe),
2012        Type::Tuple(tuple) => tuple.iter().any(|t| occurs(tvar, t)),
2013        Type::Adt(_, args) => args.iter().any(|a| occurs(tvar, a)),
2014        Type::Variable(tvar2) => tvar == tvar2,
2015        _ => false,
2016    }
2017}

2019/// The names in scope while one declaration is annotated, and the counter its fresh
2020/// type variables come from.
2021///
2022/// Two kinds of name, looked up differently:
2023///
2024/// - A **binder** — a parameter, a `case` binding — is introduced by the term being
2025///   annotated. Its type is one type: every use of it shares its variables, which is
2026///   what lets inference learn a parameter's type from how the body uses it.
2027/// - A **global** — a declaration of this module or of an imported one, or a
2028///   constructor — comes from `type_check`'s environment. Its type is read as quantified
2029///   over every variable in it, and [`by_name`](Self::by_name) hands each use a copy
2030///   with those variables replaced by fresh ones. That is what lets one declaration use
2031///   `Just` at `Maybe Int` and `Nothing` at `Maybe Char`, or `identity` at two types.
2032///
2033/// Reading every variable of a global as quantified is only right because each one was
2034/// translated from a declared type — an annotation or a union — on its own, so none of
2035/// its variables is shared with anything else in the environment. A binder shadows a
2036/// global of the same name.
2037struct Types {
2038    counter: u32,
2039    env: HashMap<String, Type>,
2040    globals: HashMap<String, Type>,
2041}
2042
2043impl Types {
2044    fn new() -> Types {
2045        let counter = 10;
2046        let env = HashMap::new();
2047        let globals = HashMap::new();
2048
2049        Types {
2050            counter,
2051            env,
2052            globals,
2053        }
2054    }
2055
2056    /// Put the declarations of an outer scope in reach, as globals.
2057    fn extends_with(&mut self, global: HashMap<String, Type>) {
2058        self.globals.extend(global)
2059    }
2060
2061    fn fresh_var(&mut self) -> Type {
2062        self.counter += 1;
2063
2064        Type::Variable(TypeVariable { id: self.counter })
2065    }
2066
2067    fn add_binder(&mut self, binding: TypeBinder) {
2068        self.env.insert(binding.name, binding.tpe);
2069    }
2070
2071    fn remove_binder(&mut self, name: &str) {
2072        self.env.remove(name);
2073    }
2074
2075    /// The type of one use of `name`: a binder's own type, or a fresh instance of a
2076    /// global's.
2077    fn by_name(&mut self, name: &String) -> Option<Type> {
2078        if let Some(tpe) = self.env.get(name) {
2079            return Some(tpe.clone());
2080        }
2081
2082        let scheme = self.globals.get(name)?.clone();
2083        let mut fresh = HashMap::new();
2084        Some(self.instantiate(scheme, &mut fresh))
2085    }
2086
2087    /// `tpe` with each of its variables replaced by a fresh one, the same variable by
2088    /// the same fresh one throughout.
2089    fn instantiate(&mut self, tpe: Type, fresh: &mut HashMap<TypeVariable, Type>) -> Type {
2090        match tpe {
2091            Type::Literal(_) | Type::Number => tpe,
2092            Type::Variable(tvar) => {
2093                if let Some(replacement) = fresh.get(&tvar) {
2094                    return replacement.clone();
2095                }
2096                let replacement = self.fresh_var();
2097                fresh.insert(tvar, replacement.clone());
2098                replacement
2099            }
2100            Type::Fun {
2101                param_tpe,
2102                return_tpe,
2103            } => Type::Fun {
2104                param_tpe: Box::new(self.instantiate(*param_tpe, fresh)),
2105                return_tpe: Box::new(self.instantiate(*return_tpe, fresh)),
2106            },
2107            Type::Tuple(Tuple::Two(a, b)) => Type::Tuple(Tuple::two(
2108                self.instantiate(*a, fresh),
2109                self.instantiate(*b, fresh),
2110            )),
2111            Type::Tuple(Tuple::Three(a, b, c)) => Type::Tuple(Tuple::three(
2112                self.instantiate(*a, fresh),
2113                self.instantiate(*b, fresh),
2114                self.instantiate(*c, fresh),
2115            )),
2116            Type::Adt(name, args) => Type::Adt(
2117                name,
2118                args.into_iter()
2119                    .map(|arg| self.instantiate(arg, fresh))
2120                    .collect(),
2121            ),
2122        }
2123    }

2126/// infer the type of the given term given known function defined in the outer scopes.
2127/// This is a translation of the algorithm demonstrated by
2128/// [Ionut Gan at I T.A.K.E Unconference 2015](https://www.youtube.com/watch?v=oPVTNxiMcSU)
2129pub fn infer(term: Term, global: HashMap<String, Type>) -> Result<Type, ErrorKind> {
2130    infer_annotated(term, global, None).map(|term| term.tpe)
2131}

2133/// [`infer`], with the declaration's type annotation as a constraint of its own, and
2134/// answering with the whole solved term rather than only its type.
2135///
2136/// The annotation is put *first*, before the constraints the body generates, and that
2137/// ordering is the point of the function. `unify` solves constraints in order, so the
2138/// annotated type is substituted into the body's constraints before any of them are
2139/// solved; when one of them then fails, its [`Origin::explanation`] names the annotation,
2140/// and the diagnostic can say `Int` was expected *because of the annotation* rather
2141/// than merely that the declaration as a whole does not check.
2142///
2143/// The term handed back is zonked — see [`Substitution::apply_term`] — so every node
2144/// carries the type inference solved for it and not the variable `annotate` gave it.
2145fn infer_annotated(
2146    term: Term,
2147    global: HashMap<String, Type>,
2148    annotation: Option<Annotation>,
2149) -> Result<TypedTerm, ErrorKind> {
2150    let mut env = Types::new();
2151    env.extends_with(global);
2152
2153    let typed_term = annotate::annotate(term, &mut env)?;
2154    debug!("typed term: {:#?}", typed_term);
2155
2156    let mut constraints = Vec::new();
2157
2158    if let Some(annotation) = annotation {
2159        // Left is the annotation's type, because left is the type of the text the
2160        // span points at — see `Constraint`.
2161        constraints.push(Constraint::new(
2162            annotation.tpe,
2163            typed_term.tpe.clone(),
2164            Reason::Annotation,
2165            annotation.span,
2166        ));
2167    }
2168
2169    constraints.extend(constraint::collect(&typed_term));
2170    debug!("Constraints: {:#?}", constraints);
2171
2172    let substitution = unifier::unify(constraints)?;
2173
2174    Ok(substitution.apply_term(typed_term))
2175}

2472    /// The four things `canonical::ExpressionKind` calls a name stay four things in the
2473    /// IR.
2474    ///
2475    /// They used to be one: every one of them became `TermKind::Identifier(String)`, and
2476    /// the string is bare for a local and qualified for the other three, so nothing
2477    /// downstream could tell a parameter from an import from a constructor
2478    /// ([`DEC-18` decision
2479    /// 1](../../../docs/decisions/dec-18.md#1--the-backend-reads-a-typed-ir-and-the-typer-is-what-produces-it)).
2480    ///
2481    /// Written against hand-built canonical expressions rather than against source, so
2482    /// that all four are read off one translation with no other module needing to be
2483    /// checked first.
2484    ///
2485    /// Mutation-checked by giving the `VarForeign` and `VarTopLevel` arms of
2486    /// `canonical_expr_to_term` the same `ReferenceKind`, which is the collapse this
2487    /// pins: the two assertions naming them then read the same value.
2488    #[test]
2489    fn the_four_kinds_of_name_stay_apart() {
2490        let (union_name, union) = main_size();
2491        let translation = Translation::of_types(HashMap::from([(union_name.clone(), &union)]));
2492
2493        let reference = |kind: canonical::ExpressionKind| {
2494            let mut counter = 0;
2495            let expression = canonical::Expression::bare(kind);
2496
2497            match canonical_expr_to_term(&expression, &translation, &mut counter) {
2498                Some(Term {
2499                    kind: TermKind::Identifier(reference),
2500                    ..
2501                }) => reference,
2502                other => panic!("expected a name, got {:?}", other),
2503            }
2504        };
2505
2506        assert_eq!(
2507            reference(canonical::ExpressionKind::VarLocal("size".into())).kind,
2508            ReferenceKind::Local
2509        );
2510        assert_eq!(
2511            reference(canonical::ExpressionKind::VarTopLevel(qual("Main", "size"))).kind,
2512            ReferenceKind::TopLevel(qual("Main", "size"))
2513        );
2514        assert_eq!(
2515            reference(canonical::ExpressionKind::VarForeign(
2516                qual("Lib", "size"),
2517                canonical::Type::Variable("a".into())
2518            ))
2519            .kind,
2520            ReferenceKind::Foreign(qual("Lib", "size"))
2521        );
2522        assert_eq!(
2523            reference(canonical::ExpressionKind::VarConstructor(
2524                qual("Main", "Big"),
2525                canonical::Type::Type(union_name.clone(), vec![])
2526            ))
2527            .kind,
2528            ReferenceKind::Constructor(Constructor {
2529                union: union_name,
2530                name: "Big".into(),
2531                index: 0,
2532                arity: 0,
2533            })
2534        );

2605    /// `Display for Type` is the text diagnostics quote back to the user, so its two
2606    /// parenthesisation rules and its inference-variable spelling are user-visible
2607    /// output. Nothing else pins them: the pipeline test that renders a type mismatch
2608    /// only ever reaches the `Literal` arms. Dropping a parenthesis here would print
2609    /// `Maybe Maybe Int`, which reads as a different type, with the suite still green.
2610    #[test]
2611    fn display_writes_types_the_way_the_source_spells_them() {
2612        let cases: Vec<(Type, &str)> = vec![
2613            // A function in *parameter* position is parenthesised, because `->` is
2614            // right-associative: `(a -> b) -> c` and `a -> b -> c` are different types.
2615            (
2616                fun_t(fun_t(int_t(), bool_t()), int_t()),
2617                "(Int -> Bool) -> Int",
2618            ),
2619            // In *return* position it is not, for the same reason: the chain already
2620            // re-reads as itself.
2621            (
2622                fun_t(int_t(), fun_t(bool_t(), char_t())),
2623                "Int -> Bool -> Char",
2624            ),
2625            // An applied `Adt` nested inside another needs parens to survive a re-read.
2626            (
2627                adt("Maybe", vec![adt("Maybe", vec![int_t()])]),
2628                "Maybe (Maybe Int)",
2629            ),
2630            // So does a function used as an `Adt` argument.
2631            (
2632                adt("Maybe", vec![fun_t(int_t(), bool_t())]),
2633                "Maybe (Int -> Bool)",
2634            ),
2635            // A *nullary* `Adt` argument does not: there is nothing to mis-group.
2636            (adt("List", vec![adt("Never", vec![])]), "List Never"),
2637            // And an applied `Adt` in parameter position does not either — application
2638            // binds tighter than `->`.
2639            (
2640                fun_t(adt("Maybe", vec![int_t()]), bool_t()),
2641                "Maybe Int -> Bool",
2642            ),
2643            (Type::Tuple(Tuple::two(int_t(), bool_t())), "( Int, Bool )"),
2644            (
2645                Type::Tuple(Tuple::three(int_t(), bool_t(), char_t())),
2646                "( Int, Bool, Char )",
2647            ),
2648            // Inference variables have no source syntax; Elm spells them `t{n}`.
2649            (Type::Variable(TypeVariable { id: 7 }), "t7"),
2650            (Type::Number, "number"),
2651        ];
2652
2653        for (tpe, expected) in cases {
2654            assert_eq!(format!("{}", tpe), expected, "rendering {:?}", tpe);
2655        }
2656    }

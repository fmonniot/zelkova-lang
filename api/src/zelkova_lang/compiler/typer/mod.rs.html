1//! This module contains the type checker pass of the language
2//!
3//! It works with the source AST and will perform two jobs:
4//! - type checks the different declarations and expression
5//! - infer the types when not declared in the source
6//!
7//! I have no idea how that works; so bear with me while I explore
8//! the space, make mistake and (hopefully) learn something :)
9//!
10//! Some papers on type inference:
11//! - http://steshaw.org/hm/hindley-milner.pdf
12//! - https://pdfs.semanticscholar.org/8983/233b3dff2c5b94efb31235f62bddc22dc899.pdf
13//! - http://gallium.inria.fr/~fpottier/publis/fpottier-elaboration.pdf
14//! - http://gallium.inria.fr/~fpottier/publis/emlti-final.pdf
15//!
16//! A type inference problem consists of a type environment Γ , an expression t, and a type T of kind ?
17//!
18//! Constraint generation rules:
19//!
20//! - Equation 1: ⟦x : T⟧ = x ≼ T
21//!   "x has type T if and only if T is an instance of the type scheme associated with x"
22//!   Important part: There is no relation to the typing environment Γ, instead x appears free (and will be bound to Γ later)
23//!
24//! - Equation 2: ⟦λz.t : T⟧ = ∃X1X2.(let z : X1 in ⟦t : X2⟧ ∧ X1 → X2 ≤ T)
25//!   "λz.t has type T if and only if, for some X1 and X2,
26//!   (i) under the assumption that z has type X1, t has type X2, and
27//!   (ii) T is a supertype of X1 → X2."
28//!   z and t types must be fresh (can't generally guess them). They are _existentially_ bound because we are going to
29//!   solve their values. Note that z is _not_ fresh in the condition (i).
30//!
31//! - Equation 3: ⟦t1 t2 : T⟧ = ∃X2.(⟦t1 : X2 → T⟧ ∧ ⟦t2 : X2⟧)
32//!   "t1 t2 has type T if and only if, for some X2, t1 has type X2 → T and t2 has type X2"
33//!
34//! - Equation 4: ⟦let z = t1 in t2 : T⟧ = let z : ∀X[⟦t1 : X⟧].X in ⟦t2 : T⟧
35//!   "let z = t1 in t2 has type T if and only if, under the assumption that z has every type X such that ⟦t1 : X⟧ holds, t2 has type T"
36//!
37//!

49/// Why two types were required to match.
50///
51/// A constraint on its own is a pair of types, which is everything inference needs
52/// and nothing a reader can act on. The reason is what turns "cannot match `Bool`
53/// with `Int`" into something located: *this branch* has type `Bool`, and `Int` is
54/// what was expected *because of this annotation*. Elm calls this a `Reason`, rustc
55/// an `ObligationCause`.
56#[derive(Debug, Clone, Copy, PartialEq, Eq)]
57pub enum Reason {
58    /// A declaration's type annotation, which its body has to satisfy.
59    Annotation,
60    /// A literal's own type: `'a'` is a `Char`.
61    Literal,
62    /// A function's type is its parameter's type arrow its body's type.
63    FunctionShape,
64    /// What is applied has to be a function from the argument's type.
65    Application,
66    /// The condition of an `if` has to be a `Bool`.
67    IfCondition,
68    /// Every branch of an `if` has the type of the whole `if`.
69    IfBranch,
70    /// Every branch of a `case` has the type of the whole `case`.
71    CaseBranch,
72    /// A pattern has to match the type of the expression being matched on.
73    CasePattern,
74    /// A `let` binding has the type of the value bound to it.
75    LetBinding,
76    /// A `let` has the type of its body.
77    LetBody,
78    /// A tuple's type is the tuple of its elements' types.
79    TupleElements,
80}
81
82impl Reason {
83    /// What goes under the caret when the constraint carrying this reason is the one
84    /// that failed: what the underlined text *is*, in the vocabulary of the source.
85    ///
86    /// It names no type, and that is not terseness. By the time a constraint fails,
87    /// `unify` has substituted solutions into both of its sides and may have
88    /// decomposed it into a component of the types the source actually mentions — so
89    /// neither side is reliably "the type of the text under this caret" any more.
90    /// The headline already prints both types; a label that named the wrong one
91    /// would be worse than a label that names none. The example that forced this:
92    /// `answer : Int` with body `true` fails on the literal's own constraint *after*
93    /// `Int` was substituted into it, and reading the type off that side produced
94    /// "this literal has type `Int`".
95    fn describes(&self) -> &'static str {
96        match self {
97            Reason::Annotation => "this type annotation",
98            Reason::Literal => "this literal",
99            Reason::FunctionShape => "this function",
100            Reason::Application => "the expression being applied",
101            Reason::IfCondition => "this condition",
102            Reason::IfBranch => "this branch of the `if`",
103            Reason::CaseBranch => "this branch of the `case`",
104            Reason::CasePattern => "this pattern",
105            Reason::LetBinding => "the value bound here",
106            Reason::LetBody => "the body of this `let`",
107            Reason::TupleElements => "this tuple",
108        }
109    }
110
111    /// What goes under the caret when this reason is not the failure itself but the
112    /// explanation for one side of it — see [`Origin::explanation`].
113    ///
114    /// No type is interpolated here, deliberately. Provenance records which
115    /// constraint brought a type into another one; it does not prove that the type
116    /// printed in the failing constraint is still literally the one this constraint
117    /// carried, and a label is not the place to guess.
118    fn explains(&self) -> &'static str {
119        match self {
120            Reason::Annotation => "expected because of this type annotation",
121            Reason::Literal => "expected because of this literal",
122            Reason::FunctionShape => "expected because of this function",
123            Reason::Application => "expected because of this application",
124            Reason::IfCondition => "expected because this is an `if` condition",
125            Reason::IfBranch => "expected because of this branch",
126            Reason::CaseBranch => "expected because of this branch",
127            Reason::CasePattern => "expected because of this pattern",
128            Reason::LetBinding => "expected because of this value",
129            Reason::LetBody => "expected because of this `let` body",
130            Reason::TupleElements => "expected because of this tuple",
131        }
132    }
133
134    /// The rule that was broken, when naming it says something the labels do not.
135    fn note(&self) -> Option<&'static str> {
136        match self {
137            Reason::Annotation => {
138                Some("a declaration's body must have the type its annotation declares")
139            }
140            Reason::IfCondition => Some("the condition of an `if` must be a `Bool`"),
141            Reason::IfBranch => Some("every branch of an `if` must have the same type"),
142            Reason::CaseBranch => Some("every branch of a `case` must have the same type"),
143            Reason::CasePattern => Some(
144                "every pattern of a `case` must match the type of the expression it matches on",
145            ),
146            _ => None,
147        }
148    }
149}

151/// One piece of source text, and why it required a type: the answer to "where did
152/// this type come from".
153///
154/// Deliberately flat — no chain. A cause names the constraint that *introduced* a
155/// type, never one that relayed it, and that is arranged when the cause is built
156/// rather than by walking a chain afterwards. See `Origin::cause_of`.
157// No `Eq`: `NodeSpan`'s `PartialEq` is deliberately blind (see its documentation), so
158// equality here is a claim about the reason and not about the position.
159#[derive(Debug, Clone, Copy, PartialEq)]
160pub struct Cause {
161    /// Why that constraint required the two types to match.
162    pub reason: Reason,
163    /// The source text it is about.
164    pub span: NodeSpan,
165}
166
167/// Which side of a constraint a type sits on.
168///
169/// Unification is symmetric and does not care; provenance does. Whether a solved
170/// type was read off the left or the right of the constraint that solved it is what
171/// decides whether that constraint is where the type came from, or merely where a
172/// substitution put it — see [`Origin::cause_of`].
173#[derive(Debug, Clone, Copy, PartialEq, Eq)]
174enum Side {
175    Left,
176    Right,
177}
178
179/// Where a constraint came from, and — once inference has moved types around — where
180/// each of its two types came from.
181#[derive(Debug, Clone, PartialEq)]
182pub struct Origin {
183    /// Why the two types were required to match.
184    pub reason: Reason,
185    /// The source text this constraint is about. A failure here draws its caret at
186    /// this span.
187    pub span: NodeSpan,
188    /// The constraint whose solution first rewrote the left type, if any.
189    ///
190    /// `unify` solves a constraint by substituting a type for a variable and then
191    /// applying that substitution to every constraint left. Once that has happened,
192    /// this side holds a type this constraint never mentioned: it came from wherever
193    /// the substitution did. That is the fact a reader needs — "`Int`, because of the
194    /// annotation two lines up" — and it is what the secondary label of a type error
195    /// renders.
196    ///
197    /// The *first* rewrite is kept and later ones are dropped: the first substitution
198    /// to reach a side is the one that brought a foreign type into it, and later ones
199    /// only rewrite what is already there.
200    left_from: Option<Cause>,
201    /// The same, for the right type.
202    right_from: Option<Cause>,
203}
204
205impl Origin {
206    fn new(reason: Reason, span: NodeSpan) -> Origin {
207        Origin {
208            reason,
209            span,
210            left_from: None,
211            right_from: None,
212        }
213    }
214
215    /// This constraint, as the explanation for a type it introduced itself.
216    fn own_cause(&self) -> Cause {
217        Cause {
218            reason: self.reason,
219            span: self.span,
220        }
221    }
222
223    /// Where the type on `side` came from: whatever rewrote that side, or — when
224    /// nothing did — this constraint itself.
225    ///
226    /// This is the whole of the provenance rule, and it is a rule about *sides*.
227    /// When `unify` solves `t := T` from a constraint, `T` was read off one side of
228    /// it. If that side is a type the constraint was written with, the constraint is
229    /// the answer. If a previous solution had rewritten that side, the constraint is
230    /// only relaying a type, and the answer is whatever rewrote it — which is already
231    /// flat, so the credit passes straight through and no chain is ever built.
232    ///
233    /// Crediting a constraint that merely relayed a type is how `result : Bool` /
234    /// `result = not 42` came to blame the annotation: the `Bool` that `42` fails
235    /// against is `not`'s parameter type, which the application constraint carries on
236    /// its *left*, while the annotation had only rewritten its right.
237    fn cause_of(&self, side: Side) -> Cause {
238        let rewritten = match side {
239            Side::Left => self.left_from,
240            Side::Right => self.right_from,
241        };
242
243        rewritten.unwrap_or_else(|| self.own_cause())
244    }
245
246    /// Record that a solution rewrote one side of this constraint, keeping whatever
247    /// was already recorded for that side. See [`Origin::left_from`].
248    fn rewritten(&mut self, side: Side, cause: Cause) {
249        let slot = match side {
250            Side::Left => &mut self.left_from,
251            Side::Right => &mut self.right_from,
252        };
253
254        if slot.is_none() {
255            *slot = Some(cause);
256        }
257    }
258
259    /// Where the type this constraint clashed on came from, when the constraint is
260    /// not itself where it came from.
261    ///
262    /// This is what a diagnostic's secondary label and its rule note are both read
263    /// off. The left side is preferred because that is the declared or expected side
264    /// by convention (see `Constraint`), so "expected because of …" reads about the
265    /// right one; the right side answers when only it was rewritten.
266    pub fn explanation(&self) -> Option<Cause> {
267        self.left_from.or(self.right_from)
268    }
269}

271/// What went wrong, and — for everything unification can raise — where.
272///
273/// Each variant carries the [`Origin`] of the constraint that failed, so a type
274/// error can put its caret under the sub-expression that disagrees instead of across
275/// the declaration containing it. The declaration is still named by [`Error`], one
276/// level up, which is the frame that knows it.
277#[derive(Debug)]
278pub enum ErrorKind {
279    /// Two types were required to match and do not.
280    ///
281    /// The two are written in the order the message reads them out — declared side
282    /// first where the source had one. Neither is reliably "the type of the text the
283    /// origin points at": by the time a constraint fails, unification has substituted
284    /// into both sides and may have decomposed the pair the source actually wrote.
285    // The origins are boxed because an `ErrorKind` is the `Err` half of a `Result`
286    // threaded through the whole of `unify`, and every *successful* return pays for
287    // the size of the largest variant. `Origin` carries a `Cause` per side.
288    UnificationFailed {
289        left: Type,
290        right: Type,
291        origin: Box<Origin>,
292    },
293    /// A type variable would have to occur inside its own solution.
294    CircularType {
295        /// The type the variable would have had to contain itself in.
296        tpe: Type,
297        origin: Box<Origin>,
298    },
299    /// A name the typer's environment does not know. See [`type_check`] for why this
300    /// one is not reported to the user today.
301    UnboundVariable {
302        name: String,
303        /// Where the name was written.
304        span: NodeSpan,
305    },
306}
307
308impl ErrorKind {
309    /// The provenance of the constraint that failed, when the failure came from
310    /// unification at all.
311    fn origin(&self) -> Option<&Origin> {
312        match self {
313            ErrorKind::UnificationFailed { origin, .. }
314            | ErrorKind::CircularType { origin, .. } => Some(origin.as_ref()),
315            ErrorKind::UnboundVariable { .. } => None,
316        }
317    }
318
319    fn message(&self) -> String {
320        match self {
321            ErrorKind::UnificationFailed { left, right, .. } => {
322                format!("cannot match `{}` with `{}`", left, right)
323            }
324            ErrorKind::CircularType { tpe, .. } => format!(
325                "circular type: a type variable would have to contain itself in `{}`",
326                tpe
327            ),
328            ErrorKind::UnboundVariable { name, .. } => {
329                format!("cannot find a value named `{}`", name)
330            }
331        }
332    }
333}

335/// A type error: what went wrong, where, and in which declaration.
336///
337/// # Where the positions come from
338///
339/// The typer does not check the canonical AST directly — it translates it into its
340/// own [`Term`] language — but that translation now carries the canonical node's
341/// [`NodeSpan`] along, and each constraint keeps the span of the term that produced
342/// it. So a unification failure knows the sub-expression it is about, and
343/// [`labels`](PhaseError::labels) draws the caret there.
344///
345/// The declaration's own span is still kept as `span`, for the one case that has no
346/// finer answer: an error whose origin came from a hand-built term, or from a
347/// canonical node the parser never spanned. Then the label degrades to the whole
348/// declaration rather than disappearing.
349#[derive(Debug)]
350pub struct Error {
351    pub kind: ErrorKind,
352    /// Where the declaration the error was found in was written. Used only as the
353    /// fallback described above.
354    pub span: NodeSpan,
355    /// That declaration's name, which is what the note names: in a module of a
356    /// hundred declarations, a caret is not much use without it when the diagnostic
357    /// is rendered without a file (see `compile_package`).
358    pub declaration: Name,
359}

361/// Type errors are about types, and [`Type`]'s `Display` writes them the way the
362/// source does — so the message can name both sides of a mismatch instead of
363/// dumping the typer's internal representation.
364impl PhaseError for Error {
365    fn message(&self) -> String {
366        self.kind.message()
367    }
368
369    fn notes(&self) -> Vec<String> {
370        let mut notes = vec![format!("in the declaration of `{}`", self.declaration)];
371
372        // The rule that was broken, taken from the failing constraint and, if that
373        // one has nothing to add, from what explains it. `if`'s "every branch must
374        // have the same type" is on the branch constraint; "a body must have the
375        // type its annotation declares" is on the annotation, which is normally the
376        // explanation rather than the failure.
377        if let Some(origin) = self.kind.origin() {
378            let rule = origin
379                .reason
380                .note()
381                .or_else(|| origin.explanation().and_then(|c| c.reason.note()));
382
383            if let Some(rule) = rule {
384                notes.push(rule.to_owned());
385            }
386        }
387
388        notes
389    }
390
391    /// A caret under the text that disagrees, and — when inference can say where the
392    /// type it disagrees with came from — a second, secondary one under that.
393    ///
394    /// Falls back to underlining the whole declaration when the failing constraint
395    /// has no position, which is what a term built by hand, or one translated from a
396    /// canonical node with no span, produces.
397    fn labels(&self) -> Vec<SpanLabel> {
398        let mut labels = Vec::new();
399
400        match &self.kind {
401            ErrorKind::UnboundVariable { span, .. } => {
402                if let Some(span) = span.span() {
403                    labels.push(SpanLabel {
404                        span,
405                        message: "not found in this scope".to_owned(),
406                        primary: true,
407                        file: None,
408                    });
409                }
410            }
411            kind => {
412                if let Some(origin) = kind.origin() {
413                    if let Some(span) = origin.span.span() {
414                        labels.push(SpanLabel {
415                            span,
416                            message: origin.reason.describes().to_owned(),
417                            primary: true,
418                            file: None,
419                        });
420                    }
421
422                    // Only worth drawing once the primary one exists: on its own it
423                    // would be a caret under the annotation with nothing to contrast
424                    // it against. And not worth drawing at all when it lands on the
425                    // same text — a literal that is its own explanation renders as two
426                    // carets under one word saying the same thing twice.
427                    if let (Some(primary), Some(because)) = (labels.first(), origin.explanation()) {
428                        match because.span.span() {
429                            Some(span) if span != primary.span => labels.push(SpanLabel {
430                                span,
431                                message: because.reason.explains().to_owned(),
432                                primary: false,
433                                file: None,
434                            }),
435                            _ => (),
436                        }
437                    }
438                }
439            }
440        }
441
442        if labels.is_empty() {
443            if let Some(span) = self.span.span() {
444                labels.push(SpanLabel {
445                    span,
446                    message: format!("in `{}`", self.declaration),
447                    primary: true,
448                    file: None,
449                });
450            }
451        }
452
453        labels
454    }
455}

457/// Type check one canonical module, reporting every value whose inference produced a
458/// reportable error rather than stopping at the first — the shape `compile_package` is
459/// built to accumulate.
460///
461/// # The two failures this pass still swallows
462///
463/// Both are in the third pass below, and neither is a statement about the user's
464/// source, which is why neither is reported as an error against it.
465///
466/// **An unsupported construct.** `value_to_term_and_annotation` returns `None` for
467/// anything the term language cannot express — a `VarKernel` or `VarForeign`
468/// reference, a nested pattern inside a `case`. The declaration is then not checked
469/// at all. Reporting that as an error would fail 53 of the declarations in
470/// `std/core/src` that are simply beyond today's inference; it is a gap in the
471/// typer, and what it wants is a warning, which the compiler does not have yet
472/// (`ERR-8`, see `docs/tickets/README.md`).
473///
474/// **An unbound variable.** [`ErrorKind::UnboundVariable`] means the name is missing
475/// from the environment the two passes above build — and that environment is
476/// assembled from *this module alone*: its own annotated values and its own type
477/// constructors. Anything that crossed a module boundary is therefore absent even
478/// though canonicalization resolved it perfectly well: `Maybe.isJust` referring to
479/// `True`, which `Basics` declares, or `Basics.negate` referring to `-`, which the
480/// infix declaration aliases to `sub`. Five declarations in `std/core/src` hit this
481/// today and not one of them is a mistake in the source. A name that genuinely does
482/// not exist is caught earlier, by canonicalization, as
483/// `canonical::Error::VariableNotFound`, with a caret under the name — so nothing a
484/// user can write reaches the user only through this path. The error nonetheless
485/// carries its span now, so the day the typer's environment spans modules it can be
486/// reported without further plumbing.
487///
488/// # Where an [`Error`] is built
489///
490/// Here, and only here: this is the last frame that holds the `canonical::Value`, so
491/// it is the only one that knows the declaration's name and its span. What went
492/// *wrong* and *where inside the declaration* comes up from inference on the
493/// [`ErrorKind`]; see [`Error`].
494pub fn type_check(module: &Module) -> Result<(), Vec<Error>> {
495    // JavaScript binding modules use synthetic placeholder bodies — skip type checking.
496    if module.binding_javascript {
497        return Ok(());
498    }
499
500    // Start at a high offset to avoid collisions with the counter inside
501    // Types::new() (which starts at 10) used during inference.
502    let mut counter = 10_000u32;
503
504    // First pass: build global env from all TypedValues' declared types.
505    // This allows values to reference other module-level typed values.
506    let mut global: HashMap<String, Type> = HashMap::new();
507    for (name, value) in &module.values {
508        if let canonical::Value::TypedValue { tpe, .. } = value {
509            let mut var_map = HashMap::new();
510            if let Some(typer_tpe) = canonical_type_to_typer_type(tpe, &mut var_map, &mut counter) {
511                // Add both qualified (e.g. "Test.not") and unqualified (e.g. "not") names
512                let qname = module
513                    .name
514                    .qualify_name(name)
515                    .to_name()
516                    .as_str()
517                    .to_string();
518                global.insert(qname, typer_tpe.clone());
519                global.insert(name.as_str().to_string(), typer_tpe);
520            }
521        }
522    }
523
524    // Second pass: add constructor types to global from module.types
525    for (type_name, union_type) in &module.types {
526        // Fresh type vars for each ADT type parameter (e.g. "a" in Maybe a)
527        let mut adt_var_map: HashMap<String, TypeVariable> = HashMap::new();
528        for tv_name in &union_type.variables {
529            counter += 1;
530            adt_var_map.insert(tv_name.as_str().to_string(), TypeVariable { id: counter });
531        }
532
533        // Build result type: Adt(type_name, [TypeVar for each param])
534        let result_args: Vec<Type> = union_type
535            .variables
536            .iter()
537            .map(|v| Type::Variable(adt_var_map[v.as_str()].clone()))
538            .collect();
539        let result_type = Type::Adt(type_name.as_str().to_string(), result_args);
540
541        for ctor in &union_type.variants {
542            let ctor_type = if ctor.type_parameters.is_empty() {
543                result_type.clone()
544            } else {
545                let mut translate_var_map = adt_var_map.clone();
546                let params: Vec<Type> = ctor
547                    .type_parameters
548                    .iter()
549                    .filter_map(|t| {
550                        canonical_type_to_typer_type(t, &mut translate_var_map, &mut counter)
551                    })
552                    .collect();
553                if params.len() != ctor.type_parameters.len() {
554                    continue; // untranslatable param type, skip this constructor
555                }
556                // Build Fun type: p1 -> p2 -> ... -> result_type
557                params
558                    .into_iter()
559                    .rev()
560                    .fold(result_type.clone(), |acc, p| Type::Fun {
561                        param_tpe: Box::new(p),
562                        return_tpe: Box::new(acc),
563                    })
564            };
565
566            // Register under both unqualified ("Just") and qualified ("Test.Just") names
567            global.insert(ctor.name.as_str().to_string(), ctor_type.clone());
568            let qname = module
569                .name
570                .qualify_name(&ctor.name)
571                .to_name()
572                .as_str()
573                .to_string();
574            global.insert(qname, ctor_type);
575        }
576    }
577
578    // Third pass: check each value. A value that fails is recorded and the pass
579    // moves on, so one broken declaration cannot hide the others.
580    let mut errors: Vec<Error> = vec![];
581    for (name, value) in &module.values {
582        let Some((term, annotation)) =
583            value_to_term_and_annotation(value, &module.types, &mut counter)
584        else {
585            continue; // unsupported construct — see this function's documentation
586        };
587
588        // The annotation is handed to inference rather than checked against its
589        // result afterwards, which is what lets a mismatch *inside* the body know
590        // that the type it failed against came from the annotation. Checking the two
591        // whole types at the end could only ever say "this declaration is `Int` and
592        // its body is something else", with the caret across the lot.
593        match infer_annotated(term, global.clone(), annotation) {
594            // See this function's documentation: an unbound variable here is a hole
595            // in the typer's environment, not a mistake in the source.
596            Err(ErrorKind::UnboundVariable { .. }) => continue,
597            Err(kind) => errors.push(Error {
598                kind,
599                span: value.span(),
600                declaration: name.clone(),
601            }),
602            Ok(_) => (),
603        }
604    }
605
606    if errors.is_empty() {
607        Ok(())
608    } else {
609        Err(errors)
610    }
611}

615/// Convert a canonical type to the typer's simplified Type representation.
616/// The match covers all four `canonical::Type` variants — `Variable`, `Arrow`, `Tuple`
617/// (either arity), and `Type` including named types with parameters — and every arm's
618/// own base case returns `Some`; a `None` only ever arises by propagating up from a
619/// nested recursive call. As of today no `canonical::Type` shape actually reaches such
620/// a case, so the function always returns `Some`. The `Option` return stays in place for
621/// when a genuinely unrepresentable variant (e.g. records, aliases) is added.
622///
623/// `var_map` maps named type variables (e.g. "a") to consistent TypeVariable
624/// ids, so that `a -> a` produces the same variable on both sides.
625fn canonical_type_to_typer_type(
626    tpe: &canonical::Type,
627    var_map: &mut HashMap<String, TypeVariable>,
628    counter: &mut u32,
629) -> Option<Type> {
630    match tpe {
631        canonical::Type::Type(name, args) if args.is_empty() && name.as_str() == "Int" => {
632            Some(Type::Literal(TypeLiteral::Int))
633        }
634        canonical::Type::Type(name, args) if args.is_empty() && name.as_str() == "Bool" => {
635            Some(Type::Literal(TypeLiteral::Bool))
636        }
637        canonical::Type::Type(name, args) if args.is_empty() && name.as_str() == "Char" => {
638            Some(Type::Literal(TypeLiteral::Char))
639        }
640        canonical::Type::Type(name, args) if args.is_empty() && name.as_str() == "Float" => {
641            Some(Type::Literal(TypeLiteral::Float))
642        }
643        canonical::Type::Variable(name) => {
644            let tv = var_map.entry(name.as_str().to_string()).or_insert_with(|| {
645                *counter += 1;
646                TypeVariable { id: *counter }
647            });
648            Some(Type::Variable(tv.clone()))
649        }
650        canonical::Type::Arrow(a, b) => {
651            let a = canonical_type_to_typer_type(a, var_map, counter)?;
652            let b = canonical_type_to_typer_type(b, var_map, counter)?;
653            Some(Type::Fun {
654                param_tpe: Box::new(a),
655                return_tpe: Box::new(b),
656            })
657        }
658        // `Tuple::try_map` keeps the arity attached to the value instead of
659        // re-deriving it here, the same way the parser → canonical conversions
660        // in `canonical/mod.rs` do.
661        canonical::Type::Tuple(tuple) => {
662            let elements = tuple
663                .try_map(|elem| canonical_type_to_typer_type(elem, var_map, counter).ok_or(()))
664                .ok()?;
665            Some(Type::Tuple(elements))
666        }
667        canonical::Type::Type(name, args) => {
668            let converted: Option<Vec<Type>> = args
669                .iter()
670                .map(|a| canonical_type_to_typer_type(a, var_map, counter))
671                .collect();
672            Some(Type::Adt(name.as_str().to_string(), converted?))
673        }
674    }
675}

677/// Convert a canonical expression to a Term, keeping the position it was written at.
678///
679/// Returns None for constructs the inference engine doesn't yet handle
680/// (VarKernel, VarForeign, complex patterns inside Case).
681///
682/// Every arm attaches `expr.span` to the term it builds. That is the whole of what
683/// `ERR-4` needed from this function: a constraint can only point at a
684/// sub-expression if the term that produced it remembers where it came from.
685fn canonical_expr_to_term(
686    expr: &canonical::Expression,
687    module_types: &HashMap<Name, canonical::UnionType>,
688    counter: &mut u32,
689) -> Option<Term> {
690    let kind = match &expr.kind {
691        canonical::ExpressionKind::Int(i) => TermKind::Int(*i as u32),
692        canonical::ExpressionKind::Bool(b) => TermKind::Bool(*b),
693        canonical::ExpressionKind::Char(c) => TermKind::Char(*c),
694        canonical::ExpressionKind::Float(f) => TermKind::Float(*f),
695        canonical::ExpressionKind::VarLocal(name) => {
696            TermKind::Identifier(name.as_str().to_string())
697        }
698        canonical::ExpressionKind::VarTopLevel(qname) => {
699            TermKind::Identifier(qname.to_name().as_str().to_string())
700        }
701        canonical::ExpressionKind::Apply(f, a) => {
702            let fun = canonical_expr_to_term(f, module_types, counter)?;
703            let arg = canonical_expr_to_term(a, module_types, counter)?;
704            TermKind::Apply {
705                fun: Box::new(fun),
706                arg: Box::new(arg),
707            }
708        }
709        canonical::ExpressionKind::If(cond, t, f) => {
710            let cond = canonical_expr_to_term(cond, module_types, counter)?;
711            let t = canonical_expr_to_term(t, module_types, counter)?;
712            let f = canonical_expr_to_term(f, module_types, counter)?;
713            TermKind::If {
714                cond: Box::new(cond),
715                true_branch: Box::new(t),
716                false_branch: Box::new(f),
717            }
718        }
719        canonical::ExpressionKind::Tuple(tuple) => {
720            let elements = tuple
721                .try_map(|elem| canonical_expr_to_term(elem, module_types, counter).ok_or(()))
722                .ok()?;
723            TermKind::Tuple(elements)
724        }
725        canonical::ExpressionKind::Case(scrutinee_expr, branches) => {
726            let scrutinee = canonical_expr_to_term(scrutinee_expr, module_types, counter)?;
727            let term_branches: Vec<(TermPattern, Box<Term>)> = branches
728                .iter()
729                .map(|cb| {
730                    let (pattern, _bindings) =
731                        translate_pattern(&cb.pattern, module_types, counter)?;
732                    let body = canonical_expr_to_term(&cb.expression, module_types, counter)?;
733                    Some((pattern, Box::new(body)))
734                })
735                .collect::<Option<Vec<_>>>()?;
736            TermKind::Case {
737                scrutinee: Box::new(scrutinee),
738                branches: term_branches,
739            }
740        }
741        // Constructors are resolved as identifiers looked up in the global env.
742        canonical::ExpressionKind::VarConstructor(qname, _) => {
743            TermKind::Identifier(qname.to_name().as_str().to_string())
744        }
745        // VarForeign: not in the module's global env, skip
746        canonical::ExpressionKind::VarForeign(_, _) => return None,
747        // Not yet supported: VarKernel
748        _ => return None,
749    };
750
751    Some(Term {
752        span: expr.span,
753        kind,
754    })
755}

757/// Translate a canonical pattern into a `TermPattern` plus any variable bindings
758/// introduced by the pattern.  Returns `None` for unsupported pattern shapes.
759///
760/// The pattern keeps its own span, separate from the branch body's: a `case` branch
761/// whose pattern does not match what is being matched on is about the pattern, and
762/// the caret belongs there rather than under the expression in the `case … of` line.
763fn translate_pattern(
764    pattern: &canonical::Pattern,
765    module_types: &HashMap<Name, canonical::UnionType>,
766    counter: &mut u32,
767) -> Option<(TermPattern, Vec<(String, Type)>)> {
768    let (kind, bindings) = match &pattern.kind {
769        canonical::PatternKind::Anything => (TermPatternKind::Anything, vec![]),
770        canonical::PatternKind::Variable(name) => {
771            // The binding's actual type will be unified with the scrutinee type in annotate.
772            (TermPatternKind::Bind(name.as_str().to_string()), vec![])
773        }
774        canonical::PatternKind::Bool(_) => (
775            TermPatternKind::Literal(Type::Literal(TypeLiteral::Bool)),
776            vec![],
777        ),
778        canonical::PatternKind::Int(_) => (
779            TermPatternKind::Literal(Type::Literal(TypeLiteral::Int)),
780            vec![],
781        ),
782        canonical::PatternKind::Char(_) => (
783            TermPatternKind::Literal(Type::Literal(TypeLiteral::Char)),
784            vec![],
785        ),
786        canonical::PatternKind::Constructor { ctor, args } => {
787            // Look up the parent union type to get its type variables.
788            let union_type = module_types.get(&ctor.tpe)?;
789
790            // Create fresh type vars for each ADT type parameter.
791            let mut adt_var_map: HashMap<String, TypeVariable> = HashMap::new();
792            for tv_name in &union_type.variables {
793                *counter += 1;
794                adt_var_map.insert(tv_name.as_str().to_string(), TypeVariable { id: *counter });
795            }
796
797            // Build the ADT result type args from the fresh vars.
798            let adt_args: Vec<Type> = union_type
799                .variables
800                .iter()
801                .map(|v| Type::Variable(adt_var_map[v.as_str()].clone()))
802                .collect();
803
804            // Translate each constructor type parameter (reuses the same fresh vars).
805            let param_types: Vec<Type> = ctor
806                .type_parameters
807                .iter()
808                .filter_map(|t| canonical_type_to_typer_type(t, &mut adt_var_map, counter))
809                .collect();
810            if param_types.len() != ctor.type_parameters.len() {
811                return None;
812            }
813
814            // Build bindings from arg patterns.
815            let mut bindings: Vec<(String, Type)> = vec![];
816            for (arg_pattern, param_type) in args.iter().zip(param_types.iter()) {
817                match &arg_pattern.kind {
818                    canonical::PatternKind::Variable(name) => {
819                        bindings.push((name.as_str().to_string(), param_type.clone()));
820                    }
821                    canonical::PatternKind::Anything => {} // no binding needed
822                    _ => return None, // nested complex patterns not yet supported
823                }
824            }
825
826            let kind = TermPatternKind::Constructor {
827                adt_name: ctor.tpe.as_str().to_string(),
828                adt_args,
829                bindings: bindings.clone(),
830            };
831            (kind, bindings)
832        }
833        _ => return None, // Tuple, Float patterns — not yet supported
834    };
835
836    Some((
837        TermPattern {
838            span: pattern.span,
839            kind,
840        },
841        bindings,
842    ))
843}

884                    span: *annotation_span,

891/// Wrap a body Term in nested Fun nodes for each pattern, outermost first.
892/// Returns None if any pattern is not translatable (e.g. constructor patterns).
893///
894/// Each `Fun` spans its parameter through the body it wraps, so a function whose
895/// declared shape does not match its definition is underlined from the parameter
896/// that starts it rather than across the annotation as well.
897fn wrap_with_patterns<'a>(
898    patterns: impl Iterator<Item = &'a canonical::Pattern>,
899    body: Term,
900) -> Option<Term> {
901    let names: Vec<(String, NodeSpan)> = patterns
902        .map(|p| match &p.kind {
903            canonical::PatternKind::Variable(name) => Some((name.as_str().to_string(), p.span)),
904            canonical::PatternKind::Anything => Some(("_".to_string(), p.span)),
905            _ => None,
906        })
907        .collect::<Option<Vec<_>>>()?;
908
909    let term = names.iter().rev().fold(body, |acc, (param, span)| Term {
910        span: span.merge(acc.span),
911        kind: TermKind::Fun {
912            param: param.clone(),
913            body: Box::new(acc),
914        },
915    });
916
917    Some(term)
918}

1070/// This is deliberately not `Debug`: `Debug` prints the typer's own vocabulary
1071/// (`Lit(Int)`, `Var(#3)`), which is what you want at a breakpoint and never what
1072/// you want in a message the user reads. Inference variables have no source syntax
1073/// at all, so they are written `t3` — Elm's convention for an unsolved variable.

1075    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
1076        match self {
1077            Type::Literal(TypeLiteral::Int) => write!(f, "Int"),
1078            Type::Literal(TypeLiteral::Bool) => write!(f, "Bool"),
1079            Type::Literal(TypeLiteral::Char) => write!(f, "Char"),
1080            Type::Literal(TypeLiteral::Float) => write!(f, "Float"),
1081            Type::Number => write!(f, "number"),
1082            Type::Variable(TypeVariable { id }) => write!(f, "t{}", id),
1083            // The parameter of a function type is parenthesised when it is itself a
1084            // function, because `->` is right-associative: `(a -> b) -> c` and
1085            // `a -> b -> c` are different types.
1086            Type::Fun {
1087                param_tpe,
1088                return_tpe,
1089            } => match **param_tpe {
1090                Type::Fun { .. } => write!(f, "({}) -> {}", param_tpe, return_tpe),
1091                _ => write!(f, "{} -> {}", param_tpe, return_tpe),
1092            },
1093            Type::Tuple(Tuple::Two(a, b)) => write!(f, "( {}, {} )", a, b),
1094            Type::Tuple(Tuple::Three(a, b, c)) => write!(f, "( {}, {}, {} )", a, b, c),
1095            Type::Adt(name, args) if args.is_empty() => write!(f, "{}", name),
1096            Type::Adt(name, args) => {
1097                write!(f, "{}", name)?;
1098                for arg in args {
1099                    // Same reason as above: an argument that is itself applied or a
1100                    // function needs parentheses to stay the same type when re-read.
1101                    match arg {
1102                        Type::Adt(_, inner) if !inner.is_empty() => write!(f, " ({})", arg)?,
1103                        Type::Fun { .. } => write!(f, " ({})", arg)?,
1104                        _ => write!(f, " {}", arg)?,
1105                    }
1106                }
1107                Ok(())
1108            }
1109        }
1110    }

1170/// Two types that have to match, and why.
1171///
1172/// # What the two sides mean
1173///
1174/// Unification treats them symmetrically; the order is a rendering convention, and
1175/// the only thing that reads it is the headline — *cannot match `left` with `right`*.
1176/// `collect` therefore writes the declared or expected side first where the source
1177/// has one (an annotation's type, a pattern's type) and the inferred side second, so
1178/// the sentence comes out in the order a reader expects.
1179///
1180/// What the sides are emphatically *not* is a way to tell which type belongs to the
1181/// text at `origin.span`. By the time a constraint fails, `unify` has substituted
1182/// other constraints' solutions into both sides, and may have decomposed it into a
1183/// component of the types the source mentioned. That is why the labels name the
1184/// [`Reason`] and leave the types to the headline — see [`Reason::describes`].
1185///
1186/// # Why these are held in a `Vec` and not a `HashSet`
1187///
1188/// They used to be a `HashSet<Constraint>`, back when a constraint was a bare pair of
1189/// types. An origin makes that collection wrong twice over. Deduplication now
1190/// discards *provenance*: two constraints with equal types but different origins are
1191/// one entry, and which origin survives is whichever was inserted first. And the
1192/// order a `HashSet` yields is unspecified, so which constraint `unify` reaches first
1193/// — and therefore which one is reported when several are unsatisfiable — would vary
1194/// between runs of the same compiler on the same file.
1195///
1196/// A `Vec` fixes both, and buys a third thing: source order. The annotation is
1197/// pushed first, so its type is substituted into the body's constraints before they
1198/// are solved, which is what lets a mismatch deep in the body say that `Int` came
1199/// from the annotation. The cost is that duplicate constraints are no longer
1200/// collapsed, which is a few more `unify` steps on terms that repeat a type.
1201#[derive(Debug, Clone, PartialEq)]
1202struct Constraint {
1203    left: Type,
1204    right: Type,
1205    origin: Origin,
1206}
1207
1208impl Constraint {
1209    fn new(left: Type, right: Type, reason: Reason, span: NodeSpan) -> Constraint {
1210        Constraint {
1211            left,
1212            right,
1213            origin: Origin::new(reason, span),
1214        }
1215    }
1216
1217    /// A constraint between two components of this one — the parameters of two
1218    /// function types being matched, say — which is about the same source text and
1219    /// was required for the same reason.
1220    ///
1221    /// The origin is inherited whole, side provenance included: if a substitution
1222    /// rewrote the left type, it rewrote whatever the left type decomposes into. That
1223    /// is an over-approximation — a substitution that reached only the return half of
1224    /// an arrow is credited with the parameter half too — but it errs towards naming
1225    /// a constraint that did carry a type in, which is the direction that keeps a
1226    /// caret on the source rather than on the compiler's working.
1227    fn component(&self, left: Type, right: Type) -> Constraint {
1228        Constraint {
1229            left,
1230            right,
1231            origin: self.origin.clone(),
1232        }
1233    }
1234}

1236/// What one type variable was solved to, and where that type came from.
1237///
1238/// The cause is not used by inference at all. It is carried so that when this
1239/// solution is substituted into another constraint and *that* constraint then fails,
1240/// the failure can say where the type it failed against came from — see
1241/// [`Origin::left_from`].
1242///
1243/// It is a [`Cause`] and not an `Origin` because the question it answers is already
1244/// settled: [`Origin::cause_of`] resolved, at the moment the variable was solved,
1245/// whether the solving constraint introduced this type or was handed it. Nothing
1246/// downstream has to walk anything.
1247// No `Eq`: a `Cause` holds a `NodeSpan`, whose `PartialEq` is deliberately blind
1248// (see its documentation), so equality here is a claim about the types and the
1249// reasons, not about the positions.
1250#[derive(Debug, PartialEq, Clone)]
1251struct Solution {
1252    tpe: Type,
1253    cause: Cause,
1254}
1255
1256#[derive(Debug, PartialEq)]
1257struct Substitution {
1258    solutions: HashMap<TypeVariable, Solution>,
1259}
1260
1261impl Substitution {
1262    // constructors
1263
1264    fn empty() -> Substitution {
1265        Substitution {
1266            solutions: HashMap::new(),
1267        }
1268    }
1269
1270    fn one(tvar: TypeVariable, tpe: Type, cause: Cause) -> Substitution {
1271        let mut sub = Substitution::empty();
1272
1273        sub.solutions.insert(tvar, Solution { tpe, cause });
1274
1275        sub
1276    }
1277
1278    // methods
1279
1280    /// Rewrite a constraint with everything solved so far, recording for each side
1281    /// which solution first reached it.
1282    ///
1283    /// The two sides are tracked apart, because that is what tells a relayed type
1284    /// from an introduced one later on — see [`Origin::cause_of`]. A solution that
1285    /// rewrites only the right side has said nothing about the left, and treating it
1286    /// as though it had is what made a type error blame an annotation that was not
1287    /// load-bearing.
1288    ///
1289    /// The solutions are visited in type-variable order rather than in `HashMap`
1290    /// order: "first" has to mean the same thing on every run, or the secondary label
1291    /// of a diagnostic would move between compilations of an unchanged file. Ids are
1292    /// handed out as inference walks the term, so that order is roughly the order the
1293    /// user wrote things in.
1294    fn apply(&self, c: &Constraint) -> Constraint {
1295        let mut origin = c.origin.clone();
1296
1297        let mut solutions: Vec<_> = self.solutions.iter().collect();
1298        solutions.sort_by_key(|(tvar, _)| tvar.id);
1299
1300        for (tvar, solution) in solutions {
1301            // A solution that does not mention a variable a side uses rewrites
1302            // nothing there, and explains nothing about it either.
1303            if occurs(tvar, &c.left) {
1304                origin.rewritten(Side::Left, solution.cause);
1305            }
1306            if occurs(tvar, &c.right) {
1307                origin.rewritten(Side::Right, solution.cause);
1308            }
1309        }
1310
1311        Constraint {
1312            left: self.apply_type(&c.left),
1313            right: self.apply_type(&c.right),
1314            origin,
1315        }
1316    }
1317
1318    fn apply_type(&self, tpe: &Type) -> Type {
1319        self.solutions
1320            .iter()
1321            .fold(tpe.clone(), |tpe, (tvar, solution)| {
1322                Substitution::substitute(tpe, tvar, &solution.tpe)
1323            })
1324    }
1325
1326    fn substitute(tpe: Type, tvar: &TypeVariable, replacement: &Type) -> Type {
1327        match tpe {
1328            Type::Literal(_) | Type::Number => tpe,
1329            Type::Fun {
1330                param_tpe,
1331                return_tpe,
1332            } => Type::Fun {
1333                param_tpe: Box::new(Substitution::substitute(*param_tpe, tvar, replacement)),
1334                return_tpe: Box::new(Substitution::substitute(*return_tpe, tvar, replacement)),
1335            },
1336            Type::Tuple(Tuple::Two(a, b)) => Type::Tuple(Tuple::two(
1337                Substitution::substitute(*a, tvar, replacement),
1338                Substitution::substitute(*b, tvar, replacement),
1339            )),
1340            Type::Tuple(Tuple::Three(a, b, c)) => Type::Tuple(Tuple::three(
1341                Substitution::substitute(*a, tvar, replacement),
1342                Substitution::substitute(*b, tvar, replacement),
1343                Substitution::substitute(*c, tvar, replacement),
1344            )),
1345            Type::Adt(name, args) => Type::Adt(
1346                name,
1347                args.into_iter()
1348                    .map(|a| Substitution::substitute(a, tvar, replacement))
1349                    .collect(),
1350            ),
1351            Type::Variable(tvar2) if tvar == &tvar2 => replacement.clone(),
1352            tpe @ Type::Variable(_) => tpe,
1353        }
1354    }
1355
1356    fn merge(&self, other: Substitution) -> Substitution {
1357        // This merge means we should try sub_tail first, and then sub_head
1358        // Merging other in self means we apply `other` substitution to `self` solutions
1359        // When merging, we want `other` solutions to take precedences over `self` solutions
1360
1361        let self_solutions = self.solutions.iter().map(|(k, v)| {
1362            (
1363                k.clone(),
1364                Solution {
1365                    tpe: other.apply_type(&v.tpe),
1366                    // The cause says where this variable's type came from, which
1367                    // rewriting the type does not change.
1368                    cause: v.cause,
1369                },
1370            )
1371        });
1372
1373        let mut sub = Substitution::empty();
1374
1375        sub.solutions.extend(self_solutions);
1376        sub.solutions.extend(other.solutions);
1377
1378        sub
1379    }
1380}

1382/// Does `tvar` appear anywhere inside `tpe`?
1383///
1384/// Two callers, for two different reasons: `unify_variable` uses it as the occurs
1385/// check that keeps it from building an infinite type, and `Substitution::apply` uses
1386/// it to tell whether a solution actually rewrites a given constraint — which is what
1387/// decides whether that solution explains one of the constraint's types.
1388fn occurs(tvar: &TypeVariable, tpe: &Type) -> bool {
1389    match tpe {
1390        Type::Fun {
1391            param_tpe,
1392            return_tpe,
1393        } => occurs(tvar, param_tpe) || occurs(tvar, return_tpe),
1394        Type::Tuple(tuple) => tuple.iter().any(|t| occurs(tvar, t)),
1395        Type::Adt(_, args) => args.iter().any(|a| occurs(tvar, a)),
1396        Type::Variable(tvar2) => tvar == tvar2,
1397        _ => false,
1398    }
1399}

1438/// infer the type of the given term given known function defined in the outer scopes.
1439/// This is a translation of the algorithm demonstrated by
1440/// [Ionut Gan at I T.A.K.E Unconference 2015](https://www.youtube.com/watch?v=oPVTNxiMcSU)
1441pub fn infer(term: Term, global: HashMap<String, Type>) -> Result<Type, ErrorKind> {
1442    infer_annotated(term, global, None)
1443}
1444
1445/// [`infer`], with the declaration's type annotation as a constraint of its own.
1446///
1447/// The annotation is put *first*, before the constraints the body generates, and that
1448/// ordering is the point of the function. `unify` solves constraints in order, so the
1449/// annotated type is substituted into the body's constraints before any of them are
1450/// solved; when one of them then fails, its [`Origin::explanation`] names the annotation,
1451/// and the diagnostic can say `Int` was expected *because of the annotation* rather
1452/// than merely that the declaration as a whole does not check.
1453fn infer_annotated(
1454    term: Term,
1455    global: HashMap<String, Type>,
1456    annotation: Option<Annotation>,
1457) -> Result<Type, ErrorKind> {
1458    let mut env = Types::new();
1459    env.extends_with(global);
1460
1461    let typed_term = annotate::annotate(term, &mut env)?;
1462    debug!("typed term: {:#?}", typed_term);
1463
1464    let mut constraints = Vec::new();
1465
1466    if let Some(annotation) = annotation {
1467        // Left is the annotation's type, because left is the type of the text the
1468        // span points at — see `Constraint`.
1469        constraints.push(Constraint::new(
1470            annotation.tpe,
1471            typed_term.tpe.clone(),
1472            Reason::Annotation,
1473            annotation.span,
1474        ));
1475    }
1476
1477    constraints.extend(constraint::collect(&typed_term));
1478    debug!("Constraints: {:#?}", constraints);
1479
1480    let substitution = unifier::unify(constraints)?;
1481
1482    Ok(substitution.apply_type(&typed_term.tpe))
1483}

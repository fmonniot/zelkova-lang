1//! This module contains the type checker pass of the language
2//!
3//! It works with the source AST and will perform two jobs:
4//! - type checks the different declarations and expression
5//! - infer the types when not declared in the source
6//!
7//! I have no idea how that works; so bear with me while I explore
8//! the space, make mistake and (hopefully) learn something :)
9//!
10//! Some papers on type inference:
11//! - http://steshaw.org/hm/hindley-milner.pdf
12//! - https://pdfs.semanticscholar.org/8983/233b3dff2c5b94efb31235f62bddc22dc899.pdf
13//! - http://gallium.inria.fr/~fpottier/publis/fpottier-elaboration.pdf
14//! - http://gallium.inria.fr/~fpottier/publis/emlti-final.pdf
15//!
16//! A type inference problem consists of a type environment Γ , an expression t, and a type T of kind ?
17//!
18//! Constraint generation rules:
19//!
20//! - Equation 1: ⟦x : T⟧ = x ≼ T
21//!   "x has type T if and only if T is an instance of the type scheme associated with x"
22//!   Important part: There is no relation to the typing environment Γ, instead x appears free (and will be bound to Γ later)
23//!
24//! - Equation 2: ⟦λz.t : T⟧ = ∃X1X2.(let z : X1 in ⟦t : X2⟧ ∧ X1 → X2 ≤ T)
25//!   "λz.t has type T if and only if, for some X1 and X2,
26//!   (i) under the assumption that z has type X1, t has type X2, and
27//!   (ii) T is a supertype of X1 → X2."
28//!   z and t types must be fresh (can't generally guess them). They are _existentially_ bound because we are going to
29//!   solve their values. Note that z is _not_ fresh in the condition (i).
30//!
31//! - Equation 3: ⟦t1 t2 : T⟧ = ∃X2.(⟦t1 : X2 → T⟧ ∧ ⟦t2 : X2⟧)
32//!   "t1 t2 has type T if and only if, for some X2, t1 has type X2 → T and t2 has type X2"
33//!
34//! - Equation 4: ⟦let z = t1 in t2 : T⟧ = let z : ∀X[⟦t1 : X⟧].X in ⟦t2 : T⟧
35//!   "let z = t1 in t2 has type T if and only if, under the assumption that z has every type X such that ⟦t1 : X⟧ holds, t2 has type T"
36//!
37//!

50/// Why two types were required to match.
51///
52/// A constraint on its own is a pair of types, which is everything inference needs
53/// and nothing a reader can act on. The reason is what turns "cannot match `Bool`
54/// with `Int`" into something located: *this branch* has type `Bool`, and `Int` is
55/// what was expected *because of this annotation*. Elm calls this a `Reason`, rustc
56/// an `ObligationCause`.
57#[derive(Debug, Clone, Copy, PartialEq, Eq)]
58pub enum Reason {
59    /// A declaration's type annotation, which its body has to satisfy.
60    Annotation,
61    /// A literal's own type: `'a'` is a `Char`.
62    Literal,
63    /// A function's type is its parameter's type arrow its body's type.
64    FunctionShape,
65    /// What is applied has to be a function from the argument's type.
66    Application,
67    /// The condition of an `if` has to be a `Bool`.
68    IfCondition,
69    /// Every branch of an `if` has the type of the whole `if`.
70    IfBranch,
71    /// Every branch of a `case` has the type of the whole `case`.
72    CaseBranch,
73    /// A pattern has to match the type of the expression being matched on.
74    CasePattern,
75    /// A `let` binding has the type of the value bound to it.
76    LetBinding,
77    /// A `let` has the type of its body.
78    LetBody,
79    /// A tuple's type is the tuple of its elements' types.
80    TupleElements,
81}
82
83impl Reason {
84    /// What goes under the caret when the constraint carrying this reason is the one
85    /// that failed: what the underlined text *is*, in the vocabulary of the source.
86    ///
87    /// It names no type, and that is not terseness. By the time a constraint fails,
88    /// `unify` has substituted solutions into both of its sides and may have
89    /// decomposed it into a component of the types the source actually mentions — so
90    /// neither side is reliably "the type of the text under this caret" any more.
91    /// The headline already prints both types; a label that named the wrong one
92    /// would be worse than a label that names none. The example that forced this:
93    /// `answer : Int` with body `true` fails on the literal's own constraint *after*
94    /// `Int` was substituted into it, and reading the type off that side produced
95    /// "this literal has type `Int`".
96    fn describes(&self) -> &'static str {
97        match self {
98            Reason::Annotation => "this type annotation",
99            Reason::Literal => "this literal",
100            Reason::FunctionShape => "this function",
101            Reason::Application => "the expression being applied",
102            Reason::IfCondition => "this condition",
103            Reason::IfBranch => "this branch of the `if`",
104            Reason::CaseBranch => "this branch of the `case`",
105            Reason::CasePattern => "this pattern",
106            Reason::LetBinding => "the value bound here",
107            Reason::LetBody => "the body of this `let`",
108            Reason::TupleElements => "this tuple",
109        }
110    }
111
112    /// What goes under the caret when this reason is not the failure itself but the
113    /// explanation for one side of it — see [`Origin::explanation`].
114    ///
115    /// No type is interpolated here, deliberately. Provenance records which
116    /// constraint brought a type into another one; it does not prove that the type
117    /// printed in the failing constraint is still literally the one this constraint
118    /// carried, and a label is not the place to guess.
119    fn explains(&self) -> &'static str {
120        match self {
121            Reason::Annotation => "expected because of this type annotation",
122            Reason::Literal => "expected because of this literal",
123            Reason::FunctionShape => "expected because of this function",
124            Reason::Application => "expected because of this application",
125            Reason::IfCondition => "expected because this is an `if` condition",
126            Reason::IfBranch => "expected because of this branch",
127            Reason::CaseBranch => "expected because of this branch",
128            Reason::CasePattern => "expected because of this pattern",
129            Reason::LetBinding => "expected because of this value",
130            Reason::LetBody => "expected because of this `let` body",
131            Reason::TupleElements => "expected because of this tuple",
132        }
133    }
134
135    /// The rule that was broken, when naming it says something the labels do not.
136    fn note(&self) -> Option<&'static str> {
137        match self {
138            Reason::Annotation => {
139                Some("a declaration's body must have the type its annotation declares")
140            }
141            Reason::IfCondition => Some("the condition of an `if` must be a `Bool`"),
142            Reason::IfBranch => Some("every branch of an `if` must have the same type"),
143            Reason::CaseBranch => Some("every branch of a `case` must have the same type"),
144            Reason::CasePattern => Some(
145                "every pattern of a `case` must match the type of the expression it matches on",
146            ),
147            _ => None,
148        }
149    }
150}

152/// One piece of source text, and why it required a type: the answer to "where did
153/// this type come from".
154///
155/// Deliberately flat — no chain. A cause names the constraint that *introduced* a
156/// type, never one that relayed it, and that is arranged when the cause is built
157/// rather than by walking a chain afterwards. See `Origin::cause_of`.
158// No `Eq`: `NodeSpan`'s `PartialEq` is deliberately blind (see its documentation), so
159// equality here is a claim about the reason and not about the position.
160#[derive(Debug, Clone, Copy, PartialEq)]
161pub struct Cause {
162    /// Why that constraint required the two types to match.
163    pub reason: Reason,
164    /// The source text it is about.
165    pub span: NodeSpan,
166}
167
168/// Which side of a constraint a type sits on.
169///
170/// Unification is symmetric and does not care; provenance does. Whether a solved
171/// type was read off the left or the right of the constraint that solved it is what
172/// decides whether that constraint is where the type came from, or merely where a
173/// substitution put it — see [`Origin::cause_of`].
174#[derive(Debug, Clone, Copy, PartialEq, Eq)]
175enum Side {
176    Left,
177    Right,
178}
179
180/// Where a constraint came from, and — once inference has moved types around — where
181/// each of its two types came from.
182#[derive(Debug, Clone, PartialEq)]
183pub struct Origin {
184    /// Why the two types were required to match.
185    pub reason: Reason,
186    /// The source text this constraint is about. A failure here draws its caret at
187    /// this span.
188    pub span: NodeSpan,
189    /// The constraint whose solution first rewrote the left type, if any.
190    ///
191    /// `unify` solves a constraint by substituting a type for a variable and then
192    /// applying that substitution to every constraint left. Once that has happened,
193    /// this side holds a type this constraint never mentioned: it came from wherever
194    /// the substitution did. That is the fact a reader needs — "`Int`, because of the
195    /// annotation two lines up" — and it is what the secondary label of a type error
196    /// renders.
197    ///
198    /// The *first* rewrite is kept and later ones are dropped: the first substitution
199    /// to reach a side is the one that brought a foreign type into it, and later ones
200    /// only rewrite what is already there.
201    left_from: Option<Cause>,
202    /// The same, for the right type.
203    right_from: Option<Cause>,
204}
205
206impl Origin {
207    fn new(reason: Reason, span: NodeSpan) -> Origin {
208        Origin {
209            reason,
210            span,
211            left_from: None,
212            right_from: None,
213        }
214    }
215
216    /// This constraint, as the explanation for a type it introduced itself.
217    fn own_cause(&self) -> Cause {
218        Cause {
219            reason: self.reason,
220            span: self.span,
221        }
222    }
223
224    /// Where the type on `side` came from: whatever rewrote that side, or — when
225    /// nothing did — this constraint itself.
226    ///
227    /// This is the whole of the provenance rule, and it is a rule about *sides*.
228    /// When `unify` solves `t := T` from a constraint, `T` was read off one side of
229    /// it. If that side is a type the constraint was written with, the constraint is
230    /// the answer. If a previous solution had rewritten that side, the constraint is
231    /// only relaying a type, and the answer is whatever rewrote it — which is already
232    /// flat, so the credit passes straight through and no chain is ever built.
233    ///
234    /// Crediting a constraint that merely relayed a type is how `result : Bool` /
235    /// `result = not 42` came to blame the annotation: the `Bool` that `42` fails
236    /// against is `not`'s parameter type, which the application constraint carries on
237    /// its *left*, while the annotation had only rewritten its right.
238    fn cause_of(&self, side: Side) -> Cause {
239        let rewritten = match side {
240            Side::Left => self.left_from,
241            Side::Right => self.right_from,
242        };
243
244        rewritten.unwrap_or_else(|| self.own_cause())
245    }
246
247    /// Record that a solution rewrote one side of this constraint, keeping whatever
248    /// was already recorded for that side. See [`Origin::left_from`].
249    fn rewritten(&mut self, side: Side, cause: Cause) {
250        let slot = match side {
251            Side::Left => &mut self.left_from,
252            Side::Right => &mut self.right_from,
253        };
254
255        if slot.is_none() {
256            *slot = Some(cause);
257        }
258    }
259
260    /// Where the type this constraint clashed on came from, when the constraint is
261    /// not itself where it came from.
262    ///
263    /// This is what a diagnostic's secondary label and its rule note are both read
264    /// off. The left side is preferred because that is the declared or expected side
265    /// by convention (see `Constraint`), so "expected because of …" reads about the
266    /// right one; the right side answers when only it was rewritten.
267    pub fn explanation(&self) -> Option<Cause> {
268        self.left_from.or(self.right_from)
269    }
270}

272/// What went wrong, and — for everything unification can raise — where.
273///
274/// Each variant carries the [`Origin`] of the constraint that failed, so a type
275/// error can put its caret under the sub-expression that disagrees instead of across
276/// the declaration containing it. The declaration is still named by [`Error`], one
277/// level up, which is the frame that knows it.
278#[derive(Debug)]
279pub enum ErrorKind {
280    /// Two types were required to match and do not.
281    ///
282    /// The two are written in the order the message reads them out — declared side
283    /// first where the source had one. Neither is reliably "the type of the text the
284    /// origin points at": by the time a constraint fails, unification has substituted
285    /// into both sides and may have decomposed the pair the source actually wrote.
286    // The origins and the types are boxed because an `ErrorKind` is the `Err` half of
287    // a `Result` threaded through the whole of `unify`, and every *successful* return
288    // pays for the size of the largest variant. `Origin` carries a `Cause` per side,
289    // and a `Type` carries a whole `QualName` for every union it names.
290    UnificationFailed {
291        left: Box<Type>,
292        right: Box<Type>,
293        origin: Box<Origin>,
294    },
295    /// A type variable would have to occur inside its own solution.
296    CircularType {
297        /// The type the variable would have had to contain itself in. Boxed for the
298        /// reason above.
299        tpe: Box<Type>,
300        origin: Box<Origin>,
301    },
302    /// A name the typer's environment does not know. See [`type_check`] for why this
303    /// one is not reported to the user today.
304    UnboundVariable {
305        name: String,
306        /// Where the name was written.
307        span: NodeSpan,
308    },
309}
310
311impl ErrorKind {
312    /// The provenance of the constraint that failed, when the failure came from
313    /// unification at all.
314    fn origin(&self) -> Option<&Origin> {
315        match self {
316            ErrorKind::UnificationFailed { origin, .. }
317            | ErrorKind::CircularType { origin, .. } => Some(origin.as_ref()),
318            ErrorKind::UnboundVariable { .. } => None,
319        }
320    }
321
322    fn message(&self) -> String {
323        match self {
324            ErrorKind::UnificationFailed { left, right, .. } => {
325                // Two modules may each declare a union of the same name, and the
326                // source spells both of them bare. Written that way the sentence
327                // uses one word for two declarations and says nothing; the module
328                // that declared each is what tells them apart, so both sides take
329                // it — both, because a sentence that qualifies one side and not the
330                // other reads as if only one of them had a module.
331                if AdtNames::for_all([left.as_ref(), right.as_ref()]) == AdtNames::Qualified {
332                    format!(
333                        "cannot match `{}` with `{}`",
334                        Qualified(left),
335                        Qualified(right)
336                    )
337                } else {
338                    format!("cannot match `{}` with `{}`", left, right)
339                }
340            }
341            // One type, but it can hold the collision on its own: the variable's
342            // solution is built out of whatever it was unified against, which may
343            // be two same-named unions from two modules.
344            ErrorKind::CircularType { tpe, .. } => {
345                let tpe: &dyn std::fmt::Display = match AdtNames::for_all([tpe.as_ref()]) {
346                    AdtNames::Qualified => &Qualified(tpe),
347                    AdtNames::Unqualified => &**tpe,
348                };
349
350                format!(
351                    "circular type: a type variable would have to contain itself in `{}`",
352                    tpe
353                )
354            }
355            ErrorKind::UnboundVariable { name, .. } => {
356                format!("cannot find a value named `{}`", name)
357            }
358        }
359    }
360}

362/// A type error: what went wrong, where, and in which declaration.
363///
364/// # Where the positions come from
365///
366/// The typer does not check the canonical AST directly — it translates it into its
367/// own [`Term`] language — but that translation now carries the canonical node's
368/// [`NodeSpan`] along, and each constraint keeps the span of the term that produced
369/// it. So a unification failure knows the sub-expression it is about, and
370/// [`labels`](PhaseError::labels) draws the caret there.
371///
372/// The declaration's own span is still kept as `span`, for the one case that has no
373/// finer answer: an error whose origin came from a hand-built term, or from a
374/// canonical node the parser never spanned. Then the label degrades to the whole
375/// declaration rather than disappearing.
376#[derive(Debug)]
377pub struct Error {
378    pub kind: ErrorKind,
379    /// Where the declaration the error was found in was written. Used only as the
380    /// fallback described above.
381    pub span: NodeSpan,
382    /// That declaration's name, which is what the note names: in a module of a
383    /// hundred declarations, a caret is not much use without it when the diagnostic
384    /// is rendered without a file (see `compile_package`).
385    pub declaration: Name,
386}
387
388/// Type errors are about types, and [`Type`]'s `Display` writes them the way the
389/// source does — so the message can name both sides of a mismatch instead of
390/// dumping the typer's internal representation.
391impl PhaseError for Error {
392    fn message(&self) -> String {
393        self.kind.message()
394    }
395
396    fn notes(&self) -> Vec<String> {
397        let mut notes = vec![format!("in the declaration of `{}`", self.declaration)];
398
399        // The rule that was broken, taken from the failing constraint and, if that
400        // one has nothing to add, from what explains it. `if`'s "every branch must
401        // have the same type" is on the branch constraint; "a body must have the
402        // type its annotation declares" is on the annotation, which is normally the
403        // explanation rather than the failure.
404        if let Some(origin) = self.kind.origin() {
405            let rule = origin
406                .reason
407                .note()
408                .or_else(|| origin.explanation().and_then(|c| c.reason.note()));
409
410            if let Some(rule) = rule {
411                notes.push(rule.to_owned());
412            }
413        }
414
415        notes
416    }
417
418    /// A caret under the text that disagrees, and — when inference can say where the
419    /// type it disagrees with came from — a second, secondary one under that.
420    ///
421    /// Falls back to underlining the whole declaration when the failing constraint
422    /// has no position, which is what a term built by hand, or one translated from a
423    /// canonical node with no span, produces.
424    fn labels(&self) -> Vec<SpanLabel> {
425        let mut labels = Vec::new();
426
427        match &self.kind {
428            ErrorKind::UnboundVariable { span, .. } => {
429                if let Some(span) = span.span() {
430                    labels.push(SpanLabel {
431                        span,
432                        message: "not found in this scope".to_owned(),
433                        primary: true,
434                        file: None,
435                    });
436                }
437            }
438            kind => {
439                if let Some(origin) = kind.origin() {
440                    if let Some(span) = origin.span.span() {
441                        labels.push(SpanLabel {
442                            span,
443                            message: origin.reason.describes().to_owned(),
444                            primary: true,
445                            file: None,
446                        });
447                    }
448
449                    // Only worth drawing once the primary one exists: on its own it
450                    // would be a caret under the annotation with nothing to contrast
451                    // it against. And not worth drawing at all when it lands on the
452                    // same text — a literal that is its own explanation renders as two
453                    // carets under one word saying the same thing twice.
454                    if let (Some(primary), Some(because)) = (labels.first(), origin.explanation()) {
455                        match because.span.span() {
456                            Some(span) if span != primary.span => labels.push(SpanLabel {
457                                span,
458                                message: because.reason.explains().to_owned(),
459                                primary: false,
460                                file: None,
461                            }),
462                            _ => (),
463                        }
464                    }
465                }
466            }
467        }
468
469        if labels.is_empty() {
470            if let Some(span) = self.span.span() {
471                labels.push(SpanLabel {
472                    span,
473                    message: format!("in `{}`", self.declaration),
474                    primary: true,
475                    file: None,
476                });
477            }
478        }
479
480        labels
481    }
482}

484/// Type check one canonical module, reporting every value whose inference produced a
485/// reportable error rather than stopping at the first — the shape `compile_package` is
486/// built to accumulate.
487///
488/// # The two failures this pass still swallows
489///
490/// Both are in the third pass below, and neither is a statement about the user's
491/// source, which is why neither is reported as an error against it.
492///
493/// **An unsupported construct.** `value_to_term_and_annotation` returns `None` for
494/// anything the term language cannot express — a `VarKernel` or `VarForeign`
495/// reference, a nested pattern inside a `case`. The declaration is then not checked
496/// at all. Reporting that as an error would fail 53 of the declarations in
497/// `std/core/src` that are simply beyond today's inference; it is a gap in the
498/// typer, and what it wants is a warning, which the compiler does not have yet
499/// (`ERR-8`, see `docs/tickets/README.md`).
500///
501/// **An unbound variable.** [`ErrorKind::UnboundVariable`] means the name is missing
502/// from the environment the two passes above build — and that environment is
503/// assembled from *this module alone*: its own annotated values and its own type
504/// constructors. Anything that crossed a module boundary is therefore absent even
505/// though canonicalization resolved it perfectly well: `Maybe.isJust` referring to
506/// `True`, which `Basics` declares, or `Basics.negate` referring to `-`, which the
507/// infix declaration aliases to `sub`. Five declarations in `std/core/src` hit this
508/// today and not one of them is a mistake in the source. A name that genuinely does
509/// not exist is caught earlier, by canonicalization, as
510/// `canonical::Error::VariableNotFound`, with a caret under the name — so nothing a
511/// user can write reaches the user only through this path. The error nonetheless
512/// carries its span now, so the day the typer's environment spans modules it can be
513/// reported without further plumbing.
514///
515/// # Where an [`Error`] is built
516///
517/// Here, and only here: this is the last frame that holds the `canonical::Value`, so
518/// it is the only one that knows the declaration's name and its span. What went
519/// *wrong* and *where inside the declaration* comes up from inference on the
520/// [`ErrorKind`]; see [`Error`].
521pub fn type_check(module: &Module) -> Result<(), Vec<Error>> {
522    // A `module foreign` facade uses synthetic placeholder bodies — skip type checking.
523    if module.binding_foreign {
524        return Ok(());
525    }
526
527    // Start at a high offset to avoid collisions with the counter inside
528    // Types::new() (which starts at 10) used during inference.
529    let mut counter = 10_000u32;
530
531    // First pass: build global env from all TypedValues' declared types.
532    // This allows values to reference other module-level typed values.
533    let mut global: HashMap<String, Type> = HashMap::new();
534    for (name, value) in &module.values {
535        if let canonical::Value::TypedValue { tpe, .. } = value {
536            let mut var_map = HashMap::new();
537            if let Some(typer_tpe) = canonical_type_to_typer_type(tpe, &mut var_map, &mut counter) {
538                // Add both qualified (e.g. "Test.not") and unqualified (e.g. "not") names
539                let qname = module
540                    .name
541                    .qualify_name(name)
542                    .to_name()
543                    .as_str()
544                    .to_string();
545                global.insert(qname, typer_tpe.clone());
546                global.insert(name.as_str().to_string(), typer_tpe);
547            }
548        }
549    }
550
551    // This module's unions, keyed by the qualified name that identifies each
552    // declaration rather than by the spelling the `type` line wrote. A constructor
553    // carries the qualified name of the type it builds, so this key is what decides
554    // whether the constructor in a pattern is one of *these* declarations' — see
555    // `translate_pattern`.
556    let module_types: ModuleTypes = module
557        .types
558        .iter()
559        .map(|(name, union_type)| (module.name.qualify_name(name), union_type))
560        .collect();
561
562    // Second pass: add constructor types to global from this module's unions
563    for (type_name, union_type) in &module_types {
564        // Fresh type vars for each ADT type parameter (e.g. "a" in Maybe a)
565        let mut adt_var_map: HashMap<String, TypeVariable> = HashMap::new();
566        for tv_name in &union_type.variables {
567            counter += 1;
568            adt_var_map.insert(tv_name.as_str().to_string(), TypeVariable { id: counter });
569        }
570
571        // Build result type: Adt(type_name, [TypeVar for each param])
572        let result_args: Vec<Type> = union_type
573            .variables
574            .iter()
575            .map(|v| Type::Variable(adt_var_map[v.as_str()].clone()))
576            .collect();
577        let result_type = Type::Adt(type_name.clone(), result_args);
578
579        for ctor in &union_type.variants {
580            let ctor_type = if ctor.type_parameters.is_empty() {
581                result_type.clone()
582            } else {
583                let mut translate_var_map = adt_var_map.clone();
584                let params: Vec<Type> = ctor
585                    .type_parameters
586                    .iter()
587                    .filter_map(|t| {
588                        canonical_type_to_typer_type(t, &mut translate_var_map, &mut counter)
589                    })
590                    .collect();
591                if params.len() != ctor.type_parameters.len() {
592                    continue; // untranslatable param type, skip this constructor
593                }
594                // Build Fun type: p1 -> p2 -> ... -> result_type
595                params
596                    .into_iter()
597                    .rev()
598                    .fold(result_type.clone(), |acc, p| Type::Fun {
599                        param_tpe: Box::new(p),
600                        return_tpe: Box::new(acc),
601                    })
602            };
603
604            // Register under both unqualified ("Just") and qualified ("Test.Just") names
605            global.insert(ctor.name.as_str().to_string(), ctor_type.clone());
606            let qname = module
607                .name
608                .qualify_name(&ctor.name)
609                .to_name()
610                .as_str()
611                .to_string();
612            global.insert(qname, ctor_type);
613        }
614    }
615
616    // Third pass: check each value. A value that fails is recorded and the pass
617    // moves on, so one broken declaration cannot hide the others.
618    let mut errors: Vec<Error> = vec![];
619    for (name, value) in &module.values {
620        let Some((term, annotation)) =
621            value_to_term_and_annotation(value, &module_types, &mut counter)
622        else {
623            continue; // unsupported construct — see this function's documentation
624        };
625
626        // The annotation is handed to inference rather than checked against its
627        // result afterwards, which is what lets a mismatch *inside* the body know
628        // that the type it failed against came from the annotation. Checking the two
629        // whole types at the end could only ever say "this declaration is `Int` and
630        // its body is something else", with the caret across the lot.
631        match infer_annotated(term, global.clone(), annotation) {
632            // See this function's documentation: an unbound variable here is a hole
633            // in the typer's environment, not a mistake in the source.
634            Err(ErrorKind::UnboundVariable { .. }) => continue,
635            Err(kind) => errors.push(Error {
636                kind,
637                span: value.span(),
638                declaration: name.clone(),
639            }),
640            Ok(_) => (),
641        }
642    }
643
644    if errors.is_empty() {
645        Ok(())
646    } else {
647        Err(errors)
648    }
649}

662/// Convert a canonical type to the typer's simplified Type representation.
663/// The match covers all four `canonical::Type` variants — `Variable`, `Arrow`, `Tuple`
664/// (either arity), and `Type` including named types with parameters — and every arm's
665/// own base case returns `Some`; a `None` only ever arises by propagating up from a
666/// nested recursive call. As of today no `canonical::Type` shape actually reaches such
667/// a case, so the function always returns `Some`. The `Option` return stays in place for
668/// when a genuinely unrepresentable variant (e.g. records, aliases) is added.
669///
670/// `var_map` maps named type variables (e.g. "a") to consistent TypeVariable
671/// ids, so that `a -> a` produces the same variable on both sides.
672///
673/// # How a name decides which type it becomes
674///
675/// A [`canonical::Type::Type`] names its declaration in full — `Widget.Size`, not
676/// `Size` (`AST-4`) — and the whole of that name is what picks the arm. A nullary
677/// type whose qualified name is [a scalar the compiler
678/// knows](super::scalars) becomes the matching [`Type::Literal`]; everything else
679/// becomes a [`Type::Adt`], carrying that name whole.
680///
681/// For a name that resolved, that is the declaring module's. For one that resolved to
682/// nothing it is the module under check, applied to the whole *written* spelling, dots
683/// and all, because `Type::from_parser_type` hands the undivided name to
684/// `ModuleName::qualify_name` rather than splitting a module half off it — so an
685/// unresolved `Missing.Thing` written in `Test` arrives here as `Test.Missing.Thing`
686/// and does not collapse onto the local `Test.Thing`.
687fn canonical_type_to_typer_type(
688    tpe: &canonical::Type,
689    var_map: &mut HashMap<String, TypeVariable>,
690    counter: &mut u32,
691) -> Option<Type> {
692    match tpe {
693        canonical::Type::Variable(name) => {
694            let tv = var_map.entry(name.as_str().to_string()).or_insert_with(|| {
695                *counter += 1;
696                TypeVariable { id: *counter }
697            });
698            Some(Type::Variable(tv.clone()))
699        }
700        canonical::Type::Arrow(a, b) => {
701            let a = canonical_type_to_typer_type(a, var_map, counter)?;
702            let b = canonical_type_to_typer_type(b, var_map, counter)?;
703            Some(Type::Fun {
704                param_tpe: Box::new(a),
705                return_tpe: Box::new(b),
706            })
707        }
708        // `Tuple::try_map` keeps the arity attached to the value instead of
709        // re-deriving it here, the same way the parser → canonical conversions
710        // in `canonical/mod.rs` do.
711        canonical::Type::Tuple(tuple) => {
712            let elements = tuple
713                .try_map(|elem| canonical_type_to_typer_type(elem, var_map, counter).ok_or(()))
714                .ok()?;
715            Some(Type::Tuple(elements))
716        }
717        canonical::Type::Type(name, args) => {
718            if args.is_empty() {
719                if let Some(literal) = scalar_literal(name) {
720                    return Some(Type::Literal(literal));
721                }
722            }
723
724            let converted: Option<Vec<Type>> = args
725                .iter()
726                .map(|a| canonical_type_to_typer_type(a, var_map, counter))
727                .collect();
728            Some(Type::Adt(name.clone(), converted?))
729        }
730    }
731}

733/// The literal type the typer gives a [scalar](super::scalars), if `name` is the
734/// qualified name of one it has a literal type for.
735///
736/// Three of the five appear here. The other two do not:
737///
738/// - [`scalars::BOOL`] is a scalar *and* an ordinary union, so `Bool` in an annotation
739///   takes the [`Type::Adt`] path every other declaration takes and meets `True` and
740///   `False` there. [`bool_type`] is the same type, built for the three `Bool`s no
741///   source spells.
742/// - [`scalars::STRING`] would gain a [`TypeLiteral`] variant the day there is a string
743///   literal to give a type to; a variant nothing constructs would be a type the
744///   unifier could name in an error and no source could produce.
745fn scalar_literal(name: &QualName) -> Option<TypeLiteral> {
746    const LITERALS: &[(scalars::Scalar, TypeLiteral)] = &[
747        (scalars::INT, TypeLiteral::Int),
748        (scalars::FLOAT, TypeLiteral::Float),
749        (scalars::CHAR, TypeLiteral::Char),
750    ];
751
752    LITERALS
753        .iter()
754        .find(|(scalar, _)| scalar.declares(name))
755        .map(|(_, literal)| literal.clone())
756}

758/// The type of a `Bool`: the union [`scalars::BOOL`] names, with no arguments.

760/// `Bool` is [a scalar and an ordinary union at

775/// Convert a canonical expression to a Term, keeping the position it was written at.
776///
777/// Returns None for constructs the inference engine doesn't yet handle
778/// (VarKernel, VarForeign, complex patterns inside Case).
779///
780/// Every arm attaches `expr.span` to the term it builds. That is the whole of what
781/// `ERR-4` needed from this function: a constraint can only point at a
782/// sub-expression if the term that produced it remembers where it came from.
783fn canonical_expr_to_term(
784    expr: &canonical::Expression,
785    module_types: &ModuleTypes,
786    counter: &mut u32,
787) -> Option<Term> {
788    let kind = match &expr.kind {
789        canonical::ExpressionKind::Int(i) => TermKind::Int(*i as u32),
790        canonical::ExpressionKind::Bool(b) => TermKind::Bool(*b),
791        canonical::ExpressionKind::Char(c) => TermKind::Char(*c),
792        canonical::ExpressionKind::Float(f) => TermKind::Float(*f),
793        canonical::ExpressionKind::VarLocal(name) => {
794            TermKind::Identifier(name.as_str().to_string())
795        }
796        canonical::ExpressionKind::VarTopLevel(qname) => {
797            TermKind::Identifier(qname.to_name().as_str().to_string())
798        }
799        canonical::ExpressionKind::Apply(f, a) => {
800            let fun = canonical_expr_to_term(f, module_types, counter)?;
801            let arg = canonical_expr_to_term(a, module_types, counter)?;
802            TermKind::Apply {
803                fun: Box::new(fun),
804                arg: Box::new(arg),
805            }
806        }
807        canonical::ExpressionKind::If(cond, t, f) => {
808            let cond = canonical_expr_to_term(cond, module_types, counter)?;
809            let t = canonical_expr_to_term(t, module_types, counter)?;
810            let f = canonical_expr_to_term(f, module_types, counter)?;
811            TermKind::If {
812                cond: Box::new(cond),
813                true_branch: Box::new(t),
814                false_branch: Box::new(f),
815            }
816        }
817        canonical::ExpressionKind::Tuple(tuple) => {
818            let elements = tuple
819                .try_map(|elem| canonical_expr_to_term(elem, module_types, counter).ok_or(()))
820                .ok()?;
821            TermKind::Tuple(elements)
822        }
823        canonical::ExpressionKind::Case(scrutinee_expr, branches) => {
824            let scrutinee = canonical_expr_to_term(scrutinee_expr, module_types, counter)?;
825            let term_branches: Vec<(TermPattern, Box<Term>)> = branches
826                .iter()
827                .map(|cb| {
828                    let (pattern, _bindings) =
829                        translate_pattern(&cb.pattern, module_types, counter)?;
830                    let body = canonical_expr_to_term(&cb.expression, module_types, counter)?;
831                    Some((pattern, Box::new(body)))
832                })
833                .collect::<Option<Vec<_>>>()?;
834            TermKind::Case {
835                scrutinee: Box::new(scrutinee),
836                branches: term_branches,
837            }
838        }
839        // Constructors are resolved as identifiers looked up in the global env.
840        canonical::ExpressionKind::VarConstructor(qname, _) => {
841            TermKind::Identifier(qname.to_name().as_str().to_string())
842        }
843        // VarForeign: not in the module's global env, skip
844        canonical::ExpressionKind::VarForeign(_, _) => return None,
845        // Not yet supported: VarKernel
846        _ => return None,
847    };
848
849    Some(Term {
850        span: expr.span,
851        kind,
852    })
853}

855/// Translate a canonical pattern into a `TermPattern` plus any variable bindings
856/// introduced by the pattern.  Returns `None` for unsupported pattern shapes.
857///
858/// The pattern keeps its own span, separate from the branch body's: a `case` branch
859/// whose pattern does not match what is being matched on is about the pattern, and
860/// the caret belongs there rather than under the expression in the `case … of` line.
861fn translate_pattern(
862    pattern: &canonical::Pattern,
863    module_types: &ModuleTypes,
864    counter: &mut u32,
865) -> Option<(TermPattern, Vec<(String, Type)>)> {
866    let (kind, bindings) = match &pattern.kind {
867        canonical::PatternKind::Anything => (TermPatternKind::Anything, vec![]),
868        canonical::PatternKind::Variable(name) => {
869            // The binding's actual type will be unified with the scrutinee type in annotate.
870            (TermPatternKind::Bind(name.as_str().to_string()), vec![])
871        }
872        canonical::PatternKind::Bool(_) => (TermPatternKind::Literal(bool_type()), vec![]),
873        canonical::PatternKind::Int(_) => (
874            TermPatternKind::Literal(Type::Literal(TypeLiteral::Int)),
875            vec![],
876        ),
877        canonical::PatternKind::Char(_) => (
878            TermPatternKind::Literal(Type::Literal(TypeLiteral::Char)),
879            vec![],
880        ),
881        canonical::PatternKind::Constructor { ctor, args } => {
882            // Look up the parent union to get its type variables. `ctor.tpe` names
883            // the declaration the constructor builds, module included, and
884            // `module_types` is keyed the same way — so this finds a union only when
885            // the constructor is one of this module's own. A constructor of an
886            // imported type finds nothing, and the whole declaration goes unchecked;
887            // giving the typer the imported unions is `BUG-36`.
888            let union_type = module_types.get(&ctor.tpe)?;
889
890            // Create fresh type vars for each ADT type parameter.
891            let mut adt_var_map: HashMap<String, TypeVariable> = HashMap::new();
892            for tv_name in &union_type.variables {
893                *counter += 1;
894                adt_var_map.insert(tv_name.as_str().to_string(), TypeVariable { id: *counter });
895            }
896
897            // Build the ADT result type args from the fresh vars.
898            let adt_args: Vec<Type> = union_type
899                .variables
900                .iter()
901                .map(|v| Type::Variable(adt_var_map[v.as_str()].clone()))
902                .collect();
903
904            // Translate each constructor type parameter (reuses the same fresh vars).
905            let param_types: Vec<Type> = ctor
906                .type_parameters
907                .iter()
908                .filter_map(|t| canonical_type_to_typer_type(t, &mut adt_var_map, counter))
909                .collect();
910            if param_types.len() != ctor.type_parameters.len() {
911                return None;
912            }
913
914            // Build bindings from arg patterns.
915            let mut bindings: Vec<(String, Type)> = vec![];
916            for (arg_pattern, param_type) in args.iter().zip(param_types.iter()) {
917                match &arg_pattern.kind {
918                    canonical::PatternKind::Variable(name) => {
919                        bindings.push((name.as_str().to_string(), param_type.clone()));
920                    }
921                    canonical::PatternKind::Anything => {} // no binding needed
922                    _ => return None, // nested complex patterns not yet supported
923                }
924            }
925
926            let kind = TermPatternKind::Constructor {
927                adt_name: ctor.tpe.clone(),
928                adt_args,
929                bindings: bindings.clone(),
930            };
931            (kind, bindings)
932        }
933        _ => return None, // Tuple, Float patterns — not yet supported
934    };
935
936    Some((
937        TermPattern {
938            span: pattern.span,
939            kind,
940        },
941        bindings,
942    ))
943}

956/// Convert a canonical Value into a (Term, optional annotation) pair.
957/// The body is wrapped in nested Fun nodes for each pattern parameter.
958/// Returns None if any part of the value cannot be translated.
959fn value_to_term_and_annotation(
960    value: &canonical::Value,
961    module_types: &ModuleTypes,
962    counter: &mut u32,
963) -> Option<(Term, Option<Annotation>)> {
964    match value {
965        canonical::Value::Value { patterns, body, .. } => {
966            let body_term = canonical_expr_to_term(body, module_types, counter)?;
967            let term = wrap_with_patterns(patterns.iter(), body_term)?;
968            Some((term, None))
969        }
970        canonical::Value::TypedValue {
971            patterns,
972            body,
973            tpe,
974            annotation_span,
975            ..
976        } => {
977            let body_term = canonical_expr_to_term(body, module_types, counter)?;
978            let pattern_iter = patterns.iter().map(|(p, _)| p);
979            let term = wrap_with_patterns(pattern_iter, body_term)?;
980            let mut var_map = HashMap::new();
981            let annotation =
982                canonical_type_to_typer_type(tpe, &mut var_map, counter).map(|tpe| Annotation {
983                    tpe,
984                    span: *annotation_span,
985                });
986            Some((term, annotation))
987        }
988    }
989}

991/// Wrap a body Term in nested Fun nodes for each pattern, outermost first.
992/// Returns None if any pattern is not translatable (e.g. constructor patterns).
993///
994/// Each `Fun` spans its parameter through the body it wraps, so a function whose
995/// declared shape does not match its definition is underlined from the parameter
996/// that starts it rather than across the annotation as well.
997fn wrap_with_patterns<'a>(
998    patterns: impl Iterator<Item = &'a canonical::Pattern>,
999    body: Term,
1000) -> Option<Term> {
1001    let names: Vec<(String, NodeSpan)> = patterns
1002        .map(|p| match &p.kind {
1003            canonical::PatternKind::Variable(name) => Some((name.as_str().to_string(), p.span)),
1004            canonical::PatternKind::Anything => Some(("_".to_string(), p.span)),
1005            _ => None,
1006        })
1007        .collect::<Option<Vec<_>>>()?;
1008
1009    let term = names.iter().rev().fold(body, |acc, (param, span)| Term {
1010        span: span.merge(acc.span),
1011        kind: TermKind::Fun {
1012            param: param.clone(),
1013            body: Box::new(acc),
1014        },
1015    });
1016
1017    Some(term)
1018}

1171/// How the name of a [`Type::Adt`] is written out.
1172///
1173/// A type is normally quoted the way the source spells it, which for a union is its
1174/// bare name. That is ambiguous exactly when one message names two declarations that
1175/// share a spelling, and [`ErrorKind::message`] switches to the qualified form there.
1176#[derive(Debug, Clone, Copy, PartialEq, Eq)]
1177enum AdtNames {
1178    /// `Size`.
1179    Unqualified,
1180    /// `Widget.Size`.
1181    Qualified,
1182}
1183
1184impl AdtNames {
1185    /// How the types a single message is about have to be written for that message
1186    /// to distinguish the declarations it names.
1187    ///
1188    /// The question is not whether the *types* are spelled alike — `A.Size` against
1189    /// `Lib.Size A.Size` renders as `Size` against `Size Size`, which differs as
1190    /// text while still using one word for three declarations. It is whether any two
1191    /// of the unions named anywhere in those types are different declarations with
1192    /// the same bare name; if so every union in the sentence is qualified, since
1193    /// qualifying only the colliding pair would read as if the rest had no module.
1194    fn for_all<'a>(types: impl IntoIterator<Item = &'a Type>) -> AdtNames {
1195        let mut names: Vec<&QualName> = Vec::new();
1196
1197        for tpe in types {
1198            tpe.collect_adt_names(&mut names);
1199        }
1200
1201        let collides = names.iter().enumerate().any(|(i, name)| {
1202            names[i + 1..]
1203                .iter()
1204                .any(|other| *other != *name && other.unqualified_name() == name.unqualified_name())
1205        });
1206
1207        if collides {
1208            AdtNames::Qualified
1209        } else {
1210            AdtNames::Unqualified
1211        }
1212    }
1213}

1257impl Type {
1258    /// Every union named anywhere in this type, outermost first, appended to `out`.
1259    ///
1260    /// A union's arguments are types in their own right and may name unions of their
1261    /// own, so this recurses rather than reading the head alone. [`AdtNames::for_all`]
1262    /// is what it exists for: deciding how to write a type means looking at every
1263    /// name the rendering will contain, not only the one at the top.
1264    fn collect_adt_names<'a>(&'a self, out: &mut Vec<&'a QualName>) {
1265        match self {
1266            Type::Literal(_) | Type::Number | Type::Variable(_) => {}
1267            Type::Fun {
1268                param_tpe,
1269                return_tpe,
1270            } => {
1271                param_tpe.collect_adt_names(out);
1272                return_tpe.collect_adt_names(out);
1273            }
1274            Type::Tuple(Tuple::Two(a, b)) => {
1275                a.collect_adt_names(out);
1276                b.collect_adt_names(out);
1277            }
1278            Type::Tuple(Tuple::Three(a, b, c)) => {
1279                a.collect_adt_names(out);
1280                b.collect_adt_names(out);
1281                c.collect_adt_names(out);
1282            }
1283            Type::Adt(name, args) => {
1284                out.push(name);
1285                for arg in args {
1286                    arg.collect_adt_names(out);
1287                }
1288            }
1289        }
1290    }
1291
1292    /// The body of both renderings: the same text either way, except for how a union
1293    /// is named. See [`AdtNames`], and [`Display`](std::fmt::Display) for why unions
1294    /// are normally written bare.
1295    fn write(&self, f: &mut std::fmt::Formatter<'_>, names: AdtNames) -> std::fmt::Result {
1296        /// The same, wrapped in parentheses.
1297        fn parenthesised(
1298            tpe: &Type,
1299            f: &mut std::fmt::Formatter<'_>,
1300            names: AdtNames,
1301        ) -> std::fmt::Result {
1302            write!(f, "(")?;
1303            tpe.write(f, names)?;
1304            write!(f, ")")
1305        }
1306
1307        match self {
1308            Type::Literal(TypeLiteral::Int) => write!(f, "Int"),
1309            Type::Literal(TypeLiteral::Char) => write!(f, "Char"),
1310            Type::Literal(TypeLiteral::Float) => write!(f, "Float"),
1311            Type::Number => write!(f, "number"),
1312            Type::Variable(TypeVariable { id }) => write!(f, "t{}", id),
1313            // The parameter of a function type is parenthesised when it is itself a
1314            // function, because `->` is right-associative: `(a -> b) -> c` and
1315            // `a -> b -> c` are different types.
1316            Type::Fun {
1317                param_tpe,
1318                return_tpe,
1319            } => {
1320                match **param_tpe {
1321                    Type::Fun { .. } => parenthesised(param_tpe, f, names)?,
1322                    _ => param_tpe.write(f, names)?,
1323                }
1324                write!(f, " -> ")?;
1325                return_tpe.write(f, names)
1326            }
1327            Type::Tuple(Tuple::Two(a, b)) => {
1328                write!(f, "( ")?;
1329                a.write(f, names)?;
1330                write!(f, ", ")?;
1331                b.write(f, names)?;
1332                write!(f, " )")
1333            }
1334            Type::Tuple(Tuple::Three(a, b, c)) => {
1335                write!(f, "( ")?;
1336                a.write(f, names)?;
1337                write!(f, ", ")?;
1338                b.write(f, names)?;
1339                write!(f, ", ")?;
1340                c.write(f, names)?;
1341                write!(f, " )")
1342            }
1343            Type::Adt(name, args) => {
1344                match names {
1345                    AdtNames::Unqualified => write!(f, "{}", name.unqualified_name())?,
1346                    AdtNames::Qualified => write!(f, "{}", name.to_name())?,
1347                }
1348                for arg in args {
1349                    // Same reason as above: an argument that is itself applied or a
1350                    // function needs parentheses to stay the same type when re-read.
1351                    write!(f, " ")?;
1352                    match arg {
1353                        Type::Adt(_, inner) if !inner.is_empty() => parenthesised(arg, f, names)?,
1354                        Type::Fun { .. } => parenthesised(arg, f, names)?,
1355                        _ => arg.write(f, names)?,
1356                    }
1357                }
1358                Ok(())
1359            }
1360        }
1361    }
1362}

1421/// Two types that have to match, and why.
1422///
1423/// # What the two sides mean
1424///
1425/// Unification treats them symmetrically; the order is a rendering convention, and
1426/// the only thing that reads it is the headline — *cannot match `left` with `right`*.
1427/// `collect` therefore writes the declared or expected side first where the source
1428/// has one (an annotation's type, a pattern's type) and the inferred side second, so
1429/// the sentence comes out in the order a reader expects.
1430///
1431/// What the sides are emphatically *not* is a way to tell which type belongs to the
1432/// text at `origin.span`. By the time a constraint fails, `unify` has substituted
1433/// other constraints' solutions into both sides, and may have decomposed it into a
1434/// component of the types the source mentioned. That is why the labels name the
1435/// [`Reason`] and leave the types to the headline — see [`Reason::describes`].
1436///
1437/// # Why these are held in a `Vec` and not a `HashSet`
1438///
1439/// They used to be a `HashSet<Constraint>`, back when a constraint was a bare pair of
1440/// types. An origin makes that collection wrong twice over. Deduplication now
1441/// discards *provenance*: two constraints with equal types but different origins are
1442/// one entry, and which origin survives is whichever was inserted first. And the
1443/// order a `HashSet` yields is unspecified, so which constraint `unify` reaches first
1444/// — and therefore which one is reported when several are unsatisfiable — would vary
1445/// between runs of the same compiler on the same file.
1446///
1447/// A `Vec` fixes both, and buys a third thing: source order. The annotation is
1448/// pushed first, so its type is substituted into the body's constraints before they
1449/// are solved, which is what lets a mismatch deep in the body say that `Int` came
1450/// from the annotation. The cost is that duplicate constraints are no longer
1451/// collapsed, which is a few more `unify` steps on terms that repeat a type.
1452#[derive(Debug, Clone, PartialEq)]
1453struct Constraint {
1454    left: Type,
1455    right: Type,
1456    origin: Origin,
1457}
1458
1459impl Constraint {
1460    fn new(left: Type, right: Type, reason: Reason, span: NodeSpan) -> Constraint {
1461        Constraint {
1462            left,
1463            right,
1464            origin: Origin::new(reason, span),
1465        }
1466    }
1467
1468    /// A constraint between two components of this one — the parameters of two
1469    /// function types being matched, say — which is about the same source text and
1470    /// was required for the same reason.
1471    ///
1472    /// The origin is inherited whole, side provenance included: if a substitution
1473    /// rewrote the left type, it rewrote whatever the left type decomposes into. That
1474    /// is an over-approximation — a substitution that reached only the return half of
1475    /// an arrow is credited with the parameter half too — but it errs towards naming
1476    /// a constraint that did carry a type in, which is the direction that keeps a
1477    /// caret on the source rather than on the compiler's working.
1478    fn component(&self, left: Type, right: Type) -> Constraint {
1479        Constraint {
1480            left,
1481            right,
1482            origin: self.origin.clone(),
1483        }
1484    }
1485}

1487/// What one type variable was solved to, and where that type came from.
1488///
1489/// The cause is not used by inference at all. It is carried so that when this
1490/// solution is substituted into another constraint and *that* constraint then fails,
1491/// the failure can say where the type it failed against came from — see
1492/// [`Origin::left_from`].
1493///
1494/// It is a [`Cause`] and not an `Origin` because the question it answers is already
1495/// settled: [`Origin::cause_of`] resolved, at the moment the variable was solved,
1496/// whether the solving constraint introduced this type or was handed it. Nothing
1497/// downstream has to walk anything.
1498// No `Eq`: a `Cause` holds a `NodeSpan`, whose `PartialEq` is deliberately blind
1499// (see its documentation), so equality here is a claim about the types and the
1500// reasons, not about the positions.
1501#[derive(Debug, PartialEq, Clone)]
1502struct Solution {
1503    tpe: Type,
1504    cause: Cause,
1505}
1506
1507#[derive(Debug, PartialEq)]
1508struct Substitution {
1509    solutions: HashMap<TypeVariable, Solution>,
1510}
1511
1512impl Substitution {
1513    // constructors
1514
1515    fn empty() -> Substitution {
1516        Substitution {
1517            solutions: HashMap::new(),
1518        }
1519    }
1520
1521    fn one(tvar: TypeVariable, tpe: Type, cause: Cause) -> Substitution {
1522        let mut sub = Substitution::empty();
1523
1524        sub.solutions.insert(tvar, Solution { tpe, cause });
1525
1526        sub
1527    }
1528
1529    // methods
1530
1531    /// Rewrite a constraint with everything solved so far, recording for each side
1532    /// which solution first reached it.
1533    ///
1534    /// The two sides are tracked apart, because that is what tells a relayed type
1535    /// from an introduced one later on — see [`Origin::cause_of`]. A solution that
1536    /// rewrites only the right side has said nothing about the left, and treating it
1537    /// as though it had is what made a type error blame an annotation that was not
1538    /// load-bearing.
1539    ///
1540    /// The solutions are visited in type-variable order rather than in `HashMap`
1541    /// order: "first" has to mean the same thing on every run, or the secondary label
1542    /// of a diagnostic would move between compilations of an unchanged file. Ids are
1543    /// handed out as inference walks the term, so that order is roughly the order the
1544    /// user wrote things in.
1545    fn apply(&self, c: &Constraint) -> Constraint {
1546        let mut origin = c.origin.clone();
1547
1548        let mut solutions: Vec<_> = self.solutions.iter().collect();
1549        solutions.sort_by_key(|(tvar, _)| tvar.id);
1550
1551        for (tvar, solution) in solutions {
1552            // A solution that does not mention a variable a side uses rewrites
1553            // nothing there, and explains nothing about it either.
1554            if occurs(tvar, &c.left) {
1555                origin.rewritten(Side::Left, solution.cause);
1556            }
1557            if occurs(tvar, &c.right) {
1558                origin.rewritten(Side::Right, solution.cause);
1559            }
1560        }
1561
1562        Constraint {
1563            left: self.apply_type(&c.left),
1564            right: self.apply_type(&c.right),
1565            origin,
1566        }
1567    }
1568
1569    fn apply_type(&self, tpe: &Type) -> Type {
1570        self.solutions
1571            .iter()
1572            .fold(tpe.clone(), |tpe, (tvar, solution)| {
1573                Substitution::substitute(tpe, tvar, &solution.tpe)
1574            })
1575    }
1576
1577    fn substitute(tpe: Type, tvar: &TypeVariable, replacement: &Type) -> Type {
1578        match tpe {
1579            Type::Literal(_) | Type::Number => tpe,
1580            Type::Fun {
1581                param_tpe,
1582                return_tpe,
1583            } => Type::Fun {
1584                param_tpe: Box::new(Substitution::substitute(*param_tpe, tvar, replacement)),
1585                return_tpe: Box::new(Substitution::substitute(*return_tpe, tvar, replacement)),
1586            },
1587            Type::Tuple(Tuple::Two(a, b)) => Type::Tuple(Tuple::two(
1588                Substitution::substitute(*a, tvar, replacement),
1589                Substitution::substitute(*b, tvar, replacement),
1590            )),
1591            Type::Tuple(Tuple::Three(a, b, c)) => Type::Tuple(Tuple::three(
1592                Substitution::substitute(*a, tvar, replacement),
1593                Substitution::substitute(*b, tvar, replacement),
1594                Substitution::substitute(*c, tvar, replacement),
1595            )),
1596            Type::Adt(name, args) => Type::Adt(
1597                name,
1598                args.into_iter()
1599                    .map(|a| Substitution::substitute(a, tvar, replacement))
1600                    .collect(),
1601            ),
1602            Type::Variable(tvar2) if tvar == &tvar2 => replacement.clone(),
1603            tpe @ Type::Variable(_) => tpe,
1604        }
1605    }
1606
1607    fn merge(&self, other: Substitution) -> Substitution {
1608        // This merge means we should try sub_tail first, and then sub_head
1609        // Merging other in self means we apply `other` substitution to `self` solutions
1610        // When merging, we want `other` solutions to take precedences over `self` solutions
1611
1612        let self_solutions = self.solutions.iter().map(|(k, v)| {
1613            (
1614                k.clone(),
1615                Solution {
1616                    tpe: other.apply_type(&v.tpe),
1617                    // The cause says where this variable's type came from, which
1618                    // rewriting the type does not change.
1619                    cause: v.cause,
1620                },
1621            )
1622        });
1623
1624        let mut sub = Substitution::empty();
1625
1626        sub.solutions.extend(self_solutions);
1627        sub.solutions.extend(other.solutions);
1628
1629        sub
1630    }
1631}

1633/// Does `tvar` appear anywhere inside `tpe`?
1634///
1635/// Two callers, for two different reasons: `unify_variable` uses it as the occurs
1636/// check that keeps it from building an infinite type, and `Substitution::apply` uses
1637/// it to tell whether a solution actually rewrites a given constraint — which is what
1638/// decides whether that solution explains one of the constraint's types.
1639fn occurs(tvar: &TypeVariable, tpe: &Type) -> bool {
1640    match tpe {
1641        Type::Fun {
1642            param_tpe,
1643            return_tpe,
1644        } => occurs(tvar, param_tpe) || occurs(tvar, return_tpe),
1645        Type::Tuple(tuple) => tuple.iter().any(|t| occurs(tvar, t)),
1646        Type::Adt(_, args) => args.iter().any(|a| occurs(tvar, a)),
1647        Type::Variable(tvar2) => tvar == tvar2,
1648        _ => false,
1649    }
1650}

1690/// This is a translation of the algorithm demonstrated by
1691/// [Ionut Gan at I T.A.K.E Unconference 2015](https://www.youtube.com/watch?v=oPVTNxiMcSU)

1695
1696/// [`infer`], with the declaration's type annotation as a constraint of its own.
1697///
1698/// The annotation is put *first*, before the constraints the body generates, and that
1699/// ordering is the point of the function. `unify` solves constraints in order, so the
1700/// annotated type is substituted into the body's constraints before any of them are
1701/// solved; when one of them then fails, its [`Origin::explanation`] names the annotation,
1702/// and the diagnostic can say `Int` was expected *because of the annotation* rather
1703/// than merely that the declaration as a whole does not check.
1704fn infer_annotated(
1705    term: Term,
1706    global: HashMap<String, Type>,
1707    annotation: Option<Annotation>,
1708) -> Result<Type, ErrorKind> {
1709    let mut env = Types::new();
1710    env.extends_with(global);
1711
1712    let typed_term = annotate::annotate(term, &mut env)?;
1713    debug!("typed term: {:#?}", typed_term);
1714
1715    let mut constraints = Vec::new();
1716
1717    if let Some(annotation) = annotation {
1718        // Left is the annotation's type, because left is the type of the text the
1719        // span points at — see `Constraint`.
1720        constraints.push(Constraint::new(
1721            annotation.tpe,
1722            typed_term.tpe.clone(),
1723            Reason::Annotation,
1724            annotation.span,
1725        ));
1726    }
1727
1728    constraints.extend(constraint::collect(&typed_term));
1729    debug!("Constraints: {:#?}", constraints);
1730
1731    let substitution = unifier::unify(constraints)?;
1732
1733    Ok(substitution.apply_type(&typed_term.tpe))
1734}

1801                Type::Literal(TypeLiteral::Char) => "Char".to_owned(),
1802                Type::Literal(TypeLiteral::Float) => "Float".to_owned(),
1803                Type::Number => "number".to_owned(),
1804                Type::Variable(TypeVariable { id }) => {
1805                    if let Some(name) = self.known.get(&id) {
1806                        name.clone()
1807                    } else {
1808                        let name = self.counter_as_letter();
1809                        self.counter += 1;
1810
1811                        self.known.insert(id, name.clone());
1812
1813                        name
1814                    }
1815                }
1816                Type::Fun {
1817                    param_tpe,
1818                    return_tpe,
1819                } => {
1820                    let is_param_fun = matches!(param_tpe.as_ref(), Type::Fun { .. });
1821                    let param = self.type_signature(*param_tpe);
1822                    let retur = self.type_signature(*return_tpe);
1823
1824                    if is_param_fun {
1825                        format!("({}) -> {}", param, retur)
1826                    } else {
1827                        format!("{} -> {}", param, retur)
1828                    }
1829                }
1830                Type::Tuple(Tuple::Two(a, b)) => {
1831                    format!("({}, {})", self.type_signature(*a), self.type_signature(*b))
1832                }
1833                Type::Tuple(Tuple::Three(a, b, c)) => {
1834                    format!(
1835                        "({}, {}, {})",
1836                        self.type_signature(*a),
1837                        self.type_signature(*b),
1838                        self.type_signature(*c)
1839                    )
1840                }
1841                Type::Adt(name, args) if args.is_empty() => name.unqualified_name().to_string(),
1842                Type::Adt(name, args) => {
1843                    let arg_strs: Vec<String> =
1844                        args.into_iter().map(|a| self.type_signature(a)).collect();
1845                    format!("{} {}", name.unqualified_name(), arg_strs.join(" "))
1846                }
1847            }
1848        }

1850        fn counter_as_letter(&self) -> String {
1851            let m = self.counter % 26;
1852            let d = self.counter / 26;
1853
1854            let m_char = (97 + m) as char; // 97 is 'a'
1855            let d_char = (96 + d) as char; // -1 because we start at 1
1856
1857            if d > 0 {
1858                format!("{}{}", d_char, m_char)
1859            } else {
1860                format!("{}", m_char)
1861            }
1862        }
1863    }

2100    /// `Display for Type` is the text diagnostics quote back to the user, so its two
2101    /// parenthesisation rules and its inference-variable spelling are user-visible
2102    /// output. Nothing else pins them: the pipeline test that renders a type mismatch
2103    /// only ever reaches the `Literal` arms. Dropping a parenthesis here would print
2104    /// `Maybe Maybe Int`, which reads as a different type, with the suite still green.
2105    #[test]
2106    fn display_writes_types_the_way_the_source_spells_them() {
2107        let cases: Vec<(Type, &str)> = vec![
2108            // A function in *parameter* position is parenthesised, because `->` is
2109            // right-associative: `(a -> b) -> c` and `a -> b -> c` are different types.
2110            (
2111                fun_t(fun_t(int_t(), bool_t()), int_t()),
2112                "(Int -> Bool) -> Int",
2113            ),
2114            // In *return* position it is not, for the same reason: the chain already
2115            // re-reads as itself.
2116            (
2117                fun_t(int_t(), fun_t(bool_t(), char_t())),
2118                "Int -> Bool -> Char",
2119            ),
2120            // An applied `Adt` nested inside another needs parens to survive a re-read.
2121            (
2122                adt("Maybe", vec![adt("Maybe", vec![int_t()])]),
2123                "Maybe (Maybe Int)",
2124            ),
2125            // So does a function used as an `Adt` argument.
2126            (
2127                adt("Maybe", vec![fun_t(int_t(), bool_t())]),
2128                "Maybe (Int -> Bool)",
2129            ),
2130            // A *nullary* `Adt` argument does not: there is nothing to mis-group.
2131            (adt("List", vec![adt("Never", vec![])]), "List Never"),
2132            // And an applied `Adt` in parameter position does not either — application
2133            // binds tighter than `->`.
2134            (
2135                fun_t(adt("Maybe", vec![int_t()]), bool_t()),
2136                "Maybe Int -> Bool",
2137            ),
2138            (Type::Tuple(Tuple::two(int_t(), bool_t())), "( Int, Bool )"),
2139            (
2140                Type::Tuple(Tuple::three(int_t(), bool_t(), char_t())),
2141                "( Int, Bool, Char )",
2142            ),
2143            // Inference variables have no source syntax; Elm spells them `t{n}`.
2144            (Type::Variable(TypeVariable { id: 7 }), "t7"),
2145            (Type::Number, "number"),
2146        ];
2147
2148        for (tpe, expected) in cases {
2149            assert_eq!(format!("{}", tpe), expected, "rendering {:?}", tpe);
2150        }
2151    }

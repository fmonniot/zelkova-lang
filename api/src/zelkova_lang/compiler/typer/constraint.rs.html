1//! Turn a typed term into the constraints its shape requires, each with its reason.
2//!
3//! This is where a constraint's [`Reason`] is chosen, because this is the last place
4//! the *structure* of the expression is still visible: by the time `unify` has a pair
5//! of types in hand, nothing says whether they were brought together by an annotation,
6//! by two branches of an `if`, or by an argument being passed.
7//!
8//! Two rules hold at every site below, and both are load-bearing for diagnostics:
9//!
10//! - a term's own constraints come before its children's, so that a type known from
11//!   the outside — a declaration's annotation, and through it a function's parameter
12//!   and result types — has been substituted into the inner constraints before any of
13//!   them is solved. The failure is then reported at the innermost thing that
14//!   disagrees, which is the sub-expression the user has to change, and the chain of
15//!   substitutions that got there leads back to the annotation (see
16//!   [`Origin::left_from`]). Collecting children first inverts that: the body settles
17//!   on its own type first and the mismatch surfaces at the whole function.
18//! - the sides of a constraint are ordered declared-first where the source has a
19//!   declared side, because that is the order the headline reads them out in (see
20//!   [`Constraint`]).
21
22use super::{
23    bool_type, Constraint, Reason, TermPatternKind, Type, TypeLiteral, TypedTerm, TypedTermKind,
24};
25use crate::compiler::tuple::Tuple;

27pub(super) fn collect(term: &TypedTerm) -> Vec<Constraint> {
28    let mut constraints = Vec::new();
29    let tpe = &term.tpe;
30    let span = term.span;
31
32    match &term.kind {
33        TypedTermKind::Bool(_) => {
34            constraints.push(Constraint::new(
35                tpe.clone(),
36                bool_type(),
37                Reason::Literal,
38                span,
39            ));
40        }
41        TypedTermKind::Int(_) => {
42            // Integer literals are polymorphic numeric values: they can unify
43            // with Int or Float (but not Bool, Char, etc.).
44            constraints.push(Constraint::new(
45                tpe.clone(),
46                Type::Number,
47                Reason::Literal,
48                span,
49            ));
50        }
51        TypedTermKind::Char(_) => {
52            constraints.push(Constraint::new(
53                tpe.clone(),
54                Type::Literal(TypeLiteral::Char),
55                Reason::Literal,
56                span,
57            ));
58        }
59        TypedTermKind::Float(_) => {
60            constraints.push(Constraint::new(
61                tpe.clone(),
62                Type::Literal(TypeLiteral::Float),
63                Reason::Literal,
64                span,
65            ));
66        }
67        TypedTermKind::Fun { param, body } => {
68            let param_tpe = Box::new(param.tpe.clone());
69            let return_tpe = Box::new(body.tpe.clone());
70            constraints.push(Constraint::new(
71                tpe.clone(),
72                Type::Fun {
73                    param_tpe,
74                    return_tpe,
75                },
76                Reason::FunctionShape,
77                span,
78            ));
79
80            constraints.extend(collect(body));
81        }
82        TypedTermKind::Identifier(_) => (),
83        TypedTermKind::Apply { fun, arg } => {
84            let param_tpe = Box::new(arg.tpe.clone());
85            let return_tpe = Box::new(tpe.clone());
86            // The span is the *applied* expression's, not the whole application's:
87            // "the expression being applied" is only useful pointing at the thing
88            // being applied.
89            constraints.push(Constraint::new(
90                fun.tpe.clone(),
91                Type::Fun {
92                    param_tpe,
93                    return_tpe,
94                },
95                Reason::Application,
96                fun.span,
97            ));
98
99            constraints.extend(collect(fun));
100            constraints.extend(collect(arg));
101        }
102        TypedTermKind::If {
103            cond,
104            true_branch,
105            false_branch,
106        } => {
107            // If put a constraint on the condition and the branches should resolve to the same type
108            constraints.push(Constraint::new(
109                cond.tpe.clone(),
110                bool_type(),
111                Reason::IfCondition,
112                cond.span,
113            ));
114            constraints.push(Constraint::new(
115                true_branch.tpe.clone(),
116                tpe.clone(),
117                Reason::IfBranch,
118                true_branch.span,
119            ));
120            constraints.push(Constraint::new(
121                false_branch.tpe.clone(),
122                tpe.clone(),
123                Reason::IfBranch,
124                false_branch.span,
125            ));
126
127            constraints.extend(collect(cond));
128            constraints.extend(collect(true_branch));
129            constraints.extend(collect(false_branch));
130        }
131        TypedTermKind::Let {
132            binding,
133            value,
134            body,
135        } => {
136            // The let expression has the body type.
137            constraints.push(Constraint::new(
138                tpe.clone(),
139                body.tpe.clone(),
140                Reason::LetBody,
141                span,
142            ));
143            // The binding type is the one of the value. Written value-first so that
144            // the side named by the span — the value — is `left`.
145            constraints.push(Constraint::new(
146                value.tpe.clone(),
147                binding.tpe.clone(),
148                Reason::LetBinding,
149                value.span,
150            ));
151
152            constraints.extend(collect(value));
153            constraints.extend(collect(body));
154        }
155        TypedTermKind::Case {
156            scrutinee,
157            branches,
158        } => {
159            // Every branch's own constraints, all of them, before any child is walked
160            // — including the scrutinee, which is a child like the branch bodies are.
161            // Walking it first was this arm's one departure from the rule the module
162            // doc states, and it showed: a `case` on a compound expression reported
163            // its pattern mismatch against a constraint from inside the scrutinee
164            // rather than against the pattern.
165            for (pattern, body) in branches {
166                // Every branch must return the case expression's type.
167                constraints.push(Constraint::new(
168                    body.tpe.clone(),
169                    tpe.clone(),
170                    Reason::CaseBranch,
171                    body.span,
172                ));
173                // Each pattern constrains the scrutinee type. The pattern is what the
174                // caret should sit under, so the pattern's type is `left`.
175                match &pattern.kind {
176                    TermPatternKind::Literal(lit) => {
177                        constraints.push(Constraint::new(
178                            lit.clone(),
179                            scrutinee.tpe.clone(),
180                            Reason::CasePattern,
181                            pattern.span,
182                        ));
183                    }
184                    TermPatternKind::Constructor {
185                        adt_name, adt_args, ..
186                    } => {
187                        constraints.push(Constraint::new(
188                            Type::Adt(adt_name.clone(), adt_args.clone()),
189                            scrutinee.tpe.clone(),
190                            Reason::CasePattern,
191                            pattern.span,
192                        ));
193                    }
194                    // Bind/Anything: the binding's type was already set to scrutinee.tpe
195                    // in annotate, so no extra constraint needed here.
196                    TermPatternKind::Bind(_) | TermPatternKind::Anything => {}
197                }
198            }
199
200            constraints.extend(collect(scrutinee));
201            for (_, body) in branches {
202                constraints.extend(collect(body));
203            }
204        }
205        TypedTermKind::Tuple(elements) => {
206            // The tuple type must equal the tuple of its element types.
207            let element_types = match elements {
208                Tuple::Two(a, b) => Tuple::two(a.tpe.clone(), b.tpe.clone()),
209                Tuple::Three(a, b, c) => Tuple::three(a.tpe.clone(), b.tpe.clone(), c.tpe.clone()),
210            };
211            constraints.push(Constraint::new(
212                tpe.clone(),
213                Type::Tuple(element_types),
214                Reason::TupleElements,
215                span,
216            ));
217
218            for elem in elements.iter() {
219                constraints.extend(collect(elem));
220            }
221        }
222    };
223
224    constraints
225}

233    /// Build a typed term with no position — these tests are about which constraints
234    /// come out and why, not about where. `NodeSpan`'s `PartialEq` is blind, so the
235    /// `Reason` is the part of an `Origin` these assertions actually pin.
236    fn typed(tpe: Type, kind: TypedTermKind) -> TypedTerm {
237        TypedTerm {
238            span: NodeSpan::none(),
239            tpe,
240            kind,
241        }
242    }
243
244    fn identifier(tpe: Type, name: &str) -> TypedTerm {
245        typed(tpe, TypedTermKind::Identifier(name.to_owned()))
246    }
247
248    fn constraint(left: Type, right: Type, reason: Reason) -> Constraint {
249        Constraint::new(left, right, reason, NodeSpan::none())
250    }
251
252    /// `Basics.Bool`, written out rather than taken from [`bool_type`].
253    ///
254    /// Which module declared the union is the whole of its identity, so an assertion
255    /// built from the function under test would hold for any name that function
256    /// picked — including a bare `Bool`, which would make every module's own `Bool`
257    /// an `if` condition ([`DEC-15`](../../../docs/decisions/dec-15.md) decision 1).
258    fn basics_bool() -> Type {
259        Type::Adt(
260            crate::compiler::name::QualName::parse("Basics.Bool").expect("a qualified name"),
261            vec![],
262        )
263    }

266    fn constrains_int() {
267        let t1 = Type::Variable(TypeVariable { id: 1 });
268
269        // Integer literals constrain to Number (polymorphic: can be Int or Float)
270        let expected = vec![constraint(t1.clone(), Type::Number, Reason::Literal)];
271
272        let int = typed(t1, TypedTermKind::Int(42));
273
274        assert_eq!(collect(&int), expected);

355    #[test]
356    fn constrains_if() {
357        let t1 = Type::Variable(TypeVariable { id: 1 });
358        let t2 = Type::Variable(TypeVariable { id: 2 });
359        let t3 = Type::Variable(TypeVariable { id: 3 });
360        let t4 = Type::Variable(TypeVariable { id: 4 });
361
362        // t2 === Basics.Bool (eg. the condition needs to be a boolean)
363        // t3 === t1   (eg. the if type is the same as the first branch)
364        // t4 === t1   (eg. the if type is the same as the second branch)
365        //
366        // The reasons are asserted too: they are what a diagnostic says out loud, and
367        // a condition reported as "every branch must have the same type" would be a
368        // lie the types alone cannot catch.
369        let expected = vec![
370            constraint(t2.clone(), basics_bool(), Reason::IfCondition),
371            constraint(t3.clone(), t1.clone(), Reason::IfBranch),
372            constraint(t4.clone(), t1.clone(), Reason::IfBranch),
373        ];
374
375        let cond = Box::new(identifier(t2, "condition"));
376        let true_branch = Box::new(identifier(t3, "if_true"));
377        let false_branch = Box::new(identifier(t4, "if_false"));
378        let if_else = typed(
379            t1,
380            TypedTermKind::If {
381                cond,
382                true_branch,
383                false_branch,
384            },
385        );
386
387        assert_eq!(collect(&if_else), expected);
388    }

1//! The canonical representation of a zelkova programs is a translation of a local
2//! source into the broader world.
3//!
4//! This phase is where we integrate the local parsed module into the rest of the
5//! program. We do the following steps:
6//!
7//! - Resolve all imports
8//! - Qualify all `Name` (eg. a local value `test` in a `Mod.A` module will be renamed `Mod.A.test`)
9//! - Checks that exported names are actually present in the module
10//! - Checks there is none cyclic dependency between this module and others (Might be done earlier, let's see)
11//!
12//! Note that we use `HashMap`'s a lot in this module's structures. This is because later phases will
13//! want to have cheap access to the different components of a `Module`.
14//!
15//! TODO Rename this to core ? I feel it's going to be te main internal representation of the language.

61impl Module {
62    /// Build the trimmed-down view of this module other modules import against.
63    ///
64    /// `file` is the [`SourceFileId`] this module was read from, which is what lets
65    /// a later module's diagnostic point back into *this* module's source — see
66    /// [`Interface::file`](super::Interface::file). It is a parameter rather than
67    /// something this method could work out because a `canonical::Module` does not
68    /// know it: only driver code does, and the sole caller
69    /// (`dependencies::ModuleWalker::check_in_order`) is driver code that has the
70    /// map from module name to file in hand. A caller with nothing to give — every
71    /// test that hand-builds an interface — passes `None`, and
72    /// [`Interface::source_span`](super::Interface::source_span) then declines to
73    /// build a cross-module label rather than building one at the wrong place.
74    ///
75    /// # What the `exposing (...)` header removes here
76    ///
77    /// This is the only place a module's own [`Exports`] is read, and reading it
78    /// here is what makes a declaration left out of the header private (`BUG-9`).
79    /// `self.values`, `self.types` and `self.infixes` stay complete — the typer,
80    /// exhaustiveness and every later phase run against the whole
81    /// [`Module`](Self), because a private declaration is still callable from
82    /// inside the module that wrote it. Only this external view is trimmed, and
83    /// only by three rules:
84    ///
85    /// - a value reaches the interface when the header exposes its name as
86    ///   [`ExportType::Value`], on top of the pre-existing requirement that it
87    ///   carry a type annotation (`Value::TypedValue`). `do_exports` is what
88    ///   makes that requirement real (`SPEC-5`): it rejects a module that
89    ///   exposes an unannotated declaration before this method ever runs, so
90    ///   the `Value::Value` arm in the `filter_map` below is unreachable
91    ///   through normal use — its own comment says why it stays rather than
92    ///   becoming an `unwrap`;
93    /// - an infix reaches it when the header exposes the operator;
94    /// - a union type reaches it when the header names it either way, but a
95    ///   `Size` entry ([`ExportType::UnionPrivate`]) hands over the declaration
96    ///   with its `variants` emptied. That is the opaque type: importers still
97    ///   get the name and its type variables — [`Interface::unions`] is where
98    ///   `process_import`'s `Privacy::Private` arm reads the arity from — and no
99    ///   constructor to build or match one with.
100    ///
101    /// [`Exports::Everything`] — a `exposing (..)` header — exposes every
102    /// declaration with every constructor, so nothing is dropped in that case.
103    ///
104    /// An exposed infix whose backing function is not itself separately exposed
105    /// — `infix left 6 (+) = add` with `(+)` in the header and `add` not, which
106    /// is every operator `std/core` declares — still needs `add`'s type
107    /// reachable, or an importer that brings `(+)` into scope has no type to
108    /// give a use of it. [`Interface::infix_functions`] carries exactly that,
109    /// kept out of `values` itself so `add` stays unreachable under its own
110    /// name.
111    pub fn to_interface(&self, file: Option<SourceFileId>) -> super::Interface {
112        let values: HashMap<Name, (NodeSpan, Type)> = self
113            .values
114            .iter()
115            .filter(|(name, _)| self.exports.exposes(name, &ExportType::Value))
116            .filter_map(|(name, value)| match value {
117                // Unreachable through normal use: `do_exports` (`SPEC-5`)
118                // rejects an exposed, unannotated declaration before
119                // canonicalization ever produces a `Module` for this method
120                // to run on (`BUG-14`). Kept rather than `unwrap`ped so a
121                // future caller that hands this a hand-built `Module` — a
122                // test, most likely — gets a value silently absent from the
123                // interface instead of a panic.
124                Value::Value { .. } => None,
125                Value::TypedValue { tpe, span, .. } => Some((name.clone(), (*span, tpe.clone()))),
126            })
127            .collect();
128
129        let unions = self
130            .types
131            .iter()
132            .filter_map(|(name, union)| match self.exports.union_visibility(name) {
133                UnionVisibility::Hidden => None,
134                UnionVisibility::Transparent => Some((name.clone(), union.clone())),
135                UnionVisibility::Opaque => Some((
136                    name.clone(),
137                    UnionType {
138                        variables: union.variables.clone(),
139                        variants: Vec::new(),
140                        span: union.span,
141                    },
142                )),
143            })
144            .collect();
145
146        let infixes: HashMap<Name, Infix> = self
147            .infixes
148            .iter()
149            .filter(|(name, _)| self.exports.exposes(name, &ExportType::Infix))
150            .map(|(name, infix)| (name.clone(), infix.clone()))
151            .collect();
152
153        // See `Interface::infix_functions`'s doc comment: only a backing function
154        // *not* already reaching `values` on its own needs a supplementary entry
155        // here — one that is already exposed by name is inserted twice
156        // otherwise, which `insert_foreign_value` reads as the same value
157        // imported from two places.
158        let infix_functions = infixes
159            .values()
160            .filter(|infix| !values.contains_key(&infix.function_name))
161            .filter_map(|infix| match self.values.get(&infix.function_name) {
162                Some(Value::TypedValue { tpe, span, .. }) => {
163                    Some((infix.function_name.clone(), (*span, tpe.clone())))
164                }
165                _ => None,
166            })
167            .collect();
168
169        super::Interface {
170            module_name: self.name.clone(),
171            values,
172            unions,
173            infixes,
174            infix_functions,
175            file,
176        }
177    }
178}

180#[derive(Debug, PartialEq)]
181pub enum Exports {
182    Everything,
183    /// Non qualified name to its export type
184    Specifics(HashMap<Name, ExportType>),
185}
186
187impl Exports {
188    /// Whether the `exposing (...)` header exposes `name` *as* `kind`.
189    ///
190    /// The kind is part of the question rather than a detail of the answer
191    /// because the three namespaces a module exports into are separate: a
192    /// header naming `map` exposes the value `map`, and says nothing about an
193    /// operator or a type that happened to share the spelling. Union types ask
194    /// [`union_visibility`](Self::union_visibility) instead, which has a third
195    /// answer for the opaque case.
196    fn exposes(&self, name: &Name, kind: &ExportType) -> bool {
197        match self {
198            Exports::Everything => true,
199            Exports::Specifics(specifics) => specifics.get(name) == Some(kind),
200        }
201    }
202
203    /// How far a union type declared in this module crosses the module boundary.
204    fn union_visibility(&self, name: &Name) -> UnionVisibility {
205        match self {
206            Exports::Everything => UnionVisibility::Transparent,
207            Exports::Specifics(specifics) => match specifics.get(name) {
208                Some(ExportType::UnionPublic) => UnionVisibility::Transparent,
209                Some(ExportType::UnionPrivate) => UnionVisibility::Opaque,
210                Some(ExportType::Value) | Some(ExportType::Infix) | None => UnionVisibility::Hidden,
211            },
212        }
213    }
214}

216/// The three answers [`Exports::union_visibility`] can give for one of the
217/// module's own union types, in the order a reader of an `exposing (...)` list
218/// meets them: no entry at all, a bare `Size` entry, a `Size(..)` entry.
219#[derive(Debug)]
220enum UnionVisibility {
221    /// Not in the header: other modules cannot name the type at all.
222    Hidden,
223    /// `Size` — the type's name and arity cross the boundary, its constructors
224    /// do not.
225    Opaque,
226    /// `Size(..)` — the declaration crosses whole, constructors included.
227    Transparent,
228}

279/// A canonical type.
280///
281/// # Why there is no span on this type, or on [`TypeConstructor`]
282///
283/// Every other canonical node this module builds carries a [`NodeSpan`] taken from
284/// the parser node it came from. A `Type` deliberately does not.
285///
286/// The reason `ERR-3` recorded is that a `Type` does not always come from the module
287/// being canonicalized: `Type::from_parser_type` resolves a name through the
288/// [`Environment`], which clones types straight out of the [`Interface`]s of the
289/// modules this one imports, so the `Type` handed back may well have been *written
290/// in a different file* — and a bare span on it would be read as a position in the
291/// importing module's source. `ERR-5` settled that half: a [`SpanLabel`] can now
292/// carry a file of its own, so "written in another file" is no longer the obstacle.
293///
294/// The other half is unfinished work, not an impossibility. `parser::Type` carries a
295/// [`NodeSpan`] per node like every other parser node, and `Type::from_parser_type`
296/// is the recursive walk over it — `tpe.span` is in hand at every level, and each
297/// branch of an `Arrow` arrives as its own spanned `parser::Type`. Those spans are
298/// discarded here by choice. Keeping them means a span field on every canonical
299/// `Type` node, plus a decision about what a `Type` cloned out of an `Environment`
300/// should then report — where it was written, or where it was used. Nobody has
301/// written that.
302///
303/// What exists instead is coarser and enough for the diagnostics that exist: the
304/// *declaration* a `Type` is the type of — [`Value`]'s own `span`, [`UnionType`]'s,
305/// [`Infix`]'s — is what a diagnostic wants to underline ("defined here"), not a
306/// position inside the type expression itself. [`Interface::values`] pairs each
307/// value's `Type` with that declaration's [`NodeSpan`], and
308/// [`Interface::source_span`] pairs it with the file, once [`Interface::file`] is
309/// set. A type error still points at the whole declaration that failed rather than
310/// the sub-expression that disagrees, which is `ERR-4`, a separate limit in the
311/// typer's own `Term`/`Constraint` translation.
312#[derive(Debug, Clone, PartialEq)]
313pub enum Type {
314    Variable(Name),
315    /// A named type applied to its arguments, the name being the [`QualName`] of
316    /// the declaration it resolved to.
317    ///
318    /// The module is part of the identity of the type, not decoration on it: a
319    /// `type Size` in `Widget` and a `type Size` in `Gadget` are two types, and
320    /// two values of them are interchangeable nowhere. Carrying only the
321    /// spelling made them one value here, which is what `AST-4` closed.
322    ///
323    /// The name is the *declaration's*, never the spelling that reached it.
324    /// `import Widget as W` followed by `W.Size` records `Widget.Size`, the same
325    /// rule [name resolution](../../../docs/spec/name-resolution.md) states for
326    /// values: an alias names a route to a declaration and not a second
327    /// declaration.
328    Type(QualName, Vec<Type>),
329    // Record
330    // Unit
331    Arrow(Box<Type>, Box<Type>),
332    /// A tuple type. Zelkova keeps Elm's restriction of two or three elements,
333    /// which [`Tuple`] carries in its shape.
334    Tuple(Tuple<Type>),
335    // Alias
336}
337
338impl Type {
339    /// The one conversion in this module that reads a span and produces none.
340    ///
341    /// `parser::Type` carries a [`NodeSpan`] like every other parser node, and it is
342    /// dropped here on purpose — see this type's documentation for why a canonical
343    /// `Type` holds no span at all, and what `ERR-5` built instead.
344    fn from_parser_type(env: &dyn Environment, tpe: &parser::Type) -> Result<Type, Error> {
345        match &tpe.kind {
346            parser::TypeKind::Unqualified(name, vars) => {
347                let args = vars
348                    .iter()
349                    .map(|t| Type::from_parser_type(env, t))
350                    .collect::<Result<Vec<_>, Error>>()?;
351
352                match env.find_type(name) {
353                    // `name` resolves to a declared type: `args` is what was
354                    // written after it, and has to match the declaration's own
355                    // arity (`BUG-17`) — nothing else here re-derives that check.
356                    // The head is the *declaration's* qualified name, not the one
357                    // written: `Lib.Option`, `Option` and an alias' `L.Option` are
358                    // three spellings of one entry, and every one of them records
359                    // the module that declared it.
360                    Some(declared) if declared.arity() == args.len() => {
361                        Ok(Type::Type(declared.name.clone(), args))
362                    }
363                    Some(declared) => Err(Error::TypeArityMismatch(
364                        name.clone(),
365                        declared.arity(),
366                        args.len(),
367                        tpe.span,
368                    )),
369                    // `name` resolves to nothing: BUG-16 is the ticket for reporting
370                    // this instead of fabricating a type for it. Until then the
371                    // fabricated head is attributed to the module under check —
372                    // the same fallback the `TypeConstructor` arm of
373                    // `Expression::from_parser` uses — and never to a module half
374                    // read off the written spelling. A written `W.Thing` says which
375                    // *route* was written and not which module declared anything:
376                    // under `import Widget as W` that half is an alias, and reading
377                    // it would fabricate a head in a module named `W`, inverting the
378                    // rule the `Some` arm above enforces. `qualify_name` does not
379                    // split the name it is handed, so a written `Missing.Thing`
380                    // stays one dotted unqualified half — which is what keeps it
381                    // distinct from a local `Thing` at the typer, exactly as it was
382                    // before the head became a `QualName`.
383                    None => Ok(Type::Type(env.module_name().qualify_name(name), args)),
384                }
385            }
386            parser::TypeKind::Arrow(t1, t2) => Ok(Type::Arrow(
387                Box::new(Type::from_parser_type(env, t1)?),
388                Box::new(Type::from_parser_type(env, t2)?),
389            )),
390            parser::TypeKind::Variable(n) => Ok(Type::Variable(n.clone())),
391            parser::TypeKind::Tuple(tuple) => Ok(Type::Tuple(
392                tuple.try_map(|t| Type::from_parser_type(env, t))?,
393            )),
394        }
395    }
396
397    // TODO Write some tests
398    fn to_linear_types(tpe: &Type) -> Vec<Type> {
399        match tpe {
400            Type::Arrow(a, b) => {
401                let mut next = Type::to_linear_types(b);
402
403                next.insert(0, *a.clone());
404
405                next
406            }
407            _ => vec![tpe.clone()],
408        }
409    }
410}

459/// A canonical pattern, and where it was written.
460///
461/// Same shape as [`parser::Pattern`] — a [`NodeSpan`] beside a kind — and for the
462/// same reason: the children stay plain `Pattern`s, so a reader matches `&p.kind`.
463#[derive(Debug, PartialEq)]
464pub struct Pattern {
465    pub span: NodeSpan,
466    pub kind: PatternKind,
467}
468
469#[derive(Debug, PartialEq)]
470pub enum PatternKind {
471    Anything,
472    Variable(Name), // TODO Name or QualName ?
473    Int(i64),
474    Float(f64),
475    Char(char),
476    Bool(bool),
477    /// A tuple pattern. Zelkova keeps Elm's restriction of two or three
478    /// elements, which [`Tuple`] carries in its shape.
479    Tuple(Tuple<Pattern>),
480
481    Constructor {
482        ctor: TypeConstructor,
483        args: Vec<Pattern>,
484    },
485}
486
487impl Pattern {
488    /// A pattern canonicalized from source, keeping the parser node's position.
489    pub fn new(span: NodeSpan, kind: PatternKind) -> Pattern {
490        Pattern { span, kind }
491    }
492
493    /// A pattern with no position — hand-built by a test. See [`NodeSpan`].
494    pub fn bare(kind: PatternKind) -> Pattern {
495        Pattern {
496            span: NodeSpan::none(),
497            kind,
498        }
499    }
500
501    fn from_parser(p: &parser::Pattern, env: &dyn Environment) -> Result<Pattern, Error> {
502        let kind = match &p.kind {
503            parser::PatternKind::Anything => PatternKind::Anything,
504            parser::PatternKind::Variable(name) => PatternKind::Variable(name.clone()),
505            parser::PatternKind::Literal(parser::Literal::Int(i)) => PatternKind::Int(*i),
506            parser::PatternKind::Literal(parser::Literal::Float(f)) => PatternKind::Float(*f),
507            parser::PatternKind::Literal(parser::Literal::Char(c)) => PatternKind::Char(*c),
508            parser::PatternKind::Literal(parser::Literal::Bool(b)) => PatternKind::Bool(*b),
509            parser::PatternKind::Tuple(tuple) => {
510                PatternKind::Tuple(tuple.try_map(|p| Pattern::from_parser(p, env))?)
511            }
512            parser::PatternKind::Constructor(name, args) => {
513                // `p.span` covers the constructor and its arguments, which is the
514                // text a "no such constructor" caret should sit under.
515                let ctor = env
516                    .find_type_constructor(name)
517                    .ok_or_else(|| {
518                        let suggestion =
519                            suggest_name(name, env.type_constructor_names().into_iter());
520                        Error::VariantNotFound(
521                            env.module_name().qualify_name(name),
522                            p.span,
523                            suggestion,
524                        )
525                    })?
526                    .clone();
527
528                let args = args
529                    .iter()
530                    .map(|p| Pattern::from_parser(p, env))
531                    .collect::<Result<Vec<_>, Error>>()?;
532
533                PatternKind::Constructor { ctor, args }
534            }
535        };
536
537        Ok(Pattern::new(p.span, kind))
538    }
539}

551// TODO Find a way to detect recursive functions (even indirect recursivity,
552// eg. `a` calls `b` calls `a`)
553/// Expression is an optimized version for checks and caches.
554///
555/// Elm declare those expressions:
556/// ```haskell
557/// data Expr_
558///   = VarLocal Name
559///   | VarTopLevel ModuleName.Canonical Name
560///   | VarKernel Name Name
561///   | VarForeign ModuleName.Canonical Name Annotation
562///   | VarCtor CtorOpts ModuleName.Canonical Name Index.ZeroBased Annotation
563///   | VarDebug ModuleName.Canonical Name Annotation
564///   | VarOperator Name ModuleName.Canonical Name Annotation -- CACHE real name for optimization
565///   | Chr ES.String
566///   | Str ES.String
567///   | Int Int
568///   | Float EF.Float
569///   | List [Expr]
570///   | Negate Expr
571///   | Binop Name ModuleName.Canonical Name Annotation Expr Expr -- CACHE real name for optimization
572///   | Lambda [Pattern] Expr
573///   | Call Expr [Expr]
574///   | If [(Expr, Expr)] Expr
575///   | Let Def Expr
576///   | LetRec [Def] Expr
577///   | LetDestruct Pattern Expr Expr
578///   | Case Expr [CaseBranch]
579///   | Accessor Name
580///   | Access Expr (A.Located Name)
581///   | Update Name Expr (Map.Map Name FieldUpdate)
582///   | Record (Map.Map Name Expr)
583///   | Unit
584///   | Tuple Expr Expr (Maybe Expr)
585/// ```
586#[derive(Debug, PartialEq)]
587pub enum ExpressionKind {
588    VarLocal(Name),
589    VarTopLevel(QualName),
590    VarKernel(QualName),
591    VarForeign(QualName, Type),
592    VarConstructor(QualName, Type),
593    Char(char),
594    Int(i64),
595    Float(f64),
596    Bool(bool),
597    // List
598    // Lambda
599    Apply(Box<Expression>, Box<Expression>),
600    If(Box<Expression>, Box<Expression>, Box<Expression>),
601    // Let

603    // LetDestruct (eg. `(a,b) = someTuple`)

629    fn from_parser(e: &parser::Expression, env: &dyn Environment) -> Result<Expression, Error> {
630        // Every arm builds a kind and every kind gets `e.span`, so a name the
631        // environment cannot resolve is underlined where it was written rather than
632        // somewhere up the tree.
633        let kind = match &e.kind {
634            parser::ExpressionKind::Lit(parser::Literal::Int(i)) => ExpressionKind::Int(*i),
635            parser::ExpressionKind::Lit(parser::Literal::Float(f)) => ExpressionKind::Float(*f),
636            parser::ExpressionKind::Lit(parser::Literal::Char(c)) => ExpressionKind::Char(*c),
637            parser::ExpressionKind::Lit(parser::Literal::Bool(b)) => ExpressionKind::Bool(*b),
638            parser::ExpressionKind::Variable(name) => {
639                match env.find_value(name).ok_or_else(|| {
640                    let suggestion = suggest_name(name, env.value_names().into_iter());
641                    Error::VariableNotFound(
642                        env.module_name().qualify_name(name),
643                        e.span,
644                        suggestion,
645                    )
646                })? {
647                    ValueType::Local => ExpressionKind::VarLocal(name.clone()),
648                    ValueType::TopLevel => {
649                        ExpressionKind::VarTopLevel(env.module_name().qualify_name(name))
650                    }
651                    ValueType::Foreign(m, _source, tpe, _origin) => {
652                        ExpressionKind::VarForeign(m.qualify_name(name), tpe.clone())
653                    }
654                    ValueType::Foreigns(candidates) => {
655                        return Err(Error::AmbiguousVariables(
656                            name.clone(),
657                            candidates.clone(),
658                            e.span,
659                        ))
660                    }
661                }
662            }
663            parser::ExpressionKind::TypeConstructor(name) => {
664                let ctor = env.find_type_constructor(name).ok_or_else(|| {
665                    let suggestion = suggest_name(name, env.type_constructor_names().into_iter());
666                    Error::VariantNotFound(env.module_name().qualify_name(name), e.span, suggestion)
667                })?;
668
669                let tpe = if ctor.type_parameters.is_empty() {
670                    Type::Type(ctor.tpe.clone(), vec![])
671                } else {
672                    // TODO Rework that part. ctor.types is only for the type parameters of the constructor, not for the overall type.
673                    let mut iter = ctor.type_parameters.iter().rev();
674                    let first = iter.next().unwrap().clone();
675
676                    // TODO tests this, 99.999% I'm wrong about it (like everytime I try to implement arrows, plus
677                    // the foldr in this particular case)
678                    let tpe = iter.fold(first, |acc, t| {
679                        Type::Arrow(Box::new(t.clone()), Box::new(acc))
680                    });
681
682                    Type::Arrow(
683                        Box::new(Type::Type(ctor.tpe.clone(), vec![])),
684                        Box::new(tpe),
685                    )
686                };
687
688                let name = name
689                    .to_qual()
690                    .unwrap_or_else(|| env.module_name().qualify_name(name));
691
692                ExpressionKind::VarConstructor(name, tpe)
693            }
694            parser::ExpressionKind::Application(a, b) => {
695                let a = Expression::from_parser(a, env)?;
696                let b = Expression::from_parser(b, env)?;
697
698                ExpressionKind::Apply(Box::new(a), Box::new(b))
699            }
700            parser::ExpressionKind::InfixChain(first, rest) => {
701                let first = Expression::from_parser(first, env)?;
702
703                let rest = rest
704                    .iter()
705                    .map(|(op_name, op_span, operand)| {
706                        let op = resolve_infix_operator(op_name, *op_span, env)?;
707                        let operand = Expression::from_parser(operand, env)?;
708                        Ok((op, operand))
709                    })
710                    .collect::<Result<Vec<_>, Error>>()?;
711
712                let mut rest = rest.into_iter().peekable();
713                let reassociated = reassociate_infix_chain(first, &mut rest, 0)?;
714
715                reassociated.kind
716            }
717            parser::ExpressionKind::Tuple(tuple) => {
718                ExpressionKind::Tuple(tuple.try_map(|e| Expression::from_parser(e, env))?)
719            }
720            parser::ExpressionKind::Case(expr, branches) => {
721                let expr = Expression::from_parser(expr, env)?;
722
723                let b = branches.iter().map::<Result<CaseBranch, Error>, _>(|cb| {
724                    let pattern = Pattern::from_parser(&cb.pattern, env)?;
725                    let mut scoped = env.new_scope();
726
727                    scoped.expose_pattern(&pattern);
728
729                    let expression = Expression::from_parser(&cb.expression, &scoped)?;
730
731                    Ok(CaseBranch {
732                        pattern,
733                        expression,
734                        span: cb.span,
735                    })
736                });
737
738                let branches = collect_accumulate(b)?;
739
740                ExpressionKind::Case(Box::new(expr), branches)
741            }
742            parser::ExpressionKind::If(cond, then, els) => {
743                let cond = Expression::from_parser(cond, env)?;
744                let then = Expression::from_parser(then, env)?;
745                let els = Expression::from_parser(els, env)?;
746
747                ExpressionKind::If(Box::new(cond), Box::new(then), Box::new(els))
748            }
749        };
750
751        Ok(Expression::new(e.span, kind))
752    }

755/// One operator of an `ExpressionKind::InfixChain`, resolved against the infix
756/// environment: its own `Infix` (precedence, associativity, and where it was
757/// declared) alongside the canonical `Expression` it resolves to as a value —
758/// a reference to the function its `infix` declaration names, qualified with the
759/// module that wrote that declaration. `reassociate_infix_chain` consults `infix`
760/// to decide how to nest, and `span` and `expr` to build the invented
761/// `Application` nodes.
762struct ResolvedInfixOp {
763    name: Name,
764    /// Where the operator itself was written — not its `infix` declaration.
765    span: NodeSpan,
766    expr: Expression,
767    infix: Infix,
768    /// Where the `infix` declaration was written, and in which module's file —
769    /// the operator may well have been declared by an imported module, in which
770    /// case `infix.span` is a byte range in *that* module's source. See
771    /// [`InfixDeclaration`].
772    declaration: InfixDeclaration,
773}

775/// Resolves one operator of an `InfixChain` into the function its `infix`
776/// declaration names, plus the `Infix` re-association needs.
777///
778/// The `infixes` map is the whole of an operator's scope: nothing else can hold
779/// a key spelled like one, since a function's own name always comes from
780/// `VarIdent`, a distinct token from `Op`. So `find_infix` failing is exactly
781/// "no operator by that name is in scope", and gets the `VariableNotFound` a
782/// reader would expect for a name they wrote and nothing declares.
783///
784/// The function behind the operator is resolved from the entry
785/// ([`InfixFunction`]) rather than looked up in the scope the operator is *used*
786/// in. An operator has no qualified spelling, so naming it in an `exposing` list
787/// is the only way to reach one across a module boundary, and whether the
788/// exporting module's backing function is separately in scope is neither the
789/// importer's choice nor visible to them. The resulting `VarTopLevel`/`VarForeign`
790/// is qualified with the module that wrote the `infix` declaration and names the
791/// function, not the operator symbol, so it matches the binding a later phase
792/// looks it up against.
793fn resolve_infix_operator(
794    name: &Name,
795    span: NodeSpan,
796    env: &dyn Environment,
797) -> Result<ResolvedInfixOp, Error> {
798    let entry = env.find_infix(name).cloned().ok_or_else(|| {
799        let suggestion = suggest_name(name, env.value_names().into_iter());
800        Error::VariableNotFound(env.module_name().qualify_name(name), span, suggestion)
801    })?;
802
803    let function_name = &entry.infix.function_name;
804    let kind = match &entry.function {
805        InfixFunction::Local => {
806            ExpressionKind::VarTopLevel(env.module_name().qualify_name(function_name))
807        }
808        InfixFunction::Imported(module, tpe) => {
809            ExpressionKind::VarForeign(module.qualify_name(function_name), tpe.clone())
810        }
811        // The exporting module declared the backing function without an
812        // annotation, so its interface carries no type for it. The name that
813        // failed to resolve is the function's, and that is what is reported —
814        // the operator is in scope, and saying otherwise would send the reader
815        // to check their own `import` line.
816        InfixFunction::ImportedUntyped(module) => {
817            return Err(Error::VariableNotFound(
818                module.qualify_name(function_name),
819                span,
820                None,
821            ))
822        }
823    };
824
825    Ok(ResolvedInfixOp {
826        name: name.clone(),
827        span,
828        expr: Expression::new(span, kind),
829        infix: entry.infix,
830        declaration: entry.declaration,
831    })
832}

834/// Re-associates a flat run of operator applications into a tree of
835/// `Application` nodes, in one pass over the operators — no operator or operand
836/// is visited twice — using precedence climbing (a standard algorithm; see e.g.
837/// https://en.wikipedia.org/wiki/Operator-precedence_parser#Precedence_climbing_method).
838///
839/// `min_prec` is the lowest precedence this call is willing to fold into `lhs`;
840/// the outer, top-level call passes 0, admitting every operator. A recursive
841/// call raises it to bind only the tighter operators that belong on the right
842/// of the operator being folded — `op.infix.precedence + 1` for strictly higher
843/// precedence, or `op.infix.precedence` itself when the next operator ties and
844/// both associate right, which is what lets a right-associative run keep
845/// folding at the same level (`a <| b <| c` → `a <| (b <| c)`).
846///
847/// Two adjacent operators of equal precedence are rejected rather than guessed
848/// at unless they agree — both `left` (fold `lhs`, the usual case) or both
849/// `right` (recurse into `rhs`). Anything else — a `left` against a `right`, or
850/// either against an `infix non` operator, including one against itself — is
851/// `Error::AmbiguousOperatorPrecedence`: nothing here says which one should
852/// bind first, and guessing would silently pick a grouping the user did not
853/// write.
854fn reassociate_infix_chain(
855    mut lhs: Expression,
856    rest: &mut std::iter::Peekable<impl Iterator<Item = (ResolvedInfixOp, Expression)>>,
857    min_prec: u8,
858) -> Result<Expression, Error> {
859    while let Some((op, mut rhs)) = rest.next_if(|(op, _)| op.infix.precedence >= min_prec) {
860        loop {
861            let next_min = match rest.peek() {
862                None => None,
863                Some((next_op, _)) => {
864                    if next_op.infix.precedence > op.infix.precedence {
865                        Some(op.infix.precedence + 1)
866                    } else if next_op.infix.precedence == op.infix.precedence {
867                        match (op.infix.associativity, next_op.infix.associativity) {
868                            (Associativity::Left, Associativity::Left) => None,
869                            (Associativity::Right, Associativity::Right) => {
870                                Some(op.infix.precedence)
871                            }
872                            _ => {
873                                return Err(Error::AmbiguousOperatorPrecedence(
874                                    AmbiguousOperator::new(&op),
875                                    AmbiguousOperator::new(next_op),
876                                    op.span.merge(next_op.span),
877                                ))
878                            }
879                        }
880                    } else {
881                        None
882                    }
883                }
884            };
885
886            match next_min {
887                Some(min) => rhs = reassociate_infix_chain(rhs, rest, min)?,
888                None => break,
889            }
890        }
891
892        lhs = apply_infix(op, lhs, rhs);
893    }
894
895    Ok(lhs)
896}

898/// Builds the two `Application` nodes one step of infix re-association adds,
899/// following the same span convention the grammar's right-recursive rewrite
900/// used before this: the partial application of the operator to `lhs` — a node
901/// with nothing in the user's text of its own — takes the operator's own span,
902/// and the outer application, which stands for `lhs op rhs` as the user wrote
903/// it, spans from wherever `lhs` starts to wherever `rhs` ends.
904fn apply_infix(op: ResolvedInfixOp, lhs: Expression, rhs: Expression) -> Expression {
905    let span = lhs.span.merge(rhs.span);
906
907    let partial = Expression::new(
908        op.span,
909        ExpressionKind::Apply(Box::new(op.expr), Box::new(lhs)),
910    );
911
912    Expression::new(
913        span,
914        ExpressionKind::Apply(Box::new(partial), Box::new(rhs)),
915    )
916}

958///
959/// # Why only some variants carry a span
960///
961/// A variant carries a [`NodeSpan`] when its construction site has one in hand — it
962/// is looking at a `parser::Import`, `parser::Infix`, `parser::UnionType`,

965/// variant below except three: the two group variants, `EnvironmentErrors` and
966/// `Many`, have no position of their own and flatten their members' labels instead;
967/// and `InvalidTupleSize` carries none for an unrelated reason — see its own doc
968/// comment. Writing `NodeSpan::none()` into a variant that lacks a real position
969/// would say "this error has a position we happen not to know", which is a lie;

978    /// or implicitly by `exposing (..)` — whose declaration carries no type
979    /// annotation (`SPEC-5`: an exposed declaration must be annotated).
980    ///
981    /// The first [`NodeSpan`] is where the `exposing` list names it — a real
982    /// position for an explicit list, `NodeSpan::none()` for `exposing (..)`,

984    /// at (`parser::Exposing::Open` carries none at all). The second is the

1069    /// Nothing in the language constructs or inspects a value of one of these

1091    /// accepting the word there would make it mean nothing in half the places it
1092    /// can be written.

1121/// Canonicalization errors name source constructs — a value, a type, an operator —
1122/// so their messages can be written in the same words the user wrote.
1123///
1124/// The variants whose construction site had a declaration in hand also point at it;
1125/// see the enum's own documentation for why the others do not.
1126impl PhaseError for Error {
1127    fn message(&self) -> String {
1128        match self {
1129            Error::ExportNotFound(name, tpe, _) => format!(
1130                "`{}` is exposed by this module but no {} of that name is declared in it",
1131                name,
1132                export_type_noun(tpe)
1133            ),
1134            Error::ExportedValueNotAnnotated(name, _, _) => format!(
1135                "`{}` is exposed by this module but has no type annotation",
1136                name
1137            ),
1138            Error::EnvironmentErrors(errors) => match errors.as_slice() {
1139                [only] => only.message(),
1140                many => format!("{} of this module's imports could not be resolved", many.len()),
1141            },
1142            Error::InfixReferenceInvalidValue(infix, function, _) => format!(
1143                "the infix operator `{}` is declared as `{}`, which is not a value declared in this module",
1144                infix, function
1145            ),
1146            Error::AmbiguousOperatorPrecedence(left, right, _) => {
1147                // Three shapes, and the everyday one is the middle: `Basics`
1148                // declares six operators `infix non 4`, so `a < b > c` reaches
1149                // here with two *different* operators that do not disagree about
1150                // anything — both say they do not chain. Calling that a
1151                // disagreement about which side groups first would name a reason
1152                // that is not the reason.
1153                if left.name == right.name {
1154                    format!(
1155                        "`{}` is declared `infix non`, so it cannot be chained with itself without parentheses to say which application comes first",
1156                        left.name
1157                    )
1158                } else if left.is_non_associative() && right.is_non_associative() {
1159                    format!(
1160                        "`{}` and `{}` are both declared `infix non` at the same precedence, so neither groups the other and this needs parentheses",
1161                        left.name, right.name
1162                    )
1163                } else if left.is_non_associative() || right.is_non_associative() {
1164                    let (non, other) = if left.is_non_associative() {
1165                        (&left.name, &right.name)
1166                    } else {
1167                        (&right.name, &left.name)
1168                    };
1169
1170                    format!(
1171                        "`{}` is declared `infix non`, so it does not chain with `{}` at the same precedence without parentheses",
1172                        non, other
1173                    )
1174                } else {
1175                    format!(
1176                        "`{}` and `{}` have the same precedence but disagree on which side groups first, so this needs parentheses to say which one applies first",
1177                        left.name, right.name
1178                    )
1179                }
1180            }
1181            Error::BindingPatternsInvalidLen(_) => {
1182                "the arguments of this declaration do not line up with its type annotation"
1183                    .to_owned()
1184            }
1185            Error::NoBindings(_) => {
1186                "this declaration has a type annotation but no body".to_owned()
1187            }
1188            Error::VariableNotFound(name, _, _) => {
1189                format!("cannot find a value named `{}`", name.to_name())
1190            }
1191            Error::AmbiguousVariables(name, _, _) => {
1192                format!("`{}` is exposed by several imported modules", name)
1193            }
1194            Error::VariantNotFound(name, _, _) => {
1195                format!("cannot find a type constructor named `{}`", name.to_name())
1196            }
1197            Error::AmbiguousVariants(name, _, _) => format!(
1198                "the type constructor `{}` is exposed by several imported modules",
1199                name
1200            ),
1201            Error::InvalidTupleSize(size) => format!(
1202                "a tuple has two or three elements, this one has {}",
1203                size
1204            ),
1205            Error::MultipleBindingsUnsupported(name, _) => format!(
1206                "`{}` is declared over several bindings, which is not supported yet",
1207                name
1208            ),
1209            Error::TypeArityMismatch(name, declared, written, _) => format!(
1210                "`{}` takes {}, but is applied to {} here",
1211                name,
1212                type_argument_count(*declared),
1213                type_argument_count(*written)
1214            ),
1215            Error::InvalidVariant(kind, _) => match kind {
1216                InvalidVariantKind::LowercaseName(name) => format!(
1217                    "`{}` is not a constructor name: a constructor name begins with an uppercase letter",
1218                    name
1219                ),
1220                InvalidVariantKind::Tuple => {
1221                    "a variant is a constructor name followed by its arguments, and this one is a tuple type"
1222                        .to_owned()
1223                }
1224                InvalidVariantKind::Arrow => {
1225                    "a variant is a constructor name followed by its arguments, and this one is a function type"
1226                        .to_owned()
1227                }
1228            },
1229            Error::InvalidScalarDeclaration(name, _) => format!(
1230                "`{}` is an opaque scalar type, so its declaration must be exactly `type {} = {}`",
1231                name.to_name(),
1232                name.unqualified_name(),
1233                name.unqualified_name()
1234            ),
1235            Error::InfixDeclared(name, _) => format!(
1236                "a `module foreign` facade cannot declare an infix operator, but declares `{}`",
1237                name
1238            ),
1239            Error::TypeDeclared(name, _) => format!(
1240                "a `module foreign` facade cannot declare a type, but declares `{}`",
1241                name
1242            ),
1243            Error::NoTypeInBinding(name, _) => format!(
1244                "`{}` has no type annotation, and a `module foreign` facade is annotations only",
1245                name
1246            ),
1247            Error::UnsafeOutsideFacade(name, _) => format!(
1248                "`{}` is marked `unsafe`, which only a signature in a `module foreign` facade may be",
1249                name
1250            ),
1251            Error::Many(errors) => match errors.as_slice() {

1255        }

1258    fn labels(&self) -> Vec<SpanLabel> {
1259        // One helper per shape: a variant either has a span and one thing to say
1260        // about it, or it delegates to the errors it wraps.
1261        let primary = |span: &NodeSpan, message: &str| match span.span() {
1262            Some(span) => vec![SpanLabel {
1263                span,
1264                message: message.to_owned(),
1265                primary: true,
1266                file: None,
1267            }],
1268            None => Vec::new(),
1269        };
1270
1271        // A secondary label pointing at where one ambiguous candidate is declared,
1272        // in *its own* module's file rather than the one being checked — the
1273        // mechanism `ERR-5` adds. `None` when that candidate's `Interface` cannot
1274        // say (a hand-built interface, or one built before its file was known)
1275        // yields no label for that candidate rather than one at the wrong place.
1276        let ambiguous_candidates =
1277            |candidates: &[(ModuleName, Option<super::SourceSpan>, ImportOrigin)]| {
1278                candidates
1279                    .iter()
1280                    .filter_map(|(module, source, _origin)| {
1281                        source.map(|s| SpanLabel {
1282                            span: s.span,
1283                            message: format!("also exposed here, by `{}`", module.name()),
1284                            primary: false,
1285                            file: Some(s.file),
1286                        })
1287                    })
1288                    .collect::<Vec<_>>()
1289            };
1290
1291        match self {
1292            Error::ExportNotFound(name, _, span) => primary(
1293                span,
1294                &format!("`{}` is not declared anywhere in this module", name),
1295            ),
1296            Error::ExportedValueNotAnnotated(name, exposed_span, declared_span) => {
1297                let mut labels = primary(
1298                    exposed_span,
1299                    &format!("`{}` is exposed here but has no type annotation", name),
1300                );
1301                if labels.is_empty() {
1302                    // Reached only for `exposing (..)`: it names nothing
1303                    // individually, so there is no exposing-list position to put
1304                    // the primary label under — the declaration becomes the
1305                    // primary (and only) label instead of a secondary one.
1306                    labels = primary(
1307                        declared_span,
1308                        &format!(
1309                            "`{}` is exposed by `exposing (..)` but has no type annotation",
1310                            name
1311                        ),
1312                    );
1313                } else if let Some(span) = declared_span.span() {
1314                    labels.push(SpanLabel {
1315                        span,
1316                        message: "declared here, with no type annotation".to_owned(),
1317                        primary: false,
1318                        file: None,
1319                    });
1320                }
1321                labels
1322            }
1323            Error::InfixReferenceInvalidValue(_, _, span) => primary(span, "declared here"),
1324            // The two "declared here" labels go through `InfixDeclaration`, which
1325            // is what knows whether the declaration is in the module under check
1326            // or in an imported one. An operator's `infix` declaration is very
1327            // often not local — `Basics` declares every one the standard library
1328            // uses — and `Infix::span` is then a byte range in *that* module's
1329            // file, so a label built with `file: None` would underline unrelated
1330            // text in the importing module (`ERR-5` is the mechanism that avoids
1331            // it).
1332            Error::AmbiguousOperatorPrecedence(left, right, span) => {
1333                let mut labels = primary(span, "ambiguous without parentheses");
1334
1335                labels.extend(
1336                    left.declaration
1337                        .label(format!("`{}` declared here", left.name)),
1338                );
1339
1340                // Both sides naming the same operator is the `infix non`
1341                // self-conflict — they point at the very same declaration, so a
1342                // second label there would only repeat the first one.
1343                if left.name != right.name {
1344                    labels.extend(
1345                        right
1346                            .declaration
1347                            .label(format!("`{}` declared here", right.name)),
1348                    );
1349                }
1350
1351                labels
1352            }
1353            // The four that name an identifier: the caret sits under the name the
1354            // user wrote, which is the whole point of spanning expressions and
1355            // patterns rather than only declarations. `VariableNotFound` and
1356            // `VariantNotFound` append their suggestion, when they have one, to
1357            // this same label rather than a free-floating note, so the caret and
1358            // the suggestion agree about which name is meant (`ERR-7`).
1359            Error::VariableNotFound(_, span, suggestion) => primary(
1360                span,
1361                &format!(
1362                    "no value of this name is in scope{}",
1363                    suggestion_suffix(suggestion)
1364                ),
1365            ),
1366            Error::AmbiguousVariables(_, candidates, span) => {
1367                let mut labels = primary(span, "this name is ambiguous");
1368                labels.extend(ambiguous_candidates(candidates));
1369                labels
1370            }
1371            Error::VariantNotFound(_, span, suggestion) => primary(
1372                span,
1373                &format!(
1374                    "no type constructor of this name is in scope{}",
1375                    suggestion_suffix(suggestion)
1376                ),
1377            ),
1378            Error::AmbiguousVariants(_, candidates, span) => {
1379                let mut labels = primary(span, "this type constructor is ambiguous");
1380                labels.extend(ambiguous_candidates(candidates));
1381                labels
1382            }
1383            Error::BindingPatternsInvalidLen(span) => primary(span, "declared here"),
1384            Error::NoBindings(span) => primary(span, "this annotation has no body"),
1385            Error::MultipleBindingsUnsupported(_, span) => primary(span, "declared here"),
1386            Error::TypeArityMismatch(name, declared, written, span) => primary(
1387                span,
1388                &format!(
1389                    "`{}` takes {}, this application has {}",
1390                    name,
1391                    type_argument_count(*declared),
1392                    type_argument_count(*written)
1393                ),
1394            ),
1395            Error::InvalidVariant(kind, span) => primary(
1396                span,
1397                match kind {
1398                    InvalidVariantKind::LowercaseName(_) => "this begins with a lowercase letter",
1399                    InvalidVariantKind::Tuple => "a tuple type, written where a variant belongs",
1400                    InvalidVariantKind::Arrow => "a function type, written where a variant belongs",
1401                },
1402            ),
1403            Error::InvalidScalarDeclaration(name, span) => primary(
1404                span,
1405                &format!(
1406                    "an opaque scalar's body must be exactly `{}`, with no other variant and no arguments",
1407                    name.unqualified_name()
1408                ),
1409            ),
1410            Error::InfixDeclared(_, span) => primary(span, "declared here"),
1411            Error::TypeDeclared(_, span) => primary(span, "declared here"),
1412            Error::NoTypeInBinding(_, span) => primary(span, "declared here"),
1413            Error::UnsafeOutsideFacade(_, span) => primary(span, "marked `unsafe` here"),
1414            // A group has no position of its own; the errors it swallowed do.
1415            Error::EnvironmentErrors(errors) => errors.iter().flat_map(|e| e.labels()).collect(),
1416            Error::Many(errors) => errors.iter().flat_map(|e| e.labels()).collect(),
1417            _ => Vec::new(),
1418        }
1419    }

1461/// The `it is exposed by: …` note shared by `Error::AmbiguousVariables` and
1462/// `Error::AmbiguousVariants` — one clause for the contributors the module wrote
1463/// an `import` for, one for those supplied by [the default import
1464/// list](../../../docs/spec/modules.md#the-default-imports) because it wrote
1465/// none. Naming the latter as implicit is what removes the surprise a module can
1466/// otherwise get from colliding with an import it never wrote (`SPEC-32`).
1467///
1468/// Every candidate this ever runs over comes from `ValueType::Foreigns`, so
1469/// `written`/`implicit` are never both empty in practice — `Foreigns` has at
1470/// least two entries and each is one or the other — but the match is written to
1471/// say something sensible even if that stopped holding.
1472fn ambiguous_note(candidates: &[(ModuleName, Option<super::SourceSpan>, ImportOrigin)]) -> String {
1473    let name_of =
1474        |(m, _, _): &(ModuleName, Option<super::SourceSpan>, ImportOrigin)| m.name().to_string();
1475    let written: Vec<String> = candidates
1476        .iter()
1477        .filter(|(_, _, origin)| *origin == ImportOrigin::Written)
1478        .map(name_of)
1479        .collect();
1480    let implicit: Vec<String> = candidates
1481        .iter()
1482        .filter(|(_, _, origin)| *origin == ImportOrigin::Default)
1483        .map(name_of)
1484        .collect();
1485
1486    match (written.is_empty(), implicit.is_empty()) {
1487        (false, true) => format!("it is exposed by: {}", written.join(", ")),
1488        (true, false) => format!("it is exposed implicitly by: {}", implicit.join(", ")),
1489        (true, true) => "it is exposed by nothing this module can see".to_owned(),
1490        (false, false) => format!(
1491            "it is exposed by: {}, and implicitly by {}",
1492            written.join(", "),
1493            implicit.join(", ")
1494        ),
1495    }
1496}

1529/// Transform a given `parser::Module` into a `canonical::Module`.
1530///
1531/// `package_declares_a_default` is `check_module`'s own parameter, passed straight
1532/// through to [`new_environment`] — see its doc comment for what it means and
1533/// where it comes from.
1534pub fn canonicalize(
1535    package: &PackageName,
1536    interfaces: &HashMap<Name, Interface>,
1537    source: &parser::Module,
1538    package_declares_a_default: bool,
1539) -> Result<Module, Vec<Error>> {
1540    let name = ModuleName {
1541        package: package.clone(),
1542        name: source.name.clone(),
1543    };
1544
1545    let mut errors: Vec<Error> = vec![];
1546    let mut env = new_environment(
1547        &name,
1548        interfaces,
1549        &source.imports,
1550        package_declares_a_default,
1551    )
1552    .map_err(|e| vec![e.into()])?;
1553
1554    // `unsafe` is a claim about the companion standing behind a facade signature,
1555    // so it has nothing to say on a declaration with a body above it. The grammar
1556    // accepts the word on any annotation — it has no way to know the module's
1557    // header — which leaves this as the only place that can reject one. Reported
1558    // for every marked declaration, then canonicalization carries on, so a module
1559    // with a stray `unsafe` still reports whatever else is wrong with it.
1560    if !source.binding_foreign {
1561        errors.extend(
1562            source
1563                .functions
1564                .iter()
1565                .filter(|f| f.marked_unsafe)
1566                .map(|f| Error::UnsafeOutsideFacade(f.name.clone(), f.annotation_span)),
1567        );
1568    }
1569
1570    let (infixes, types, values) = if source.binding_foreign {
1571        // A `module foreign` facade runs a parallel canonicalization process as the constraints are a bit different:
1572        // - Only functions without bindings are authorized.
1573        // - Infixes and types are forbidden.
1574        // The idea being to have the facade stand for the companion file shipped beside it.
1575        // Assuming Json types are part of the prelude, this should goes well with the restriction on what types are available for bindings.
1576
1577        // Verify no infix present
1578        if !source.infixes.is_empty() {
1579            let e = source
1580                .infixes
1581                .iter()
1582                .map(|i| Error::InfixDeclared(i.operator.clone(), i.span));
1583            errors.extend(e);
1584        }
1585        // Verify no types present
1586        if !source.types.is_empty() {
1587            let e = source
1588                .types
1589                .iter()
1590                .map(|t| Error::TypeDeclared(t.name.clone(), t.span));
1591            errors.extend(e);
1592        }
1593
1594        // Register each binding as a top-level value before resolving any of
1595        // them, the same as the non-`foreign` branch below — otherwise
1596        // `do_exports`'s existence check (`BUG-8`) would reject a facade
1597        // exposing its own declared binding, since nothing would have told
1598        // `env` the binding exists.
1599        for f in source.functions.iter() {
1600            env.insert_top_level_value(f.name.clone());
1601        }
1602
1603        // Iterate on values
1604        let iter = source.functions.iter().map(|function| {
1605            // Make sure there is no binding
1606            if !function.bindings.is_empty() {
1607                //println!("bindings = {:?} (js module)", function.bindings);
1608                Err(Error::BindingPatternsInvalidLen(function.span))? // TODO More specific error
1609            }
1610
1611            // Make sure there is a type
1612            let tpe = function
1613                .tpe
1614                .as_ref()
1615                .ok_or_else(|| Error::NoTypeInBinding(function.name.clone(), function.span))?;
1616            let tpe = Type::from_parser_type(&env, tpe)?;
1617
1618            let name = function.name.clone();
1619            // TODO Think how it's going to be represented. Currently canonical values assume an expression is present
1620            //      I'd like to not introduce a trait or new struct for binding. Should we fake an expression or create
1621            //      a new type of value ? New type of value will be annoying for regular modules as they aren't present
1622            //      there. Fake expression might be ok as MVP. We have to make sure that binding module are removed from
1623            //      some phase of the compilation pipeline.
1624            let value = Value::TypedValue {
1625                name: name.clone(),
1626                patterns: vec![],
1627                // A `module foreign` facade has no body in the source, so this
1628                // stand-in has nothing to point at (see the TODO above).
1629                body: Expression::bare(ExpressionKind::Bool(true)),
1630                tpe,
1631                marked_unsafe: function.marked_unsafe,
1632                span: function.span,
1633                annotation_span: function.annotation_span,
1634            };
1635
1636            Ok((name, value))
1637        });
1638        let values = crate::utils::collect_accumulate(iter).unwrap_or_else(|err| {
1639            errors.extend(err);
1640            HashMap::new()
1641        });
1642
1643        (HashMap::new(), HashMap::new(), values)
1644    } else {
1645        // Because we are rewriting infixes in this phase, we must do this check before
1646        // resolving values.
1647        let infixes =
1648            do_infixes(&source.infixes, &mut env, &source.functions).unwrap_or_else(|err| {
1649                errors.extend(err);
1650                HashMap::new()
1651            });
1652
1653        let types = do_types(&env, &source.types).unwrap_or_else(|err| {
1654            errors.extend(err);
1655            HashMap::new()
1656        });
1657
1658        for (n, t) in types.iter() {
1659            env.insert_union_type(n.clone(), t.clone());
1660        }
1661
1662        trace!("Environment after do_types: {:#?}", env);
1663
1664        // TODO Should I manage infixes rewrite here too ?
1665        // Yes I should do it here
1666        let values = do_values(&mut env, &source.functions).unwrap_or_else(|err| {
1667            errors.extend(err);
1668            HashMap::new()
1669        });
1670
1671        (infixes, types, values)
1672    };
1673
1674    // We do exports at the end, and verify that all exported value do
1675    // have a reference within the current module
1676    let exports = do_exports(&source.exposing, &env, &values).unwrap_or_else(|err| {
1677        errors.extend(err);
1678        Exports::Everything // Never exposed, as we will return the errors instead
1679    });
1680
1681    if errors.is_empty() {
1682        Ok(Module {
1683            name,
1684            exports,
1685            infixes,
1686            types,
1687            values,
1688            binding_foreign: source.binding_foreign,
1689        })
1690    } else {
1691        Err(errors)
1692    }
1693}

1695fn do_values(
1696    env: &mut RootEnvironment,
1697    functions: &[parser::Function],
1698) -> Result<HashMap<Name, Value>, Vec<Error>> {
1699    // Before resolving expressions, we store the top-level values in the environment.
1700    // We do so first because their expression below could refer to them.
1701    for f in functions.iter() {
1702        env.insert_top_level_value(f.name.clone());
1703    }
1704
1705    let iter = functions.iter().map(|function| {
1706        // Bindings to expression
1707
1708        // TODO Better error message with position of mismatch
1709        // TODO Error when bindings is empty
1710        let bindings_size = function
1711            .bindings
1712            .iter()
1713            .all(|v| v.patterns.len() == function.bindings[0].patterns.len());
1714
1715        if !bindings_size {
1716            //println!("bindings = {:?} (bindings_size)", function.bindings);
1717            Err(Error::BindingPatternsInvalidLen(function.span))?
1718        }
1719
1720        let (patterns, body): (Vec<Pattern>, Expression) = match function.bindings.len() {
1721            0 => Err(Error::NoBindings(function.span)),
1722            1 => {
1723                // if one binding, we can convert directly to canonical format
1724                let binding = &function.bindings[0];
1725
1726                let mut scoped = env.new_scope();
1727
1728                let patterns: Vec<Pattern> = binding
1729                    .patterns
1730                    .iter()
1731                    .map(|p| Pattern::from_parser(p, env))
1732                    .collect::<Result<Vec<_>, Error>>()?;
1733
1734                for p in &patterns {
1735                    scoped.expose_pattern(p);
1736                }
1737
1738                // Maybe create a case_branch function and make it common with Expression::Case ?
1739                // Or maybe not at the case_branch level, as here we can have multiple patterns
1740                // whereas cases cannot.
1741                // eg. a: Int -> Int -> Int  ==>  a b c = b + c
1742                //println!("Env before transforming expression: {:?}", scoped);
1743                let body = Expression::from_parser(&binding.body, &scoped)?;
1744
1745                Ok((patterns, body))
1746            }
1747            _ => {
1748                // if multiple bindings, we need to create synthetics variables and put all bindings into a case expression
1749                Err(Error::MultipleBindingsUnsupported(
1750                    function.name.clone(),
1751                    function.span,
1752                ))
1753            }
1754        }?;
1755
1756        let name = function.name.clone();
1757
1758        match &function.tpe {
1759            Some(t) => {
1760                let tpe = Type::from_parser_type(env, t)?;
1761                let linear = Type::to_linear_types(&tpe);
1762
1763                // Linear is a list of types making the function. Because it includes the return type,
1764                // it will always be bigger than the number of patterns by one.
1765                if !patterns.is_empty() && (linear.len() - 1 != patterns.len()) {
1766                    // TODO Better error message
1767                    debug!(
1768                        "linear = {:#?}\nbindings = {:#?} (linear.len ({}) != patterns.len ({}))",
1769                        linear,
1770                        function.bindings,
1771                        linear.len(),
1772                        patterns.len()
1773                    );
1774                    Err(Error::BindingPatternsInvalidLen(function.span))?
1775                }
1776
1777                let patterns = patterns.into_iter().zip(linear).collect();
1778
1779                Ok((
1780                    function.name.clone(),
1781                    Value::TypedValue {
1782                        name,
1783                        patterns,
1784                        body,
1785                        tpe,
1786                        // `do_values` only runs for a module that is not a facade,
1787                        // and `canonicalize` has already rejected a marked
1788                        // annotation there.
1789                        marked_unsafe: false,
1790                        span: function.span,
1791                        annotation_span: function.annotation_span,
1792                    },
1793                ))
1794            }
1795            None => Ok((
1796                function.name.clone(),
1797                Value::Value {
1798                    name,
1799                    patterns,
1800                    body,
1801                    span: function.span,
1802                },
1803            )),
1804        }
1805    });
1806
1807    collect_accumulate(iter)
1808}

1810fn do_types(
1811    env: &dyn Environment,
1812    types: &[parser::UnionType],
1813) -> Result<HashMap<Name, UnionType>, Vec<Error>> {
1814    let iter = types.iter().map(|tpe| {
1815        let tpe_name = tpe.name.clone();
1816        // The declaration is this module's, so this is where its variants learn
1817        // which module the type they build belongs to (`AST-4`).
1818        let qualified_tpe_name = env.module_name().qualify_name(&tpe_name);
1819        let variables = tpe.type_arguments.clone();
1820
1821        trace!("do_types(in:{:?})", tpe);
1822
1823        // An opaque scalar's declaration writes only its own name and contributes
1824        // no constructor (`LANG-59`, `DEC-15` decision 2): nothing in the language
1825        // builds or inspects an `Int`, a `Float`, a `Char` or a `String`, so the
1826        // only body worth accepting is the one that says so. This runs on the
1827        // *qualified* name, the same way `scalars::Scalar::declares` always does —
1828        // a module other than the one a scalar is declared in shares its spelling
1829        // and nothing else (`BUG-26`), so its own `type Int = …` reaches the
1830        // ordinary path below untouched.
1831        if let Some(scalar) = scalars::opaque_scalar_of(&qualified_tpe_name) {
1832            let is_self_naming = matches!(
1833                tpe.variants.as_slice(),
1834                [variant] if matches!(
1835                    &variant.kind,
1836                    parser::TypeKind::Unqualified(name, args) if *name == tpe_name && args.is_empty()
1837                )
1838            );
1839
1840            if !is_self_naming {
1841                return Err(Error::InvalidScalarDeclaration(
1842                    scalar.qual_name(),
1843                    tpe.span,
1844                ));
1845            }
1846
1847            return Ok((
1848                tpe_name,
1849                UnionType {
1850                    variables,
1851                    variants: Vec::new(),
1852                    span: tpe.span,
1853                },
1854            ));
1855        }
1856
1857        // A variant is a constructor name and its arguments, which the parser spells
1858        // `TypeKind::Unqualified`. It is the grammar's general `Type` production that
1859        // parses a variant list, though, so the three other kinds arrive here too —
1860        // and each is a declaration the user wrote that has no meaning, not a variant
1861        // this pass may leave out. Skipping one deletes a constructor from the
1862        // declaration and reports nothing, which was `BUG-18`.
1863        //
1864        // The `collect` below short-circuits on the first `Err`, so only the first bad
1865        // variant in a single `type` declaration is reported — `type T = c | (Int,
1866        // Int)` names only `c`. The enclosing `collect_accumulate` still reports the
1867        // next `type` declaration independently, matching what the `Unqualified` arm
1868        // already did for `from_parser_type`.
1869        let variants = tpe
1870            .variants
1871            .iter()
1872            .map(|t| match &t.kind {
1873                // TODO It might actually make more sense to put Type::from_parser_type
1874                // on `Environment`.
1875                parser::TypeKind::Unqualified(name, vars) => {
1876                    let type_parameters = vars
1877                        .iter()
1878                        .map(|t| Type::from_parser_type(env, t))
1879                        .collect::<Result<Vec<_>, Error>>()?;
1880
1881                    Ok(TypeConstructor {
1882                        name: name.clone(),
1883                        type_parameters,
1884                        tpe: qualified_tpe_name.clone(),
1885                    })
1886                }
1887                parser::TypeKind::Variable(name) => Err(Error::InvalidVariant(
1888                    InvalidVariantKind::LowercaseName(name.clone()),
1889                    t.span,
1890                )),
1891                parser::TypeKind::Tuple(_) => {
1892                    Err(Error::InvalidVariant(InvalidVariantKind::Tuple, t.span))
1893                }
1894                parser::TypeKind::Arrow(_, _) => {
1895                    Err(Error::InvalidVariant(InvalidVariantKind::Arrow, t.span))
1896                }
1897            })
1898            .collect::<Result<Vec<_>, Error>>()?;
1899
1900        Ok((
1901            tpe_name,
1902            UnionType {
1903                variables,
1904                variants,
1905                span: tpe.span,
1906            },
1907        ))
1908    });
1909
1910    collect_accumulate(iter)
1911}

1913fn do_infixes(
1914    infixes: &[parser::Infix],
1915    env: &mut RootEnvironment,
1916    functions: &[parser::Function],
1917) -> Result<HashMap<Name, Infix>, Vec<Error>> {
1918    let iter = infixes.iter().map(|infix| {
1919        let op_name = infix.operator.clone();
1920        let function_name = infix.function_name.clone();
1921
1922        let function_exist = functions
1923            .iter()
1924            .find(|f| f.name == infix.function_name)
1925            .is_some();
1926
1927        if function_exist {
1928            let infix = Infix {
1929                associativity: infix.associativity,
1930                precedence: infix.precedence,
1931                function_name,
1932                span: infix.span,
1933            };
1934
1935            env.insert_local_infix(op_name.clone(), infix.clone());
1936
1937            Ok((op_name, infix))
1938        } else {
1939            Err(Error::InfixReferenceInvalidValue(
1940                op_name,
1941                function_name,
1942                infix.span,
1943            ))
1944        }
1945    });
1946
1947    collect_accumulate(iter)
1948}

1950// Every arm below checks that the name it names actually resolves in `env`
1951// before accepting it — a `Lower`/`Upper`/`Operator` name in a module's own
1952// `exposing (...)` header that nothing declares is `Error::ExportNotFound`
1953// rather than silently accepted (`BUG-8`). `Upper`'s two arms differ only in
1954// which `ExportType` they report on success: `Privacy` governs whether the
1955// type's constructors are exposed, not whether the type itself exists, so
1956// both check existence with `find_type` the same way.
1957//
1958// `values` is this module's own declarations (from `do_values`/the
1959// facade iterator above), separate from `env`: `env.find_value` also
1960// answers `Some` for a name resolved through an import, and a re-exported
1961// foreign value is already guaranteed typed by its own module's `do_exports`
1962// (`SPEC-5` applies there, transitively, through `Interface::values` only
1963// ever holding typed entries) — so `Lower`'s annotation check is scoped to a
1964// name this module declares itself, and leaves a name it does not declare to
1965// the existence check above.
1966fn do_exports(
1967    source_exposing: &parser::Exposing,
1968    env: &dyn Environment,
1969    values: &HashMap<Name, Value>,
1970) -> Result<Exports, Vec<Error>> {
1971    match source_exposing {
1972        // `exposing (..)` exposes every top-level declaration this module
1973        // makes, so `SPEC-5` applies to every one of them, not just those
1974        // named individually — and there is no per-name span in this header
1975        // to blame (`parser::Exposing::Open` carries none), so
1976        // `ExportedValueNotAnnotated`'s first span is `NodeSpan::none()` and
1977        // the declaration itself is what a diagnostic points at.
1978        parser::Exposing::Open => {
1979            let checked = values.values().map(|value| match value {
1980                Value::TypedValue { .. } => Ok(()),
1981                Value::Value { name, span, .. } => Err(Error::ExportedValueNotAnnotated(
1982                    name.clone(),
1983                    NodeSpan::none(),
1984                    *span,
1985                )),
1986            });
1987
1988            let _: Vec<()> = collect_accumulate(checked)?;
1989
1990            Ok(Exports::Everything)
1991        }
1992        parser::Exposing::Explicit(exposed) => {
1993            let specifics =
1994                exposed.iter().map(|exposed| match &exposed.kind {
1995                    parser::ExposedKind::Lower(name) => {
1996                        if env.find_value(name).is_none() {
1997                            return Err(Error::ExportNotFound(
1998                                name.clone(),
1999                                ExportType::Value,
2000                                exposed.span,
2001                            ));
2002                        }
2003
2004                        match values.get(name) {
2005                            Some(Value::Value { span, .. }) => Err(
2006                                Error::ExportedValueNotAnnotated(name.clone(), exposed.span, *span),
2007                            ),
2008                            // `Some(TypedValue)` is declared locally and annotated;
2009                            // `None` is resolved through an import instead of a
2010                            // local declaration, already covered above.
2011                            Some(Value::TypedValue { .. }) | None => {
2012                                Ok((name.clone(), ExportType::Value))
2013                            }
2014                        }
2015                    }
2016                    // Privacy governs whether the type's constructors are exposed,
2017                    // not whether the type itself exists — both arms check
2018                    // existence the same way, and only the `ExportType` they
2019                    // report on success differs.
2020                    parser::ExposedKind::Upper(name, parser::Privacy::Public) => {
2021                        if env.find_type(name).is_some() {
2022                            Ok((name.clone(), ExportType::UnionPublic))
2023                        } else {
2024                            Err(Error::ExportNotFound(
2025                                name.clone(),
2026                                ExportType::UnionPublic,
2027                                exposed.span,
2028                            ))
2029                        }
2030                    }
2031                    parser::ExposedKind::Upper(name, parser::Privacy::Private) => {
2032                        if env.find_type(name).is_some() {
2033                            Ok((name.clone(), ExportType::UnionPrivate))
2034                        } else {
2035                            Err(Error::ExportNotFound(
2036                                name.clone(),
2037                                ExportType::UnionPrivate,
2038                                exposed.span,
2039                            ))
2040                        }
2041                    }
2042                    parser::ExposedKind::Operator(name) => {
2043                        if env.local_infix_exists(name) {
2044                            Ok((name.clone(), ExportType::Infix))
2045                        } else {
2046                            Err(Error::ExportNotFound(
2047                                name.clone(),
2048                                ExportType::Infix,
2049                                exposed.span,
2050                            ))
2051                        }
2052                    }
2053                });
2054
2055            let specifics = collect_accumulate(specifics)?;
2056
2057            Ok(Exports::Specifics(specifics))
2058        }
2059    }
2060}

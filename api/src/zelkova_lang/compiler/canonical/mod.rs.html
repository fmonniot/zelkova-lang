1//! The canonical representation of a zelkova programs is a translation of a local
2//! source into the broader world.
3//!
4//! This phase is where we integrate the local parsed module into the rest of the
5//! program. We do the following steps:
6//!
7//! - Resolve all imports
8//! - Qualify all `Name` (eg. a local value `test` in a `Mod.A` module will be renamed `Mod.A.test`)
9//! - Checks that exported names are actually present in the module
10//! - Checks there is none cyclic dependency between this module and others (Might be done earlier, let's see)
11//!
12//! Note that we use `HashMap`'s a lot in this module's structures. This is because later phases will
13//! want to have cheap access to the different components of a `Module`.
14//!
15//! TODO Rename this to core ? I feel it's going to be te main internal representation of the language.

42/// A resolved module
43#[derive(Debug)]
44pub struct Module {
45    pub name: ModuleName,
46    pub exports: Exports,
47    /// Operator name to infix details
48    pub infixes: HashMap<Name, Infix>,
49    pub types: HashMap<Name, UnionType>,
50    pub values: HashMap<Name, Value>,
51    /// True when this module is a JavaScript binding module.
52    /// Such modules have synthetic placeholder bodies and must not be type-checked.
53    pub binding_javascript: bool,
54}
55
56impl Module {
57    /// Build the trimmed-down view of this module other modules import against.
58    ///
59    /// `file` is the [`SourceFileId`] this module was read from, which is what lets
60    /// a later module's diagnostic point back into *this* module's source — see
61    /// [`Interface::file`](super::Interface::file). It is a parameter rather than
62    /// something this method could work out because a `canonical::Module` does not
63    /// know it: only driver code does, and the sole caller
64    /// (`dependencies::ModuleWalker::check_in_order`) is driver code that has the
65    /// map from module name to file in hand. A caller with nothing to give — every
66    /// test that hand-builds an interface — passes `None`, and
67    /// [`Interface::source_span`](super::Interface::source_span) then declines to
68    /// build a cross-module label rather than building one at the wrong place.
69    pub fn to_interface(&self, file: Option<SourceFileId>) -> super::Interface {
70        let values = self
71            .values
72            .iter()
73            .filter_map(|(name, value)| match value {
74                Value::Value { .. } => None,
75                Value::TypedValue { tpe, span, .. } => Some((name.clone(), (*span, tpe.clone()))),
76            })
77            .collect();
78
79        super::Interface {
80            module_name: self.name.clone(),
81            values,
82            unions: self.types.clone(),
83            infixes: self.infixes.clone(),
84            file,
85        }
86    }
87}

137/// A canonical type.
138///
139/// # Why there is no span on this type, or on [`TypeConstructor`]
140///
141/// Every other canonical node this module builds carries a [`NodeSpan`] taken from
142/// the parser node it came from. A `Type` deliberately does not.
143///
144/// The reason `ERR-3` recorded is that a `Type` does not always come from the module
145/// being canonicalized: `Type::from_parser_type` resolves a name through the
146/// [`Environment`], which clones types straight out of the [`Interface`]s of the
147/// modules this one imports, so the `Type` handed back may well have been *written
148/// in a different file* — and a bare span on it would be read as a position in the
149/// importing module's source. `ERR-5` settled that half: a [`SpanLabel`] can now
150/// carry a file of its own, so "written in another file" is no longer the obstacle.
151///
152/// The other half is unfinished work, not an impossibility. `parser::Type` carries a
153/// [`NodeSpan`] per node like every other parser node, and `Type::from_parser_type`
154/// is the recursive walk over it — `tpe.span` is in hand at every level, and each
155/// branch of an `Arrow` arrives as its own spanned `parser::Type`. Those spans are
156/// discarded here by choice. Keeping them means a span field on every canonical
157/// `Type` node, plus a decision about what a `Type` cloned out of an `Environment`
158/// should then report — where it was written, or where it was used. Nobody has
159/// written that.
160///
161/// What exists instead is coarser and enough for the diagnostics that exist: the
162/// *declaration* a `Type` is the type of — [`Value`]'s own `span`, [`UnionType`]'s,
163/// [`Infix`]'s — is what a diagnostic wants to underline ("defined here"), not a
164/// position inside the type expression itself. [`Interface::values`] pairs each
165/// value's `Type` with that declaration's [`NodeSpan`], and
166/// [`Interface::source_span`] pairs it with the file, once [`Interface::file`] is
167/// set. A type error still points at the whole declaration that failed rather than
168/// the sub-expression that disagrees, which is `ERR-4`, a separate limit in the
169/// typer's own `Term`/`Constraint` translation.
170#[derive(Debug, Clone, PartialEq)]
171pub enum Type {
172    Variable(Name),
173    Type(Name, Vec<Type>),
174    // Record
175    // Unit
176    Arrow(Box<Type>, Box<Type>),
177    /// A tuple type. Zelkova keeps Elm's restriction of two or three elements,
178    /// which [`Tuple`] carries in its shape.
179    Tuple(Tuple<Type>),
180    // Alias
181}
182
183impl Type {
184    /// The one conversion in this module that reads a span and produces none.
185    ///
186    /// `parser::Type` carries a [`NodeSpan`] like every other parser node, and it is
187    /// dropped here on purpose — see this type's documentation for why a canonical
188    /// `Type` holds no span at all, and what `ERR-5` built instead.
189    fn from_parser_type(env: &dyn Environment, tpe: &parser::Type) -> Result<Type, Error> {
190        match &tpe.kind {
191            parser::TypeKind::Unqualified(name, vars) => {
192                let args = vars
193                    .iter()
194                    .map(|t| Type::from_parser_type(env, t))
195                    .collect::<Result<Vec<_>, Error>>()?;
196
197                match env.find_type(name) {
198                    // `name` resolves to a declared type: `args` is what was
199                    // written after it, and has to match the declaration's own
200                    // arity (`BUG-17`) — nothing else here re-derives that check.
201                    // The head is the *declaration's* name, not the one written:
202                    // `Lib.Option` and `Option` are two spellings of one entry, and
203                    // normalizing here is what lets the two unify downstream.
204                    Some(declared) if declared.arity() == args.len() => {
205                        Ok(Type::Type(declared.name.clone(), args))
206                    }
207                    Some(declared) => Err(Error::TypeArityMismatch(
208                        name.clone(),
209                        declared.arity(),
210                        args.len(),
211                        tpe.span,
212                    )),
213                    // `name` resolves to nothing: BUG-16 is the ticket for reporting
214                    // this instead of fabricating a type for it.
215                    None => Ok(Type::Type(name.clone(), args)),
216                }
217            }
218            parser::TypeKind::Arrow(t1, t2) => Ok(Type::Arrow(
219                Box::new(Type::from_parser_type(env, t1)?),
220                Box::new(Type::from_parser_type(env, t2)?),
221            )),
222            parser::TypeKind::Variable(n) => Ok(Type::Variable(n.clone())),
223            parser::TypeKind::Tuple(tuple) => Ok(Type::Tuple(
224                tuple.try_map(|t| Type::from_parser_type(env, t))?,
225            )),
226        }
227    }
228
229    // TODO Write some tests
230    fn to_linear_types(tpe: &Type) -> Vec<Type> {
231        match tpe {
232            Type::Arrow(a, b) => {
233                let mut next = Type::to_linear_types(b);
234
235                next.insert(0, *a.clone());
236
237                next
238            }
239            _ => vec![tpe.clone()],
240        }
241    }
242}

282/// A canonical pattern, and where it was written.
283///
284/// Same shape as [`parser::Pattern`] — a [`NodeSpan`] beside a kind — and for the
285/// same reason: the children stay plain `Pattern`s, so a reader matches `&p.kind`.
286#[derive(Debug, PartialEq)]
287pub struct Pattern {
288    pub span: NodeSpan,
289    pub kind: PatternKind,
290}
291
292#[derive(Debug, PartialEq)]
293pub enum PatternKind {
294    Anything,
295    Variable(Name), // TODO Name or QualName ?
296    Int(i64),
297    Float(f64),
298    Char(char),
299    Bool(bool),
300    /// A tuple pattern. Zelkova keeps Elm's restriction of two or three
301    /// elements, which [`Tuple`] carries in its shape.
302    Tuple(Tuple<Pattern>),
303
304    Constructor {
305        ctor: TypeConstructor,
306        args: Vec<Pattern>,
307    },
308}
309
310impl Pattern {
311    /// A pattern canonicalized from source, keeping the parser node's position.
312    pub fn new(span: NodeSpan, kind: PatternKind) -> Pattern {
313        Pattern { span, kind }
314    }
315
316    /// A pattern with no position — hand-built by a test. See [`NodeSpan`].
317    pub fn bare(kind: PatternKind) -> Pattern {
318        Pattern {
319            span: NodeSpan::none(),
320            kind,
321        }
322    }
323
324    fn from_parser(p: &parser::Pattern, env: &dyn Environment) -> Result<Pattern, Error> {
325        let kind = match &p.kind {
326            parser::PatternKind::Anything => PatternKind::Anything,
327            parser::PatternKind::Variable(name) => PatternKind::Variable(name.clone()),
328            parser::PatternKind::Literal(parser::Literal::Int(i)) => PatternKind::Int(*i),
329            parser::PatternKind::Literal(parser::Literal::Float(f)) => PatternKind::Float(*f),
330            parser::PatternKind::Literal(parser::Literal::Char(c)) => PatternKind::Char(*c),
331            parser::PatternKind::Literal(parser::Literal::Bool(b)) => PatternKind::Bool(*b),
332            parser::PatternKind::Tuple(tuple) => {
333                PatternKind::Tuple(tuple.try_map(|p| Pattern::from_parser(p, env))?)
334            }
335            parser::PatternKind::Constructor(name, args) => {
336                // `p.span` covers the constructor and its arguments, which is the
337                // text a "no such constructor" caret should sit under.
338                let ctor = env
339                    .find_type_constructor(name)
340                    .ok_or_else(|| {
341                        let suggestion =
342                            suggest_name(name, env.type_constructor_names().into_iter());
343                        Error::VariantNotFound(
344                            env.module_name().qualify_name(name),
345                            p.span,
346                            suggestion,
347                        )
348                    })?
349                    .clone();
350
351                let args = args
352                    .iter()
353                    .map(|p| Pattern::from_parser(p, env))
354                    .collect::<Result<Vec<_>, Error>>()?;
355
356                PatternKind::Constructor { ctor, args }
357            }
358        };
359
360        Ok(Pattern::new(p.span, kind))
361    }
362}

364/// A canonical expression, and where it was written.
365///
366/// Same shape as [`parser::Expression`] — a [`NodeSpan`] beside a kind — and for the
367/// same reason: the children stay `Box<Expression>`, so a reader matches `&e.kind`.
368#[derive(Debug, PartialEq)]
369pub struct Expression {
370    pub span: NodeSpan,
371    pub kind: ExpressionKind,
372}
373
374// TODO Find a way to detect recursive functions (even indirect recursivity,
375// eg. `a` calls `b` calls `a`)
376/// Expression is an optimized version for checks and caches.
377///
378/// Elm declare those expressions:
379/// ```haskell
380/// data Expr_
381///   = VarLocal Name
382///   | VarTopLevel ModuleName.Canonical Name
383///   | VarKernel Name Name
384///   | VarForeign ModuleName.Canonical Name Annotation
385///   | VarCtor CtorOpts ModuleName.Canonical Name Index.ZeroBased Annotation
386///   | VarDebug ModuleName.Canonical Name Annotation
387///   | VarOperator Name ModuleName.Canonical Name Annotation -- CACHE real name for optimization
388///   | Chr ES.String
389///   | Str ES.String
390///   | Int Int
391///   | Float EF.Float
392///   | List [Expr]
393///   | Negate Expr
394///   | Binop Name ModuleName.Canonical Name Annotation Expr Expr -- CACHE real name for optimization
395///   | Lambda [Pattern] Expr
396///   | Call Expr [Expr]
397///   | If [(Expr, Expr)] Expr
398///   | Let Def Expr
399///   | LetRec [Def] Expr
400///   | LetDestruct Pattern Expr Expr
401///   | Case Expr [CaseBranch]
402///   | Accessor Name
403///   | Access Expr (A.Located Name)
404///   | Update Name Expr (Map.Map Name FieldUpdate)
405///   | Record (Map.Map Name Expr)
406///   | Unit
407///   | Tuple Expr Expr (Maybe Expr)
408/// ```
409#[derive(Debug, PartialEq)]
410pub enum ExpressionKind {
411    VarLocal(Name),
412    VarTopLevel(QualName),
413    VarKernel(QualName),
414    VarForeign(QualName, Type),
415    VarConstructor(QualName, Type),
416    Char(char),
417    Int(i64),
418    Float(f64),
419    Bool(bool),
420    // List
421    // Lambda
422    Apply(Box<Expression>, Box<Expression>),
423    If(Box<Expression>, Box<Expression>, Box<Expression>),
424    // Let
425    // LetRec
426    // LetDestruct (eg. `(a,b) = someTuple`)
427    Case(Box<Expression>, Vec<CaseBranch>),
428    // Accessor
429    // Access
430    // Update (record)
431    // Unit
432    /// A tuple expression. Zelkova keeps Elm's restriction of two or three
433    /// elements, which [`Tuple`] carries in its shape.
434    Tuple(Tuple<Expression>),
435}
436
437impl Expression {
438    /// An expression canonicalized from source, keeping the parser node's position.
439    pub fn new(span: NodeSpan, kind: ExpressionKind) -> Expression {
440        Expression { span, kind }
441    }
442
443    /// An expression with no position: hand-built by a test, or synthesised by the
444    /// compiler with nothing in the user's text behind it. See [`NodeSpan`].
445    pub fn bare(kind: ExpressionKind) -> Expression {
446        Expression {
447            span: NodeSpan::none(),
448            kind,
449        }
450    }
451
452    fn from_parser(e: &parser::Expression, env: &dyn Environment) -> Result<Expression, Error> {
453        // Every arm builds a kind and every kind gets `e.span`, so a name the
454        // environment cannot resolve is underlined where it was written rather than
455        // somewhere up the tree.
456        let kind = match &e.kind {
457            parser::ExpressionKind::Lit(parser::Literal::Int(i)) => ExpressionKind::Int(*i),
458            parser::ExpressionKind::Lit(parser::Literal::Float(f)) => ExpressionKind::Float(*f),
459            parser::ExpressionKind::Lit(parser::Literal::Char(c)) => ExpressionKind::Char(*c),
460            parser::ExpressionKind::Lit(parser::Literal::Bool(b)) => ExpressionKind::Bool(*b),
461            parser::ExpressionKind::Variable(name) => {
462                match env.find_value(name).ok_or_else(|| {
463                    let suggestion = suggest_name(name, env.value_names().into_iter());
464                    Error::VariableNotFound(
465                        env.module_name().qualify_name(name),
466                        e.span,
467                        suggestion,
468                    )
469                })? {
470                    ValueType::Local => ExpressionKind::VarLocal(name.clone()),
471                    ValueType::TopLevel => {
472                        ExpressionKind::VarTopLevel(env.module_name().qualify_name(name))
473                    }
474                    ValueType::Foreign(m, _source, tpe) => {
475                        ExpressionKind::VarForeign(m.qualify_name(name), tpe.clone())
476                    }
477                    ValueType::Foreigns(candidates) => {
478                        return Err(Error::AmbiguousVariables(
479                            name.clone(),
480                            candidates.clone(),
481                            e.span,
482                        ))
483                    }
484                }
485            }
486            parser::ExpressionKind::TypeConstructor(name) => {
487                let ctor = env.find_type_constructor(name).ok_or_else(|| {
488                    let suggestion = suggest_name(name, env.type_constructor_names().into_iter());
489                    Error::VariantNotFound(env.module_name().qualify_name(name), e.span, suggestion)
490                })?;
491
492                let tpe = if ctor.type_parameters.is_empty() {
493                    Type::Type(ctor.tpe.clone(), vec![])
494                } else {
495                    // TODO Rework that part. ctor.types is only for the type parameters of the constructor, not for the overall type.
496                    let mut iter = ctor.type_parameters.iter().rev();
497                    let first = iter.next().unwrap().clone();
498
499                    // TODO tests this, 99.999% I'm wrong about it (like everytime I try to implement arrows, plus
500                    // the foldr in this particular case)
501                    let tpe = iter.fold(first, |acc, t| {
502                        Type::Arrow(Box::new(t.clone()), Box::new(acc))
503                    });
504
505                    Type::Arrow(
506                        Box::new(Type::Type(ctor.tpe.clone(), vec![])),
507                        Box::new(tpe),
508                    )
509                };
510
511                let name = name
512                    .to_qual()
513                    .unwrap_or_else(|| env.module_name().qualify_name(name));
514
515                ExpressionKind::VarConstructor(name, tpe)
516            }
517            parser::ExpressionKind::Application(a, b) => {
518                let a = Expression::from_parser(a, env)?;
519                let b = Expression::from_parser(b, env)?;
520
521                ExpressionKind::Apply(Box::new(a), Box::new(b))
522            }
523            parser::ExpressionKind::InfixChain(first, rest) => {
524                let first = Expression::from_parser(first, env)?;
525
526                let rest = rest
527                    .iter()
528                    .map(|(op_name, op_span, operand)| {
529                        let op = resolve_infix_operator(op_name, *op_span, env)?;
530                        let operand = Expression::from_parser(operand, env)?;
531                        Ok((op, operand))
532                    })
533                    .collect::<Result<Vec<_>, Error>>()?;
534
535                let mut rest = rest.into_iter().peekable();
536                let reassociated = reassociate_infix_chain(first, &mut rest, 0)?;
537
538                reassociated.kind
539            }
540            parser::ExpressionKind::Tuple(tuple) => {
541                ExpressionKind::Tuple(tuple.try_map(|e| Expression::from_parser(e, env))?)
542            }
543            parser::ExpressionKind::Case(expr, branches) => {
544                let expr = Expression::from_parser(expr, env)?;
545
546                let b = branches.iter().map::<Result<CaseBranch, Error>, _>(|cb| {
547                    let pattern = Pattern::from_parser(&cb.pattern, env)?;
548                    let mut scoped = env.new_scope();
549
550                    scoped.expose_pattern(&pattern);
551
552                    let expression = Expression::from_parser(&cb.expression, &scoped)?;
553
554                    Ok(CaseBranch {
555                        pattern,
556                        expression,
557                        span: cb.span,
558                    })
559                });
560
561                let branches = collect_accumulate(b)?;
562
563                ExpressionKind::Case(Box::new(expr), branches)
564            }
565            parser::ExpressionKind::If(cond, then, els) => {
566                let cond = Expression::from_parser(cond, env)?;
567                let then = Expression::from_parser(then, env)?;
568                let els = Expression::from_parser(els, env)?;
569
570                ExpressionKind::If(Box::new(cond), Box::new(then), Box::new(els))
571            }
572        };
573
574        Ok(Expression::new(e.span, kind))
575    }
576}

578/// One operator of an `ExpressionKind::InfixChain`, resolved against the infix
579/// environment: its own `Infix` (precedence, associativity, and where it was
580/// declared) alongside the canonical `Expression` it resolves to as a value —
581/// `VarLocal`, `VarTopLevel` or `VarForeign`, whichever `find_value` reports for
582/// its `function_name`, exactly as a plain reference to the operator would
583/// resolve. `reassociate_infix_chain` consults `infix` to decide how to nest,
584/// and `span` and `expr` to build the invented `Application` nodes.
585struct ResolvedInfixOp {
586    name: Name,
587    /// Where the operator itself was written — not its `infix` declaration.
588    span: NodeSpan,
589    expr: Expression,
590    infix: Infix,
591    /// Where the `infix` declaration was written, and in which module's file —
592    /// the operator may well have been declared by an imported module, in which
593    /// case `infix.span` is a byte range in *that* module's source. See
594    /// [`InfixDeclaration`].
595    declaration: InfixDeclaration,
596}
597
598/// Resolves one operator of an `InfixChain` the same way a bare reference to it
599/// would resolve as a value, plus the `Infix` re-association needs.
600///
601/// An operator symbol only ever reaches `self.variables` through the `infixes`
602/// redirect in `RootEnvironment::find_value` — nothing else can insert a key
603/// spelled like an operator, since a function's own name always comes from
604/// `VarIdent`, a distinct token from `Op`. So `find_infix` failing here is
605/// exactly the case a plain `Variable` reference to the same name would fail
606/// `find_value` on, and gets the same `VariableNotFound` a reader would expect.
607fn resolve_infix_operator(
608    name: &Name,
609    span: NodeSpan,
610    env: &dyn Environment,
611) -> Result<ResolvedInfixOp, Error> {
612    let entry = env.find_infix(name).cloned().ok_or_else(|| {
613        let suggestion = suggest_name(name, env.value_names().into_iter());
614        Error::VariableNotFound(env.module_name().qualify_name(name), span, suggestion)
615    })?;
616
617    // Resolved the same way a written `Variable(name)` would be — through the
618    // `Variable` arm of `Expression::from_parser` — so the operator picks up
619    // exactly the `VarLocal`/`VarTopLevel`/`VarForeign` shape a plain reference
620    // to it would.
621    let synthetic = parser::Expression::new(span, parser::ExpressionKind::Variable(name.clone()));
622    let expr = Expression::from_parser(&synthetic, env)?;
623
624    Ok(ResolvedInfixOp {
625        name: name.clone(),
626        span,
627        expr,
628        infix: entry.infix,
629        declaration: entry.declaration,
630    })
631}

633/// Re-associates a flat run of operator applications into a tree of
634/// `Application` nodes, in one pass over the operators — no operator or operand
635/// is visited twice — using precedence climbing (a standard algorithm; see e.g.
636/// https://en.wikipedia.org/wiki/Operator-precedence_parser#Precedence_climbing_method).
637///
638/// `min_prec` is the lowest precedence this call is willing to fold into `lhs`;
639/// the outer, top-level call passes 0, admitting every operator. A recursive
640/// call raises it to bind only the tighter operators that belong on the right
641/// of the operator being folded — `op.infix.precedence + 1` for strictly higher
642/// precedence, or `op.infix.precedence` itself when the next operator ties and
643/// both associate right, which is what lets a right-associative run keep
644/// folding at the same level (`a <| b <| c` → `a <| (b <| c)`).
645///
646/// Two adjacent operators of equal precedence are rejected rather than guessed
647/// at unless they agree — both `left` (fold `lhs`, the usual case) or both
648/// `right` (recurse into `rhs`). Anything else — a `left` against a `right`, or
649/// either against an `infix non` operator, including one against itself — is
650/// `Error::AmbiguousOperatorPrecedence`: nothing here says which one should
651/// bind first, and guessing would silently pick a grouping the user did not
652/// write.
653fn reassociate_infix_chain(
654    mut lhs: Expression,
655    rest: &mut std::iter::Peekable<impl Iterator<Item = (ResolvedInfixOp, Expression)>>,
656    min_prec: u8,
657) -> Result<Expression, Error> {
658    while let Some((op, mut rhs)) = rest.next_if(|(op, _)| op.infix.precedence >= min_prec) {
659        loop {
660            let next_min = match rest.peek() {
661                None => None,
662                Some((next_op, _)) => {
663                    if next_op.infix.precedence > op.infix.precedence {
664                        Some(op.infix.precedence + 1)
665                    } else if next_op.infix.precedence == op.infix.precedence {
666                        match (op.infix.associativity, next_op.infix.associativity) {
667                            (Associativity::Left, Associativity::Left) => None,
668                            (Associativity::Right, Associativity::Right) => {
669                                Some(op.infix.precedence)
670                            }
671                            _ => {
672                                return Err(Error::AmbiguousOperatorPrecedence(
673                                    AmbiguousOperator::new(&op),
674                                    AmbiguousOperator::new(next_op),
675                                    op.span.merge(next_op.span),
676                                ))
677                            }
678                        }
679                    } else {
680                        None
681                    }
682                }
683            };
684
685            match next_min {
686                Some(min) => rhs = reassociate_infix_chain(rhs, rest, min)?,
687                None => break,
688            }
689        }
690
691        lhs = apply_infix(op, lhs, rhs);
692    }
693
694    Ok(lhs)
695}

697/// Builds the two `Application` nodes one step of infix re-association adds,
698/// following the same span convention the grammar's right-recursive rewrite
699/// used before this: the partial application of the operator to `lhs` — a node
700/// with nothing in the user's text of its own — takes the operator's own span,
701/// and the outer application, which stands for `lhs op rhs` as the user wrote
702/// it, spans from wherever `lhs` starts to wherever `rhs` ends.
703fn apply_infix(op: ResolvedInfixOp, lhs: Expression, rhs: Expression) -> Expression {
704    let span = lhs.span.merge(rhs.span);
705
706    let partial = Expression::new(
707        op.span,
708        ExpressionKind::Apply(Box::new(op.expr), Box::new(lhs)),
709    );
710
711    Expression::new(
712        span,
713        ExpressionKind::Apply(Box::new(partial), Box::new(rhs)),
714    )
715}

717/// One side of an [`Error::AmbiguousOperatorPrecedence`]: the operator as the user
718/// spelled it, how its `infix` declaration said it associates, and where that
719/// declaration was written.
720///
721/// `associativity` is carried rather than re-derived because the message depends on
722/// it — an `infix non` operator does not chain at all, where a `left` against a
723/// `right` is a choice the user can make with parentheses — and the error outlives
724/// the environment the declaration was looked up in.
725#[derive(Debug)]
726pub struct AmbiguousOperator {
727    pub name: Name,
728    pub associativity: Associativity,
729    pub declaration: InfixDeclaration,
730}

732impl AmbiguousOperator {
733    fn new(op: &ResolvedInfixOp) -> Self {
734        AmbiguousOperator {
735            name: op.name.clone(),
736            associativity: op.infix.associativity,
737            declaration: op.declaration,
738        }
739    }
740
741    fn is_non_associative(&self) -> bool {
742        self.associativity == Associativity::None
743    }
744}

756/// Everything canonicalization can reject.
757///
758/// # Why only some variants carry a span
759///
760/// A variant carries a [`NodeSpan`] when its construction site has one in hand — it
761/// is looking at a `parser::Import`, `parser::Infix`, `parser::UnionType`,
762/// `parser::Function` or `parser::Exposed`, all of which the grammar gives a span.
763/// Those are the variants a diagnostic can put a caret under, and today that is every
764/// variant below except three: the two group variants, `EnvironmentErrors` and
765/// `Many`, have no position of their own and flatten their members' labels instead;
766/// and `InvalidTupleSize` carries none for an unrelated reason — see its own doc
767/// comment. Writing `NodeSpan::none()` into a variant that lacks a real position
768/// would say "this error has a position we happen not to know", which is a lie;
769/// leaving the field off says "this error has nowhere to point", which is true, and
770/// the reporter renders it as message-plus-notes with no caret.
771#[derive(Debug)]
772pub enum Error {
773    /// A name in the `module … exposing (…)` header that nothing in the module
774    /// declares, and where that name was written (`ERR-9`).
775    ExportNotFound(Name, ExportType, NodeSpan),
776    EnvironmentErrors(Vec<EnvError>),
777    /// (infix, function), and where the `infix` declaration was written
778    InfixReferenceInvalidValue(Name, Name, NodeSpan),
779    /// Infix re-association (`reassociate_infix_chain`) found two adjacent
780    /// operators of equal precedence that do not agree how to group: the left
781    /// one, the right one, and the ambiguous pair's own span — from the left
782    /// operator through the right one.
783    ///
784    /// Three shapes reach this, and [`PhaseError::message`] says which:
785    /// `left.name == right.name` is one `infix non` operator chained with
786    /// itself (`a == b == c`); either side being `infix non` against a
787    /// different operator is a non-associative operator that does not chain at
788    /// all (`a < b > c`); and a `left` against a `right` is the genuine
789    /// disagreement about which side groups first (`a << b >> c`).
790    AmbiguousOperatorPrecedence(AmbiguousOperator, AmbiguousOperator, NodeSpan),
791    BindingPatternsInvalidLen(NodeSpan),
792    /// A declaration with a type annotation and no body, and where the annotation
793    /// was written — which is the only part of it there is to point at.
794    NoBindings(NodeSpan),
795    /// A name used as a value that nothing in scope declares, where it was
796    /// written — the identifier alone, not the declaration around it — and,
797    /// when one name in scope is a close enough typo-distance match, a
798    /// suggestion for what was meant (`ERR-7`).
799    VariableNotFound(QualName, NodeSpan, Option<Name>),
800    /// A name exposed unqualified by more than one imported module, and where it
801    /// was used. Each candidate module is paired with where the name is declared
802    /// there — `Some` when that module's `Interface` knows both its file and the
803    /// declaration's span, `None` for a hand-built interface (a test) or one built
804    /// before its module's file was known (`ERR-5`).
805    AmbiguousVariables(Name, Vec<(ModuleName, Option<super::SourceSpan>)>, NodeSpan),
806    /// A constructor used in an expression or a pattern that nothing in scope
807    /// declares, where it was written, and an optional "did you mean …?"
808    /// suggestion (`ERR-7`).
809    VariantNotFound(QualName, NodeSpan, Option<Name>),
810    /// Nothing constructs this today — `Environment::find_type_constructor` returns
811    /// at most one constructor per name, so it has no way to report an ambiguity.
812    /// It is the designated rejection path once it can, and carries the span the
813    /// construction site would have, alongside each candidate's declaration
814    /// location — see [`Error::AmbiguousVariables`].
815    AmbiguousVariants(Name, Vec<(ModuleName, Option<super::SourceSpan>)>, NodeSpan),
816    /// A tuple type, pattern or expression had a size other than 2 or 3 (the
817    /// only sizes the language supports).
818    ///
819    /// Nothing constructs this today, and it is kept deliberately. Both ASTs
820    /// now hold their tuples in [`Tuple`], which cannot represent another
821    /// arity, and the grammar has one production per arity — so a bad tuple is
822    /// a parse error and never reaches canonicalization. This variant is the
823    /// designated rejection path should a future source of tuples (a REPL, a
824    /// desugaring pass) build one from a list.
825    InvalidTupleSize(usize),
826    /// A function was declared with multiple bindings (multi-clause definitions),
827    /// which the compiler does not support yet.
828    MultipleBindingsUnsupported(Name, NodeSpan),
829    /// A type name applied to the wrong number of arguments: the name, its
830    /// declaration's own arity, the number of arguments actually written, and
831    /// `tpe.span` — the whole application, so the caret covers every argument
832    /// along with the name (`BUG-17`).
833    TypeArityMismatch(Name, usize, usize, NodeSpan),
834
835    // Binding module
836    InfixDeclared(Name, NodeSpan),
837    TypeDeclared(Name, NodeSpan),
838    NoTypeInBinding(Name, NodeSpan),
839
840    // Utility error
841    Many(Vec<Error>),
842}

844/// Canonicalization errors name source constructs — a value, a type, an operator —
845/// so their messages can be written in the same words the user wrote.
846///
847/// The variants whose construction site had a declaration in hand also point at it;
848/// see the enum's own documentation for why the others do not.
849impl PhaseError for Error {
850    fn message(&self) -> String {
851        match self {
852            Error::ExportNotFound(name, tpe, _) => format!(
853                "`{}` is exposed by this module but no {} of that name is declared in it",
854                name,
855                export_type_noun(tpe)
856            ),
857            Error::EnvironmentErrors(errors) => match errors.as_slice() {
858                [only] => only.message(),
859                many => format!("{} of this module's imports could not be resolved", many.len()),
860            },
861            Error::InfixReferenceInvalidValue(infix, function, _) => format!(
862                "the infix operator `{}` is declared as `{}`, which is not a value declared in this module",
863                infix, function
864            ),
865            Error::AmbiguousOperatorPrecedence(left, right, _) => {
866                // Three shapes, and the everyday one is the middle: `Basics`
867                // declares six operators `infix non 4`, so `a < b > c` reaches
868                // here with two *different* operators that do not disagree about
869                // anything — both say they do not chain. Calling that a
870                // disagreement about which side groups first would name a reason
871                // that is not the reason.
872                if left.name == right.name {
873                    format!(
874                        "`{}` is declared `infix non`, so it cannot be chained with itself without parentheses to say which application comes first",
875                        left.name
876                    )
877                } else if left.is_non_associative() && right.is_non_associative() {
878                    format!(
879                        "`{}` and `{}` are both declared `infix non` at the same precedence, so neither groups the other and this needs parentheses",
880                        left.name, right.name
881                    )
882                } else if left.is_non_associative() || right.is_non_associative() {
883                    let (non, other) = if left.is_non_associative() {
884                        (&left.name, &right.name)
885                    } else {
886                        (&right.name, &left.name)
887                    };
888
889                    format!(
890                        "`{}` is declared `infix non`, so it does not chain with `{}` at the same precedence without parentheses",
891                        non, other
892                    )
893                } else {
894                    format!(
895                        "`{}` and `{}` have the same precedence but disagree on which side groups first, so this needs parentheses to say which one applies first",
896                        left.name, right.name
897                    )
898                }
899            }
900            Error::BindingPatternsInvalidLen(_) => {
901                "the arguments of this declaration do not line up with its type annotation"
902                    .to_owned()
903            }
904            Error::NoBindings(_) => {
905                "this declaration has a type annotation but no body".to_owned()
906            }
907            Error::VariableNotFound(name, _, _) => {
908                format!("cannot find a value named `{}`", name.to_name())
909            }
910            Error::AmbiguousVariables(name, _, _) => {
911                format!("`{}` is exposed by several imported modules", name)
912            }
913            Error::VariantNotFound(name, _, _) => {
914                format!("cannot find a type constructor named `{}`", name.to_name())
915            }
916            Error::AmbiguousVariants(name, _, _) => format!(
917                "the type constructor `{}` is exposed by several imported modules",
918                name
919            ),
920            Error::InvalidTupleSize(size) => format!(
921                "a tuple has two or three elements, this one has {}",
922                size
923            ),
924            Error::MultipleBindingsUnsupported(name, _) => format!(
925                "`{}` is declared over several bindings, which is not supported yet",
926                name
927            ),
928            Error::TypeArityMismatch(name, declared, written, _) => format!(
929                "`{}` takes {}, but is applied to {} here",
930                name,
931                type_argument_count(*declared),
932                type_argument_count(*written)
933            ),
934            Error::InfixDeclared(name, _) => format!(
935                "a `module javascript` facade cannot declare an infix operator, but declares `{}`",
936                name
937            ),
938            Error::TypeDeclared(name, _) => format!(
939                "a `module javascript` facade cannot declare a type, but declares `{}`",
940                name
941            ),
942            Error::NoTypeInBinding(name, _) => format!(
943                "`{}` has no type annotation, and a `module javascript` facade is annotations only",
944                name
945            ),
946            Error::Many(errors) => match errors.as_slice() {
947                [only] => only.message(),
948                many => format!("{} errors while canonicalizing this module", many.len()),
949            },
950        }
951    }
952
953    fn labels(&self) -> Vec<SpanLabel> {
954        // One helper per shape: a variant either has a span and one thing to say
955        // about it, or it delegates to the errors it wraps.
956        let primary = |span: &NodeSpan, message: &str| match span.span() {
957            Some(span) => vec![SpanLabel {
958                span,
959                message: message.to_owned(),
960                primary: true,
961                file: None,
962            }],
963            None => Vec::new(),
964        };
965
966        // A secondary label pointing at where one ambiguous candidate is declared,
967        // in *its own* module's file rather than the one being checked — the
968        // mechanism `ERR-5` adds. `None` when that candidate's `Interface` cannot
969        // say (a hand-built interface, or one built before its file was known)
970        // yields no label for that candidate rather than one at the wrong place.
971        let ambiguous_candidates = |candidates: &[(ModuleName, Option<super::SourceSpan>)]| {
972            candidates
973                .iter()
974                .filter_map(|(module, source)| {
975                    source.map(|s| SpanLabel {
976                        span: s.span,
977                        message: format!("also exposed here, by `{}`", module.name()),
978                        primary: false,
979                        file: Some(s.file),
980                    })
981                })
982                .collect::<Vec<_>>()
983        };
984
985        match self {
986            Error::ExportNotFound(name, _, span) => primary(
987                span,
988                &format!("`{}` is not declared anywhere in this module", name),
989            ),
990            Error::InfixReferenceInvalidValue(_, _, span) => primary(span, "declared here"),
991            // The two "declared here" labels go through `InfixDeclaration`, which
992            // is what knows whether the declaration is in the module under check
993            // or in an imported one. An operator's `infix` declaration is very
994            // often not local — `Basics` declares every one the standard library
995            // uses — and `Infix::span` is then a byte range in *that* module's
996            // file, so a label built with `file: None` would underline unrelated
997            // text in the importing module (`ERR-5` is the mechanism that avoids
998            // it).
999            Error::AmbiguousOperatorPrecedence(left, right, span) => {
1000                let mut labels = primary(span, "ambiguous without parentheses");
1001
1002                labels.extend(
1003                    left.declaration
1004                        .label(format!("`{}` declared here", left.name)),
1005                );
1006
1007                // Both sides naming the same operator is the `infix non`
1008                // self-conflict — they point at the very same declaration, so a
1009                // second label there would only repeat the first one.
1010                if left.name != right.name {
1011                    labels.extend(
1012                        right
1013                            .declaration
1014                            .label(format!("`{}` declared here", right.name)),
1015                    );
1016                }
1017
1018                labels
1019            }
1020            // The four that name an identifier: the caret sits under the name the
1021            // user wrote, which is the whole point of spanning expressions and
1022            // patterns rather than only declarations. `VariableNotFound` and
1023            // `VariantNotFound` append their suggestion, when they have one, to
1024            // this same label rather than a free-floating note, so the caret and
1025            // the suggestion agree about which name is meant (`ERR-7`).
1026            Error::VariableNotFound(_, span, suggestion) => primary(
1027                span,
1028                &format!(
1029                    "no value of this name is in scope{}",
1030                    suggestion_suffix(suggestion)
1031                ),
1032            ),
1033            Error::AmbiguousVariables(_, candidates, span) => {
1034                let mut labels = primary(span, "this name is ambiguous");
1035                labels.extend(ambiguous_candidates(candidates));
1036                labels
1037            }
1038            Error::VariantNotFound(_, span, suggestion) => primary(
1039                span,
1040                &format!(
1041                    "no type constructor of this name is in scope{}",
1042                    suggestion_suffix(suggestion)
1043                ),
1044            ),
1045            Error::AmbiguousVariants(_, candidates, span) => {
1046                let mut labels = primary(span, "this type constructor is ambiguous");
1047                labels.extend(ambiguous_candidates(candidates));
1048                labels
1049            }
1050            Error::BindingPatternsInvalidLen(span) => primary(span, "declared here"),
1051            Error::NoBindings(span) => primary(span, "this annotation has no body"),
1052            Error::MultipleBindingsUnsupported(_, span) => primary(span, "declared here"),
1053            Error::TypeArityMismatch(name, declared, written, span) => primary(
1054                span,
1055                &format!(
1056                    "`{}` takes {}, this application has {}",
1057                    name,
1058                    type_argument_count(*declared),
1059                    type_argument_count(*written)
1060                ),
1061            ),
1062            Error::InfixDeclared(_, span) => primary(span, "declared here"),
1063            Error::TypeDeclared(_, span) => primary(span, "declared here"),
1064            Error::NoTypeInBinding(_, span) => primary(span, "declared here"),
1065            // A group has no position of its own; the errors it swallowed do.
1066            Error::EnvironmentErrors(errors) => errors.iter().flat_map(|e| e.labels()).collect(),
1067            Error::Many(errors) => errors.iter().flat_map(|e| e.labels()).collect(),
1068            _ => Vec::new(),
1069        }
1070    }
1071
1072    fn notes(&self) -> Vec<String> {
1073        match self {
1074            Error::AmbiguousVariables(_, candidates, _)
1075            | Error::AmbiguousVariants(_, candidates, _) => {
1076                vec![format!(
1077                    "it is exposed by: {}",
1078                    candidates
1079                        .iter()
1080                        .map(|(m, _)| m.name().to_string())
1081                        .collect::<Vec<_>>()
1082                        .join(", ")
1083                )]
1084            }
1085            // A group renders as a summary, so every message it swallowed becomes a
1086            // note. A group of one is rendered by its own message and adds nothing.
1087            Error::EnvironmentErrors(errors) => match errors.as_slice() {
1088                [only] => only.notes(),
1089                many => many.iter().flat_map(|e| e.message_and_notes()).collect(),
1090            },
1091            Error::Many(errors) => match errors.as_slice() {
1092                [only] => only.notes(),
1093                many => many.iter().flat_map(|e| e.message_and_notes()).collect(),
1094            },
1095            _ => Vec::new(),
1096        }
1097    }
1098}

1144/// Transform a given `parser::Module` into a `canonical::Module`
1145pub fn canonicalize(
1146    package: &PackageName,
1147    interfaces: &HashMap<Name, Interface>,
1148    source: &parser::Module,
1149) -> Result<Module, Vec<Error>> {
1150    let name = ModuleName {
1151        package: package.clone(),
1152        name: source.name.clone(),
1153    };
1154
1155    let mut errors: Vec<Error> = vec![];
1156    let mut env =
1157        new_environment(&name, interfaces, &source.imports).map_err(|e| vec![e.into()])?;
1158
1159    let (infixes, types, values) = if source.binding_javascript {
1160        // Javascript modules run a parallel canonicalization process as the constraints are a bit different:
1161        // - Only functions without bindings are authorized.
1162        // - Infixes and types are forbidden.
1163        // The idea being to have the js module be a facade for the actual Javascript module.
1164        // Assuming Json types are part of the prelude, this should goes well with the restriction on what types are available for bindings.
1165
1166        // Verify no infix present
1167        if !source.infixes.is_empty() {
1168            let e = source
1169                .infixes
1170                .iter()
1171                .map(|i| Error::InfixDeclared(i.operator.clone(), i.span));
1172            errors.extend(e);
1173        }
1174        // Verify no types present
1175        if !source.types.is_empty() {
1176            let e = source
1177                .types
1178                .iter()
1179                .map(|t| Error::TypeDeclared(t.name.clone(), t.span));
1180            errors.extend(e);
1181        }
1182
1183        // Iterate on values
1184        let iter = source.functions.iter().map(|function| {
1185            // Make sure there is no binding
1186            if !function.bindings.is_empty() {
1187                //println!("bindings = {:?} (js module)", function.bindings);
1188                Err(Error::BindingPatternsInvalidLen(function.span))? // TODO More specific error
1189            }
1190
1191            // Make sure there is a type
1192            let tpe = function
1193                .tpe
1194                .as_ref()
1195                .ok_or_else(|| Error::NoTypeInBinding(function.name.clone(), function.span))?;
1196            let tpe = Type::from_parser_type(&env, tpe)?;
1197
1198            let name = function.name.clone();
1199            // TODO Think how it's going to be represented. Currently canonical values assume an expression is present
1200            //      I'd like to not introduce a trait or new struct for binding. Should we fake an expression or create
1201            //      a new type of value ? New type of value will be annoying for regular modules as they aren't present
1202            //      there. Fake expression might be ok as MVP. We have to make sure that binding module are removed from
1203            //      some phase of the compilation pipeline.
1204            let value = Value::TypedValue {
1205                name: name.clone(),
1206                patterns: vec![],
1207                // A `module javascript` facade has no body in the source, so this
1208                // stand-in has nothing to point at (see the TODO above).
1209                body: Expression::bare(ExpressionKind::Bool(true)),
1210                tpe,
1211                span: function.span,
1212                annotation_span: function.annotation_span,
1213            };
1214
1215            Ok((name, value))
1216        });
1217        let values = crate::utils::collect_accumulate(iter).unwrap_or_else(|err| {
1218            errors.extend(err);
1219            HashMap::new()
1220        });
1221
1222        (HashMap::new(), HashMap::new(), values)
1223    } else {
1224        // Because we are rewriting infixes in this phase, we must do this check before
1225        // resolving values.
1226        let infixes =
1227            do_infixes(&source.infixes, &mut env, &source.functions).unwrap_or_else(|err| {
1228                errors.extend(err);
1229                HashMap::new()
1230            });
1231
1232        let types = do_types(&env, &source.types).unwrap_or_else(|err| {
1233            errors.extend(err);
1234            HashMap::new()
1235        });
1236
1237        for (n, t) in types.iter() {
1238            env.insert_union_type(n.clone(), t.clone());
1239        }
1240
1241        trace!("Environment after do_types: {:#?}", env);
1242
1243        // TODO Should I manage infixes rewrite here too ?
1244        // Yes I should do it here
1245        let values = do_values(&mut env, &source.functions).unwrap_or_else(|err| {
1246            errors.extend(err);
1247            HashMap::new()
1248        });
1249
1250        (infixes, types, values)
1251    };
1252
1253    // We do exports at the end, and verify that all exported value do
1254    // have a reference within the current module
1255    let exports = do_exports(&source.exposing, &env).unwrap_or_else(|err| {
1256        errors.extend(err);
1257        Exports::Everything // Never exposed, as we will return the errors instead
1258    });
1259
1260    if errors.is_empty() {
1261        Ok(Module {
1262            name,
1263            exports,
1264            infixes,
1265            types,
1266            values,
1267            binding_javascript: source.binding_javascript,
1268        })
1269    } else {
1270        Err(errors)
1271    }
1272}

1274fn do_values(
1275    env: &mut RootEnvironment,
1276    functions: &[parser::Function],
1277) -> Result<HashMap<Name, Value>, Vec<Error>> {
1278    // Before resolving expressions, we store the top-level values in the environment.
1279    // We do so first because their expression below could refer to them.
1280    for f in functions.iter() {
1281        env.insert_top_level_value(f.name.clone());
1282    }
1283
1284    let iter = functions.iter().map(|function| {
1285        // Bindings to expression
1286
1287        // TODO Better error message with position of mismatch
1288        // TODO Error when bindings is empty
1289        let bindings_size = function
1290            .bindings
1291            .iter()
1292            .all(|v| v.patterns.len() == function.bindings[0].patterns.len());
1293
1294        if !bindings_size {
1295            //println!("bindings = {:?} (bindings_size)", function.bindings);
1296            Err(Error::BindingPatternsInvalidLen(function.span))?
1297        }
1298
1299        let (patterns, body): (Vec<Pattern>, Expression) = match function.bindings.len() {
1300            0 => Err(Error::NoBindings(function.span)),
1301            1 => {
1302                // if one binding, we can convert directly to canonical format
1303                let binding = &function.bindings[0];
1304
1305                let mut scoped = env.new_scope();
1306
1307                let patterns: Vec<Pattern> = binding
1308                    .patterns
1309                    .iter()
1310                    .map(|p| Pattern::from_parser(p, env))
1311                    .collect::<Result<Vec<_>, Error>>()?;
1312
1313                for p in &patterns {
1314                    scoped.expose_pattern(p);
1315                }
1316
1317                // Maybe create a case_branch function and make it common with Expression::Case ?
1318                // Or maybe not at the case_branch level, as here we can have multiple patterns
1319                // whereas cases cannot.
1320                // eg. a: Int -> Int -> Int  ==>  a b c = b + c
1321                //println!("Env before transforming expression: {:?}", scoped);
1322                let body = Expression::from_parser(&binding.body, &scoped)?;
1323
1324                Ok((patterns, body))
1325            }
1326            _ => {
1327                // if multiple bindings, we need to create synthetics variables and put all bindings into a case expression
1328                Err(Error::MultipleBindingsUnsupported(
1329                    function.name.clone(),
1330                    function.span,
1331                ))
1332            }
1333        }?;
1334
1335        let name = function.name.clone();
1336
1337        match &function.tpe {
1338            Some(t) => {
1339                let tpe = Type::from_parser_type(env, t)?;
1340                let linear = Type::to_linear_types(&tpe);
1341
1342                // Linear is a list of types making the function. Because it includes the return type,
1343                // it will always be bigger than the number of patterns by one.
1344                if !patterns.is_empty() && (linear.len() - 1 != patterns.len()) {
1345                    // TODO Better error message
1346                    debug!(
1347                        "linear = {:#?}\nbindings = {:#?} (linear.len ({}) != patterns.len ({}))",
1348                        linear,
1349                        function.bindings,
1350                        linear.len(),
1351                        patterns.len()
1352                    );
1353                    Err(Error::BindingPatternsInvalidLen(function.span))?
1354                }
1355
1356                let patterns = patterns.into_iter().zip(linear).collect();
1357
1358                Ok((
1359                    function.name.clone(),
1360                    Value::TypedValue {
1361                        name,
1362                        patterns,
1363                        body,
1364                        tpe,
1365                        span: function.span,
1366                        annotation_span: function.annotation_span,
1367                    },
1368                ))
1369            }
1370            None => Ok((
1371                function.name.clone(),
1372                Value::Value {
1373                    name,
1374                    patterns,
1375                    body,
1376                    span: function.span,
1377                },
1378            )),
1379        }
1380    });
1381
1382    collect_accumulate(iter)
1383}

1385fn do_types(
1386    env: &dyn Environment,
1387    types: &[parser::UnionType],
1388) -> Result<HashMap<Name, UnionType>, Vec<Error>> {
1389    let iter = types.iter().map(|tpe| {
1390        let tpe_name = tpe.name.clone();
1391        let variables = tpe.type_arguments.clone();
1392
1393        trace!("do_types(in:{:?})", tpe);
1394
1395        // variants are represented as parser::Type::Unqualified. Other types
1396        // can be safely ignored in this context.
1397
1398        let variants = tpe
1399            .variants
1400            .iter()
1401            .filter_map(|t| match &t.kind {
1402                // TODO It might actually make more sense to put Type::from_parser_type
1403                // on `Environment`.
1404                parser::TypeKind::Unqualified(name, vars) => Some((name, vars)),
1405                _ => None,
1406            })
1407            .map(|(name, vars)| {
1408                let type_parameters = vars
1409                    .iter()
1410                    .map(|t| Type::from_parser_type(env, t))
1411                    .collect::<Result<Vec<_>, Error>>()?;
1412
1413                Ok(TypeConstructor {
1414                    name: name.clone(),
1415                    type_parameters,
1416                    tpe: tpe_name.clone(),
1417                })
1418            })
1419            .collect::<Result<Vec<_>, Error>>()?;
1420
1421        Ok((
1422            tpe_name,
1423            UnionType {
1424                variables,
1425                variants,
1426                span: tpe.span,
1427            },
1428        ))
1429    });
1430
1431    collect_accumulate(iter)
1432}

1434fn do_infixes(
1435    infixes: &[parser::Infix],
1436    env: &mut RootEnvironment,
1437    functions: &[parser::Function],
1438) -> Result<HashMap<Name, Infix>, Vec<Error>> {
1439    let iter = infixes.iter().map(|infix| {
1440        let op_name = infix.operator.clone();
1441        let function_name = infix.function_name.clone();
1442
1443        let function_exist = functions
1444            .iter()
1445            .find(|f| f.name == infix.function_name)
1446            .is_some();
1447
1448        if function_exist {
1449            let infix = Infix {
1450                associativity: infix.associativity,
1451                precedence: infix.precedence,
1452                function_name,
1453                span: infix.span,
1454            };
1455
1456            env.insert_local_infix(op_name.clone(), infix.clone());
1457
1458            Ok((op_name, infix))
1459        } else {
1460            Err(Error::InfixReferenceInvalidValue(
1461                op_name,
1462                function_name,
1463                infix.span,
1464            ))
1465        }
1466    });
1467
1468    collect_accumulate(iter)
1469}

1471// `BUG-8`: only the `Operator` arm below checks that the exposed name actually
1472// exists — a `Lower` or `Upper` name in a module's own `exposing (...)` header is
1473// accepted unconditionally even when nothing by that name is declared.
1474fn do_exports(
1475    source_exposing: &parser::Exposing,
1476    env: &dyn Environment,
1477) -> Result<Exports, Vec<Error>> {
1478    match source_exposing {
1479        parser::Exposing::Open => Ok(Exports::Everything),
1480        parser::Exposing::Explicit(exposed) => {
1481            let specifics = exposed.iter().map(|exposed| match &exposed.kind {
1482                parser::ExposedKind::Lower(name) => Ok((name.clone(), ExportType::Value)),
1483                parser::ExposedKind::Upper(name, parser::Privacy::Public) => {
1484                    Ok((name.clone(), ExportType::UnionPublic))
1485                }
1486                parser::ExposedKind::Upper(name, parser::Privacy::Private) => {
1487                    Ok((name.clone(), ExportType::UnionPrivate))
1488                }
1489                parser::ExposedKind::Operator(name) => {
1490                    if env.local_infix_exists(name) {
1491                        Ok((name.clone(), ExportType::Infix))
1492                    } else {
1493                        Err(Error::ExportNotFound(
1494                            name.clone(),
1495                            ExportType::Infix,
1496                            exposed.span,
1497                        ))
1498                    }
1499                }
1500            });
1501
1502            let specifics = collect_accumulate(specifics)?;
1503
1504            Ok(Exports::Specifics(specifics))
1505        }
1506    }
1507}

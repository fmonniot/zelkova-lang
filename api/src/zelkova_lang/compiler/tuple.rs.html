1//! The tuple representation shared by the parser AST, the canonical AST and the
2//! typer.
3//!
4//! Zelkova follows Elm here: a tuple holds two or three elements — never one,
5//! never four. `Tuple` is where that rule is written down, and it is written in
6//! the *shape* of the type rather than in a check, so a tuple of any other size
7//! cannot be built in the first place. That is deliberate: the rule used to be
8//! re-derived at every parser → canonical conversion (once for types, once for
9//! patterns, once for expressions), each with its own arithmetic, and the three
10//! did not agree.
11//!
12//! The grammar (`parser/grammar.lalrpop`) has one production per arity, so a
13//! four-element tuple in a `.zel` source is a parse error and never reaches an
14//! AST at all.
15
16/// A tuple of exactly two or three elements.
17///
18/// The element type is a parameter so types, patterns and expressions — in the
19/// parser AST, the canonical AST and the typer alike — all reuse this one enum.
20/// The typer's `Type::Tuple`, `Term::Tuple` and `TypedTerm::Tuple` each hold a
21/// `Tuple<_>` of their own element type. `Hash`/`Eq` are
22/// derived alongside `PartialEq` because the typer's `Type::Tuple(Tuple<Type>)`
23/// needs both: `Type` sits in a `HashSet<Constraint>` during unification.
24#[derive(Debug, PartialEq, Eq, Hash, Clone)]
25pub enum Tuple<T> {
26    Two(Box<T>, Box<T>),
27    Three(Box<T>, Box<T>, Box<T>),
28}
29
30impl<T> Tuple<T> {
31    pub fn two(first: T, second: T) -> Tuple<T> {
32        Tuple::Two(Box::new(first), Box::new(second))
33    }
34
35    pub fn three(first: T, second: T, third: T) -> Tuple<T> {
36        Tuple::Three(Box::new(first), Box::new(second), Box::new(third))
37    }
38
39    /// The elements, in source order.
40    pub fn iter(&self) -> impl Iterator<Item = &T> {
41        let (first, second, third) = match self {
42            Tuple::Two(a, b) => (&**a, &**b, None),
43            Tuple::Three(a, b, c) => (&**a, &**b, Some(&**c)),
44        };
45
46        std::iter::once(first)
47            .chain(std::iter::once(second))
48            .chain(third)
49    }
50
51    /// Convert every element with `f`, keeping the arity, and stop at the first
52    /// error.
53    ///
54    /// This is what the parser → canonical conversions use, and what the
55    /// canonical → typer conversions (`canonical_type_to_typer_type`,
56    /// `canonical_expr_to_term`) use: the arity travels with the value, so
57    /// none of them count elements.
58    pub fn try_map<U, E, F>(&self, mut f: F) -> Result<Tuple<U>, E>
59    where
60        F: FnMut(&T) -> Result<U, E>,
61    {
62        match self {
63            Tuple::Two(a, b) => Ok(Tuple::two(f(a)?, f(b)?)),
64            Tuple::Three(a, b, c) => Ok(Tuple::three(f(a)?, f(b)?, f(c)?)),
65        }
66    }
67}

69#[cfg(test)]
70mod tests {
71    use super::*;
72
73    #[test]
74    fn iter_yields_elements_in_source_order() {
75        assert_eq!(Tuple::two(1, 2).iter().collect::<Vec<_>>(), vec![&1, &2]);
76        assert_eq!(
77            Tuple::three(1, 2, 3).iter().collect::<Vec<_>>(),
78            vec![&1, &2, &3]
79        );
80    }
81
82    #[test]
83    fn try_map_preserves_arity() {
84        let two: Result<Tuple<i32>, ()> = Tuple::two(1, 2).try_map(|i| Ok(i * 2));
85        assert_eq!(two, Ok(Tuple::two(2, 4)));
86
87        let three: Result<Tuple<i32>, ()> = Tuple::three(1, 2, 3).try_map(|i| Ok(i * 2));
88        assert_eq!(three, Ok(Tuple::three(2, 4, 6)));
89    }
90
91    #[test]
92    fn try_map_stops_at_the_first_error() {
93        let mut seen = vec![];
94        let result: Result<Tuple<i32>, i32> = Tuple::three(1, 2, 3).try_map(|i| {
95            seen.push(*i);
96            if *i == 2 {
97                Err(*i)
98            } else {
99                Ok(*i)
100            }
101        });
102
103        assert_eq!(result, Err(2));
104        assert_eq!(seen, vec![1, 2], "the third element is never visited");
105    }
106}

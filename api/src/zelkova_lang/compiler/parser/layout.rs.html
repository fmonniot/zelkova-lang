19/// Apply the offside rule to a token stream, injecting `OpenBlock`/`CloseBlock`
20/// so the parser does not have to track indentation itself.
21///
22/// **The returned iterator stops at the first `Err`.** A consumer which drains
23/// it fully therefore observes at most one error, ever, and then terminates —
24/// a layout error is not recoverable by the iterator, so replaying past it
25/// would repeat that error without bound (`BUG-4`). It is a `FusedIterator`,
26/// so `.fuse()` is a no-op. See `Layout` and its `Iterator::next` for the
27/// reasoning.
28pub fn layout<I: Iterator<Item = Result<Spanned<Position, Token>, Error>>>(
29    iter: I,
30) -> impl FusedIterator<Item = Result<(BytePos, Token, BytePos), Error>> {
31    Layout::new(iter)
32}

34/// Context represent the kind of expression we are looking at.
35///
36/// It let us associate context-aware indentation rules
37///
38/// ## Elm Rules
39///
40/// Elm has surprisingly few indentation rules:
41/// - `case <> of` must be followed by branches on indent + 1 level, and the content of each branch must be indent + 1 if on a next line
42/// - `let <> in`: the first block must be indent + 1 compared to the let keyword, and the in expression must be on indent + 1 of the _parent_ block
43///   Note that I'll probably change the in rule to be at the same level.
44/// - top level declaration body must either be one liner or be in an opened block at indent + 1 (this apply to function, custom types or type alias)
45/// - function application have no rules on where they should be. Meaning the let/in and case/of rules apply.
46///
47/// We will start with those rules, but will probably implement a "strict mode" along the road to enforce some convention on
48/// indentation. Probably something loosely based on what elm-format recommend. Let's be draconian and enforce uniformity :pirate:.
49///
50/// ## Examples
51/// Here is an example of context for a pattern matching expression
52///
53/// ```text
54///    case maybe of
55///         |---| is a CaseExpression
56/// |-   Just value ->       -|
57/// |      Just (f value)    -|- is a CaseBranch
58/// |    Nothing ->        -|
59/// |-     Nothing         -|- is a second CaseBranch
60/// |
61/// |-- is a `CaseBlock`
62/// ```
63///
64/// `Context` and `Offside` are `Copy`: `handle_next_token` has to read the
65/// current context and then mutate the context stack, so it takes a copy of
66/// the top of the stack to release the borrow. Keep any future variant small
67/// enough that this stays true.
68#[derive(Debug, PartialEq, Clone, Copy)]
69pub enum Context {
70    /// Context for the expression a pattern matching will match on. Carries
71    /// the column of the `case` keyword itself. When this context is popped
72    /// on seeing `of`, that column is stashed in `Layout::case_of_column` so
73    /// the branch block can later check its own column against it: the
74    /// branch block's minimum column is required to sit strictly right of
75    /// `case`'s own (`BUG-10`).
76    CaseExpression(usize),
77
78    /// Context for the block containing the different matches of a catch/of
79    /// A case block minimum indentation is set by the first token after the block is opened.
80    ///
81    /// The `case` keyword's own column lives in `Layout::case_of_column`
82    /// rather than on this variant: widening `CaseBlock` by a `usize` pushes
83    /// `Context` from 16 to 24 bytes, and `CompilationError` — which embeds it
84    /// several layers down — from 120 to 128, where clippy's `result_large_err`
85    /// default starts complaining.
86    CaseBlock(Option<usize>),
87
88    /// Context for a branch in a case/of expression.
89    CaseBranch,
90
91    /// Context for a let expression
92    Let,
93
94    /// Context for a top level declaration.
95    /// Those can be module, custom type, type alias or functions (type annotation/value).
96    TopLevelDeclaration,
97}
98
99impl Context {
100    /// How to name this block when talking to the person who wrote the source.
101    /// Used by `Error::diagnostic` to say which block an indentation error
102    /// belongs to.
103    pub fn description(&self) -> &'static str {
104        match self {
105            Context::CaseExpression(_) => "the expression of a `case … of`",
106            Context::CaseBlock(_) => "the branches of a `case … of`",
107            Context::CaseBranch => "the body of a `case … of` branch",
108            Context::Let => "a `let` block",
109            Context::TopLevelDeclaration => "a top level declaration",
110        }
111    }
112}

114#[derive(Debug, PartialEq, Clone, Copy)]
115pub struct Offside {
116    context: Context,
117    indent: usize, // TODO rename to min_indent
118    line: usize,
119}
120
121impl Offside {
122    /// The smallest column a token may start on without breaking this context's
123    /// indentation rule.
124    ///
125    /// A `case … of` block takes its minimum from the first token which followed
126    /// the block opening, recorded in `CaseBlock`; every other context uses its
127    /// own indentation. `handle_next_token` enforces this, and
128    /// `Error::diagnostic` reports it, so the rule is written here once.
129    pub fn min_indent(&self) -> usize {
130        match self.context {
131            Context::CaseBlock(Some(min)) => min,
132            _ => self.indent,
133        }
134    }
135
136    /// The 1-indexed line the context was opened on.
137    pub fn line(&self) -> usize {
138        self.line
139    }
140
141    /// The kind of block this context describes.
142    pub fn context(&self) -> Context {
143        self.context
144    }
145}

147struct Contexts {
148    stack: Vec<Offside>,
149}
150
151impl Contexts {
152    fn new() -> Contexts {
153        Contexts { stack: vec![] }
154    }
155
156    fn last(&self) -> Option<&Offside> {
157        self.stack.last()
158    }
159
160    fn push(&mut self, offside: Offside) {
161        self.stack.push(offside)
162    }
163
164    fn pop(&mut self) -> Option<Offside> {
165        self.stack.pop()
166    }
167}

169/// The Layout struct is an iterator over a serie of Spanned tokens
170/// which is managing some indentation rules.
171///
172/// It does so by having a context for the current token. A context
173/// represent what kind of terms we are looking at and what indentation
174/// rules we should apply.
175///
176/// The core loop does not clone tokens. A token which has to be emitted as
177/// something else (a block token) *and* reprocessed afterwards is moved into
178/// `reprocess_tokens`, and the token actually emitted is rebuilt from the
179/// original's span, which is `Copy`.
180///
181/// The iterator is fused: once `next` has returned `None` — because it hit an
182/// `Err`, or because the source ran out — every subsequent call returns `None`.
183/// Both cases go through the same `finished` latch, so `Layout` implements
184/// `FusedIterator`. See `Iterator::next` below for why an `Err` is terminal.
185struct Layout<I> {
186    /// The source iterator
187    tokens: I,
188    /// The current level of indentation
189    contexts: Contexts,
190    /// Buffer of tokens already read, but that couldn't have been emitted.
191    ///
192    /// For example, when opening a block we return the OpenBlock token and thus
193    /// we have to reprocess the original token.
194    reprocess_tokens: Vec<Spanned<Position, Token>>,
195    /// Set once `next` has returned `None`, whether because of an `Err` or
196    /// because the source is exhausted; from then on it keeps returning `None`.
197    /// This is what makes the `FusedIterator` impl below sound.
198    /// See `Iterator::next`.
199    finished: bool,
200    /// The column of the `case` keyword whose `CaseExpression` context was
201    /// just popped by seeing `of`, set aside until the first token of the
202    /// matching `CaseBlock` needs to check its own column against it
203    /// (`BUG-10`; see the `(_, Context::CaseBlock(c @ None))` arm). It is not
204    /// carried on `Context::CaseBlock` itself to avoid growing that type —
205    /// see the variant's doc comment.
206    ///
207    /// A single slot, rather than a stack, is enough even for a `case`
208    /// nested in another's scrutinee: the token stream between an `of` and
209    /// its block's first real token can only be that block's own content
210    /// (the synthetic `OpenBlock`), so a second `case … of` cannot have its
211    /// `of` popped before the first one's floor has been read and cleared.
212    case_of_column: Option<usize>,
213}
214
215impl<I> Layout<I>
216where
217    I: Iterator<Item = Result<Spanned<Position, Token>, Error>>,
218{
219    /// Create and initialize a new `Layout` iterator
220    pub fn new(iter: I) -> Layout<I> {
221        Layout {
222            tokens: iter,
223            contexts: Contexts::new(),
224            reprocess_tokens: vec![],
225            finished: false,
226            case_of_column: None,
227        }
228    }
229
230    /// A simple function which manage the internal lookahead structure
231    /// in tandem with the source iterator.
232    ///
233    /// It also convert the end of the source iterator into `Token::EndOfFile`.
234    fn next_token(&mut self) -> Result<Spanned<Position, Token>, Error> {
235        self.reprocess_tokens.pop().map(Ok).unwrap_or_else(|| {
236            self.tokens.next().unwrap_or_else(|| {
237                // The absolute part is unused (hence 0) but the column value is important
238                // (we want a value of 1 to match the first token of a line)
239                let position = Position::new(0, 1, 1);
240
241                Ok(spanned(position, position, Token::EndOfFile))
242            })
243        })
244    }
245
246    /// This is the entry point for our layout processor.
247    ///
248    /// It is called by the iterator's next on each token.
249    ///
250    fn handle_next_token(&mut self) -> Result<Spanned<Position, Token>, Error> {
251        let token = self.next_token()?;
252
253        // Short circuit handling of EOF, and verify we don't have any
254        // remaining contexts to clean.
255        if let Token::EndOfFile = token.value {
256            let Span { start, end } = token.span;
257
258            return match self.contexts.pop() {
259                Some(_) => {
260                    self.reprocess_tokens.push(token);
261                    Ok(spanned(start, end, Token::CloseBlock))
262                }
263                None => Ok(token),
264            };
265        }
266
267        // Retrieve the current offside and, if none exists, create one,
268        // put the current token on the back burner and emit the new block.
269        // In theory this should only happens when we are looking at a top
270        // level declaration (or it's a bug)
271        let offside = match self.contexts.stack.last_mut() {
272            Some(offside) => offside,
273            None => {
274                let start = token.span.start;
275                let off = Offside {
276                    context: Context::TopLevelDeclaration,
277                    indent: start.column,
278                    line: start.line,
279                };
280                self.contexts.push(off);
281
282                self.reprocess_tokens.push(token);
283                return Ok(spanned(start, start, Token::OpenBlock));
284            }
285        };
286
287        trace!("step 1: {:?}, offside: {:?}", token.value, offside);
288
289        // First, we check if we have a closing token with an associated context.
290        // If we do, let's remove the context and return the token
291        match (&token.value, &mut offside.context) {
292            (Token::Of, Context::CaseExpression(case_col)) => {
293                let Span { start, end } = token.span;
294
295                self.case_of_column = Some(*case_col);
296                self.contexts.pop();
297                self.reprocess_tokens.push(token);
298                return Ok(spanned(start, end, Token::CloseBlock));
299            }
300            (Token::OpenBlock, Context::CaseBlock(None)) => (),
301            (_, Context::CaseBlock(c @ None)) => {
302                // Here we are seeing the first token after opening the block, and
303                // this token sets the minimum indentation for the block — unless
304                // it does not clear the `case` keyword's own column, in which case
305                // the block is invalid (BUG-10): a branch level with, or left of,
306                // `case` reads as belonging to something enclosing it.
307                //
308                // `case_of_column` was set when the matching `CaseExpression` was
309                // popped by `of`, several tokens ago (the synthetic `OpenBlock`
310                // sits in between); it is `take`n rather than peeked because
311                // this is the one and only time this block needs it. It is only
312                // ever absent for a malformed `of` with no preceding `case`, in
313                // which case the check below degrades to "column 0 or more" —
314                // i.e. it never fires — rather than rejecting every column.
315                let case_col = self.case_of_column.take().unwrap_or(0);
316                let column = token.start().column;
317
318                // A token at or left of `offside.indent` closes this block in
319                // step 2 below; letting it through here is what allows that
320                // close to happen, so the grammar reports the missing branches
321                // against the `CloseBlock` instead of this arm blaming the
322                // closing token for a misindented branch it never wrote.
323                //
324                // The price is reach: since `offside.indent` is the enclosing
325                // block's indent plus one, the check below only ever fires for a
326                // `case` whose own column is strictly greater than that — and
327                // for anything shallower, a branch really written level with its
328                // `case` is left to the grammar too, whether or not it was meant
329                // as a branch. A `case` nested directly under an enclosing
330                // branch is the common shape that falls out this way.
331                if column > offside.indent && column <= case_col {
332                    // `indent` reports `case_col + 1` — the floor this error is
333                    // actually about — rather than `offside.indent`, which
334                    // stays enclosing-derived (see the `Token::Of` push arm
335                    // below) and would report the block's unrelated
336                    // implicit-close threshold instead.
337                    let error_offside = Offside {
338                        context: Context::CaseBlock(None),
339                        indent: case_col + 1,
340                        line: offside.line,
341                    };
342                    return Err(LayoutError::LayoutError {
343                        offside: error_offside,
344                        token,
345                    }
346                    .into());
347                }
348
349                c.replace(column);
350            }
351            (Token::Else, Context::CaseBlock(Some(min))) if token.start().column < *min => {
352                // `else` closes a `CaseBlock` when it sits strictly left of the
353                // column every branch pattern is required to start at (`min`) —
354                // i.e. it cannot be read as the start of another branch, which
355                // would land exactly on `min`, not left of it (`BUG-23`).
356                //
357                // This has to be a column check, not a bare-token one. An
358                // earlier version of this fix popped on `Token::Else`
359                // unconditionally, on the reasoning that `else` "cannot appear
360                // inside an unclosed `case` branch for any other reason" — but
361                // it can: a branch's body can itself be a multi-line `if …
362                // then … else`, and `if`/`then`/`else` push no context of
363                // their own (see this file's module doc comment), so the
364                // *first* `Else` seen while a `CaseBranch`/`CaseBlock` is open
365                // may belong to that nested `if` rather than to whatever
366                // encloses the `case`. Such an `else` is indented to at least
367                // the branch body's own column, i.e. at or right of `min`, so
368                // the `< *min` guard leaves it alone and the ordinary
369                // indentation rule (`min_indent_required` below) accepts it as
370                // part of the branch body instead of popping anything.
371                //
372                // `Context::CaseBranch` is deliberately not matched by this
373                // arm at all, unlike the version it replaces. Its `.indent`
374                // already equals its `min_indent()` (`CaseBlock` is the one
375                // context where the two differ — see its doc comment for why),
376                // so step 2's ordinary implicit close just below already
377                // closes a `CaseBranch` exactly when a token — `else` included
378                // — dedents out of its body, with no shallow/deep gap for a
379                // nested `if`'s `else` to be misread through. This arm exists
380                // only for the gap that's specific to `CaseBlock`.
381                //
382                // `CaseBlock(None)` — a `case` with no branches yet — is left
383                // out on purpose: that shape is already a grammar error
384                // (`CaseBranch+` requires at least one), and leaving it alone
385                // keeps this fix scoped to the block `else` can legally follow.
386                let Span { start, end } = token.span;
387
388                self.contexts.pop();
389                self.reprocess_tokens.push(token);
390                return Ok(spanned(start, end, Token::CloseBlock));
391            }
392            (Token::In, Context::Let) => {
393                // TODO akin to of/case above, we might have to create a let/in block
394                // to let the parser know when the let part ended. Not sure yet.
395                // TODO We might need to check for the `in` indentation here, needs to be
396                // same as `let`.
397                self.contexts.pop();
398                return Ok(token);
399            }
400            (Token::CloseBlock, Context::TopLevelDeclaration) => {
401                self.contexts.pop();
402                return Ok(token);
403            }
404            _ => (),
405        }
406
407        // Now that we have checked explicit context poping, let's check the implicit one.
408        // These apply to contexts which are terminated by simply having a token on a column
409        // less than the one required by the context.
410        let offside: Offside = {
411            // We repeat the contexts checking here, because we are going to remove contexts
412            // and
413            let offside = match self.contexts.last() {
414                Some(offside) => offside,
415                None => {
416                    let start = token.span.start;
417                    let off = Offside {
418                        context: Context::TopLevelDeclaration,
419                        indent: start.column,
420                        line: start.line,
421                    };
422                    self.contexts.push(off);
423
424                    self.reprocess_tokens.push(token);
425                    return Ok(spanned(start, start, Token::OpenBlock));
426                }
427            };
428
429            let token_column = token.span.start.column;
430            let context_column = offside.indent;
431
432            trace!(
433                "step 2: {:?}, token:{:?}, context:{:?}",
434                offside.context,
435                token_column,
436                context_column
437            );
438
439            match &offside.context {
440                // case branch terminates when we have a token at a level
441                Context::CaseBranch | Context::CaseBlock(_) if token_column <= context_column => {
442                    //   value // token
443                    // Nothing // context
444                    // i i
445                    // Here we have a token on an indentation level lower than the case
446                    // context, so we close that context.
447                    let Span { start, end } = token.span;
448
449                    self.contexts.pop();
450                    self.reprocess_tokens.push(token);
451                    return Ok(spanned(start, end, Token::CloseBlock));
452                }
453
454                // let and top level declaration aren't managed here
455                // although tld could be.
456                _ => (),
457            };
458
459            // we release the reference on self.contexts because we need to
460            // mutate it down the line. `Offside` is `Copy`, so this is a few
461            // words on the stack and not an allocation.
462            *offside
463        };
464
465        // Second, we enforce the indentation rule we have on record
466        let min_indent_required = offside.min_indent();
467
468        if token.span.start.column.cmp(&min_indent_required) == Ordering::Less {
469            // The token is moved into the error and is *not* pushed onto
470            // `reprocess_tokens`. Nothing in this branch mutates `self.contexts`,
471            // so replaying the token would re-run this exact comparison against
472            // this exact context and produce the same error forever. `next`
473            // fuses the iterator on `Err` anyway, so the token has no reader
474            // left; keeping it buffered would only make the loop reachable
475            // again for anyone who removes that fuse.
476            return Err(LayoutError::LayoutError { offside, token }.into());
477        };
478
479        // Third, we create new tokens, new contexts and emit block tokens as required
480
481        trace!(
482            "step 3: {:?} ({}:{}), context: {:?}",
483            token.value,
484            token.start().column,
485            token.end().column,
486            offside.context
487        );
488        match (&token.value, &offside.context) {
489            (Token::Case, _) => {
490                self.contexts.push(Offside {
491                    context: Context::CaseExpression(token.start().column),
492                    indent: token.start().column + 1,
493                    line: token.start().line,
494                });
495                self.reprocess_tokens
496                    .push(spanned(*token.end(), *token.end(), Token::OpenBlock));
497            }
498            (Token::Of, _) => {
499                // `case_of_column`, set by the matching `Context::CaseExpression`
500                // pop above, is left set here on purpose: the BUG-10 check in the
501                // `(_, Context::CaseBlock(c @ None))` arm reads and clears it when
502                // this block's first real token arrives.
503                //
504                // `indent` is derived from the enclosing block rather than from
505                // `case`'s column, because it is the threshold step 2's implicit
506                // close keys on: a dedent back to the enclosing block's level has
507                // to *close* this block, not merely violate it.
508                self.contexts.push(Offside {
509                    context: Context::CaseBlock(None),
510                    indent: offside.indent + 1,
511                    line: token.start().line,
512                });
513                self.reprocess_tokens
514                    .push(spanned(*token.end(), *token.end(), Token::OpenBlock));
515            }
516            (Token::Let, _) => self.contexts.push(Offside {
517                context: Context::Let,
518                indent: token.start().column + 1,
519                line: token.start().line,
520            }),
521            (Token::Arrow, Context::CaseBlock(Some(min_indent))) => {
522                self.contexts.push(Offside {
523                    context: Context::CaseBranch,
524                    indent: min_indent + 1,
525                    line: token.start().line,
526                });
527                self.reprocess_tokens
528                    .push(spanned(*token.end(), *token.end(), Token::OpenBlock));
529            }
530            (Token::OpenBlock, _) => (),
531            _ => {
532                if token.span.start.column == 1 && token.span.start.line > offside.line {
533                    // Here we have a token which isn't OpenBlock (special case above)
534                    // but which is at the beginning of a new line. This most probably
535                    // mean we have reached the end of the previous block and are
536                    // starting a new one.
537
538                    let start = token.span.start;
539                    self.reprocess_tokens.push(token);
540
541                    // Furthermore in case of implicitely terminated block,
542                    // pop the context from the stack and let the parser complain
543                    // about the invalid syntax. We do this to break an infinite
544                    // loop where we would always be checking the current token
545                    // against the current context.
546                    if offside.context == Context::TopLevelDeclaration {
547                        self.contexts.pop();
548                    }
549
550                    return Ok(spanned(start, start, Token::CloseBlock));
551                }
552            }
553        }
554
555        Ok(token)
556    }
557}

559impl<I> Iterator for Layout<I>
560where
561    I: Iterator<Item = Result<Spanned<Position, Token>, Error>>,
562{
563    type Item = Result<(BytePos, Token, BytePos), Error>;
564
565    /// Yields one layout-processed token per call, and stops — returns `None`
566    /// forever — at the first of two events: `Token::EndOfFile`, or an `Err`.
567    ///
568    /// Both events set the same `finished` latch, which is what makes "forever"
569    /// unconditional. It is worth being precise about this, because the two
570    /// events do not arrive the same way. The `Err` is terminal by decision.
571    /// `EndOfFile` is *re-derived* on each call — `next_token` finds
572    /// `reprocess_tokens` empty, asks the source for another token, gets `None`
573    /// and synthesises a fresh `EndOfFile` — and `Iterator`'s contract permits a
574    /// source to yield `Some` again after a `None`. Every source used here is in
575    /// fact fused, so latching changes nothing in practice; it just means the
576    /// guarantee is a property of this type rather than of its callers, which is
577    /// what the `FusedIterator` impl below asserts.
578    ///
579    /// The error fuse matters because layout errors are not, in general,
580    /// recoverable *by this iterator*: an indentation violation is diagnosed
581    /// without changing `self.contexts`, so there is no state transition that
582    /// would let the same input be read differently on a second attempt. Errors
583    /// from the tokenizer, propagated through `handle_next_token`, are fused
584    /// the same way. A consumer which drains this iterator fully therefore sees
585    /// at most one error and then terminates, rather than the same error
586    /// repeated without bound.
587    ///
588    /// This deliberately forecloses accumulating layout diagnostics. The phases
589    /// after parsing do accumulate — canonicalization, type checking and
590    /// exhaustiveness all return `Result<_, Vec<Error>>` so one bad declaration
591    /// cannot hide the next — and should layout ever want the same, this fuse is
592    /// what has to change: the branch above would need a state transition that
593    /// guarantees forward progress before the error could be resumed from.
594    fn next(&mut self) -> Option<Self::Item> {
595        if self.finished {
596            return None;
597        }
598
599        let res = self.handle_next_token();
600        trace!("step 4: {:?}", res);
601
602        match res {
603            Ok(Spanned {
604                value: Token::EndOfFile,
605                ..
606            }) => {
607                self.finished = true;
608                None
609            }
610            Ok(Spanned { value, span }) => {
611                Some(Ok((span.start.absolute, value, span.end.absolute)))
612            }
613            Err(err) => {
614                self.finished = true;
615                Some(Err(err))
616            }
617        }
618    }
619}

622/// returned `None` it cannot return `Some` again regardless of what the source
623/// iterator does. That is exactly `FusedIterator`'s contract, and stating it
624/// makes `.fuse()` a no-op for callers.

635    // Create an approximation for the token position in the stream.
636    // We don't count the spaces between tokens, but it gives us enough
637    // to understand where a failure happened.
638    fn tokens_to_spanned(tokens: &[Token]) -> Vec<Result<Spanned<Position, Token>, Error>> {
639        let mut pos = Position::new(0, 1, 1);
640
641        tokens
642            .iter()
643            .cloned()
644            .filter_map(|token| {
645                let start = pos;
646                let inc = match &token {
647                    Token::Module => 6,
648                    Token::UpperIdentifier(name) => name.len(),
649                    Token::Exposing => 8,
650                    Token::LPar | Token::RPar => 1,
651                    Token::Comma => 1,
652                    Token::Pipe => 1,
653                    Token::Equal => 1,
654                    Token::Type | Token::Case => 4,
655                    Token::Of | Token::Arrow => 2,
656                    _ => 0,
657                };
658
659                // Hack to simulate new lines and indentation
660                let emit = if let Token::LowerIdentifier(name) = &token {
661                    match name.as_str() {
662                        "\n" => {
663                            pos.new_line();
664                            false
665                        }
666                        "  " => {
667                            pos.increment_by(2);
668                            false
669                        }
670                        _ => {
671                            pos.increment_by(name.len());
672                            true
673                        }
674                    }
675                } else {
676                    pos.increment_by(inc);
677                    true
678                };
679
680                let end = pos;
681
682                if emit {
683                    Some(Ok(spanned(start, end, token)))
684                } else {
685                    None
686                }
687            })
688            .collect()
689    }

1172    /// Poll `iter` up to `cap` times, stopping early if it terminates.
1173    ///
1174    /// The cap is what makes a non-terminating iterator show up as a failed
1175    /// assertion rather than as a hung test process.
1176    fn drain_bounded<I: Iterator>(iter: &mut I, cap: usize) -> Vec<I::Item> {
1177        let mut items = Vec::with_capacity(cap);
1178
1179        for _ in 0..cap {
1180            match iter.next() {
1181                Some(item) => items.push(item),
1182                None => break,
1183            }
1184        }
1185
1186        items
1187    }

10/// Walk one of a package's source roots into a database of its own.
11///
12/// Nothing else can be in that database, so a file is named by its place in the
13/// package and nothing more — see [`load_package_sources_into`] for the build case,
14/// where that name is not unique.
15// We don't support non-UTF8 characters in path
16pub fn load_package_sources(
17    package_dir: &Path,
18    root: SourceRoot,
19) -> Result<SourceFiles, CompilationError> {
20    let mut sources = SourceFiles::new();
21    load_package_sources_into(package_dir, root, None, &mut sources)?;
22    Ok(sources)
23}

25/// Walk one source root of the package rooted at `package_dir` for `.zel` files, add
26/// each one to `files`, and return the id each was given.
27///
28/// A package has two source roots and they are walked separately, because a module's
29/// name is its path under *its own* root: `src/Model.zel` and `tests/Model.zel` are
30/// both `Model` ([*Source
31/// roots*](../../../../docs/spec/packages.md#source-roots)). Each file carries the root
32/// it came from, so a diagnostic can tell the two apart.
33///
34/// A package must have a `src/`, and a walk of one that is not there is the error
35/// `WalkDir` reports. A `tests/` that is not there is a package that holds no tests,
36/// which is not an error and loads as no modules at all.
37///
38/// A build holds one file database for every package in it, because a
39/// [`SourceFileId`](files::SourceFileId) travels inside an `Interface` and outlives the
40/// package it came from (`ERR-5`), so the ids of two packages have to be drawn from one
41/// sequence. This is the entry point that appends into that shared database;
42/// [`load_package_sources`] is the same walk over a database of its own, which is what
43/// a caller compiling exactly one source root wants.
44///
45/// `package` names the package `package_dir` holds, and every caller sharing a database
46/// across packages has to pass it: a path relative to a package's own `src/` is not
47/// unique in a build — two packages may each hold a `Size.zel` — so without it a
48/// diagnostic cannot say which package it is about.
49///
50/// The ids come back rather than the files themselves: `files` is borrowed mutably for
51/// the walk, and the caller needs it borrowed immutably afterwards to read the sources
52/// back out.
53pub fn load_package_sources_into(
54    package_dir: &Path,
55    root: SourceRoot,
56    package: Option<&PackageName>,
57    sources: &mut SourceFiles,
58) -> Result<Vec<files::SourceFileId>, CompilationError> {
59    let mut loaded = vec![];
60    let mut errors = vec![];
61
62    let root_dir = package_dir.join(root.directory());
63
64    // A package that holds no tests has no `tests/` to walk, and that is not a failure
65    // the way a missing `src/` is. Asking first is what keeps the walk's own
66    // does-not-exist error meaning what it says everywhere else.
67    if root == SourceRoot::Tests && !root_dir.exists() {
68        return Ok(loaded);
69    }
70
71    // `WalkDir`'s iterator advances past every entry it yields — including an `Err`
72    // one — and terminates on its own (a root that doesn't exist yields exactly one
73    // `Err` and then ends), so this loop cannot spin on a single failing entry the
74    // way the invariant in `CLAUDE.md` about `Result`-yielding iterators warns
75    // against.
76    for entry in WalkDir::new(&root_dir).follow_links(true) {
77        match entry {
78            Ok(entry) => {
79                let path = entry.path().to_path_buf();
80
81                match path.extension() {
82                    Some(ext) if ext == "zel" => {}
83                    _ => continue,
84                }
85
86                match SourceFile::load(path, &root_dir, root, package) {
87                    Ok(src) => {
88                        loaded.push(sources.add_file(src));
89                    }
90                    Err(err) => {
91                        errors.push(err);
92                    }
93                }
94            }
95            Err(walk_err) => {
96                errors.push(SourceFileError::from_walk_error(&root_dir, walk_err));
97            }
98        }
99    }
100
101    if errors.is_empty() {
102        Ok(loaded)
103    } else {
104        Err(errors.into())
105    }
106}

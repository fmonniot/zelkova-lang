3/// Levenshtein edit distance between two strings, counted in `char`s rather
4/// than bytes so a suggestion built over multi-byte UTF-8 identifiers isn't
5/// skewed by how many bytes each character happens to take.
6fn levenshtein_distance(a: &str, b: &str) -> usize {
7    let a: Vec<char> = a.chars().collect();
8    let b: Vec<char> = b.chars().collect();
9    let lb = b.len();
10
11    // One row of the edit-distance matrix, updated in place: `row[j]` holds
12    // the distance between the prefix of `a` seen so far and `b[..j]`.
13    let mut row: Vec<usize> = (0..=lb).collect();
14    for (i, &ca) in a.iter().enumerate() {
15        let mut prev_diag = row[0];
16        row[0] = i + 1;
17        for (j, &cb) in b.iter().enumerate() {
18            let temp = row[j + 1];
19            let cost = if ca == cb { 0 } else { 1 };
20            row[j + 1] = (row[j + 1] + 1).min(row[j] + 1).min(prev_diag + cost);
21            prev_diag = temp;
22        }
23    }
24    row[lb]
25}

27/// The candidate in `candidates` closest to `target`, if it is close enough
28/// to be worth suggesting as a "did you mean …?".
29///
30/// A bad suggestion is worse than none — it sends the reader off to check
31/// something irrelevant — so the threshold scales with `target`'s length
32/// rather than being a flat constant: one typo'd character in a three-letter
33/// name is proportionally as far as three typos in a ten-letter one, and both
34/// should still suggest, while a same-length but otherwise unrelated word
35/// should not. A third of the length, rounded up, with a floor of one so a
36/// one- or two-character name can still get a same-distance suggestion, is
37/// what `ERR-7` was tuned against (`widthDefault` → `withDefault`, distance
38/// 1 over 12 characters).
39///
40/// Equally-distant candidates are broken by comparing the candidates
41/// themselves, so the answer depends only on the *set* of candidates and not
42/// on the order they arrive in. That matters because every caller in the
43/// compiler feeds this `HashMap` key iteration, whose order is randomised per
44/// instance: without the tie-break, `foldl`/`foldr` or two sibling
45/// constructors would produce a different suggestion from one run to the next
46/// for unchanged source.
47pub fn suggest<'a, I>(target: &str, candidates: I) -> Option<&'a str>
48where
49    I: IntoIterator<Item = &'a str>,
50{
51    let threshold = ((target.chars().count() as f64) / 3.0).ceil().max(1.0) as usize;
52
53    candidates
54        .into_iter()
55        .filter(|c| *c != target)
56        .map(|c| (c, levenshtein_distance(target, c)))
57        .filter(|(_, distance)| *distance <= threshold)
58        .min_by(|(a, da), (b, db)| da.cmp(db).then_with(|| a.cmp(b)))
59        .map(|(c, _)| c)
60}

62/// This let use collect an iterator of result into a result of vectors.
63/// This let us partition an iteration of result into an iterator of result or a vector of errors if any.
64///
65/// The generic signature, assuming `I: FromIterator` would look like `Iterator<Result<T, E>> -> Result<I<T>, I<E>>`.
66/// I wish this was part of the standard library, but as it is not here is my custom version.
67pub fn collect_accumulate<T, E, I, R>(iterator: I) -> Result<R, Vec<E>>
68where
69    I: Iterator<Item = Result<T, E>>,
70    R: FromIterator<T>,
71{
72    let mut errors = vec![];
73
74    let r = iterator
75        .filter_map(|i| match i {
76            Ok(t) => Some(t),
77            Err(e) => {
78                errors.push(e);
79                None
80            }
81        })
82        .collect();
83
84    if errors.is_empty() {
85        Ok(r)
86    } else {
87        Err(errors)
88    }
89}

92mod tests {
93    use super::*;

95    #[test]
96    fn levenshtein_distance_identical_strings_is_zero() {
97        assert_eq!(levenshtein_distance("withDefault", "withDefault"), 0);
98    }
99
100    #[test]
101    fn levenshtein_distance_one_insertion() {
102        // `widthDefault` has one extra `d` compared to `withDefault`.
103        assert_eq!(levenshtein_distance("widthDefault", "withDefault"), 1);
104    }
105
106    #[test]
107    fn levenshtein_distance_counts_chars_not_bytes() {
108        // Both are single-character strings in `char`s, but multiple bytes in
109        // UTF-8 — a byte-wise distance would overcount.
110        assert_eq!(levenshtein_distance("é", "è"), 1);
111    }
112
113    #[test]
114    fn suggest_finds_a_close_candidate() {
115        let candidates = vec!["andThen", "map", "withDefault"];
116        assert_eq!(
117            suggest("widthDefault", candidates),
118            Some("withDefault"),
119            "a one-character typo should suggest the near-miss"
120        );
121    }
122
123    #[test]
124    fn suggest_finds_nothing_for_an_unrelated_name() {
125        let candidates = vec!["andThen", "map", "withDefault"];
126        assert_eq!(
127            suggest("frobnicate", candidates),
128            None,
129            "a name resembling nothing in scope must not produce a suggestion"
130        );
131    }
132
133    #[test]
134    fn suggest_ignores_exact_matches() {
135        // `suggest` is only ever called after a lookup by the exact name has
136        // already failed, but it should not recommend the target back to
137        // itself if a caller ever hands it one anyway.
138        let candidates = vec!["map"];
139        assert_eq!(suggest("map", candidates), None);
140    }
141
142    #[test]
143    fn suggest_picks_the_closest_of_several_candidates() {
144        // Distances from "withDefault" are 4, 2 and 1 respectively, and the
145        // threshold for an 11-character target is 4 — so all three survive the
146        // filter and only the distance comparison separates them. The nearest
147        // is deliberately *last*: returning the first candidate under the
148        // threshold, or dropping the comparison entirely, picks "withDef".
149        let candidates = vec!["withDef", "withDefau", "withDefaults"];
150        assert_eq!(suggest("withDefault", candidates), Some("withDefaults"));
151    }
152
153    #[test]
154    fn suggest_breaks_ties_on_the_candidate_not_on_iteration_order() {
155        // All three are distance 1 from "maq", so only the tie-break decides.
156        // Callers hand `suggest` `HashMap` key iteration, whose order is
157        // randomised per instance; the answer must depend on the set alone.
158        assert_eq!(suggest("maq", vec!["mad", "map", "mac"]), Some("mac"));
159        assert_eq!(suggest("maq", vec!["map", "mac", "mad"]), Some("mac"));
160        assert_eq!(suggest("maq", vec!["mac", "mad", "map"]), Some("mac"));
161    }

1//! Cut a module's raw token stream into one chunk per top-level declaration, so that a
2//! syntax error in one declaration does not end the parse of the ones after it.
3//!
4//! A top-level declaration begins in column 1, and a line indented past column 1
5//! continues the declaration above it (`docs/spec/layout.md`, *Top-level
6//! declarations*). [`Chunks`] reads that rule literally, on the tokenizer's output and
7//! before the layout pass: a token starts a new chunk when it sits in column 1, is not
8//! the first token of the file, and [`can_start_declaration`] accepts it. The first
9//! chunk is the module header. Because the cut is made before layout, nothing a
10//! declaration contains — an unclosed `case`, a missing `of` — can move where the next
11//! one begins.
12//!
13//! What the cut deliberately does not do:
14//!
15//! - A column-1 token no declaration starts with (`1`, `|`, `=`, `)`) stays in its
16//!   chunk. The layout pass closes the declaration at it as it would anywhere, and the
17//!   mistake costs that chunk one error.
18//! - A tokenizer error never starts a chunk. It belongs to the chunk it falls in, and
19//!   the tokens after it are still read, so the declarations after it are still cut.
20//!
21//! Each chunk is then given its own [`Layout`](super::layout::Layout) and its own
22//! grammar entry point by `parser::parse_recovering`.
23

30/// One item of the tokenizer's output, its error already converted to the parser's.
31pub(crate) type RawToken = Result<Spanned<Position, Token>, Error>;
32
33/// Whether a declaration can begin with `token`: the tokens the grammar's `Decl`
34/// alternatives open on, soft keywords included, since each of those is also a name a
35/// function can have.
36///
37/// [`Chunks`] cuts before a column-1 token this accepts, and `Layout::explain` asks it
38/// whether an indented line could have been meant as a declaration of its own.
39pub(crate) fn can_start_declaration(token: &Token) -> bool {
40    matches!(
41        token,
42        Token::LowerIdentifier(_)
43            | Token::Left
44            | Token::Right
45            | Token::Non
46            | Token::Foreign
47            | Token::Unsafe
48            | Token::Where
49            | Token::Derived
50            | Token::Class
51            | Token::Instance
52            | Token::Type
53            | Token::Import
54            | Token::Infix
55    )
56}

58/// The tokens of one top-level declaration, or of the module header, as the tokenizer
59/// produced them, errors included.
60#[derive(Debug, Clone, PartialEq)]
61pub(crate) struct Chunk {
62    /// The tokenizer's items for this chunk, in source order.
63    pub(crate) tokens: Vec<RawToken>,
64    /// Where this chunk's text begins: byte 0 for the first chunk, and the first token's
65    /// start for every other. The chunks of a file therefore tile it.
66    pub(crate) start: BytePos,
67    /// Where this chunk's input ends: the start of the next chunk's first token, or the
68    /// end of the source for the last chunk. The layout pass closes the chunk's open
69    /// blocks here.
70    pub(crate) end: Position,
71}
72
73impl Chunk {
74    /// A chunk with no tokens, ending at `end`. What the header chunk of a file with no
75    /// tokens at all is.
76    pub(crate) fn empty(end: Position) -> Chunk {
77        Chunk {
78            tokens: vec![],
79            start: end.absolute,
80            end,
81        }
82    }
83
84    /// The source text this chunk covers.
85    pub(crate) fn span(&self) -> Span<BytePos> {
86        Span {
87            start: self.start,
88            end: self.end.absolute,
89        }
90    }
91
92    /// The value this chunk is a declaration of, read off its first item alone: the name
93    /// of a `LowerIdentifier` there, and `None` for any other first item, a tokenizer
94    /// error included. Only a declaration chunk is asked; the header is not.
95    pub(crate) fn declares(&self) -> Option<Name> {
96        match self.tokens.first() {
97            Some(Ok(Spanned {
98                value: Token::LowerIdentifier(name),
99                ..
100            })) => Some(Name::new(name.as_str())),
101            _ => None,
102        }
103    }
104}
105
106/// The position just past the last character of `source`, counted the way the tokenizer
107/// counts: lines and columns are 1-indexed, and a column advances by a character's byte
108/// length.
109pub(crate) fn end_of(source: &str) -> Position {
110    let line = source.matches('\n').count() + 1;
111    let column = match source.rfind('\n') {
112        Some(newline) => source.len() - newline,
113        None => source.len() + 1,
114    };
115
116    Position::new(source.len(), column, line)
117}

119/// An iterator adaptor that takes the tokenizer's output and yields one [`Chunk`] per
120/// top-level declaration, the module header first. See the module documentation for
121/// where it cuts.
122///
123/// # Advance or stop
124///
125/// This is the first consumer that reads the tokenizer past an error, which is the
126/// operation `BUG-4` and `BUG-5` were about: a tokenizer path that reports an error
127/// without consuming input reports it again on every poll. `Chunks` therefore ends the
128/// stream at the second of two consecutive equal errors, keeping the first in its chunk.
129/// A tokenizer that repeats one error truncates the parse there, where it would
130/// otherwise never end (`CLAUDE.md` — *A `Result`-yielding iterator must advance or
131/// stop*). Two consecutive errors that differ are both kept. The check catches only an
132/// error repeated back to back, which is the `BUG-4`/`BUG-5` shape: a cycle of different
133/// errors, or of an error and a token that consumed nothing, is not caught.
134///
135/// Once `next` has returned `None` it keeps returning `None`, whatever the source does,
136/// so `Chunks` is a `FusedIterator`.
137pub(crate) struct Chunks<I> {
138    tokens: I,
139    /// Where the source ends, which is where the last chunk's input ends.
140    end_of_source: Position,
141    /// The token that started the next chunk, read while the previous one was still
142    /// being collected.
143    pending: Option<Spanned<Position, Token>>,
144    /// Whether any item has been read yet. The first token of the file never cuts, so
145    /// the header chunk always opens on it.
146    read_any: bool,
147    /// The previous item read, when it was an error: the error a second, equal one is
148    /// compared against.
149    last_error: Option<Error>,
150    /// Set once the source is exhausted, or repeated an error.
151    finished: bool,
152}
153
154impl<I> Chunks<I>
155where
156    I: Iterator<Item = RawToken>,
157{
158    /// Cut `tokens`, a source's tokenizer output, into chunks. `end_of_source` is where
159    /// the last chunk's input ends — see [`end_of`].
160    pub(crate) fn new(tokens: I, end_of_source: Position) -> Chunks<I> {
161        Chunks {
162            tokens,
163            end_of_source,
164            pending: None,
165            read_any: false,
166            last_error: None,
167            finished: false,
168        }
169    }
170
171    /// Whether `token` opens a new chunk rather than continuing the current one.
172    fn cuts(&self, token: &Spanned<Position, Token>) -> bool {
173        self.read_any && token.span.start.column == 1 && can_start_declaration(&token.value)
174    }
175}
176
177impl<I> Iterator for Chunks<I>
178where
179    I: Iterator<Item = RawToken>,
180{
181    type Item = Chunk;
182
183    fn next(&mut self) -> Option<Chunk> {
184        let mut chunk = match self.pending.take() {
185            Some(first) => Chunk {
186                start: first.span.start.absolute,
187                tokens: vec![Ok(first)],
188                end: self.end_of_source,
189            },
190            None if self.finished => return None,
191            None => Chunk {
192                tokens: vec![],
193                start: BytePos(0),
194                end: self.end_of_source,
195            },
196        };
197
198        while !self.finished {
199            match self.tokens.next() {
200                None => self.finished = true,
201                Some(Ok(token)) => {
202                    self.last_error = None;
203                    if self.cuts(&token) {
204                        chunk.end = token.span.start;
205                        self.pending = Some(token);
206                        return Some(chunk);
207                    }
208                    self.read_any = true;
209                    chunk.tokens.push(Ok(token));
210                }
211                Some(Err(error)) => {
212                    if self.last_error.as_ref() == Some(&error) {
213                        self.finished = true;
214                    } else {
215                        self.read_any = true;
216                        self.last_error = Some(error.clone());
217                        chunk.tokens.push(Err(error));
218                    }
219                }
220            }
221        }
222
223        // The source ran out with nothing read since the last cut: there is no chunk to
224        // hand back. That only happens to a source with no items at all, since a cut
225        // always leaves its token pending.
226        if chunk.tokens.is_empty() {
227            return None;
228        }
229
230        Some(chunk)
231    }
232}

234/// `next` returns `None` only once `finished` is set and nothing is pending, and nothing
235/// clears `finished` or fills `pending` afterwards.
236impl<I> FusedIterator for Chunks<I> where I: Iterator<Item = RawToken> {}

247    fn chunks_of(source: &str) -> Vec<Chunk> {
248        let tokens = make_tokenizer(source).map(|r| r.map_err(Error::from));
249        Chunks::new(tokens, end_of(source)).collect()
250    }
251
252    /// The first token of every chunk, or `None` for one that opens on an error.
253    fn first_tokens(chunks: &[Chunk]) -> Vec<Option<Token>> {
254        chunks
255            .iter()
256            .map(|chunk| match chunk.tokens.first() {
257                Some(Ok(token)) => Some(token.value.clone()),
258                _ => None,
259            })
260            .collect()
261    }

263    fn lower(name: &str) -> Option<Token> {
264        Some(Token::LowerIdentifier(name.to_string()))
265    }

267    /// A column-1 token a declaration can start with cuts; an indented one and a
268    /// column-1 one no declaration starts with do not. The chunks tile the source, and
269    /// the last one ends at its end.
270    ///
271    /// Verified to fail by making `cuts` return `false`: the source is then one chunk.
272    #[test]
273    fn a_column_1_declaration_start_cuts_and_nothing_else_does() {
274        let source = indoc! {"
275            module M exposing (..)
276
277            f x =
278              g x
279            | oops
280            type T = A
281        "};
282
283        let chunks = chunks_of(source);
284
285        assert_eq!(
286            first_tokens(&chunks),
287            vec![Some(Token::Module), lower("f"), Some(Token::Type)]
288        );
289
290        let f_start = source.find("f x").unwrap();
291        let type_start = source.find("type").unwrap();
292        assert_eq!(chunks[0].span().to_range(), 0..f_start);
293        assert_eq!(chunks[1].span().to_range(), f_start..type_start);
294        assert_eq!(chunks[2].span().to_range(), type_start..source.len());
295        assert_eq!(chunks[1].end.column, 1);
296    }

325    /// A source that yields one token and then the same error without end is cut short
326    /// at the second error, so polling the adaptor ends.
327    ///
328    /// The source counts its polls and panics past a cap, which is how the test turns
329    /// a loop that would never return into a failure. Verified to fail by removing the
330    /// `self.last_error.as_ref() == Some(&error)` check from `next`: the first poll then
331    /// reads the source past its cap.
332    #[test]
333    fn a_repeated_tokenizer_error_ends_the_stream() {
334        const SOURCE_CAP: usize = 64;
335        const POLLS: usize = 4;
336
337        let polls = Rc::new(Cell::new(0));
338        let counter = Rc::clone(&polls);
339        let position = Position::new(0, 1, 1);
340        let error = Error::Tokenizer(TokenizerError {
341            error: spanned(
342                BytePos(1),
343                BytePos(2),
344                crate::parser::tokenizer::TokenizerErrorType::TabError,
345            ),
346        });
347        let mut first = Some(Ok(spanned(position, position, Token::Module)));
348        let source = std::iter::from_fn(move || {
349            counter.set(counter.get() + 1);
350            assert!(
351                counter.get() <= SOURCE_CAP,
352                "the adaptor kept reading a source that repeats one error"
353            );
354            Some(first.take().unwrap_or_else(|| Err(error.clone())))
355        });
356
357        let mut chunks = Chunks::new(source, position);
358        let items: Vec<Chunk> = (0..POLLS).map_while(|_| chunks.next()).collect();
359
360        assert_eq!(items.len(), 1, "expected one chunk, got {:?}", items);
361        assert_eq!(items[0].tokens.len(), 2, "the token and the first error");
362        assert!(
363            chunks.next().is_none(),
364            "the adaptor restarted after ending"
365        );
366        assert!(
367            polls.get() <= 3,
368            "read {} items from the source",
369            polls.get()
370        );
371    }

373    /// `can_start_declaration` and the grammar's `Decl` alternatives agree on what a
374    /// declaration opens with. `Decls` reads `Decl+`, so a start the grammar accepts and
375    /// this function does not would not fail: that declaration would silently join the
376    /// previous chunk. The grammar's own first set is the `expected` list of the error a
377    /// stray `)` in column 1 raises, and each name in it is mapped to its token here.
378    ///
379    /// Verified to fail by removing `Token::Infix` from `can_start_declaration` (the
380    /// assertion on `infix` goes red).
381    #[test]
382    fn can_start_declaration_matches_the_grammars_declaration_starts() {
383        use crate::parser::{parse_recovering, Error};
384        use codespan_reporting::files::SimpleFile;
385
386        let source = "module Main exposing (..)\n\nf = 1\n)\n";
387        let file = SimpleFile::new("test".to_owned(), source.to_owned());
388        let parsed = parse_recovering(&file);
389        let expected = match parsed.failures.first().map(|f| &f.error) {
390            Some(Error::UnexpectedToken { expected, .. }) => expected.clone(),
391            other => panic!("expected an unexpected-token error, got {:?}", other),
392        };
393
394        // Every token kind a declaration could plausibly start with, by the name the
395        // grammar gives it, so a new one has to be added here to be compared.
396        let candidates: Vec<(&str, Token)> = vec![
397            ("lo_ident", Token::LowerIdentifier("f".to_string())),
398            ("left", Token::Left),
399            ("right", Token::Right),
400            ("non", Token::Non),
401            ("foreign", Token::Foreign),
402            ("unsafe", Token::Unsafe),
403            ("where", Token::Where),
404            ("derived", Token::Derived),
405            ("type", Token::Type),
406            ("import", Token::Import),
407            ("infix", Token::Infix),
408            ("class", Token::Class),
409            ("instance", Token::Instance),
410        ];
411
412        for (name, token) in &candidates {
413            let in_grammar = expected.iter().any(|e| e.trim_matches('"') == *name);
414            assert_eq!(
415                can_start_declaration(token),
416                in_grammar,
417                "`{}`: can_start_declaration disagrees with the grammar's expected {:?}",
418                name,
419                expected
420            );
421        }
422        for e in &expected {
423            let e = e.trim_matches('"');
424            assert!(
425                candidates.iter().any(|(name, _)| *name == e),
426                "the grammar accepts `{}` to start a declaration and this test does not list it",
427                e
428            );
429        }
430    }

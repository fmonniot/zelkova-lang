1//! Cut a module's raw token stream into one chunk per top-level declaration, so that a
2//! syntax error in one declaration does not end the parse of the ones after it.
3//!
4//! A top-level declaration begins in column 1, and a line indented past column 1
5//! continues the declaration above it (`docs/spec/layout.md`, *Top-level
6//! declarations*). [`Chunks`] reads that rule literally, on the tokenizer's output and
7//! before the layout pass: a token starts a new chunk when it sits in column 1, is not
8//! the first token of the file, and [`can_start_declaration`] accepts it. The first
9//! chunk is the module header. Because the cut is made before layout, nothing a
10//! declaration contains — an unclosed `case`, a missing `of` — can move where the next
11//! one begins.
12//!
13//! What the cut deliberately does not do:
14//!
15//! - A column-1 token no declaration starts with (`1`, `|`, `=`, `)`) stays in its
16//!   chunk. The layout pass closes the declaration at it as it would anywhere, and the
17//!   mistake costs that chunk one error.
18//! - A tokenizer error never starts a chunk. It belongs to the chunk it falls in, and
19//!   the tokens after it are still read, so the declarations after it are still cut.
20//!
21//! Each chunk is then given its own [`Layout`](super::layout::Layout) and its own
22//! grammar entry point by `parser::parse_recovering`.
23

30/// One item of the tokenizer's output, its error already converted to the parser's.
31pub(crate) type RawToken = Result<Spanned<Position, Token>, Error>;
32
33/// Whether a declaration can begin with `token`: the tokens the grammar's `Decl`
34/// alternatives open on, soft keywords included, since each of those is also a name a
35/// function can have.
36///
37/// [`Chunks`] cuts before a column-1 token this accepts, and `Layout::explain` asks it
38/// whether an indented line could have been meant as a declaration of its own.
39pub(crate) fn can_start_declaration(token: &Token) -> bool {
40    matches!(
41        token,
42        Token::LowerIdentifier(_)
43            | Token::Left
44            | Token::Right
45            | Token::Non
46            | Token::Foreign
47            | Token::Unsafe
48            | Token::Type
49            | Token::Import
50            | Token::Infix
51    )
52}

54/// The tokens of one top-level declaration, or of the module header, as the tokenizer
55/// produced them, errors included.
56#[derive(Debug, Clone, PartialEq)]
57pub(crate) struct Chunk {
58    /// The tokenizer's items for this chunk, in source order.
59    pub(crate) tokens: Vec<RawToken>,
60    /// Where this chunk's text begins: byte 0 for the first chunk, and the first token's
61    /// start for every other. The chunks of a file therefore tile it.
62    pub(crate) start: BytePos,
63    /// Where this chunk's input ends: the start of the next chunk's first token, or the
64    /// end of the source for the last chunk. The layout pass closes the chunk's open
65    /// blocks here.
66    pub(crate) end: Position,
67}
68
69impl Chunk {
70    /// A chunk with no tokens, ending at `end`. What the header chunk of a file with no
71    /// tokens at all is.
72    pub(crate) fn empty(end: Position) -> Chunk {
73        Chunk {
74            tokens: vec![],
75            start: end.absolute,
76            end,
77        }
78    }
79
80    /// The source text this chunk covers.
81    pub(crate) fn span(&self) -> Span<BytePos> {
82        Span {
83            start: self.start,
84            end: self.end.absolute,
85        }
86    }
87
88    /// The value this chunk is a declaration of, read off its first item alone: the name
89    /// of a `LowerIdentifier` there, and `None` for any other first item, a tokenizer
90    /// error included. Only a declaration chunk is asked; the header is not.
91    pub(crate) fn declares(&self) -> Option<Name> {
92        match self.tokens.first() {
93            Some(Ok(Spanned {
94                value: Token::LowerIdentifier(name),
95                ..
96            })) => Some(Name::new(name.as_str())),
97            _ => None,
98        }
99    }
100}

102/// The position just past the last character of `source`, counted the way the tokenizer
103/// counts: lines and columns are 1-indexed, and a column advances by a character's byte
104/// length.
105pub(crate) fn end_of(source: &str) -> Position {
106    let line = source.matches('\n').count() + 1;
107    let column = match source.rfind('\n') {
108        Some(newline) => source.len() - newline,
109        None => source.len() + 1,
110    };
111
112    Position::new(source.len(), column, line)
113}

115/// An iterator adaptor that takes the tokenizer's output and yields one [`Chunk`] per
116/// top-level declaration, the module header first. See the module documentation for
117/// where it cuts.
118///
119/// # Advance or stop
120///
121/// This is the first consumer that reads the tokenizer past an error, which is the
122/// operation `BUG-4` and `BUG-5` were about: a tokenizer path that reports an error
123/// without consuming input reports it again on every poll. `Chunks` therefore ends the
124/// stream at the second of two consecutive equal errors, keeping the first in its chunk.
125/// A tokenizer that repeats one error truncates the parse there, where it would
126/// otherwise never end (`CLAUDE.md` — *A `Result`-yielding iterator must advance or
127/// stop*). Two consecutive errors that differ are both kept. The check catches only an
128/// error repeated back to back, which is the `BUG-4`/`BUG-5` shape: a cycle of different
129/// errors, or of an error and a token that consumed nothing, is not caught.
130///
131/// Once `next` has returned `None` it keeps returning `None`, whatever the source does,
132/// so `Chunks` is a `FusedIterator`.
133pub(crate) struct Chunks<I> {
134    tokens: I,
135    /// Where the source ends, which is where the last chunk's input ends.
136    end_of_source: Position,
137    /// The token that started the next chunk, read while the previous one was still
138    /// being collected.
139    pending: Option<Spanned<Position, Token>>,
140    /// Whether any item has been read yet. The first token of the file never cuts, so
141    /// the header chunk always opens on it.
142    read_any: bool,
143    /// The previous item read, when it was an error: the error a second, equal one is
144    /// compared against.
145    last_error: Option<Error>,
146    /// Set once the source is exhausted, or repeated an error.
147    finished: bool,
148}
149
150impl<I> Chunks<I>
151where
152    I: Iterator<Item = RawToken>,
153{
154    /// Cut `tokens`, a source's tokenizer output, into chunks. `end_of_source` is where
155    /// the last chunk's input ends — see [`end_of`].
156    pub(crate) fn new(tokens: I, end_of_source: Position) -> Chunks<I> {
157        Chunks {
158            tokens,
159            end_of_source,
160            pending: None,
161            read_any: false,
162            last_error: None,
163            finished: false,
164        }
165    }
166
167    /// Whether `token` opens a new chunk rather than continuing the current one.
168    fn cuts(&self, token: &Spanned<Position, Token>) -> bool {
169        self.read_any && token.span.start.column == 1 && can_start_declaration(&token.value)
170    }
171}

173impl<I> Iterator for Chunks<I>
174where
175    I: Iterator<Item = RawToken>,
176{
177    type Item = Chunk;
178
179    fn next(&mut self) -> Option<Chunk> {
180        let mut chunk = match self.pending.take() {
181            Some(first) => Chunk {
182                start: first.span.start.absolute,
183                tokens: vec![Ok(first)],
184                end: self.end_of_source,
185            },
186            None if self.finished => return None,
187            None => Chunk {
188                tokens: vec![],
189                start: BytePos(0),
190                end: self.end_of_source,
191            },
192        };
193
194        while !self.finished {
195            match self.tokens.next() {
196                None => self.finished = true,
197                Some(Ok(token)) => {
198                    self.last_error = None;
199                    if self.cuts(&token) {
200                        chunk.end = token.span.start;
201                        self.pending = Some(token);
202                        return Some(chunk);
203                    }
204                    self.read_any = true;
205                    chunk.tokens.push(Ok(token));
206                }
207                Some(Err(error)) => {
208                    if self.last_error.as_ref() == Some(&error) {
209                        self.finished = true;
210                    } else {
211                        self.read_any = true;
212                        self.last_error = Some(error.clone());
213                        chunk.tokens.push(Err(error));
214                    }
215                }
216            }
217        }
218
219        // The source ran out with nothing read since the last cut: there is no chunk to
220        // hand back. That only happens to a source with no items at all, since a cut
221        // always leaves its token pending.
222        if chunk.tokens.is_empty() {
223            return None;
224        }
225
226        Some(chunk)
227    }
228}

230/// `next` returns `None` only once `finished` is set and nothing is pending, and nothing
231/// clears `finished` or fills `pending` afterwards.
232impl<I> FusedIterator for Chunks<I> where I: Iterator<Item = RawToken> {}

243    fn chunks_of(source: &str) -> Vec<Chunk> {
244        let tokens = make_tokenizer(source).map(|r| r.map_err(Error::from));
245        Chunks::new(tokens, end_of(source)).collect()
246    }

248    /// The first token of every chunk, or `None` for one that opens on an error.
249    fn first_tokens(chunks: &[Chunk]) -> Vec<Option<Token>> {
250        chunks
251            .iter()
252            .map(|chunk| match chunk.tokens.first() {
253                Some(Ok(token)) => Some(token.value.clone()),
254                _ => None,
255            })
256            .collect()
257    }

259    fn lower(name: &str) -> Option<Token> {
260        Some(Token::LowerIdentifier(name.to_string()))
261    }

263    /// A column-1 token a declaration can start with cuts; an indented one and a
264    /// column-1 one no declaration starts with do not. The chunks tile the source, and
265    /// the last one ends at its end.
266    ///
267    /// Verified to fail by making `cuts` return `false`: the source is then one chunk.
268    #[test]
269    fn a_column_1_declaration_start_cuts_and_nothing_else_does() {
270        let source = indoc! {"
271            module M exposing (..)
272
273            f x =
274              g x
275            | oops
276            type T = A
277        "};
278
279        let chunks = chunks_of(source);
280
281        assert_eq!(
282            first_tokens(&chunks),
283            vec![Some(Token::Module), lower("f"), Some(Token::Type)]
284        );
285
286        let f_start = source.find("f x").unwrap();
287        let type_start = source.find("type").unwrap();
288        assert_eq!(chunks[0].span().to_range(), 0..f_start);
289        assert_eq!(chunks[1].span().to_range(), f_start..type_start);
290        assert_eq!(chunks[2].span().to_range(), type_start..source.len());
291        assert_eq!(chunks[1].end.column, 1);
292    }

321    /// A source that yields one token and then the same error without end is cut short
322    /// at the second error, so polling the adaptor ends.
323    ///
324    /// The source counts its polls and panics past a cap, which is how the test turns
325    /// a loop that would never return into a failure. Verified to fail by removing the
326    /// `self.last_error.as_ref() == Some(&error)` check from `next`: the first poll then
327    /// reads the source past its cap.
328    #[test]
329    fn a_repeated_tokenizer_error_ends_the_stream() {
330        const SOURCE_CAP: usize = 64;
331        const POLLS: usize = 4;
332
333        let polls = Rc::new(Cell::new(0));
334        let counter = Rc::clone(&polls);
335        let position = Position::new(0, 1, 1);
336        let error = Error::Tokenizer(TokenizerError {
337            error: spanned(
338                BytePos(1),
339                BytePos(2),
340                crate::parser::tokenizer::TokenizerErrorType::TabError,
341            ),
342        });
343        let mut first = Some(Ok(spanned(position, position, Token::Module)));
344        let source = std::iter::from_fn(move || {
345            counter.set(counter.get() + 1);
346            assert!(
347                counter.get() <= SOURCE_CAP,
348                "the adaptor kept reading a source that repeats one error"
349            );
350            Some(first.take().unwrap_or_else(|| Err(error.clone())))
351        });
352
353        let mut chunks = Chunks::new(source, position);
354        let items: Vec<Chunk> = (0..POLLS).map_while(|_| chunks.next()).collect();
355
356        assert_eq!(items.len(), 1, "expected one chunk, got {:?}", items);
357        assert_eq!(items[0].tokens.len(), 2, "the token and the first error");
358        assert!(
359            chunks.next().is_none(),
360            "the adaptor restarted after ending"
361        );
362        assert!(
363            polls.get() <= 3,
364            "read {} items from the source",
365            polls.get()
366        );
367    }

369    /// `can_start_declaration` and the grammar's `Decl` alternatives agree on what a
370    /// declaration opens with. `Decls` reads `Decl+`, so a start the grammar accepts and
371    /// this function does not would not fail: that declaration would silently join the
372    /// previous chunk. The grammar's own first set is the `expected` list of the error a
373    /// stray `)` in column 1 raises, and each name in it is mapped to its token here.
374    ///
375    /// Verified to fail by removing `Token::Infix` from `can_start_declaration` (the
376    /// assertion on `infix` goes red).
377    #[test]
378    fn can_start_declaration_matches_the_grammars_declaration_starts() {
379        use crate::parser::{parse_recovering, Error};
380        use codespan_reporting::files::SimpleFile;
381
382        let source = "module Main exposing (..)\n\nf = 1\n)\n";
383        let file = SimpleFile::new("test".to_owned(), source.to_owned());
384        let parsed = parse_recovering(&file);
385        let expected = match parsed.failures.first().map(|f| &f.error) {
386            Some(Error::UnexpectedToken { expected, .. }) => expected.clone(),
387            other => panic!("expected an unexpected-token error, got {:?}", other),
388        };
389
390        // Every token kind a declaration could plausibly start with, by the name the
391        // grammar gives it, so a new one has to be added here to be compared.
392        let candidates: Vec<(&str, Token)> = vec![
393            ("lo_ident", Token::LowerIdentifier("f".to_string())),
394            ("left", Token::Left),
395            ("right", Token::Right),
396            ("non", Token::Non),
397            ("foreign", Token::Foreign),
398            ("unsafe", Token::Unsafe),
399            ("type", Token::Type),
400            ("import", Token::Import),
401            ("infix", Token::Infix),
402        ];
403
404        for (name, token) in &candidates {
405            let in_grammar = expected.iter().any(|e| e.trim_matches('"') == *name);
406            assert_eq!(
407                can_start_declaration(token),
408                in_grammar,
409                "`{}`: can_start_declaration disagrees with the grammar's expected {:?}",
410                name,
411                expected
412            );
413        }
414        for e in &expected {
415            let e = e.trim_matches('"');
416            assert!(
417                candidates.iter().any(|(name, _)| *name == e),
418                "the grammar accepts `{}` to start a declaration and this test does not list it",
419                e
420            );
421        }
422    }

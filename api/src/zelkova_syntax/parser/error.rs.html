10#[derive(Debug, PartialEq, Clone)]
11pub enum Error {
12    Tokenizer(TokenizerError),
13    Layout(LayoutError),
14    // Errors coming from the parser
15    InvalidToken(BytePos),
16    UnexpectedEOF {
17        position: BytePos,
18        expected: Vec<String>, // The kind of token the parser was expecting
19    },
20    UnexpectedToken {
21        token: Spanned<BytePos, Token>,
22        expected: Vec<String>, // The kind of token the parser was expecting
23    },
24    ExtraToken {
25        token: Spanned<BytePos, Token>,
26    },
27    /// An `infix` declaration's precedence does not fit in a `u8` — `infix left 300
28    /// (op) = f`. The grammar's `Infix` production parses the precedence as the
29    /// tokenizer's `"integer"` (an `i64`) and narrows it, so this is a distinct
30    /// variant from every `TokenizerErrorType`: the literal itself tokenized fine,
31    /// and the grammar is what rejects it (`BUG-12`).
32    InfixPrecedenceOutOfRange {
33        precedence: Spanned<BytePos, i64>,
34    },
35    /// A `.` the tokenizer did not read as `Token::Dot`, where the grammar did not take
36    /// it: one with whitespace after it — `Widget . size`, `Widget. size`, `. name`,
37    /// `import Ui . Widget` — or before an uppercase name, `Widget .Size`, which are
38    /// `Token::SpacedDot`; or an accessor's `.` where no expression may stand, as in
39    /// `import Ui .widget`, which is `Token::AccessorDot`. No production consumes the
40    /// first, and the second only where an expression begins, so the parser reports
41    /// either as an unexpected token and `From<ParseError>` turns that into this. `dot`
42    /// is the `.`.
43    ///
44    /// `continues_a_name` is whether an uppercase name stands right before the `.` and the
45    /// parser, at the point it met the `.`, would have accepted a `Token::Dot` there: a
46    /// `.` would have extended that name into a qualified one, and the whitespace is what
47    /// stopped it. Only then does the message speak of a qualified name; `main = . name`,
48    /// `main = 1 .` and `main = a . b` have no qualified name in them, and advice about
49    /// one would be about something the user did not write. `From<ParseError>` sees only
50    /// the second half, which a field access also satisfies, and `parser::parse_chunk`,
51    /// which has the tokens, adds the first.
52    SpacedDot {
53        dot: Span<BytePos>,
54        continues_a_name: bool,
55    },
56}

58impl Error {
59    pub fn diagnostic<Id: Copy>(&self, name: Id) -> Diagnostic<Id> {
60        match self {
61            Error::UnexpectedToken { token, expected } => {
62                Diagnostic::error()
63                    .with_message(format!("unexpected token: `{:?}`", token.value)) // TODO display instead of debug
64                    .with_labels(vec![Label::primary(name, token.span.to_range())
65                        .with_message("unexpected token")])
66                    .with_notes(vec![format!(
67                        "we were expecting one of the following tokens: {:?}",
68                        expected
69                    )
70                    .to_owned()])
71            }
72            Error::Tokenizer(err) => {
73                let diag = Diagnostic::error();
74                match err.error.value {
75                    TokenizerErrorType::CharNotClosedError(None) => {
76                        diag.with_message("char sequence opened but never closed")
77                            .with_labels(vec![
78                                Label::primary(name, err.error.span.to_range())
79                                    .with_message("The char is declared here but not closed")
80                            ])
81                    }
82                    TokenizerErrorType::CharNotClosedError(Some(_)) => {
83                        // `span.start` is the opening quote and `span.end` falls
84                        // inside the character sitting where the closing quote
85                        // should have been (tokenizer.rs, the
86                        // `CharNotClosedError(Some(_))` arm). Both are single
87                        // `BytePos`es, so `one_byte_at` widens each into a
88                        // visible one-byte range the same way `InvalidToken` and
89                        // `UnexpectedEOF` do (`BUG-7`).
90                        let open = err.error.span.start;
91                        let close = err.error.span.end;
92                        diag.with_message("char sequence opened but never closed")
93                            .with_labels(vec![
94                                Label::primary(name, one_byte_at(close))
95                                    .with_message("we were expecting a single quote here"),
96                                Label::secondary(name, one_byte_at(open))
97                                    .with_message("for the opening quote here")
98                            ])
99                    }
100                    TokenizerErrorType::StringNotClosedError => diag
101                        .with_message("string opened but never closed")
102                        .with_labels(vec![
103                            Label::primary(name, err.error.span.to_range())
104                                .with_message("this string reaches the end of its line without a closing `\"`")
105                        ])
106                        .with_notes(vec![
107                            "a string may not contain a line ending; write `\\n` for one".to_owned()
108                        ]),
109                    TokenizerErrorType::InvalidEscape => diag
110                        .with_message("this is not an escape sequence")
111                        .with_labels(vec![Label::primary(name, err.error.span.to_range())
112                            .with_message("unknown escape sequence")])
113                        .with_notes(vec![
114                            "the escape sequences are `\\n`, `\\r`, `\\t`, `\\\\`, `\\'`, `\\\"` and `\\u{…}`".to_owned()
115                        ]),
116                    TokenizerErrorType::UnicodeError => diag
117                        .with_message("this unicode escape sequence could not be read")
118                        .with_labels(vec![Label::primary(name, err.error.span.to_range())
119                            .with_message("this escape sequence")])
120                        .with_notes(vec![
121                            "a unicode escape is `\\u{`, one to six hexadecimal digits naming a Unicode scalar value, and `}`".to_owned()
122                        ]),
123                    TokenizerErrorType::IndentationError => {
124                        diag.with_message("Invalid indentation level")
125                            .with_labels(vec![
126                                Label::primary(name, err.error.span.to_range())
127                            ])
128                            .with_notes(vec![
129                                "Zelkova use exclusively two spaces to denote indentation but an odd number of spaces was found".to_owned()
130                            ])
131                    }
132                    TokenizerErrorType::TabError => {
133                        diag.with_message("Tab found")
134                        .with_labels(vec![
135                            Label::primary(name, err.error.span.to_range())
136                        ])
137                        .with_notes(vec!["Zelkova use exclusively two spaces to denote indentation and forbid the usage of tabs".to_owned()])
138                    }
139                    TokenizerErrorType::UnrecognizedToken { tok } => {
140                        Diagnostic::error()
141                        .with_message("Unexpected token found")
142                        .with_labels(vec![Label::primary(name, err.error.span.to_range())
143                            .with_message(format!("Unrecognized token {} found", tok))])
144                    }
145                    TokenizerErrorType::IntegerOverflow => diag
146                        .with_message("this integer is too large")
147                        .with_labels(vec![Label::primary(name, err.error.span.to_range())
148                            .with_message("this literal does not fit in a 64-bit signed integer")])
149                        .with_notes(vec![
150                            "an integer literal must fit in a 64-bit signed integer; Zelkova guarantees every integer in -2^31 .. 2^31 - 1 is representable on every target".to_owned()
151                        ]),
152                    TokenizerErrorType::MultipleDecimalPoints => diag
153                        .with_message("a number has one decimal point")
154                        .with_labels(vec![Label::primary(name, err.error.span.to_range())
155                            .with_message("this literal has more than one `.`")]),
156                    TokenizerErrorType::NonAsciiDigit { digit } => diag
157                        .with_message("a numeric literal is written with the digits 0-9")
158                        .with_labels(vec![Label::primary(name, err.error.span.to_range())
159                            .with_message(format!("`{}` is not one of the digits 0-9", digit))]),
160                    TokenizerErrorType::MalformedNumber => diag
161                        .with_message("this is not a number")
162                        .with_labels(vec![Label::primary(name, err.error.span.to_range())
163                            .with_message("this literal cannot be read as an integer or a float")]),
164                    TokenizerErrorType::UnclosedBlockComment => diag
165                        .with_message("this block comment is never closed")
166                        .with_labels(vec![Label::primary(name, non_empty(err.error.span.to_range()))
167                            .with_message("this comment opens here, but no matching `-}` was found before the end of the file")])
168                        .with_notes(vec![
169                            "block comments nest, so every `{-` needs its own `-}`".to_owned()
170                        ]),
171                }
172            }
173
174            Error::Layout(LayoutError::LayoutError { offside, token }) => {
175                // A layout token carries `Position`s rather than `BytePos`, so the
176                // range is built from their byte offsets. The layout pass injects
177                // `OpenBlock`/`CloseBlock` tokens whose start and end are equal and
178                // an indentation error can land on one of those, hence `non_empty`.
179                let start = token.span.start.absolute.0 as usize;
180                let end = token.span.end.absolute.0 as usize;
181
182                Diagnostic::error()
183                    .with_message("this line is not indented far enough")
184                    .with_labels(vec![Label::primary(name, non_empty(start..end))
185                        .with_message(format!(
186                        "this token starts at column {}, but its block requires column {} or more",
187                        token.span.start.column,
188                        offside.min_indent()
189                    ))])
190                    .with_notes(vec![format!(
191                        "the block it belongs to ({}) starts on line {}",
192                        offside.context().description(),
193                        offside.line()
194                    )])
195            }
196
197            Error::Layout(LayoutError::IndentedDeclaration {
198                token,
199                declaration_line,
200            }) => {
201                let start = token.span.start.absolute.0 as usize;
202                let end = token.span.end.absolute.0 as usize;
203
204                Diagnostic::error()
205                    .with_message("this line is indented, so it continues the declaration above it")
206                    .with_labels(vec![Label::primary(name, non_empty(start..end))
207                        .with_message(format!(
208                            "this token starts at column {}, so it is read as part of the declaration on line {}, which was already complete",
209                            token.span.start.column, declaration_line
210                        ))])
211                    .with_notes(vec![
212                        "a top-level declaration begins in column 1; move this line there if it starts a new declaration".to_owned()
213                    ])
214            }
215
216            Error::Layout(LayoutError::MemberOnOpeningLine { token }) => Diagnostic::error()
217                .with_message("the first member of this body is on the line that opens it")
218                .with_labels(vec![Label::primary(name, non_empty(token_range(token)))
219                    .with_message("this member starts on the same line as `where`")])
220                .with_notes(vec![
221                    "the members of a `class` or `instance` start on the next line, indented past the declaration".to_owned()
222                ]),
223
224            Error::Layout(LayoutError::WhereOnItsOwnLine { token }) => Diagnostic::error()
225                .with_message("`where` is alone at the start of its line")
226                .with_labels(vec![Label::primary(name, non_empty(token_range(token)))
227                    .with_message("this `where` begins a line")])
228                .with_notes(vec![
229                    "put `where` at the end of the line holding the head of the `class` or `instance`".to_owned()
230                ]),
231
232            Error::Layout(LayoutError::MisalignedMember {
233                token,
234                member_column,
235            }) => Diagnostic::error()
236                .with_message("this line is not on the column of the members above it")
237                .with_labels(vec![Label::primary(name, non_empty(token_range(token)))
238                    .with_message(format!(
239                        "this token starts at column {}, but the members start at column {}",
240                        token.span.start.column, member_column
241                    ))])
242                .with_notes(vec![
243                    "every member of a `class` or `instance` starts on the column of the first one".to_owned()
244                ]),
245
246            Error::InvalidToken(position) => Diagnostic::error()
247                .with_message("the parser could not read this token")
248                .with_labels(vec![Label::primary(name, one_byte_at(*position))
249                    .with_message("the parser stopped here")]),
250
251            Error::UnexpectedEOF { position, expected } => Diagnostic::error()
252                .with_message("the file ended before the declaration did")
253                .with_labels(vec![
254                    Label::primary(name, one_byte_at(*position)).with_message("the file ends here")
255                ])
256                .with_notes(vec![format!(
257                    "we were expecting one of the following tokens here: {}",
258                    expected.join(", ")
259                )]),
260
261            Error::ExtraToken { token } => Diagnostic::error()
262                .with_message("the module continues past its end")
263                .with_labels(vec![Label::primary(name, non_empty(token.span.to_range()))
264                    .with_message("this token comes after the module was complete")]),
265
266            Error::InfixPrecedenceOutOfRange { precedence } => Diagnostic::error()
267                .with_message("this infix declaration's precedence is out of range")
268                .with_labels(vec![Label::primary(name, non_empty(precedence.span.to_range()))
269                    .with_message(format!(
270                        "{} does not fit between 0 and 255",
271                        precedence.value
272                    ))])
273                .with_notes(vec![
274                    "Zelkova represents an operator's precedence as one byte, so it must be between 0 and 255".to_owned()
275                ]),
276
277            Error::SpacedDot {
278                dot,
279                continues_a_name: true,
280            } => Diagnostic::error()
281                .with_message("a qualified name is written with no spaces around its `.`")
282                .with_labels(vec![Label::primary(name, non_empty(dot.to_range()))
283                    .with_message("this `.` has whitespace next to it")])
284                .with_notes(vec![
285                    "write the name in one piece, as in `Dict.get` or `Ui.Widget`".to_owned()
286                ]),
287
288            Error::SpacedDot {
289                dot,
290                continues_a_name: false,
291            } => Diagnostic::error()
292                .with_message("unexpected `.`")
293                .with_labels(vec![Label::primary(name, non_empty(dot.to_range()))
294                    .with_message("a `.` cannot be used here")]),
295        }
296    }

299/// Widen a single position into a one byte range.
300///
301/// `BytePos::to_range` is zero width by construction (`u..u`), which
302/// codespan-reporting renders as a caret with no character above it. The errors
303/// which only know one position (`InvalidToken`, `UnexpectedEOF`) therefore
304/// underline the single byte starting there: that is the character the parser
305/// stopped on, and the range stays valid even at the end of the file.
306fn one_byte_at(position: BytePos) -> Range<usize> {
307    let start = position.0 as usize;
308
309    start..start + 1
310}
311
312/// Keep a span-derived range visible, for the same reason as `one_byte_at`: a
313/// range whose start equals its end underlines nothing.
314fn non_empty(range: Range<usize>) -> Range<usize> {
315    if range.start < range.end {
316        range
317    } else {
318        range.start..range.start + 1
319    }
320}

322/// lalrpop expected tokens in error are wrapped in double quote, which we don't really want
323fn unquote_tokens(mut tokens: Vec<String>) -> Vec<String> {
324    for token in &mut tokens {
325        if token.starts_with('"') {
326            token.remove(0);
327        }
328
329        if token.ends_with('"') {
330            token.pop();
331        }
332    }
333
334    tokens.to_vec()
335}

337impl From<ParseError<BytePos, Token, Error>> for Error {
338    fn from(e: ParseError<BytePos, Token, Error>) -> Self {
339        match e {
340            ParseError::InvalidToken { location } => Error::InvalidToken(location),
341            ParseError::UnrecognizedEof { location, expected } => Error::UnexpectedEOF {
342                position: location,
343                expected: unquote_tokens(expected),
344            },
345            ParseError::UnrecognizedToken {
346                token: (start, Token::SpacedDot | Token::AccessorDot, end),
347                expected,
348            } => Error::SpacedDot {
349                dot: Span { start, end },
350                // The terminal is spelled `"."` in the grammar, and lalrpop quotes it. A
351                // `Dot` is accepted after every operand a field access may be written on
352                // as well as after a name it would qualify, so this is only half of what
353                // the field says; `parser::parse_chunk` narrows it.
354                continues_a_name: expected.iter().any(|terminal| terminal == "\".\""),
355            },
356            ParseError::UnrecognizedToken { token, expected } => Error::UnexpectedToken {
357                token: token.into(),
358                expected: unquote_tokens(expected),
359            },
360            ParseError::ExtraToken { token } => Error::ExtraToken {
361                token: token.into(),
362            },
363            ParseError::User { error } => error,
364        }
365    }
366}

373/// The byte range of a layout-processed token, whose span carries `Position`s.
374fn token_range(token: &Spanned<Position, Token>) -> std::ops::Range<usize> {
375    token.span.start.absolute.0 as usize..token.span.end.absolute.0 as usize
376}

425        let tokens =
426            crate::parser::tokenizer::make_tokenizer(source).map(|r| r.map_err(|e| e.into()));

428        let end = crate::parser::chunk::end_of(source);
429        let mut errors: Vec<Error> = crate::parser::layout::layout(tokens, end)
430            .filter_map(|item| item.err())
431            .collect();
432
433        match errors.pop() {
434            Some(e) => e,
435            None => panic!("expected the source to fail the layout pass"),
436        }
437    }

629    /// must render as visible one-byte ranges, and the label at the opening

631    ///
632    /// Verified to fail two ways (`BUG-7`): reverting `one_byte_at` back to
633    /// `.to_range()` turns both ranges zero-width and reddens
634    /// `assert_points_at_source`; swapping the two `.with_message` calls back
635    /// puts "we were expecting a single quote here" on the opening-quote
636    /// label, which reddens the `opening_label` assertion below.
637    #[test]
638    fn char_not_closed_with_extra_char_labels_open_and_expected_close() {
639        let error: Error = crate::parser::tokenizer::make_tokenizer("'ab")
640            .collect::<Result<Vec<_>, _>>()
641            .expect_err("expected the source to fail the tokenizer")
642            .into();
643
644        let diagnostic = error.diagnostic(());
645
646        assert_prose_message(&diagnostic, "char sequence opened but never closed");
647        assert_points_at_source(&diagnostic);
648        assert_eq!(diagnostic.labels.len(), 2);
649
650        let opening_label = diagnostic
651            .labels
652            .iter()
653            .find(|label| label.range == (0..1))
654            .expect("no label at the opening quote (byte 0)");
655        assert_eq!(opening_label.message, "for the opening quote here");
656
657        let expected_close_label = diagnostic
658            .labels
659            .iter()
660            .find(|label| label.range == (2..3))
661            .expect("no label at the expected closing quote (byte 2)");
662        assert_eq!(
663            expected_close_label.message,
664            "we were expecting a single quote here"
665        );

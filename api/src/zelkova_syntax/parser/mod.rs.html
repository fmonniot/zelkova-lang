37/// Parse `source_file` into a `Module`, or report the first syntax error in it.
38///
39/// This is [`parse_recovering`] reduced to its first failure: a source with none is
40/// `Ok` of its module, and any other is `Err` of the first failure's error, in source
41/// order. Every caller that only wants to know whether a file parses, and why not, keeps
42/// that meaning.
43pub fn parse(source_file: &SimpleFile<String, String>) -> Result<Module, Error> {
44    let (header, declarations, failures) = parse_chunks(source_file.source());
45
46    match (header, failures.into_iter().next()) {
47        (Err(failure), _) | (Ok(_), Some(failure)) => Err(failure.error),
48        (Ok((modifier, name, exposing, exposing_span)), None) => Ok(Module::from_declarations(
49            modifier,
50            name,
51            exposing,
52            exposing_span,
53            declarations,
54            vec![],
55        )),
56    }
57}

59/// What [`parse_recovering`] makes of a source: every declaration that parsed, and every
60/// chunk that did not.
61#[derive(Debug)]
62pub struct Parsed {
63    /// The module, holding every declaration that parsed and, in
64    /// [`Module::failed`], a record of every declaration that did not. `None` exactly
65    /// when the module header is among the `failures`.
66    pub module: Option<Module>,
67    /// Every chunk that failed to parse, in source order: the header first if it failed,
68    /// then each declaration.
69    pub failures: Vec<Failure>,
70}
71
72/// One top-level declaration, or the module header, that failed to parse.
73#[derive(Debug, Clone, PartialEq)]
74pub struct Failure {
75    /// The source text of the chunk: from its first token to the first token of the
76    /// next chunk, or to the end of the source. The header's chunk starts at byte 0.
77    pub span: Span<BytePos>,
78    /// The first error the chunk's tokenizer, layout pass or grammar raised. Nothing
79    /// after it in the same chunk is reported.
80    pub error: Error,
81    /// The value the chunk declares, when its tokens say so: `Some(name)` for a
82    /// declaration chunk whose first item is the token `LowerIdentifier(name)`, `None`
83    /// for every other chunk and for the header.
84    ///
85    /// A chunk opening on a lowercase identifier is the annotation or a binding of the
86    /// value of that name, because those are the only two declaration forms the grammar
87    /// opens on one. Every other opening — `type`, `infix`, `import`, `unsafe` or another
88    /// soft keyword — is read as naming nothing: past the first token what the chunk
89    /// would have declared is a guess (`docs/decisions/dec-23.md`, decision 4).
90    pub declares: Option<Name>,
91}

93/// A record, kept on [`Module::failed`], of one declaration chunk that failed to parse.
94#[derive(Debug, Clone, PartialEq)]
95pub struct Failed {
96    /// The chunk's source text, as [`Failure::span`].
97    pub span: Span<BytePos>,
98    /// The value the chunk declares, as [`Failure::declares`].
99    pub declares: Option<Name>,
100}

102/// Parse `source_file`, carrying on past a syntax error to the next top-level
103/// declaration.
104///
105/// The token stream is cut into one chunk per top-level declaration before the layout
106/// pass sees it (`chunk.rs` has where it cuts and why there), and each chunk gets its own
107/// `Layout` and its own grammar entry point: `Header` for the first, `Decls` for every
108/// other. A chunk reports the first error raised in it and nothing after, which is the
109/// rule a whole module followed before the cut. The other chunks are parsed regardless,
110/// for their declarations or their errors.
111pub fn parse_recovering(source_file: &SimpleFile<String, String>) -> Parsed {
112    let (header, declarations, failures) = parse_chunks(source_file.source());
113
114    match header {
115        Ok((modifier, name, exposing, exposing_span)) => Parsed {
116            module: Some(Module::from_declarations(
117                modifier,
118                name,
119                exposing,
120                exposing_span,
121                declarations,
122                failures
123                    .iter()
124                    .map(|failure| Failed {
125                        span: failure.span,
126                        declares: failure.declares.clone(),
127                    })
128                    .collect(),
129            )),
130            failures,
131        },
132        Err(failure) => Parsed {
133            module: None,
134            failures: std::iter::once(failure).chain(failures).collect(),
135        },
136    }
137}

139/// The parts of a module header the `Header` entry point parses: the arguments
140/// `Module::from_declarations` takes before the declarations.
141type Header = (Option<tokenizer::Token>, Name, Exposing, NodeSpan);
142
143/// Cut `source` into chunks and parse each: the header, the declarations that parsed in
144/// source order, and the failures among the chunks after the header, in source order.
145fn parse_chunks(source: &str) -> (Result<Header, Failure>, Vec<Declaration>, Vec<Failure>) {
146    let end = chunk::end_of(source);
147    let tokens = tokenizer::make_tokenizer(source).map(|r| r.map_err(Error::from));
148    let mut chunks = chunk::Chunks::new(tokens, end);
149    let mut declarations = vec![];
150    let mut failures = vec![];
151
152    // A source with no tokens at all still has a header to fail, at its end.
153    let header_chunk = chunks.next().unwrap_or_else(|| chunk::Chunk::empty(end));
154    let header = match parse_chunk(&header_chunk, |tokens| {
155        grammar::HeaderParser::new().parse(tokens)
156    }) {
157        Ok((header, header_declarations)) => {
158            declarations.extend(header_declarations);
159            Ok(header)
160        }
161        Err(error) => Err(Failure {
162            span: header_chunk.span(),
163            error,
164            declares: None,
165        }),
166    };
167
168    for chunk in chunks {
169        match parse_chunk(&chunk, |tokens| grammar::DeclsParser::new().parse(tokens)) {
170            Ok(chunk_declarations) => declarations.extend(chunk_declarations),
171            Err(error) => failures.push(Failure {
172                span: chunk.span(),
173                error,
174                declares: chunk.declares(),
175            }),
176        }
177    }
178
179    (header, declarations, failures)
180}

182/// The tokens a grammar entry point reads: the layout pass's output.
183type LayoutItem = Result<(BytePos, tokenizer::Token, BytePos), Error>;
184
185/// Run one chunk through its own `Layout` and the grammar entry point `entry`.
186///
187/// A grammar error can be a layout rule seen from the grammar's side, which only the
188/// layout pass can recognise, so it gets to explain the error first. What it asks back —
189/// whether the chunk's tokens before a line parse on their own — is answered with the
190/// same entry point, on a `Layout` that ends where that line starts.
191fn parse_chunk<T>(
192    chunk: &chunk::Chunk,
193    entry: impl Fn(
194        &mut dyn Iterator<Item = LayoutItem>,
195    ) -> Result<T, ParseError<BytePos, tokenizer::Token, Error>>,
196) -> Result<T, Error> {
197    let mut indented = layout::Layout::new(chunk.tokens.iter().cloned(), chunk.end);
198
199    entry(&mut indented).map_err(|e| {
200        let error = after_a_name(e.into(), &chunk.tokens);
201
202        indented.explain(error, |line_start| {
203            let before = chunk.tokens.iter().take_while(|item| {
204                matches!(item, Ok(token) if token.span.start.absolute.0 < line_start.absolute.0)
205            });
206            entry(&mut layout::Layout::new(before.cloned(), line_start)).is_ok()
207        })
208    })
209}

211/// `error`, with an [`Error::SpacedDot`]'s `continues_a_name` kept only when the token
212/// right before the `.` is an uppercase name.
213///
214/// `From<ParseError>` sets the flag when the grammar would have accepted a `Dot` where it
215/// met the `.`, and that holds after every operand a field access may be written on —
216/// `a . b`, `1 .`, `r. name` — as well as after a name a qualification continues. Only
217/// the second has a qualified name for the message to speak of, and the grammar's
218/// expected tokens cannot tell the two apart, so this reads the token itself.
219fn after_a_name(error: Error, tokens: &[chunk::RawToken]) -> Error {
220    match error {
221        Error::SpacedDot {
222            dot,
223            continues_a_name: true,
224        } => {
225            let before = tokens
226                .iter()
227                .filter_map(|item| item.as_ref().ok())
228                .take_while(|token| token.span.start.absolute.0 < dot.start.0)
229                .last();
230
231            Error::SpacedDot {
232                dot,
233                continues_a_name: matches!(
234                    before,
235                    Some(token) if matches!(token.value, tokenizer::Token::UpperIdentifier(_))
236                ),
237            }
238        }
239        error => error,
240    }
241}

243/// A part of a declared type. This is also used in type annotations.
244///
245/// Types with no arguments will be composed of exactly one `Type`.
246/// As their name indicates, a type with arguments will requires more
247/// `Type` as arguments. For example the optional type will require
248/// one no-arg type: `Maybe Int` (this is another name for higher-kinded types)
249///
250/// # Why the span is a field rather than a wrapper
251///
252/// The three recursive nodes of this AST — `Type`, [`Pattern`] and [`Expression`] —
253/// are each a struct pairing a [`NodeSpan`] with a `…Kind` enum, rather than the
254/// enum wrapped in a `Spanned<BytePos, _>`. The difference shows up in the children:
255/// here they stay `Box<Type>` and `Vec<Type>`, each carrying its own span, so a
256/// reader matches `&t.kind` once per function instead of unwrapping a `.value` at
257/// every child. `canonical/mod.rs` is that reader.
258#[derive(Debug, PartialEq, Clone)]
259pub struct Type {
260    pub span: NodeSpan,
261    pub kind: TypeKind,
262}
263
264#[derive(Debug, PartialEq, Clone)]
265pub enum TypeKind {
266    /// Type constructor
267    Unqualified(Name, Vec<Type>),
268    // TODO Qualified type eg Maybe.Maybe (or is it already merged in Unqualified ?)
269    /// Type constructor →
270    ///
271    /// Applications of the type constructor → are written infix and
272    /// associate to the right, so T → T' → T" stands for T → (T' → T").
273    Arrow(Box<Type>, Box<Type>),
274    /// Type variable
275    Variable(Name),
276    /// A tuple type, of two or three elements — see [`Tuple`].
277    Tuple(Tuple<Type>),
278    /// The unit type, `()`. Not a tuple: [`Tuple`] has no arity of zero.
279    Unit,
280    /// A record type, `{ label : Type, … }`, its fields in the order they were
281    /// written. The grammar never builds one with no field.
282    ///
283    /// The order means nothing — a record type is a set of fields
284    /// (`docs/spec/records.md`) — and it is kept only because this is the source as
285    /// written: canonicalization is where the list becomes a set and where a repeated
286    /// label is reported.
287    Record(Vec<Field<Type>>),
288}
289
290/// One field of a record type, a record or an update, or one entry of a record
291/// pattern: a label and what it was given, `label : Type`, `label = expr` or `label =
292/// pattern`.
293#[derive(Debug, PartialEq, Clone)]
294pub struct Field<T> {
295    pub label: Name,
296    /// Where the label alone was written. A diagnostic about the label — a repeated
297    /// one — points here, and the value carries a span of its own.
298    pub label_span: NodeSpan,
299    pub value: T,
300}
301
302impl<T> Field<T> {
303    /// A field the parser built, its label at `label_span`.
304    pub fn new(label: Name, label_span: NodeSpan, value: T) -> Field<T> {
305        Field {
306            label,
307            label_span,
308            value,
309        }
310    }
311}

336/// A Module is the top-level structure for a source file.
337///
338/// It contains everything in a source file.
339///
340///
341/// The elm compiler declare a module as follow:
342/// ```haskell
343/// data Module =
344///   Module
345///     { _name    :: ModuleName.Canonical
346///     , _exports :: Exports
347///     , _docs    :: Src.Docs
348///     , _decls   :: Decls
349///     , _unions  :: Map.Map Name Union
350///     , _aliases :: Map.Map Name Alias
351///     , _binops  :: Map.Map Name Binop
352///     , _effects :: Effects
353///     }
354/// ```
355#[derive(Debug, PartialEq)]
356pub struct Module {
357    pub name: Name,
358    pub binding_foreign: bool,
359    pub exposing: Exposing,
360    /// Where the header's `exposing (...)` was written, the keyword and the list together.
361    /// A diagnostic about what the module exposes as a whole, rather than about one name
362    /// in the list, points here.
363    pub exposing_span: NodeSpan,
364    pub imports: Vec<Import>,
365    pub infixes: Vec<Infix>,
366    pub types: Vec<UnionType>,
367    /// Every `class` declaration, in source order.
368    pub classes: Vec<ClassDecl>,
369    /// Every `instance` declaration, in source order.
370    pub instances: Vec<InstanceDecl>,
371    pub functions: Vec<Function>,
372    /// The declaration chunks that failed to parse, in source order.
373    ///
374    /// Only [`parse_recovering`] fills it: [`parse`] has no module to return when any
375    /// declaration failed. A failed chunk that names a value is how canonicalization
376    /// knows the name exists while nothing of it parsed, and one that names nothing is
377    /// how it knows a name could be missing from the module's scope.
378    pub failed: Vec<Failed>,
379}
380
381impl Module {
382    fn from_declarations(
383        modifier: Option<tokenizer::Token>,
384        name: Name,
385        exposing: Exposing,
386        exposing_span: NodeSpan,
387        declarations: Vec<Declaration>,
388        failed: Vec<Failed>,
389    ) -> Module {
390        let binding_foreign = matches!(modifier, Some(tokenizer::Token::Foreign));
391
392        let mut imports = vec![];
393        let mut types = vec![];
394        let mut infixes = vec![];
395        let mut classes = vec![];
396        let mut instances = vec![];
397        let mut functions = HashMap::<Name, Vec<Declaration>>::new();
398
399        for declaration in declarations {
400            match declaration {
401                Declaration::Function(FunBinding { ref name, .. })
402                | Declaration::FunctionType(FunType { ref name, .. }) => {
403                    match functions.get_mut(name) {
404                        Some(decls) => decls.push(declaration),
405                        None => {
406                            functions.insert(name.clone(), vec![declaration]);
407                        }
408                    }
409                }
410                Declaration::Import(i) => imports.push(i),
411                Declaration::Infix(i) => infixes.push(i),
412                Declaration::Union(t) => types.push(t),
413                Declaration::Class(c) => classes.push(c),
414                Declaration::Instance(i) => instances.push(i),
415            }
416        }
417
418        let functions = functions.into_iter().map(|(name, decls)| {
419            let mut tpe = None;
420            let mut context = None;
421            let mut marked_unsafe = false;
422            let mut bindings = vec![];
423            // The declarations that make one function were parsed independently, so
424            // the function's span is the union of theirs: the annotation merged with
425            // every binding. `merge` tolerates a missing half, which is what an
426            // annotation with no body (a `module foreign` facade) needs.
427            let mut span = NodeSpan::none();
428            // The annotation on its own, kept beside the merged span because a type
429            // error points at the annotation to say where the expected type came from.
430            let mut annotation_span = NodeSpan::none();
431
432            // TODO Error if more than function type is defined
433
434            for d in decls {
435                match d {
436                    Declaration::Function(b) => {
437                        span = span.merge(b.span);
438                        bindings.push(b.pattern);
439                    }
440                    Declaration::FunctionType(t) => {
441                        span = span.merge(t.span);
442                        annotation_span = t.span;
443                        marked_unsafe = t.marked_unsafe;
444                        context = t.context;
445                        tpe.replace(t.tpe);
446                    }
447                    _ => panic!("Invalid kind of declaration used in functions, report this error ({:?})", d),
448                }
449            }
450
451            Function { name, tpe, context, marked_unsafe, bindings, span, annotation_span }
452        }).collect::<Vec<_>>();
453
454        Module {
455            name,
456            binding_foreign,
457            exposing,
458            exposing_span,
459            imports,
460            infixes,
461            types,
462            classes,
463            instances,
464            functions,
465            failed,
466        }
467    }
468}

470/// `Function` represent a function declaration in the source code.
471///
472/// In _zelkova_, like in _Haskell_ but as opposed to _Elm_, we can
473/// use pattern matching and multiple line to declare a function:
474///
475/// ```zel
476/// map : (a -> b) -> Maybe a -> Maybe b
477/// map f (Just value) = f value
478/// map _ Nothing => Nothing
479/// ```
480///
481/// Because of this syntax, a function is defined as a list of `Match`
482/// statement, which each entry specifying a line. The parser will happily let
483/// us define a function which doesn't match all cases, a later phase will
484/// need to check this.
485#[derive(Debug, PartialEq)]
486pub struct Function {
487    pub name: Name,
488    pub tpe: Option<Type>,
489    /// What the annotation wrote in front of `=>`, if anything — see
490    /// [`FunType::context`]. Always `None` when `tpe` is.
491    pub context: Option<Context>,
492    /// True when this function's annotation was written `unsafe name : Type`.
493    ///
494    /// Read off the [`FunType`] that contributed the annotation, so a function
495    /// with no annotation is never marked. See [`FunType::marked_unsafe`].
496    pub marked_unsafe: bool,
497    pub bindings: Vec<Match>,
498    /// The annotation and every binding, merged into one span.
499    ///
500    /// A `Function` is assembled in `Module::from_declarations` out of
501    /// declarations the grammar saw separately, so this covers the annotation
502    /// *and* the body rather than either alone — which is what a type mismatch
503    /// between the two is actually about.
504    pub span: NodeSpan,
505    /// Where the `name : Type` annotation alone was written, or
506    /// [`NodeSpan::none`] when the function has no annotation.
507    ///
508    /// The merged `span` above cannot answer "where does the expected type come
509    /// from", because it also covers the body that contradicts it. A type error
510    /// draws its secondary label — *expected because of this type annotation* —
511    /// here.
512    pub annotation_span: NodeSpan,
513}

515/// Exposing represent whether an import (or export) expose terms.
516///
517/// `Exposing::Explicit` represents a selection of term exported
518/// (or imported) for a given module.
519///
520/// `Exposing::Open` means every top-level terms are exported, or when

522#[derive(Debug, PartialEq)]
523pub enum Exposing {
524    Open,
525    Explicit(Vec<Exposed>),

528/// A single name in an `exposing` list — either a module's header
529/// (`module Foo exposing (bar)`) or an import's (`import Foo exposing (bar)`).

537    pub span: NodeSpan,
538    pub kind: ExposedKind,

563/// Privacy control how a custom type is exposed.
564///
565/// For example, given the following type:

597/// A `class` declaration: `class Eq a => Comparable a where`, and the members under it.

673/// Represents the type signature of a particular function
674#[derive(Debug, PartialEq)]
675pub struct FunType {
676    pub name: Name,
677    /// The type after `=>`, or the whole annotation when there is no context.
678    pub tpe: Type,
679    /// What was written in front of `=>` — `Comparable a` in
680    /// `min : Comparable a => a -> a -> a` — or `None` for an unconstrained
681    /// annotation. See [`Context`] for why its elements are types.
682    ///
683    /// A context is a property of a *signature*, so it lives on the signature
684    /// rather than as a [`TypeKind`] variant. `TypeKind` is the shape of every type
685    /// the parser builds, variants and nested arguments included, and a
686    /// `Constrained` case there would be one that every match over it had to handle
687    /// while only ever being legal at the top of an annotation. Here the grammar
688    /// cannot put one anywhere else, and a `class` or `instance` head — the other
689    /// place `=>` is written — reuses the same production and the same pair.
690    pub context: Option<Context>,
691    /// True when the annotation was written `unsafe name : Type`.
692    ///
693    /// The word only means something on a [facade](../../docs/spec/interop.md)
694    /// signature, where it declares a plain function instead of the effect a
695    /// facade declares by default. The grammar accepts it on any annotation, and
696    /// canonicalization is what rejects one outside a `module foreign` header.
697    pub marked_unsafe: bool,
698    /// Where the annotation — `unsafe name : Type`, modifier included — was
699    /// written.
700    pub span: NodeSpan,
701}
702
703/// The constraints written in front of an annotation's `=>`: one, or a
704/// parenthesised, comma-separated list of any length.
705///
706/// Each element is a [`Type`] and not yet a constraint, because the grammar cannot
707/// tell the two apart: `(Comparable k, Eq v)` is the same tokens as a two-tuple
708/// type, so a context of up to three is parsed as a type and split here by
709/// `Context::from_type`, and anything type-shaped can be an element, `Int -> Int`
710/// included. Canonicalization is what checks that each one is shaped like a
711/// constraint.
712///
713/// A list of any length rather than a [`Tuple`]: a context has no arity limit,
714/// and a tuple type keeps its own.
715#[derive(Debug, PartialEq)]
716pub struct Context {
717    /// What was written between the parentheses, in order, or the one constraint
718    /// written without them.
719    pub constraints: Vec<Type>,
720    /// Where the context was written: a list's parentheses included, a single
721    /// constraint's own span otherwise.
722    pub span: NodeSpan,
723}
724
725impl Context {
726    /// Read a context the grammar parsed as a type. A tuple type is the list, its
727    /// elements the constraints; any other type is one constraint. Only the
728    /// outermost tuple is the list — one nested inside it stays an element, for
729    /// canonicalization to reject.
730    fn from_type(context: Type) -> Context {
731        let span = context.span;
732        let constraints = match context.kind {
733            TypeKind::Tuple(Tuple::Two(a, b)) => vec![*a, *b],
734            TypeKind::Tuple(Tuple::Three(a, b, c)) => vec![*a, *b, *c],
735            kind => vec![Type { kind, span }],
736        };
737
738        Context { constraints, span }
739    }
740}

777/// A `FunBinding` is one of the (possibly multiple) function declaration.
778///
779/// ## Examples
780///
781///
782/// `const = 42` in Zelkova will result in the following AST:
783/// ```text
784/// Declaration::Function(
785///     FunBinding {
786///         name: Name("const"),
787///         patterns: Match {
788///             pattern: [],
789///             body: Expression::Lit(Literal::Int(42)),
790///         }
791///     }
792/// )
793/// ```
794///
795///
796/// `identity x y = x` in Zelkova will result in the following AST:
797/// ```text
798/// Declaration::Function(
799///     FunBinding {
800///         name: Name("identity"),
801///         pattern: Match {
802///             patterns: [ Pattern::Var(Name("x")), Pattern::Var(Name("y")) ],
803///             body: Expression::Var(Name("x")),
804///         ]
805///     }
806/// )
807/// ```
808#[derive(Debug, PartialEq)]
809pub struct FunBinding {
810    pub name: Name,
811    pub pattern: Match,
812    /// Where this one binding — patterns and body — was written.
813    pub span: NodeSpan,
814}
815
816impl FunType {
817    /// Build a signature out of the pieces the grammar captured. The grammar has
818    /// one production per way of spelling the name and the `unsafe` modifier, and
819    /// this is the body they share; `constrained` is what `ConstrainedType`
820    /// produced, the context first.
821    fn assemble(
822        name: Name,
823        constrained: (Option<Context>, Type),
824        marked_unsafe: bool,
825        span: NodeSpan,
826    ) -> FunType {
827        let (context, tpe) = constrained;
828
829        FunType {
830            name,
831            tpe,
832            context,
833            marked_unsafe,
834            span,
835        }
836    }
837}
838
839impl FunBinding {
840    /// Build a binding out of the pieces the grammar captured.
841    ///
842    /// The grammar has one production per way of spelling the name — an ordinary
843    /// lowercase name, or the soft keyword `unsafe` used as one — and this is the
844    /// body they share. `l` is the start of the name, `ml` the start of the
845    /// patterns; both ends come from the expression rather than an `@R`, for the
846    /// reason `FunBinding`'s productions in `grammar.lalrpop` set out.
847    fn assemble(
848        name: Name,
849        l: crate::position::BytePos,
850        ml: crate::position::BytePos,
851        patterns: Vec<Pattern>,
852        expr: Expression,
853    ) -> FunBinding {
854        let span = NodeSpan::to_end_of(l, expr.span);
855        let match_span = NodeSpan::to_end_of(ml, expr.span);
856
857        FunBinding {
858            name,
859            pattern: Match {
860                patterns,
861                body: expr,
862                span: match_span,
863            },
864            span,
865        }
866    }
867}

869/// The match structure is composed of a serie of patterns and an associated expression
870#[derive(Debug, PartialEq)]
871pub struct Match {
872    pub patterns: Vec<Pattern>,
873    pub body: Expression,
874    /// The patterns and the body, from the first pattern to the last byte of the
875    /// expression. It excludes the function's name, which the enclosing
876    /// [`FunBinding`] span covers.
877    pub span: NodeSpan,

880/// A pattern is the left handside of a pattern-match expression
881///
882/// A pattern-matching expression can be present in function declaration
883/// or as part of the `case of` syntax.
884///
885/// ## Missing Patterns
886/// - `Alias Pattern (Name)`
887/// - `Ctor Name [Pattern]`
888/// - `CtorQual Name Name [Pattern]`
889/// - `List [Pattern]`
890/// - `Cons Pattern Pattern`
891#[derive(Debug, PartialEq, Clone)]
892pub struct Pattern {
893    pub span: NodeSpan,
894    pub kind: PatternKind,
895}
896
897#[derive(Debug, PartialEq, Clone)]
898pub enum PatternKind {
899    Variable(Name),
900    Literal(Literal),
901    /// A tuple pattern, of two or three elements — see [`Tuple`].
902    Tuple(Tuple<Pattern>),
903    /// The unit pattern, `()`, which matches the one value of the unit type.
904    Unit,
905    Constructor(Name, Vec<Pattern>),
906    Anything,
907    /// A record pattern, `{ label = pattern, … }`, its entries in the order they were
908    /// written. The grammar never builds one with no entry.
909    ///
910    /// The shorthand `{ label }` arrives here already desugared to `label = label`, a
911    /// [`Variable`](PatternKind::Variable) of the label's own name spanned at the
912    /// label, so this is the one shape a record pattern has. A pattern names a subset
913    /// of a record's fields (`docs/spec/records.md`), so the entries are not a record
914    /// type's whole set; canonicalization is where a repeated label is reported.
915    Record(Vec<Field<Pattern>>),
916}
917
918impl Pattern {
919    /// A pattern the parser built, at the position its production captured.
920    pub fn new(span: NodeSpan, kind: PatternKind) -> Pattern {
921        Pattern { span, kind }
922    }
923
924    /// A pattern with no position — hand-built by a test. See [`NodeSpan`].
925    pub fn bare(kind: PatternKind) -> Pattern {
926        Pattern {
927            span: NodeSpan::none(),
928            kind,
929        }
930    }
931}

933/// An Expression
934///
935/// Like [`Type`] and [`Pattern`], this is a span plus a kind rather than a spanned
936/// enum; see [`Type`] for why.
937#[derive(Debug, PartialEq, Clone)]
938pub struct Expression {
939    pub span: NodeSpan,
940    pub kind: ExpressionKind,
941}
942
943#[derive(Debug, PartialEq, Clone)]
944pub enum ExpressionKind {
945    Lit(Literal),                                  // Literal, as other are fully named
946    Application(Box<Expression>, Box<Expression>), // TODO Rename Apply ?
947    Variable(Name),                                // TODO Qualified variable
948    TypeConstructor(Name),
949    /// A tuple expression, of two or three elements — see [`Tuple`].
950    Tuple(Tuple<Expression>),
951    /// The unit value, `()`.
952    Unit,
953    Case(Box<Expression>, Vec<CaseBranch>),
954    If(Box<Expression>, Box<Expression>, Box<Expression>),
955    /// A run of infix operator applications, exactly as the grammar saw them —
956    /// `a * b + c` is `InfixChain(a, [(*, _, b), (+, _, c)])`, not a tree.
957    ///
958    /// The grammar has no operator table: an operator's `infix` declaration may
959    /// live in another module, resolved only once canonicalization has that
960    /// module's `Interface` in hand, so `InfixExpr` cannot decide precedence or
961    /// associativity while parsing. It hands this flat sequence to
962    /// canonicalization instead — each tuple is one operator's name, the span it
963    /// was written at (which an invented `Application` node takes, the same way
964    /// the previous right-recursive rewrite did), and the operand to its right —
965    /// and canonicalization re-associates it into nested `Application` nodes
966    /// once it has resolved every operator through the infix environment. See
967    /// `canonical::Expression::from_parser`'s `InfixChain` arm.
968    InfixChain(Box<Expression>, Vec<(Name, NodeSpan, Expression)>),
969    /// A record, `{ label = expr, … }`. The grammar never builds one with no field.
970    ///
971    /// The fields stay in the order they were written, here and in the canonical
972    /// AST. Unlike a record type's, that order means something: each field is a
973    /// subexpression, and subexpressions are evaluated left to right
974    /// (`docs/spec/evaluation-semantics.md`).
975    Record(Vec<Field<Expression>>),
976    /// An update, `{ expr | label = expr, … }`: the record `expr` evaluates to, with
977    /// the named fields replaced. The fields keep their written order, as a
978    /// [`Record`](ExpressionKind::Record)'s do.
979    Update(Box<Expression>, Vec<Field<Expression>>),
980    /// A field access, `r.name`: the record, the label, and where the label alone was
981    /// written. The expression's own span covers the whole `r.name`.
982    ///
983    /// The record is never a bare constructor name: `Widget.size` is a qualified name,
984    /// and the grammar's `Accessible` says why that leaves the constructor out.
985    Access(Box<Expression>, Name, NodeSpan),
986    /// An accessor, `.name`, the function reading that field: the label, and where the
987    /// label alone was written. The expression's own span covers the `.` as well.
988    Accessor(Name, NodeSpan),
989}
990
991impl Expression {
992    /// An expression the parser built, at the position its production captured.
993    pub fn new(span: NodeSpan, kind: ExpressionKind) -> Expression {
994        Expression { span, kind }
995    }
996
997    /// An expression with no position — hand-built by a test. See [`NodeSpan`].
998    pub fn bare(kind: ExpressionKind) -> Expression {
999        Expression {
1000            span: NodeSpan::none(),
1001            kind,
1002        }
1003    }
1004}

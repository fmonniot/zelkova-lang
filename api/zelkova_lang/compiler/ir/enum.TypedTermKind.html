pub enum TypedTermKind {
    Int(i64),
    Bool(bool),
    Char(char),
    Float(f64),
    Identifier(Reference),
    Fun {

        body: Box<TypedTerm>,
    },
    Apply {
        fun: Box<TypedTerm>,
        arg: Box<TypedTerm>,
        saturation: Saturation,
    },
    If {

        false_branch: Box<TypedTerm>,
    },

        body: Box<TypedTerm>,
    },
    Tuple(Tuple<TypedTerm>),

        scrutinee: Box<TypedTerm>,
        branches: Vec<(TermPattern, Box<TypedTerm>)>,
        form: CaseForm,
    },
}
